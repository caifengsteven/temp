class SimpleFinancialDataGenerator:
    """Base class for generating financial time series data"""
    
    def generate(self, n_samples, n_timesteps):
        """Generate synthetic financial time series data"""
        raise NotImplementedError("Subclasses must implement generate method")
    
    def calculate_statistics(self, data):
        """Calculate trend and realized volatility for the given data"""
        log_returns = np.diff(np.log(data)) * 100
        trend = np.sum(log_returns)
        realized_vol = np.sum(log_returns**2)
        return trend, realized_vol
    
    def calculate_autocorrelation(self, data, lags=[1, 5, 10, 20, 30]):
        """Calculate autocorrelation of absolute returns for given lags"""
        log_returns = np.diff(np.log(data))
        abs_returns = np.abs(log_returns)
        acorrs = []
        
        for lag in lags:
            if lag >= len(abs_returns):
                acorrs.append(0)
                continue
                
            acorr = np.corrcoef(abs_returns[:-lag], abs_returns[lag:])[0, 1]
            acorrs.append(acorr)
            
        return acorrs
    
    def calculate_hill_index(self, data, q=0.05):
        """Calculate Hill index for the given data"""
        log_returns = np.diff(np.log(data))
        abs_returns = np.abs(log_returns)
        sorted_returns = np.sort(abs_returns)
        n = len(sorted_returns)
        k = int(n * q)
        
        threshold = sorted_returns[n-k]
        excesses = sorted_returns[n-k:]
        log_excesses = np.log(excesses/threshold)
        
        hill_index = 1 / np.mean(log_excesses) if np.mean(log_excesses) > 0 else np.nan
        return hill_index

class GBMGenerator(SimpleFinancialDataGenerator):
    """Geometric Brownian Motion generator"""
    
    def __init__(self, mu=0, sigma=1):
        self.mu = mu
        self.sigma = sigma
        
    def generate(self, n_samples, n_timesteps):
        """Generate GBM price paths"""
        dt = 1/n_timesteps
        paths = []
        
        for _ in range(n_samples):
            # Start with price of 100
            price = np.zeros(n_timesteps)
            price[0] = 100
            
            # Generate Wiener process increments
            dW = np.random.normal(0, np.sqrt(dt), n_timesteps-1)
            
            # Generate price path
            for t in range(1, n_timesteps):
                price[t] = price[t-1] * np.exp((self.mu - 0.5 * self.sigma**2) * dt + self.sigma * dW[t-1])
                
            paths.append(price)
            
        return np.array(paths)

class GARCHGenerator(SimpleFinancialDataGenerator):
    """GARCH(1,1) generator"""
    
    def __init__(self, omega=0.1, alpha=0.1, beta=0.8):
        self.omega = omega
        self.alpha = alpha
        self.beta = beta
        
    def generate(self, n_samples, n_timesteps):
        """Generate GARCH price paths"""
        paths = []
        
        for _ in range(n_samples):
            # Initialize arrays
            returns = np.zeros(n_timesteps)
            variance = np.zeros(n_timesteps)
            price = np.zeros(n_timesteps)
            price[0] = 100
            variance[0] = self.omega / (1 - self.alpha - self.beta)  # unconditional variance
            
            # Generate returns
            for t in range(1, n_timesteps):
                # Generate random innovation
                z = np.random.normal(0, 1)
                
                # Calculate variance according to GARCH(1,1)
                variance[t] = self.omega + self.alpha * (returns[t-1]**2) + self.beta * variance[t-1]
                
                # Calculate return
                returns[t] = np.sqrt(variance[t]) * z
                
                # Calculate price
                price[t] = price[t-1] * np.exp(returns[t])
                
            paths.append(price)
            
        return np.array(paths)
