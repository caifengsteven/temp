# Technical indicators
def calculate_sma(prices: List[float], window: int) -> Optional[float]:
    """Calculate Simple Moving Average"""
    if len(prices) < window:
        return None
    return sum(prices[-window:]) / window

def calculate_rsi(prices: List[float], window: int = 14) -> Optional[float]:
    """Calculate Relative Strength Index"""
    if len(prices) <= window:
        return None
    
    # Calculate price changes
    deltas = [prices[i] - prices[i-1] for i in range(1, len(prices))]
    
    # Get gains and losses
    gains = [delta if delta > 0 else 0 for delta in deltas]
    losses = [-delta if delta < 0 else 0 for delta in deltas]
    
    # Calculate average gains and losses
    avg_gain = sum(gains[-window:]) / window
    avg_loss = sum(losses[-window:]) / window
    
    if avg_loss == 0:
        return 100
    
    rs = avg_gain / avg_loss
    rsi = 100 - (100 / (1 + rs))
    
    return rsi

def calculate_stochastic(prices: List[float], highs: List[float], lows: List[float], k_window: int = 14, d_window: int = 3) -> Tuple[Optional[float], Optional[float]]:
    """Calculate Stochastic Oscillator (%K and %D)"""
    if len(prices) < k_window:
        return None, None
    
    # Calculate %K
    current_close = prices[-1]
    lowest_low = min(lows[-k_window:])
    highest_high = max(highs[-k_window:])
    
    if highest_high - lowest_low == 0:
        k = 50
    else:
        k = 100 * ((current_close - lowest_low) / (highest_high - lowest_low))
    
    # Calculate %D (3-period SMA of %K)
    k_values = []
    for i in range(min(d_window, len(prices) - k_window + 1)):
        c = prices[-(i+1)]
        l = min(lows[-(i+k_window):-(i) if i > 0 else None])
        h = max(highs[-(i+k_window):-(i) if i > 0 else None])
        
        if h - l == 0:
            k_values.append(50)
        else:
            k_values.append(100 * ((c - l) / (h - l)))
    
    d = sum(k_values) / len(k_values) if k_values else None
    
    return k, d
