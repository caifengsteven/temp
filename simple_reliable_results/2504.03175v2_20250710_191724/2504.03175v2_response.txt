class ExtendedBlackScholesPDE:
    def __init__(self):
        self.name = "Extended Black-Scholes PDE Model"
    
    def d1(self, S, K, r, sigma, T):
        return (np.log(S/K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    
    def d2(self, S, K, r, sigma, T):
        return self.d1(S, K, r, sigma, T) - sigma * np.sqrt(T)
    
    def european_call(self, S, K, r, sigma, T):
        """Standard Black-Scholes formula for European call option"""
        if T <= 0:
            return max(0, S - K)
        d1_val = self.d1(S, K, r, sigma, T)
        d2_val = self.d2(S, K, r, sigma, T)
        return S * norm.cdf(d1_val) - K * np.exp(-r * T) * norm.cdf(d2_val)
    
    def finite_difference_solver(self, S, K, r, sigma, T, S_max, time_steps, space_steps):
        """Solve extended Black-Scholes PDE using implicit finite difference method"""
        # Grid setup
        dt = T / time_steps
        dS = S_max / space_steps
        
        # Initialize grid for option prices
        grid = np.zeros((time_steps + 1, space_steps + 1))
        
        # Boundary condition at expiry (payoff)
        for j in range(space_steps + 1):
            S_j = j * dS
            grid[time_steps, j] = max(0, S_j - K)  # Call option payoff
        
        # Boundary conditions at S = 0 and S = S_max
        for i in range(time_steps + 1):
            grid[i, 0] = 0  # Option value is 0 when S = 0
            grid[i, space_steps] = S_max - K * np.exp(-r * (T - i * dt))  # Linear for large S
        
        # Main loop - backward in time
        for i in range(time_steps - 1, -1, -1):
            t = i * dt
            # Stochastic volatility and interest rate - here simplified to vary with time
            sigma_t = sigma * (1 + 0.1 * np.sin(t * np.pi / T))  # Example time-varying volatility
            r_t = r * (1 + 0.05 * np.cos(t * np.pi / T))  # Example time-varying interest rate
            
            # Build tridiagonal system
            alpha = np.zeros(space_steps - 1)
            beta = np.zeros(space_steps - 1)
            gamma = np.zeros(space_steps - 1)
            rhs = np.zeros(space_steps - 1)
            
            for j in range(1, space_steps):
                S_j = j * dS
                alpha[j-1] = 0.5 * dt * (sigma_t**2 * S_j**2 / dS**2 - r_t * S_j / dS)
                beta[j-1] = 1 + dt * (sigma_t**2 * S_j**2 / dS**2 + r_t)
                gamma[j-1] = 0.5 * dt * (sigma_t**2 * S_j**2 / dS**2 + r_t * S_j / dS)
                rhs[j-1] = grid[i+1, j]
            
            # Adjust RHS for boundary conditions
            rhs[0] -= alpha[0] * grid[i, 0]
            rhs[-1] -= gamma[-1] * grid[i, space_steps]
            
            # Solve tridiagonal system
            grid[i, 1:space_steps] = self.solve_tridiagonal(alpha, beta, gamma, rhs)
        
        # Linear interpolation to get option price at S
        j = int(S / dS)
        if j >= space_steps:
            return grid[0, space_steps]
        weight = (S - j * dS) / dS
        return grid[0, j] * (1 - weight) + grid[0, j+1] * weight
    
    def solve_tridiagonal(self, a, b, c, d):
        """Solve a tridiagonal system Ax = d where A has diagonals a, b, c"""
        n = len(b)
        x = np.zeros(n)
        
        # Forward elimination
        for i in range(1, n):
            m = a[i-1] / b[i-1]
            b[i] -= m * c[i-1]
            d[i] -= m * d[i-1]
        
        # Back substitution
        x[-1] = d[-1] / b[-1]
        for i in range(n-2, -1, -1):
            x[i] = (d[i] - c[i] * x[i+1]) / b[i]
        
        return x
    
    def predict(self, S, K, r, sigma, T, method='closed_form'):
        """Predict option price using either closed-form or finite difference"""
        if method == 'closed_form':
            return self.european_call(S, K, r, sigma, T)
        elif method == 'finite_difference':
            S_max = 3 * K  # Upper bound for stock price grid
            time_steps = 100
            space_steps = 100
            return self.finite_difference_solver(S, K, r, sigma, T, S_max, time_steps, space_steps)
        else:
            raise ValueError("Method must be 'closed_form' or 'finite_difference'")


class LSTMOptionPricing:
    def __init__(self):
        self.name = "LSTM Option Pricing Model"
        self.model = None
        self.scaler_X = MinMaxScaler() if TENSORFLOW_AVAILABLE else None
        self.scaler_y = MinMaxScaler() if TENSORFLOW_AVAILABLE else None
    
    def create_sequences(self, X, y, time_steps):
        """Create sequences for LSTM input"""
        X_seq, y_seq = [], []
        for i in range(len(X) - time_steps):
            X_seq.append(X[i:i + time_steps])
            y_seq.append(y[i + time_steps])
        return np.array(X_seq), np.array(y_seq)
    
    def build_model(self, input_shape):
        """Build LSTM model architecture"""
        if not TENSORFLOW_AVAILABLE:
            print("TensorFlow not available. Cannot build LSTM model.")
            return None
            
        model = Sequential()
        model.add(LSTM(50, return_sequences=True, input_shape=input_shape))
        model.add(Dropout(0.2))
        model.add(LSTM(50))
        model.add(Dropout(0.2))
        model.add(Dense(1))
        model.compile(optimizer='adam', loss='mean_squared_error')
        return model
    
    def train(self, X_train, y_train, time_steps=30, epochs=50, batch_size=32):
        """Train LSTM model"""
        if not TENSORFLOW_AVAILABLE:
            print("TensorFlow not available. Cannot train LSTM model.")
            return
            
        # Scale data
        X_scaled = self.scaler_X.fit_transform(X_train)
        y_scaled = self.scaler_y.fit_transform(y_train.reshape(-1, 1))
        
        # Create sequences
        X_seq, y_seq = self.create_sequences(X_scaled, y_scaled, time_steps)
        
        # Build and train model
        self.model = self.build_model((X_seq.shape[1], X_seq.shape[2]))
        self.model.fit(X_seq, y_seq, epochs=epochs, batch_size=batch_size, verbose=0)
    
    def predict(self, X_test, time_steps=30):
        """Predict option prices using trained LSTM model"""
        if not TENSORFLOW_AVAILABLE or self.model is None:
            print("TensorFlow not available or model not trained. Cannot make predictions.")
            return np.zeros(len(X_test) - time_steps)
            
        # Scale input data
        X_scaled = self.scaler_X.transform(X_test)
        
        # Create sequences
        X_seq = []
        for i in range(len(X_scaled) - time_steps):
            X_seq.append(X_scaled[i:i + time_steps])
        X_seq = np.array(X_seq)
        
        # Make predictions
        y_pred_scaled = self.model.predict(X_seq)
        
        # Inverse scale predictions
        y_pred = self.scaler_y.inverse_transform(y_pred_scaled)
        
        return y_pred.flatten()


def generate_simulated_data(days=500, seed=42):
    """Generate simulated stock price and option data"""
    np.random.seed(seed)
    
    # Generate dates
    base_date = datetime(2023, 1, 1)
    dates = [base_date + timedelta(days=i) for i in range(days)]
    
    # Generate stock price with stochastic volatility
    S0 = 100  # Initial stock price
    mu = 0.05  # Drift
    sigma0 = 0.2  # Initial volatility
    kappa = 2.0  # Mean reversion speed for volatility
    theta = 0.2  # Long-term volatility
    xi = 0.1  # Volatility of volatility
    
    stock_prices = [S0]
    volatilities = [sigma0]
    
    for i in range(1, days):
        dt = 1/252  # Trading days in a year
        
        # Update volatility using Heston model
        vol_drift = kappa * (theta - volatilities[-1]) * dt
        vol_diffusion = xi * np.sqrt(volatilities[-1]) * np.sqrt(dt) * np.random.normal()
        new_vol = max(0.05, volatilities[-1] + vol_drift + vol_diffusion)  # Ensure positive volatility
        volatilities.append(new_vol)
        
        # Update stock price
        price_drift = mu * stock_prices[-1] * dt
        price_diffusion = volatilities[-1] * stock_prices[-1] * np.sqrt(dt) * np.random.normal()
        new_price = stock_prices[-1] + price_drift + price_diffusion
        stock_prices.append(max(1, new_price))  # Ensure positive price
    
    # Generate interest rates using Vasicek model
    r0 = 0.03  # Initial interest rate
    a = 0.1  # Mean reversion speed
    b = 0.04  # Long-term mean
    s = 0.01  # Volatility of interest rate
    
    interest_rates = [r0]
    
    for i in range(1, days):
        dt = 1/252
        r_drift = a * (b - interest_rates[-1]) * dt
        r_diffusion = s * np.sqrt(dt) * np.random.normal()
        new_rate = interest_rates[-1] + r_drift + r_diffusion
        interest_rates.append(max(0.001, new_rate))  # Ensure positive rate
    
    # Generate option data for different strike prices and maturities
    strikes = np.linspace(0.8 * S0, 1.2 * S0, 5)
    maturities = [30/365, 60/365, 90/365]  # 30, 60, 90 days in years
    
    option_data = []
    for day in range(days):
        for K in strikes:
            for T in maturities:
                # Use Black-Scholes to calculate theoretical option price
                if day + int(T * 365) < days:  # Ensure maturity is within data range
                    S = stock_prices[day]
                    sigma = volatilities[day]
                    r = interest_rates[day]
                    
                    # Calculate BS price with some noise to simulate market prices
                    bs = ExtendedBlackScholesPDE()
                    bs_price = bs.european_call(S, K, r, sigma, T)
                    market_price = bs_price * (1 + 0.05 * np.random.normal())  # Add some noise
                    
                    option_data.append({
                        'Date': dates[day],
                        'StockPrice': S,
                        'Strike': K,
                        'Maturity': T,
                        'DaysToExpiry': int(T * 365),
                        'Volatility': sigma,
                        'InterestRate': r,
                        'OptionPrice': max(0.01, market_price)  # Ensure positive price
                    })
    
    # Create DataFrames
    stock_df = pd.DataFrame({
        'Date': dates,
        'StockPrice': stock_prices,
        'Volatility': volatilities,
        'InterestRate': interest_rates
    })
    
    option_df = pd.DataFrame(option_data)
    
    return stock_df, option_df

def plot_3d_option_surface(model, S_range, T_range, K, r, sigma):
    """Create a 3D plot of option prices vs stock price and time to maturity"""
    S_mesh, T_mesh = np.meshgrid(S_range, T_range)
    Z = np.zeros_like(S_mesh)
    
    for i in range(len(T_range)):
        for j in range(len(S_range)):
            Z[i, j] = model.predict(S_mesh[i, j], K, r, sigma, T_mesh[i, j])
    
    fig = plt.figure(figsize=(12, 8))
    ax = fig.add_subplot(111, projection='3d')
    surf = ax.plot_surface(S_mesh, T_mesh, Z, cmap='viridis', alpha=0.8)
    
    ax.set_xlabel('Stock Price')
    ax.set_ylabel('Time to Maturity (years)')
    ax.set_zlabel('Option Price')
    ax.set_title('3D Option Price Surface')
    
    plt.colorbar(surf, ax=ax, shrink=0.5, aspect=5)
    plt.savefig('option_price_surface.png')
