class TimeSeriesPreprocessor:
    @staticmethod
    def preprocess_price_series(series, start_price=None):
        """
        Preprocess price series as described in the paper:
        - Apply logarithmic transformation
        - Normalize to a base value (100)
        """
        if start_price is None:
            start_price = series[0]
        
        # Apply log transformation and normalize
        log_series = np.log(series / start_price) + 100
        return log_series
    
    @staticmethod
    def preprocess_volume_series(series):
        """
        Preprocess volume series as described in the paper
        """
        # Apply log(V + 1) transformation
        log_series = np.log(series + 1)
        return log_series


class MetricsCalculator:
    @staticmethod
    def calculate_mse(y_true, y_pred):
        """Calculate Mean Squared Error"""
        return mean_squared_error(y_true, y_pred)
    
    @staticmethod
    def calculate_mae(y_true, y_pred):
        """Calculate Mean Absolute Error"""
        return mean_absolute_error(y_true, y_pred)
    
    @staticmethod
    def calculate_msIC(y_true, y_pred):
        """
        Calculate mean sequential correlation (msIC) as defined in the paper
        """
        correlations = []
        
        # For multivariate time series
        if len(y_true.shape) == 3:  # (batch, time, features)
            batch_size, _, num_features = y_true.shape
            for i in range(batch_size):
                for j in range(num_features):
                    # Calculate rank correlation for this time series
                    corr, _ = spearmanr(y_true[i, :, j], y_pred[i, :, j])
                    if not np.isnan(corr):
                        correlations.append(corr)
        else:  # For univariate time series
            batch_size = y_true.shape[0]
            for i in range(batch_size):
                corr, _ = spearmanr(y_true[i], y_pred[i])
                if not np.isnan(corr):
                    correlations.append(corr)
                    
        # Return mean correlation
        return np.mean(correlations) if correlations else 0
    
    @staticmethod
    def calculate_msIR(y_true, y_pred):
        """
        Calculate correlation stability ratio (msIR) as defined in the paper
        """
        # First calculate the msIC values per sample
        msIC_values = []
        
        # For multivariate time series
        if len(y_true.shape) == 3:  # (batch, time, features)
            batch_size, _, num_features = y_true.shape
            for i in range(batch_size):
                sample_corrs = []
                for j in range(num_features):
                    corr, _ = spearmanr(y_true[i, :, j], y_pred[i, :, j])
                    if not np.isnan(corr):
                        sample_corrs.append(corr)
                if sample_corrs:
                    msIC_values.append(np.mean(sample_corrs))
        else:  # For univariate time series
            batch_size = y_true.shape[0]
            for i in range(batch_size):
                corr, _ = spearmanr(y_true[i], y_pred[i])
                if not np.isnan(corr):
                    msIC_values.append(corr)
        
        if not msIC_values:
            return 0
            
        # Calculate mean and standard deviation
        mean_msIC = np.mean(msIC_values)
        std_msIC = np.std(msIC_values)
        
        # Return msIR
        return mean_msIC / std_msIC if std_msIC > 0 else 0

class TimingStrategy:
    """
    Implementation of the timing strategy described in the paper
    """
    def __init__(self, window_size=63):
        self.window_size = window_size
        
    def generate_signals(self, predictions, actual_prices):
        """
        Generate trading signals based on model predictions
        """
        # Calculate the predicted price change
        pred_changes = predictions[:, -1] - actual_prices[:, -1]
        
        # Calculate the rolling average (if we have enough data)
        if len(pred_changes) >= self.window_size:
            rolling_avg = np.convolve(pred_changes, 
                                     np.ones(self.window_size)/self.window_size, 
                                     mode='valid')
            # Pad the beginning to maintain the same length
            padding = np.full(self.window_size-1, rolling_avg[0])
            rolling_avg = np.concatenate([padding, rolling_avg])
            
            # Calculate difference in difference
            diff_in_diff = pred_changes - rolling_avg
        else:
            # If not enough data, just use the raw signal
            diff_in_diff = pred_changes
            
        # Generate signals: 1 for buy, 0 for hold cash
        signals = np.zeros_like(diff_in_diff)
        signals[diff_in_diff > 0] = 1
        
        return signals
    
    def backtest(self, signals, price_returns, initial_capital=1.0):
        """
        Backtest the strategy using the generated signals
        """
        # Initialize portfolio values
        portfolio_values = [initial_capital]
        current_value = initial_capital
        
        # Apply signals to returns
        for i in range(len(signals)):
            if i < len(price_returns):
                # If signal is 1, we're exposed to the market return
                # If signal is 0, we hold cash (0% return)
                current_value *= (1 + signals[i] * price_returns[i])
                portfolio_values.append(current_value)
        
        return np.array(portfolio_values)


def calculate_strategy_metrics(portfolio_values, risk_free_rate=0.02):
    """
    Calculate strategy performance metrics
    """
    # Convert to daily returns
    daily_returns = np.diff(portfolio_values) / portfolio_values[:-1]
    
    # Total return
    total_return = portfolio_values[-1] / portfolio_values[0] - 1
    
    # Annualized return (assuming 252 trading days)
    years = len(daily_returns) / 252
    annual_return = (1 + total_return) ** (1 / years) - 1
    
    # Annualized volatility
    annual_volatility = np.std(daily_returns) * np.sqrt(252)
    
    # Maximum drawdown
    cumulative_returns = portfolio_values / portfolio_values[0]
    running_max = np.maximum.accumulate(cumulative_returns)
    drawdowns = (cumulative_returns - running_max) / running_max
    max_drawdown = np.min(drawdowns)
    
    # Sharpe ratio
    sharpe_ratio = (annual_return - risk_free_rate) / annual_volatility if annual_volatility > 0 else 0
    
    # Calmar ratio
    calmar_ratio = annual_return / abs(max_drawdown) if max_drawdown < 0 else 0
    
    # Sortino ratio (downside deviation)
    downside_returns = daily_returns[daily_returns < 0]
    downside_deviation = np.std(downside_returns) * np.sqrt(252) if len(downside_returns) > 0 else 0
    sortino_ratio = (annual_return - risk_free_rate) / downside_deviation if downside_deviation > 0 else 0
    
    return {
        "annual_return": annual_return,
        "cumulative_return": total_return,
        "annual_volatility": annual_volatility,
        "sharpe_ratio": sharpe_ratio,
        "calmar_ratio": calmar_ratio,
        "max_drawdown": max_drawdown,
        "sortino_ratio": sortino_ratio
    }


def simulate_financial_time_series(n_samples=1000, n_features=5, trend_strength=0.01, 
                                  volatility=0.02, seasonality_amp=0.005, freq=20):
    """
    Generate synthetic financial time series data similar to the GSMI dataset
    """
    # Time steps
    t = np.arange(n_samples)
    
    # Base series with random walk (trend + noise)
    base_series = np.zeros((n_samples, n_features))
    
    # Initial values (around 100)
    base_series[0] = 100 * (1 + 0.1 * np.random.randn(n_features))
    
    # Generate random walk with drift
    for i in range(1, n_samples):
        # Random drift
        drift = trend_strength * (0.5 + np.random.randn(n_features))
        
        # Random shock
        shock = volatility * np.random.randn(n_features)
        
        # Seasonality component
        season = seasonality_amp * np.sin(2 * np.pi * t[i] / freq + np.random.randn(n_features))
        
        # Update with drift and shock
        base_series[i] = base_series[i-1] * (1 + drift + shock + season)
    
    # Create OHLC data
    # High = Close * (1 + random fluctuation)
    high_series = base_series * (1 + 0.005 * np.random.rand(n_samples, n_features))
    
    # Low = Close * (1 - random fluctuation)
    low_series = base_series * (1 - 0.005 * np.random.rand(n_samples, n_features))
    
    # Open = Previous Close * (1 + overnight change)
    open_series = np.zeros_like(base_series)
    open_series[0] = base_series[0] * (1 - 0.001 * np.random.rand(n_features))
    open_series[1:] = base_series[:-1] * (1 + 0.002 * np.random.randn(n_samples-1, n_features))
    
    # Volume (log-normal distribution)
    volume_series = np.exp(8 + 0.5 * np.random.randn(n_samples, n_features))
    
    # Ensure OHLC relationship: Low <= Open, Close <= High
    high_series = np.maximum(high_series, np.maximum(open_series, base_series))
    low_series = np.minimum(low_series, np.minimum(open_series, base_series))
    
    # Pack into a dataset dictionary
    data = {
        'close': base_series,
        'open': open_series,
        'high': high_series,
        'low': low_series,
        'volume': volume_series
    }
    
    return data


def prepare_sequences(data, input_length=512, forecast_horizon=5, stride=5):
    """
    Prepare input-output sequences from time series data
    """
    n_samples = data['close'].shape[0]
    n_features = data['close'].shape[1]
    
    # Calculate number of sequences
    n_sequences = (n_samples - input_length - forecast_horizon) // stride + 1
    
    # Initialize arrays for inputs and targets
    X = np.zeros((n_sequences, input_length, n_features * 5))  # 5 for OHLCV
    y = np.zeros((n_sequences, forecast_horizon, n_features))  # Only predict close prices
    
    # Prepare the data
    for i in range(n_sequences):
        start_idx = i * stride
        end_idx = start_idx + input_length
        
        # Stack all features for input
        features = []
        for key in ['open', 'high', 'low', 'close', 'volume']:
            # Apply preprocessing
            if key == 'volume':
                processed = TimeSeriesPreprocessor.preprocess_volume_series(data[key][start_idx:end_idx])
            else:
                processed = TimeSeriesPreprocessor.preprocess_price_series(data[key][start_idx:end_idx])
            features.append(processed)
        
        X[i] = np.column_stack(features).reshape(input_length, n_features * 5)
        
        # For target, only use close prices
        target_start = end_idx
        target_end = target_start + forecast_horizon
        y[i] = TimeSeriesPreprocessor.preprocess_price_series(
            data['close'][target_start:target_end], 
            start_price=data['close'][target_start-1]
        )
    
    return X, y
