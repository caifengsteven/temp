import numpy as np

class VolatilityModels:

    def ads_model(self, moneyness, params, k_min=1.0):
        """
        Angelini-di Sciorio (AdS) model for implied volatility
        
        Args:
            moneyness: S/K ratio (underlying price / strike price)
            params: tuple (alpha, beta, delta, epsilon)
            k_min: Moneyness value at which volatility is minimized
            
        Returns:
            Implied volatility according to AdS model
        """
        alpha, beta, delta, epsilon = params
        
        # Calculate Hurst exponent based on moneyness
        h_factor = 0.5 * (1 + (1 - k_min)**delta)
        h = h_factor / (1 + (moneyness - k_min)**delta)
        
        # Calculate quadratic term
        quad_term = (moneyness - k_min)**2
        
        # Calculate exponential decay term
        exp_term = np.exp(-beta * h * (moneyness - k_min))
        
        # Complete AdS model
        sigma = alpha * quad_term * exp_term + epsilon
        
        return sigma, h
    
    def sabr_model(self, moneyness, params, t=1.0):
        """
        SABR model approximation for implied volatility
        
        Args:
            moneyness: S/K ratio
            params: tuple (alpha, beta, rho, nu)
            t: time to expiration
            
        Returns:
            Implied volatility according to SABR model
        """
        alpha, beta, rho, nu = params
        f = moneyness  # Forward price (simplified)
        k = 1.0  # Normalized strike
        
        # SABR formula (simplified Hagan approximation)
        log_fk = np.log(f/k)
        
        # Handle the case when moneyness is close to 1
        if abs(log_fk) < 1e-8:
            z = nu * f**(1-beta) * log_fk / alpha
        else:
            z = nu * f**(1-beta) * log_fk / alpha
            
        chi = np.log((np.sqrt(1 - 2*rho*z + z**2) + z - rho) / (1 - rho))
        
        if abs(1 - beta) < 1e-8:
            correction = 1 + ((1/24)*nu**2*t + (1/4)*rho*beta*nu*alpha*t + ((2-3*rho**2)/24)*nu**2*t)
        else:
            correction = 1 + ((1/24)*(1-beta)**2*log_fk**2 + (1/4)*rho*beta*nu*alpha*t + ((2-3*rho**2)/24)*nu**2*t)
        
        sigma = (alpha / (f**(1-beta))) * (z / chi) * correction
        
        return sigma
    
    def fsabr_model(self, moneyness, params, t=1.0, h=0.3):
        """
        Fractional SABR model for implied volatility
        
        Args:
            moneyness: S/K ratio
            params: tuple (alpha, beta, rho, nu)
            t: time to expiration
            h: Hurst exponent
            
        Returns:
            Implied volatility according to fSABR model
        """
        alpha, beta, rho, nu = params
        f = moneyness  # Forward price
        k = 1.0  # Normalized strike
        
        # Basic SABR vol
        sabr_vol = self.sabr_model(moneyness, params, t)
        
        # Apply fractional adjustment
        if h == 0.5:
            return sabr_vol
        
        # Fractional correction (simplified)
        h_correction = t**(h - 0.5)
        
        # Scale volatility by the fractional correction
        fsabr_vol = sabr_vol * h_correction
        
        return fsabr_vol
    
    def rmse(self, observed, predicted):
        """Calculate Root Mean Square Error"""
        return np.sqrt(np.mean((observed - predicted)**2))
    
    def mae(self, observed, predicted):
        """Calculate Mean Absolute Error"""
        return np.mean(np.abs(observed - predicted))
    
    def mse(self, observed, predicted):
        """Calculate Mean Squared Error"""
        return np.mean((observed - predicted)**2)
    
    def calculate_curvature(self, moneyness, volatility):
        """
        Calculate numerical approximation of second derivative (curvature)
        """
        n = len(moneyness)
        curvature = np.zeros(n)
        
        # Use central difference for interior points
        for i in range(1, n-1):
            h = (moneyness[i+1] - moneyness[i-1]) / 2
            curvature[i] = (volatility[i+1] - 2*volatility[i] + volatility[i-1]) / h**2
            
        # Forward and backward differences for endpoints
        if n > 2:
            h_0 = moneyness[1] - moneyness[0]
            curvature[0] = (volatility[2] - 2*volatility[1] + volatility[0]) / h_0**2
            
            h_n = moneyness[-1] - moneyness[-2]
            curvature[-1] = (volatility[-3] - 2*volatility[-2] + volatility[-1]) / h_n**2
            
        return curvature
    
    def rmsce(self, observed_curv, predicted_curv):
        """Calculate Root Mean Square Curvature Error"""
        return np.sqrt(np.mean((observed_curv - predicted_curv)**2))
    
    def ace(self, observed_curv, predicted_curv):
        """Calculate Absolute Curvature Error"""
        return np.mean(np.abs(observed_curv - predicted_curv))
    
    def calibrate_ads(self, moneyness, implied_volatility, k_min=1.0):
        """
        Calibrate AdS model to market data
        """
        def objective(params):
            pred_vol, _ = self.ads_model(moneyness, params, k_min)
            return self.rmse(implied_volatility, pred_vol)
        
        # Initial guess for params: alpha, beta, delta, epsilon
        initial_guess = [0.1, 0.5, 0.5, 0.01]
        
        # Parameter bounds
        bounds = [(0.001, 10), (-1, 1), (0, 1), (0.001, 1)]
        
        # Optimization
        result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B')
        
        return result.x
    
    def calibrate_sabr(self, moneyness, implied_volatility, t=1.0):
        """
        Calibrate SABR model to market data
        """
        def objective(params):
            pred_vol = np.array([self.sabr_model(m, params, t) for m in moneyness])

        # Optimization
        result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B')
        
        return result.x
    
    def calibrate_fsabr(self, moneyness, implied_volatility, t=1.0, fixed_h=None):
        """
        Calibrate fSABR model to market data
        """
        def objective(params):
            if fixed_h is None:

            else:
                alpha, beta, rho, nu = params
                h = fixed_h
            
            pred_vol = np.array([self.fsabr_model(m, (alpha, beta, rho, nu), t, h) for m in moneyness])

        if fixed_h is None:
            # Initial guess for params: alpha, beta, rho, nu, h
            initial_guess = [0.2, 0.5, 0, 0.4, 0.4]

            initial_guess = [0.2, 0.5, 0, 0.4]

            bounds = [(0.001, 1), (0, 1), (-0.999, 0.999), (0.001, 1)]

        result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B')
