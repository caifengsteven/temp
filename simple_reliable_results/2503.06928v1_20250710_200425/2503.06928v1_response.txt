class FinancialDataGenerator:
    """Generates synthetic financial time series data similar to those described in the paper."""

        data = {}
        
        # Generate base trend with random walk
        base_trend = 100 + np.cumsum(np.random.normal(0.0005, 0.01, n_samples))

            # Create individual index with base trend plus noise
            noise = np.random.normal(0, 0.02, n_samples)

            # Generate OHLC data
            close_price = trend
            high_price = close_price * (1 + np.abs(np.random.normal(0, 0.005, n_samples)))
            low_price = close_price * (1 - np.abs(np.random.normal(0, 0.005, n_samples)))
            open_price = low_price + np.random.random(n_samples) * (high_price - low_price)
            volume = np.random.lognormal(10, 1, n_samples)
            
            # Add to dataframe with index name
            index_name = f"INDEX_{i+1}"
            data[f"{index_name}_close"] = close_price
            data[f"{index_name}_open"] = open_price
            data[f"{index_name}_high"] = high_price
            data[f"{index_name}_low"] = low_price
            data[f"{index_name}_volume"] = volume

class TimeSeriesForecaster:
    """Simple forecasting models to test with the financial data."""
    
    def __init__(self):
        pass
    
    def naive_forecast(self, X: np.ndarray) -> np.ndarray:
        """Naive forecasting model that predicts the last observed value."""
        return np.tile(X[:, -1:], (1, X.shape[2], 1)).transpose(0, 2, 1)
    
    def linear_trend_forecast(self, X: np.ndarray, horizon: int) -> np.ndarray:
        """Simple linear trend forecasting."""
        batch_size, seq_len, n_features = X.shape
        result = np.zeros((batch_size, horizon, n_features))
        
        for b in range(batch_size):
            for f in range(n_features):
                # Fit linear trend to last 30 points
                x = np.arange(30)
                y = X[b, -30:, f]
                coeffs = np.polyfit(x, y, 1)
                
                # Forecast next 'horizon' points
                for h in range(horizon):
                    result[b, h, f] = coeffs[0] * (30 + h) + coeffs[1]
        
        return result
    
    def mean_forecast(self, X: np.ndarray, horizon: int) -> np.ndarray:
        """Forecasting using the mean of the last few observations."""
        batch_size, seq_len, n_features = X.shape
        result = np.zeros((batch_size, horizon, n_features))
        
        for b in range(batch_size):
            for f in range(n_features):
                # Use mean of last 10 observations
                mean_val = np.mean(X[b, -10:, f])
                result[b, :, f] = mean_val
        
        return result
    
    def psformer_simulator(self, X: np.ndarray, horizon: int) -> np.ndarray:
        """Simulate a more sophisticated model (PSformer) with trend + noise + correlation."""
        batch_size, seq_len, n_features = X.shape
        result = np.zeros((batch_size, horizon, n_features))
        
        for b in range(batch_size):
            # Base trend from linear projection
            trend = self.linear_trend_forecast(X[b:b+1], horizon)[0]
            
            # Add correlations between features
            for f in range(n_features):
                # Add some temporal correlation
                base_signal = trend[:, f] + np.random.normal(0, 0.01, horizon)
                
                # Add momentum effect (common in financial data)
                momentum = np.convolve(np.diff(X[b, -10:, f], prepend=X[b, -11, f]), 
                                      np.ones(3)/3, mode='valid')
                direction = np.sign(np.mean(momentum))
                
                # Final forecast with slight bias in the direction of recent momentum
                result[b, :, f] = base_signal + direction * np.linspace(0, 0.05, horizon)
        
        return result

class EvaluationMetrics:
    """Implements evaluation metrics from the paper including msIC and msIR."""
    
    def __init__(self):
        pass
    
    def calculate_mse(self, y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """Calculate Mean Squared Error."""
        return mean_squared_error(y_true.flatten(), y_pred.flatten())
    
    def calculate_mae(self, y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """Calculate Mean Absolute Error."""
        return mean_absolute_error(y_true.flatten(), y_pred.flatten())
    
    def calculate_msIC(self, y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """
        Calculate mean sequential correlation (msIC) as described in the paper.
        
        This measures temporal correlation between predicted and true time series.
        """
        B, F, C = y_true.shape  # Batch, Forecast length, Channels/Features
        correlations = []
        
        for i in range(B):
            for j in range(C):
                # Calculate rank correlation for this sample and variable
                true_series = y_true[i, :, j]
                pred_series = y_pred[i, :, j]
                
                # Handle constant predictions
                if np.std(pred_series) == 0:
                    correlations.append(0)
                else:
                    corr, _ = spearmanr(true_series, pred_series)
                    if np.isnan(corr):
                        corr = 0
                    correlations.append(corr)
        
        return np.mean(correlations)
    
    def calculate_msIR(self, y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """
        Calculate correlation stability ratio (msIR) as described in the paper.
        
        This measures the ratio between effective correlation and correlation noise.
        """
        B, F, C = y_true.shape  # Batch, Forecast length, Channels/Features
        msIC_values = []
        
        for i in range(B):
            # Calculate msIC for this sample
            sample_msIC = 0
            count = 0
            for j in range(C):
                true_series = y_true[i, :, j]
                pred_series = y_pred[i, :, j]
                
                # Handle constant predictions
                if np.std(pred_series) == 0:
                    corr = 0
                else:
                    corr, _ = spearmanr(true_series, pred_series)
                    if np.isnan(corr):
                        corr = 0
                
                sample_msIC += corr
                count += 1
            
            msIC_values.append(sample_msIC / count if count > 0 else 0)
        
        # Calculate overall msIC
        msIC = np.mean(msIC_values)
        
        # Calculate standard deviation of msIC sequence
        msIC_std = np.std(msIC_values)
        
        # Calculate msIR
        if msIC_std == 0:
            return 0
        else:
            return msIC / msIC_std
    
    def evaluate_model(self, model_name: str, y_true: np.ndarray, y_pred: np.ndarray) -> Dict:
        """Evaluate model performance using multiple metrics."""
        return {
            'model': model_name,
            'MSE': self.calculate_mse(y_true, y_pred),
            'MAE': self.calculate_mae(y_true, y_pred),
            'msIC': self.calculate_msIC(y_true, y_pred),
            'msIR': self.calculate_msIR(y_true, y_pred)
        }

class FinancialStrategy:
    """Implements financial strategies based on model predictions."""
    
    def __init__(self):
        pass
    
    def timing_strategy(self, y_pred: np.ndarray, y_true: np.ndarray, 
                       threshold: float = 0.1) -> Tuple[np.ndarray, Dict]:
        """
        Implement timing strategy as described in the paper.
        
        Returns:
            - Cumulative returns
            - Strategy metrics
        """
        n_samples = y_true.shape[0]
        returns = np.zeros(n_samples)
        positions = np.zeros(n_samples)  # 0 = cash, 1 = long
        
        # Calculate true returns from the ground truth
        true_returns = np.diff(y_true[:, 0, 0], prepend=y_true[0, 0, 0])
        
        # Implement the strategy logic
        for i in range(1, n_samples):
            # Calculate predicted price change
            pred_change = y_pred[i-1, -1, 0] - y_true[i-1, 0, 0]
            
            # If predicted change is greater than threshold, go long
            if pred_change > threshold:
                positions[i] = 1
                returns[i] = true_returns[i]
            # Otherwise, stay in cash
            else:
                positions[i] = 0
                returns[i] = 0
        
        # Calculate cumulative returns
        cum_returns = np.cumprod(1 + returns) - 1
        
        # Calculate strategy metrics
        annual_return = (1 + cum_returns[-1]) ** (252 / n_samples) - 1
        volatility = np.std(returns) * np.sqrt(252)
        sharpe_ratio = annual_return / volatility if volatility > 0 else 0
        
        # Calculate max drawdown
        peak = np.maximum.accumulate(1 + cum_returns)
        drawdown = (1 + cum_returns) / peak - 1
        max_drawdown = np.min(drawdown)
        
        # Calculate Calmar ratio
        calmar_ratio = annual_return / abs(max_drawdown) if max_drawdown != 0 else 0
        
        metrics = {
            'annual_return': annual_return,
            'cumulative_return': cum_returns[-1],
            'volatility': volatility,
            'sharpe_ratio': sharpe_ratio,
            'max_drawdown': max_drawdown,
            'calmar_ratio': calmar_ratio
        }
        
        return cum_returns, metrics
