class CausalStockModel:
    """
    Implementation of causal stock prediction model based on the paper
    "A Causal Perspective of Stock Prediction Models"
    """
    def __init__(self, k_tilde=2, lambda1=5, lambda2=1):
        """
        Initialize the causal stock prediction model
        
        Parameters:
        k_tilde (int): Dimensionality of causal features
        lambda1 (float): Hyperparameter controlling invariance strength
        lambda2 (float): Hyperparameter for alignment regularization
        """
        self.k_tilde = k_tilde
        self.lambda1 = lambda1
        self.lambda2 = lambda2
        self.nu = None
        self.xi_prime = None
        
    def discover_causal_features(self, domains):
        """
        Discover causal features across multiple domains
        
        Parameters:
        domains (list): List of domain data (X, R) tuples
        
        Returns:
        np.ndarray: Discovered causal coefficients
        """
        # Initialize xi_prime (represents 1-ξ where ξ is the residual level)
        self.xi_prime = 0.5
        
        # For each domain, compute domain-specific coefficients
        domain_coeffs = []
        for X, R in domains:
            # Simulate feature extraction (in reality this would be done by neural networks)
            Z_tilde = self._extract_features(X)
            
            # Compute domain-specific coefficients
            nu_domain = self._compute_domain_coefficients(Z_tilde, R)
            domain_coeffs.append(nu_domain)
        
        # Compute the average coefficient across domains
        self.nu = np.mean(domain_coeffs, axis=0)
        
        # Apply softmax and scaling to satisfy constraints in the paper
        self.nu = np.sqrt(self.xi_prime) * self._softmax(self.nu)
        
        return self.nu
    
    def _extract_features(self, X):
        """
        Extract causal features from input data (simplified implementation)
        In the paper, this would be done by neural networks (ϕ)
        
        Parameters:
        X (np.ndarray): Input data
        
        Returns:
        np.ndarray: Extracted causal features
        """
        # Simplified feature extraction (linear projection + standardization)
        # In reality, this would be done by neural networks with proper invariance constraints
        n_samples, n_features = X.shape
        
        # Random projection to k_tilde dimensions (simulating neural network)
        np.random.seed(42)  # For reproducibility
        W = np.random.randn(n_features, self.k_tilde)
        Z_tilde = X @ W
        
        # Standardize features to have zero mean and unit variance
        Z_tilde = (Z_tilde - np.mean(Z_tilde, axis=0)) / np.std(Z_tilde, axis=0)
        
        return Z_tilde
    
    def _compute_domain_coefficients(self, Z_tilde, R):
        """
        Compute domain-specific coefficients
        
        Parameters:
        Z_tilde (np.ndarray): Extracted causal features
        R (np.ndarray): Returns
        
        Returns:
        np.ndarray: Domain-specific coefficients
        """
        # Compute coefficients using ordinary least squares
        return np.linalg.inv(Z_tilde.T @ Z_tilde) @ Z_tilde.T @ R
    
    def _softmax(self, x):
        """
        Softmax function implementation
        
        Parameters:
        x (np.ndarray): Input array
        
        Returns:
        np.ndarray: Softmax output
        """
        e_x = np.exp(x - np.max(x))
        return e_x / e_x.sum()
    
    def predict(self, X):
        """
        Predict returns using the discovered causal features
        
        Parameters:
        X (np.ndarray): Input data
        
        Returns:
        np.ndarray: Predicted returns
        """
        if self.nu is None:
            raise ValueError("Model not trained. Call discover_causal_features first.")
        
        Z_tilde = self._extract_features(X)
        return Z_tilde @ self.nu

def evaluate_wasserstein_distance(model, domains, oos_domain):
    """
    Estimate the Wasserstein distance between source and OOS domains
    
    Parameters:
    model (CausalStockModel): Trained model
    domains (list): Source domains
    oos_domain (tuple): OOS domain
    
    Returns:
    float: Average Wasserstein distance
    """
    X_oos, R_oos = oos_domain
    pred_oos = model.predict(X_oos)
    
    # Calculate joint distribution distances
    distances = []
    for X_source, R_source in domains:
        pred_source = model.predict(X_source)
        
        # Calculate Wasserstein distance between prediction-return distributions
        # This is a simplified version of the Wasserstein term in the paper
        joint_dist_source = np.column_stack([pred_source, R_source])
        joint_dist_oos = np.column_stack([pred_oos, R_oos])
        
        # Calculate Wasserstein distance (1D approximation)
        w_dist = wasserstein_distance(pred_source, pred_oos) + wasserstein_distance(R_source, R_oos)
        distances.append(w_dist)
    
    return np.mean(distances)


def construct_portfolio(predictions, n_stocks=100):
    """
    Construct a portfolio based on predictions
    
    Parameters:
    predictions (np.ndarray): Predicted returns
    n_stocks (int): Number of stocks to include
    
    Returns:
    np.ndarray: Portfolio weights
    """
    # Rank stocks based on predictions
    ranks = np.argsort(-predictions)  # Negative to sort in descending order
    
    # Select top stocks
    selected = ranks[:n_stocks]
    
    # Create portfolio weights (equal-weighted among selected)
    weights = np.zeros_like(predictions)
    weights[selected] = 1.0 / n_stocks
    
    return weights


def evaluate_portfolio(weights, actual_returns):
    """
    Evaluate portfolio performance
    
    Parameters:
    weights (np.ndarray): Portfolio weights
    actual_returns (np.ndarray): Actual returns
    
    Returns:
    float: Portfolio return
    """
    return np.sum(weights * actual_returns)
