class StockDataGenerator:
    """Generates simulated stock price data for testing"""
    
    def __init__(self, num_stocks=10, num_days=1000):
        self.num_stocks = num_stocks
        self.num_days = num_days
        self.stock_names = [f"STOCK_{i}" for i in range(num_stocks)]
        
    def generate_data(self):
        """Generate simulated price data with realistic correlations"""
        # Base trend for all stocks (market factor)
        base_trend = np.cumsum(np.random.normal(0.001, 0.01, self.num_days))
        
        # Generate correlated stock movements
        correlation_matrix = self._generate_correlation_matrix()
        
        # Create daily returns with correlations
        daily_returns = np.random.multivariate_normal(
            mean=np.ones(self.num_stocks) * 0.0005,
            cov=correlation_matrix,
            size=self.num_days
        )
        
        # Convert returns to prices (start at different base prices)
        base_prices = np.random.uniform(50, 200, self.num_stocks)
        prices = np.zeros((self.num_days, self.num_stocks))
        
        for i in range(self.num_stocks):
            stock_returns = daily_returns[:, i]
            # Add market factor influence
            stock_returns += np.random.normal(0, 0.005, self.num_days) + 0.3 * np.diff(base_trend, prepend=0)
            # Convert returns to prices
            prices[:, i] = base_prices[i] * np.cumprod(1 + stock_returns)
        
        # Create DataFrame with dates
        dates = pd.date_range(start='2020-01-01', periods=self.num_days)
        df_dict = {'Date': dates}
        
        for i, stock in enumerate(self.stock_names):
            df_dict[stock] = prices[:, i]
            
        return pd.DataFrame(df_dict).set_index('Date')
    
    def _generate_correlation_matrix(self):
        """Generate a realistic correlation matrix for stocks"""
        # Create base correlation matrix (positive correlations in general)
        corr = np.random.uniform(0.3, 0.7, (self.num_stocks, self.num_stocks))
        # Make it symmetric
        corr = (corr + corr.T) / 2
        # Set diagonal to 1
        np.fill_diagonal(corr, 1.0)
        
        # Create clusters of higher correlation (e.g., stocks in same sector)
        num_clusters = 3
        cluster_size = self.num_stocks // num_clusters
        
        for c in range(num_clusters):
            start_idx = c * cluster_size
            end_idx = start_idx + cluster_size
            
            # Higher correlation within clusters
            for i in range(start_idx, end_idx):
                for j in range(start_idx, end_idx):
                    if i != j:
                        corr[i, j] = np.random.uniform(0.6, 0.9)
        
        # Ensure matrix is positive definite
        min_eigval = np.min(np.linalg.eigvals(corr))
        if min_eigval < 0:
            corr += -min_eigval * np.eye(self.num_stocks) * 1.1
            
        return corr


class LSTM(nn.Module):
    """LSTM component of the hybrid model"""
    
    def __init__(self, input_size=1, hidden_size=64, num_layers=2, output_size=64):
        super(LSTM, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        
        self.lstm = nn.LSTM(input_size, hidden_size, num_layers, batch_first=True)
        self.fc = nn.Linear(hidden_size, output_size)
        
    def forward(self, x):
        # Initialize hidden state with zeros
        h0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size).to(x.device)
        c0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size).to(x.device)
        
        # Forward propagate LSTM
        out, _ = self.lstm(x, (h0, c0))
        
        # Get the output from the last time step
        out = self.fc(out[:, -1, :])
        return out


class GNN(nn.Module):
    """GNN component of the hybrid model"""
    
    def __init__(self, input_size=1, hidden_size=64, output_size=64):
        super(GNN, self).__init__()
        self.conv1 = GCNConv(input_size, hidden_size)
        self.conv2 = GCNConv(hidden_size, output_size)
        self.relu = nn.ReLU()
        self.dropout = nn.Dropout(0.5)
        
    def forward(self, x, edge_index, edge_weight=None):
        x = self.conv1(x, edge_index, edge_weight)
        x = self.relu(x)
        x = self.dropout(x)
        x = self.conv2(x, edge_index, edge_weight)
        return x


class HybridLSTM_GNN(nn.Module):
    """Hybrid model combining LSTM and GNN components"""
    
    def __init__(self, lstm_input_size=1, gnn_input_size=1, hidden_size=64, output_size=1):
        super(HybridLSTM_GNN, self).__init__()
        
        # Component models
        self.lstm = LSTM(lstm_input_size, hidden_size, num_layers=2, output_size=hidden_size)
        self.gnn = GNN(gnn_input_size, hidden_size, hidden_size)
        
        # Integration layers
        self.fc1 = nn.Linear(hidden_size * 2, hidden_size)
        self.fc2 = nn.Linear(hidden_size, output_size)
        self.relu = nn.ReLU()
        
    def forward(self, lstm_x, gnn_x, edge_index, edge_weight=None):
        # Get embeddings from LSTM and GNN
        lstm_embedding = self.lstm(lstm_x)
        gnn_embedding = self.gnn(gnn_x, edge_index, edge_weight)
        
        # Concatenate embeddings
        combined = torch.cat((lstm_embedding, gnn_embedding), dim=1)
        
        # Final prediction layers
        x = self.relu(self.fc1(combined))
        x = self.fc2(x)
        
        return x


class StockPredictor:
    """Stock price prediction using the hybrid LSTM-GNN model"""
    
    def __init__(self, sequence_length=11):
        self.sequence_length = sequence_length
        self.scaler = MinMaxScaler()
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        self.hybrid_model = None
        self.lstm_model = None
        self.linear_model = None
        
    def preprocess_data(self, data):
        """Preprocess the stock data"""
        # Scale data
        scaled_data = self.scaler.fit_transform(data)
        return scaled_data
    
    def create_sequences(self, data):
        """Create sequences for LSTM input"""
        X, y = [], []
        for i in range(len(data) - self.sequence_length):
            X.append(data[i:i+self.sequence_length])
            y.append(data[i+self.sequence_length])
        return np.array(X), np.array(y)
    
    def build_graph(self, data):
        """Build a graph representing stock relationships"""
        # Calculate returns
        returns = np.diff(data, axis=0) / data[:-1]
        
        # Calculate Pearson correlation
        corr_matrix = np.corrcoef(returns.T)
        
        # Create graph edges based on strong correlations
        edges = []
        edge_weights = []
        for i in range(corr_matrix.shape[0]):
            for j in range(i+1, corr_matrix.shape[0]):
                if abs(corr_matrix[i, j]) > 0.7:  # Threshold from the paper
                    edges.append([i, j])
                    edges.append([j, i])  # Add both directions
                    edge_weights.append(abs(corr_matrix[i, j]))
                    edge_weights.append(abs(corr_matrix[i, j]))
        
        edge_index = torch.tensor(edges, dtype=torch.long).t().contiguous()
        edge_weight = torch.tensor(edge_weights, dtype=torch.float)
        
        return edge_index, edge_weight
    
    def train_hybrid_model(self, data, num_epochs=50, lr=0.005):
        """Train the hybrid LSTM-GNN model"""
        # Preprocess data
        scaled_data = self.preprocess_data(data)
        X_seq, y_seq = self.create_sequences(scaled_data)
        
        # Build graph
        edge_index, edge_weight = self.build_graph(scaled_data)
        
        # Prepare data for PyTorch
        X_lstm = torch.FloatTensor(X_seq).to(self.device)
        y = torch.FloatTensor(y_seq).to(self.device)
        
        # For GNN, we need node features
        # Use the last value of each sequence as node feature
        X_gnn = torch.FloatTensor(X_seq[:, -1, :]).to(self.device)
        
        # Initialize model
        self.hybrid_model = HybridLSTM_GNN(
            lstm_input_size=1, 
            gnn_input_size=data.shape[1], 
            hidden_size=64, 
            output_size=data.shape[1]
        ).to(self.device)
        
        # Loss and optimizer
        criterion = nn.MSELoss()
        optimizer = optim.Adam(self.hybrid_model.parameters(), lr=lr)
        
        # Training loop
        for epoch in range(num_epochs):
            # Forward pass
            outputs = self.hybrid_model(
                X_lstm.view(X_lstm.size(0), X_lstm.size(1), data.shape[1]), 
                X_gnn,
                edge_index.to(self.device),
                edge_weight.to(self.device)
            )
            
            # Calculate loss
            loss = criterion(outputs, y)
            
            # Backward and optimize
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            if (epoch+1) % 10 == 0:
                print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.6f}')
    
    def train_lstm_model(self, data, num_epochs=50, lr=0.005):
        """Train standalone LSTM model for comparison"""
        # Preprocess data
        scaled_data = self.preprocess_data(data)
        X_seq, y_seq = self.create_sequences(scaled_data)
        
        # Prepare data for PyTorch
        X = torch.FloatTensor(X_seq).to(self.device)
        y = torch.FloatTensor(y_seq).to(self.device)
        
        # Initialize model
        self.lstm_model = LSTM(
            input_size=1, 
            hidden_size=64, 
            num_layers=2, 
            output_size=data.shape[1]
        ).to(self.device)
        
        # Loss and optimizer
        criterion = nn.MSELoss()
        optimizer = optim.Adam(self.lstm_model.parameters(), lr=lr)
        
        # Training loop
        for epoch in range(num_epochs):
            # Forward pass
            outputs = self.lstm_model(X.view(X.size(0), X.size(1), data.shape[1]))
            
            # Calculate loss
            loss = criterion(outputs, y)
            
            # Backward and optimize
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            if (epoch+1) % 10 == 0:
                print(f'LSTM Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.6f}')
    
    def train_linear_model(self, data):
        """Train linear regression model for comparison"""
        # Preprocess data
        scaled_data = self.preprocess_data(data)
        X_seq, y_seq = self.create_sequences(scaled_data)
        
        # Reshape for linear regression
        X_flat = X_seq.reshape(X_seq.shape[0], -1)
        
        # Initialize and train model
        self.linear_model = LinearRegression()
        self.linear_model.fit(X_flat, y_seq)
    
    def expanding_window_test(self, data, test_days=50):
        """Test using expanding window approach"""
        # Ensure we have enough data
        if len(data) <= test_days + self.sequence_length:
            raise ValueError("Not enough data for testing")
        
        # Split data
        train_data = data[:-test_days]
        test_data = data[-test_days-self.sequence_length:]
        
        # Track predictions
        hybrid_preds = []
        lstm_preds = []
        linear_preds = []
        actuals = []
        
        # For each test day
        for i in range(test_days):
            # Current train data with expanding window
            current_train = data[:-test_days+i]
            
            # Train models
            self.train_hybrid_model(current_train, num_epochs=40)
            self.train_lstm_model(current_train, num_epochs=40)
            self.train_linear_model(current_train)
            
            # Prepare test sequence
            test_seq = test_data[i:i+self.sequence_length]
            scaled_test_seq = self.scaler.transform(test_seq)
            
            # Prepare input for hybrid model
            X_lstm = torch.FloatTensor(scaled_test_seq.reshape(1, self.sequence_length, -1)).to(self.device)
            X_gnn = torch.FloatTensor(scaled_test_seq[-1].reshape(1, -1)).to(self.device)
            
            # Get graph data
            edge_index, edge_weight = self.build_graph(self.scaler.transform(current_train))
            
            # Make predictions
            with torch.no_grad():
                self.hybrid_model.eval()
                hybrid_pred = self.hybrid_model(X_lstm, X_gnn, edge_index.to(self.device), edge_weight.to(self.device))
                hybrid_pred = hybrid_pred.cpu().numpy()
                
                self.lstm_model.eval()
                lstm_pred = self.lstm_model(X_lstm)
                lstm_pred = lstm_pred.cpu().numpy()
            
            # Linear model prediction
            X_linear = scaled_test_seq.reshape(1, -1)
            linear_pred = self.linear_model.predict(X_linear)
            
            # Inverse transform predictions
            hybrid_pred = self.scaler.inverse_transform(hybrid_pred)
            lstm_pred = self.scaler.inverse_transform(lstm_pred)
            linear_pred = self.scaler.inverse_transform(linear_pred)
            
            # Store predictions and actual
            hybrid_preds.append(hybrid_pred[0])
            lstm_preds.append(lstm_pred[0])
            linear_preds.append(linear_pred[0])
            actuals.append(test_data[i+self.sequence_length])
        
        # Calculate MSE
        hybrid_mse = mean_squared_error(np.array(actuals), np.array(hybrid_preds))
        lstm_mse = mean_squared_error(np.array(actuals), np.array(lstm_preds))
        linear_mse = mean_squared_error(np.array(actuals), np.array(linear_preds))
        
        return {
            'hybrid_mse': hybrid_mse,
            'lstm_mse': lstm_mse,
            'linear_mse': linear_mse,
            'hybrid_preds': hybrid_preds,
            'lstm_preds': lstm_preds,
            'linear_preds': linear_preds,
            'actuals': actuals
        }
