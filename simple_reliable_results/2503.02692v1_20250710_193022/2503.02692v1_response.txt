        }
        
        weights = risk_weights.get(risk_preference, risk_weights['moderate'])
        
        # Combined prediction
        combined_score = (time_series_pred * weights['time'] +

class TradingSimulator:
    """Simulates trading based on predictions"""
    def __init__(self, initial_capital: float = 100000.0):
        self.initial_capital = initial_capital
        self.capital = initial_capital
        self.holdings = 0
        self.trades = []
        self.daily_value = []
    
    def reset(self):
        """Reset the simulator"""
        self.capital = self.initial_capital
        self.holdings = 0
        self.trades = []
        self.daily_value = []
    
    def execute_strategy(self, stock_data: pd.DataFrame, predictions: List[str], 
                         risk_profile: str = 'moderate') -> Dict:
        """Execute trading strategy based on predictions and risk profile"""
        self.reset()
        
        # Risk profile determines trading parameters
        risk_params = {
            'conservative': {'buy_pct': 0.5, 'sell_pct': 1.0},
            'moderately_conservative': {'buy_pct': 0.7, 'sell_pct': 1.0},
            'moderately_aggressive': {'buy_pct': 1.0, 'sell_pct': 0.5},
            'aggressive': {'buy_pct': 1.0, 'sell_pct': 0.3}
        }
        
        params = risk_params.get(risk_profile, risk_params['moderate'])
        
        # Simulate trading
        for i, (date, row) in enumerate(stock_data.iterrows()):
            if i >= len(predictions):
                break
                
            price = row['close']
            decision = predictions[i]
            
            portfolio_value = self.capital + self.holdings * price
            self.daily_value.append(portfolio_value)
            
            if decision == "BUY" and self.capital > 0:
                # Buy stocks
                amount_to_spend = self.capital * params['buy_pct']
                shares_to_buy = amount_to_spend / price
                self.holdings += shares_to_buy
                self.capital -= amount_to_spend
                
                self.trades.append({
                    'date': date,
                    'action': 'BUY',
                    'price': price,
                    'shares': shares_to_buy,
                    'value': amount_to_spend
                })
                
            elif decision == "SELL" and self.holdings > 0:
                # Sell stocks
                shares_to_sell = self.holdings * params['sell_pct']
                amount_received = shares_to_sell * price
                self.holdings -= shares_to_sell
                self.capital += amount_received
                
                self.trades.append({
                    'date': date,
                    'action': 'SELL',
                    'price': price,
                    'shares': shares_to_sell,
                    'value': amount_received
                })
        
        # Calculate final portfolio value
        final_price = stock_data['close'].iloc[-1]
        final_value = self.capital + self.holdings * final_price
        
        # Calculate metrics
        days = len(self.daily_value)
        annualized_return = ((final_value / self.initial_capital) ** (365 / days)) - 1
        
        # Calculate daily returns for Sharpe ratio
        daily_returns = []
        for i in range(1, len(self.daily_value)):
            daily_return = (self.daily_value[i] / self.daily_value[i-1]) - 1
            daily_returns.append(daily_return)
        
        if daily_returns:
            sharpe_ratio = (np.mean(daily_returns) / np.std(daily_returns)) * np.sqrt(252)
        else:
            sharpe_ratio = 0
            
        # Calculate maximum drawdown
        running_max = 0
        drawdowns = []
        
        for value in self.daily_value:
            if value > running_max:
                running_max = value
            
            if running_max > 0:
                drawdown = (running_max - value) / running_max
                drawdowns.append(drawdown)
        
        max_drawdown = max(drawdowns) if drawdowns else 0
        
        return {
            'initial_capital': self.initial_capital,
            'final_value': final_value,
            'return': (final_value / self.initial_capital) - 1,
            'annualized_return': annualized_return,
            'sharpe_ratio': sharpe_ratio,
            'max_drawdown': max_drawdown,
            'trades': len(self.trades),
            'trade_history': self.trades
        }
