import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.stats import norm
from datetime import datetime, timedelta

class OUExchangeRateModel:
    """Implementation of the exchange rate risk premium model using Ornstein-Uhlenbeck dynamics"""
    
    def __init__(self, theta=0.3, mu=0.0, sigma_k=0.05, sigma_s=0.01):
        """
        Initialize the model with OU parameters
        
        Args:
            theta: Mean reversion speed parameter
            mu: Long-run mean of the risk premium
            sigma_k: Volatility of the risk premium process
            sigma_s: Volatility of the exchange rate process
        """
        self.theta = theta
        self.mu = mu
        self.sigma_k = sigma_k
        self.sigma_s = sigma_s
        
    def simulate_risk_premium(self, k0, dt=1/252, periods=252):
        """
        Simulate the Ornstein-Uhlenbeck process for risk premium
        
        Args:
            k0: Initial risk premium value
            dt: Time step (default: daily)
            periods: Number of periods to simulate
            
        Returns:
            Array of simulated risk premium values
        """
        k = np.zeros(periods + 1)
        k[0] = k0
        
        for t in range(1, periods + 1):
            dz = np.random.normal(0, np.sqrt(dt))
            k[t] = k[t-1] + self.theta * (self.mu - k[t-1]) * dt + self.sigma_k * dz
            
        return k
    
    def simulate_exchange_rate(self, s0, k_series, us_rate, kr_rate, dt=1/252):
        """
        Simulate exchange rate based on the risk premium and interest rates
        
        Args:
            s0: Initial exchange rate
            k_series: Risk premium series
            us_rate: US interest rate (annualized)
            kr_rate: KR interest rate (annualized)
            dt: Time step
            
        Returns:
            Array of simulated exchange rates
        """
        periods = len(k_series) - 1
        s = np.zeros(periods + 1)
        s[0] = s0
        
        for t in range(1, periods + 1):
            int_diff = (us_rate - kr_rate) * dt
            dw = np.random.normal(0, np.sqrt(dt))
            s[t] = s[t-1] * np.exp((int_diff + k_series[t-1] * dt) + self.sigma_s * dw)
            
        return s
    
    def forecast_distribution(self, s_current, k_current, us_rate, kr_rate, horizon):
        """
        Generate the forecast distribution for a given horizon
        
        Args:
            s_current: Current exchange rate
            k_current: Current risk premium value
            us_rate: US interest rate (annualized)
            kr_rate: KR interest rate (annualized)
            horizon: Forecast horizon in days
            
        Returns:
            Mean and standard deviation of the log exchange rate forecast
        """
        # Time in years
        h = horizon / 252
        
        # Expected log exchange rate
        int_diff = us_rate - kr_rate
        log_s_mean = np.log(s_current) + int_diff * h
        log_s_mean += (k_current / self.theta) * (1 - np.exp(-self.theta * h))
        log_s_mean += self.mu * (h - (1 - np.exp(-self.theta * h)) / self.theta)
        
        # Variance of log exchange rate
        log_s_var = self.sigma_s**2 * h
        log_s_var += (self.sigma_k**2 / (2 * self.theta)) * (1 - np.exp(-2 * self.theta * h))
        
        return log_s_mean, np.sqrt(log_s_var)
    
    def get_confidence_interval(self, mean, std, confidence=0.95):
        """
        Calculate confidence interval for the forecast
        
        Args:
            mean: Mean of the log distribution
            std: Standard deviation of the log distribution
            confidence: Confidence level (default: 0.95)
            
        Returns:
            Lower and upper bounds of the confidence interval
        """
        z = norm.ppf((1 + confidence) / 2)
        lower = np.exp(mean - z * std)
        upper = np.exp(mean + z * std)
        return lower, upper
    
    def backtest(self, data, horizon_days, confidence_levels=[0.5, 0.6, 0.7, 0.8, 0.9, 0.95, 0.99]):
        """
        Backtest the model on historical data
        
        Args:
            data: DataFrame with exchange rates and interest rates
            horizon_days: Forecast horizon in days
            confidence_levels: List of confidence levels to test
            
        Returns:
            Dictionary of coverage rates for each confidence level
        """
        coverage_rates = {level: 0 for level in confidence_levels}
        count = 0
        
        for i in range(len(data) - horizon_days):
            # Current data point
            s_current = data.loc[i, 'exchange_rate']
            k_current = data.loc[i, 'risk_premium']
            us_rate = data.loc[i, 'us_rate']
            kr_rate = data.loc[i, 'kr_rate']
            
            # Future realized value
            s_realized = data.loc[i + horizon_days, 'exchange_rate']
            
            # Generate forecast
            mean, std = self.forecast_distribution(s_current, k_current, us_rate, kr_rate, horizon_days)
            
            # Check coverage for each confidence level
            for level in confidence_levels:
                lower, upper = self.get_confidence_interval(mean, std, level)
                if lower <= s_realized <= upper:
                    coverage_rates[level] += 1
            
            count += 1
        
        # Convert to percentages
        for level in confidence_levels:
            coverage_rates[level] = (coverage_rates[level] / count) * 100
            
        return coverage_rates


def generate_simulated_data(start_date='2010-01-01', end_date='2025-04-09'):
    """
    Generate simulated data for testing the model
    
    Args:
        start_date: Start date for the simulation
        end_date: End date for the simulation
        
    Returns:
        DataFrame with simulated data
    """
    # Generate date range
    start = datetime.strptime(start_date, '%Y-%m-%d')
    end = datetime.strptime(end_date, '%Y-%m-%d')
    days = (end - start).days
    business_days = int(days * 5/7)  # Approximate business days
    
    # Create date index
    date_range = pd.date_range(start=start_date, periods=business_days, freq='B')
    
    # Initial values
    s0 = 1150.0  # Initial USD/KRW rate
    k0 = 0.02    # Initial risk premium
    
    # Interest rates with some variation over time
    time_array = np.linspace(0, 1, business_days)
    us_rates = 0.03 + 0.01 * np.sin(2 * np.pi * time_array / 3)
    kr_rates = 0.04 + 0.01 * np.sin(2 * np.pi * time_array / 3 + 0.5)
    
    # Create model instance
    model = OUExchangeRateModel(theta=0.2, mu=0.0, sigma_k=0.04, sigma_s=0.008)
    
    # Simulate risk premium
    k_series = model.simulate_risk_premium(k0, periods=business_days-1)
    
    # Simulate exchange rate
    s_series = model.simulate_exchange_rate(s0, k_series, us_rates.mean(), kr_rates.mean())
    
    # Calculate realized risk premium
    realized_risk_premium = np.zeros(business_days)
    for i in range(business_days-1):
        h = 1/252  # One day in years
        realized_risk_premium[i] = np.log(s_series[i+1]/s_series[i]) - (us_rates[i] - kr_rates[i]) * h
    
    # Create DataFrame
    df = pd.DataFrame({
        'date': date_range,
        'exchange_rate': s_series,
        'us_rate': us_rates,
        'kr_rate': kr_rates,
        'risk_premium': realized_risk_premium
    })
    
    return df
