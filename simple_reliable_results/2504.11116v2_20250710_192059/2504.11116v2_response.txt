class PortfolioEnvironment:

        self.n_assets = n_assets  # Number of assets
        self.n_factors = n_factors  # Number of state factors
        self.T = T  # Time horizon
        self.dt = dt  # Time step
        self.r = r  # Risk-free rate
        self.gamma = gamma  # Risk aversion
        
        # Initialize parameters
        self._setup_parameters()
        
    def _setup_parameters(self):
        # Mean reversion speed for factors
        self.kappa_Y = np.diag([2.0 + 0.5*i for i in range(self.n_factors)])
        
        # Long-term means for factors
        self.theta_Y = np.random.uniform(0.2, 0.4, self.n_factors)
        
        # Factor volatility (diagonal matrix)
        self.sigma_Y = np.diag(np.random.uniform(0.3, 0.5, self.n_factors))
        
        # Asset volatilities
        self.sigma = np.random.uniform(0.1, 0.5, self.n_assets)
        
        # Factor loadings for assets
        self.alpha = np.zeros((self.n_assets, self.n_factors))
        for i in range(self.n_assets):
            alpha_i = np.random.dirichlet(np.ones(self.n_factors))
            self.alpha[i, :] = alpha_i

        self.psi = self._generate_correlation_matrix(self.n_assets)  # Asset correlation
        self.phi = self._generate_correlation_matrix(self.n_factors)  # Factor correlation
        self.rho = np.random.uniform(-0.2, 0.2, (self.n_assets, self.n_factors))  # Cross-correlation
        
        # Ensure positive definiteness of the block matrix
        self._ensure_positive_definite_correlations()
        
    def _generate_correlation_matrix(self, size):
        """Generate a valid correlation matrix."""
        A = np.random.randn(size, size)
        # Create a positive definite matrix
        A = A @ A.T
        # Convert to correlation matrix
        D = np.sqrt(np.diag(A))
        corr = A / np.outer(D, D)
        return corr
    
    def _ensure_positive_definite_correlations(self):
        """Ensure the full correlation matrix is positive definite."""
        full_corr = np.block([[self.psi, self.rho], [self.rho.T, self.phi]])
        
        # Check eigenvalues
        eig_vals = np.linalg.eigvals(full_corr)

        # If not positive definite, add a small diagonal loading
        if min_eig <= 1e-6:
            epsilon = abs(min_eig) + 1e-6

            self.phi += np.eye(self.n_factors) * epsilon
    
    def simulate_path(self, initial_wealth, initial_state, policy_fn, n_steps=None):
        """Simulate a single path using the given policy function."""
        if n_steps is None:

        state = np.zeros((n_steps + 1, self.n_factors))

        wealth[0] = initial_wealth
        state[0] = initial_state

        full_cov = np.block([
            [np.diag(self.sigma) @ self.psi @ np.diag(self.sigma), np.diag(self.sigma) @ self.rho @ self.sigma_Y],
            [self.sigma_Y @ self.rho.T @ np.diag(self.sigma), self.sigma_Y @ self.phi @ self.sigma_Y]
        ])
        chol_factor = np.linalg.cholesky(full_cov * self.dt)
        
        # Simulation
        for t in range(n_steps):
            current_time = times[t]
            current_wealth = wealth[t]
            current_state = state[t]

            actions[t] = portfolio_weights

            correlated_noise = chol_factor @ z

            factor_noise = correlated_noise[self.n_assets:]
            
            # Update state (OU process)
            state_drift = self.kappa_Y @ (self.theta_Y - current_state) * self.dt
            state_diffusion = factor_noise
            state[t+1] = current_state + state_drift + state_diffusion

                self.r + portfolio_weights @ excess_returns
            ) * self.dt
            wealth_diffusion = current_wealth * (portfolio_weights * self.sigma) @ asset_noise
            wealth[t+1] = current_wealth + wealth_drift + wealth_diffusion
            
            # Ensure wealth stays positive
            wealth[t+1] = max(wealth[t+1], 0.1)
        
        # Get final action
        actions[n_steps] = policy_fn(times[n_steps], wealth[n_steps], state[n_steps])

            'times': times,
            'wealth': wealth,
            'state': state,

        else:

class PolicyNetwork(nn.Module):
    """Neural network for portfolio policy."""
    
    def __init__(self, n_factors, n_assets):
        super(PolicyNetwork, self).__init__()
        
        # Input: time, wealth, state factors
        input_dim = 2 + n_factors
        
        # Hidden layers
        self.layers = nn.Sequential(
            nn.Linear(input_dim, 64),
            nn.LeakyReLU(),
            nn.Linear(64, 64),
            nn.LeakyReLU(),
            nn.Linear(64, 32),
            nn.LeakyReLU(),
            nn.Linear(32, n_assets)
        )
        
    def forward(self, t, wealth, state):
        # Concatenate inputs
        x = torch.cat([t.reshape(-1, 1), wealth.reshape(-1, 1), state], dim=1)
        return self.layers(x)

    def estimate_costates(self, t, wealth, state, n_mc_samples=50):
        """Estimate costates using Monte Carlo averaging of BPTT estimates."""
        lambda_estimates = []
        dx_lambda_estimates = []
        dy_lambda_estimates = []
        
        # Create computational graph for each MC sample
        for _ in range(n_mc_samples):
            # Set up tensors with gradients
            t_tensor = torch.tensor([t], dtype=torch.float32)
            wealth_tensor = torch.tensor([wealth], dtype=torch.float32, requires_grad=True)
            state_tensor = torch.tensor([state], dtype=torch.float32, requires_grad=True)
            
            # Simplified simulation of a single trajectory to terminal time
            remaining_time = self.env.T - t
            n_remaining_steps = max(int(remaining_time / self.env.dt), 1)
            dt = remaining_time / n_remaining_steps
            
            current_t = t_tensor
            current_wealth = wealth_tensor
            current_state = state_tensor
            
            # Forward simulation with gradients
            for _ in range(n_remaining_steps):
                # Get policy action
                actions = self.policy_net(current_t, current_wealth, current_state)
                
                # Calculate expected excess returns
                excess_returns = torch.zeros(self.n_assets)
                for i in range(self.n_assets):
                    excess_returns[i] = self.env.sigma[i] * torch.sum(torch.tensor(self.env.alpha[i]) * current_state[0])
                
                # Update wealth (deterministic for gradient calculation)
                wealth_drift = current_wealth * (
                    self.env.r + torch.sum(actions * excess_returns)
                ) * dt
                
                # Update state (simplified deterministic for gradient calculation)
                kappa_tensor = torch.tensor(self.env.kappa_Y)
                theta_tensor = torch.tensor(self.env.theta_Y)
                state_drift = torch.mm((theta_tensor - current_state), kappa_tensor) * dt
                
                # Apply updates
                current_wealth = current_wealth + wealth_drift
                current_state = current_state + state_drift
                current_t = current_t + torch.tensor([dt])
            
            # Calculate terminal utility
            if self.env.gamma == 1:
                utility = torch.log(current_wealth)
            else:
                utility = (current_wealth**(1-self.env.gamma)) / (1-self.env.gamma)
            
            # Compute gradients
            utility.backward(retain_graph=True)
            
            # Extract gradients (costates)
            lambda_val = wealth_tensor.grad.item()  # ∂J/∂x
            lambda_estimates.append(lambda_val)
            
            # Reset gradients
            wealth_tensor.grad.zero_()
            state_tensor.grad.zero_()
            
            # Compute second-order derivatives by differentiating lambda
            dx_lambda = torch.autograd.grad(lambda_val, wealth_tensor, create_graph=True)[0].item()
            dy_lambda = torch.autograd.grad(lambda_val, state_tensor, create_graph=True)[0].numpy()
            
            dx_lambda_estimates.append(dx_lambda)
            dy_lambda_estimates.append(dy_lambda)
        
        # Average estimates
        lambda_avg = np.mean(lambda_estimates)
        dx_lambda_avg = np.mean(dx_lambda_estimates)
        dy_lambda_avg = np.mean(dy_lambda_estimates, axis=0)
        
        return lambda_avg, dx_lambda_avg, dy_lambda_avg
    
    def get_pmp_policy(self, t, wealth, state):
        """Get policy action using PMP first-order conditions with estimated costates."""
        # Estimate costates
        lambda_val, dx_lambda, dy_lambda = self.estimate_costates(t, wealth, state, n_mc_samples=10)
        
        # Asset return covariance matrix
        sigma_diag = np.diag(self.env.sigma)
        asset_cov = sigma_diag @ self.env.psi @ sigma_diag
        
        # Calculate risk premium
        risk_premium = np.zeros(self.n_assets)
        for i in range(self.n_assets):
            risk_premium[i] = self.env.sigma[i] * (self.env.alpha[i] @ state)
        
        # Calculate myopic component: -λ/(x*∂xλ) * Σ^(-1) * (μ-r)
        myopic_term = -lambda_val / (wealth * dx_lambda) * np.linalg.solve(asset_cov, risk_premium)
        
        # Calculate hedging component: -1/(x*∂xλ) * Σ^(-1) * σρσY * ∂Yλ
        cross_term = sigma_diag @ self.env.rho @ self.env.sigma_Y
        hedging_term = -1 / (wealth * dx_lambda) * np.linalg.solve(asset_cov, cross_term @ dy_lambda)
        
        # Combined policy
        pmp_policy = myopic_term + hedging_term
        
        # Clip to prevent extreme values
        clipped_policy = np.clip(pmp_policy, -self.max_weight, self.max_weight)
        
        return clipped_policy
