def generate_example_function_data(n_samples=50000, x_min=-10, x_max=10, 
                                  LL=-5, LI=0, LS=50, UL=5, UI=0, US=50):
    """
    Generate data for the first example function from the paper.
    
    f(x) = asymptotic(x) + x * zasymptotic(x)
    """
    # Generate random x values
    x = np.random.uniform(x_min, x_max, n_samples)
    
    # Calculate asymptotic component
    s0 = (UI - LI) / (UL - LL)
    LS_tilde = (LS - s0) / (UL - LL)
    US_tilde = (s0 - US) / (UL - LL)
    a0 = (US_tilde - LS_tilde) / (UL - LL)
    
    # Initialize arrays for function values and derivatives
    y = np.zeros_like(x)
    dy_dx = np.zeros_like(x)
    
    # Lower asymptotic region
    lower_mask = x <= LL
    y[lower_mask] = LS * (x[lower_mask] - LL) + LI
    dy_dx[lower_mask] = LS
    
    # Upper asymptotic region
    upper_mask = x >= UL
    y[upper_mask] = US * (x[upper_mask] - UL) + UI
    dy_dx[upper_mask] = US
    
    # Non-asymptotic region
    middle_mask = np.logical_and(x > LL, x < UL)
    
    # Asymptotic component in middle region
    cubic_term = (x[middle_mask] - LL) * (UL - x[middle_mask]) * (a0 * (x[middle_mask] - LL) + LS_tilde) + (s0 * (x[middle_mask] - LL) + LI)
    
    # Zasymptotic component
    zasymptotic = (x[middle_mask] - UL)**2 * (x[middle_mask] - LL)**2 * (1.0 / (LL * UL))
    
    # Non-asymptotic component (x * zasymptotic in this example)
    non_asym = x[middle_mask] * zasymptotic
    
    # Combine for middle region
    y[middle_mask] = cubic_term + non_asym
    
    # Calculate derivative for middle region
    # Derivative of cubic term
    d_cubic = (UL - x[middle_mask]) * (a0 * (x[middle_mask] - LL) + LS_tilde) + \
              (x[middle_mask] - LL) * (-1) * (a0 * (x[middle_mask] - LL) + LS_tilde) + \
              (x[middle_mask] - LL) * (UL - x[middle_mask]) * a0 + s0
              
    # Derivative of zasymptotic
    d_zasymptotic = 2 * (x[middle_mask] - UL) * (x[middle_mask] - LL)**2 * (1.0 / (LL * UL)) + \
                    (x[middle_mask] - UL)**2 * 2 * (x[middle_mask] - LL) * (1.0 / (LL * UL))
    
    # Derivative of non-asymptotic component
    d_non_asym = zasymptotic + x[middle_mask] * d_zasymptotic
    
    # Combine for derivative in middle region
    dy_dx[middle_mask] = d_cubic + d_non_asym
    
    return x, y, dy_dx


def generate_black_scholes_data(n_samples=50000, x_min=0, x_max=20, 
                               K=10, r=0.0, sigma=0.1, t=1.0, T=2.0,
                               LL=7, UL=13):
    """
    Generate data for Black-Scholes call option pricing.
    
    Parameters:
    -----------
    n_samples: int
        Number of samples to generate
    x_min, x_max: float
        Range for spot price S
    K: float
        Strike price
    r: float
        Risk-free rate
    sigma: float
        Volatility
    t: float
        Current time
    T: float
        Maturity
    LL, UL: float
        Lower and upper asymptotic boundaries
    """
    # Generate random spot prices
    S = np.random.uniform(x_min, x_max, n_samples)
    
    # Time to maturity
    tau = T - t
    
    # Black-Scholes formula for call option
    d1 = (np.log(S/K) + (r + 0.5 * sigma**2) * tau) / (sigma * np.sqrt(tau))
    d2 = d1 - sigma * np.sqrt(tau)
    
    call_price = S * stats.norm.cdf(d1) - K * np.exp(-r * tau) * stats.norm.cdf(d2)
    
    # Delta (derivative with respect to S)
    delta = stats.norm.cdf(d1)
    
    # Calculate asymptotic coefficients
    # Lower asymptotic: LS = 0, LI = 0 (call value approaches 0 as S approaches 0)
    LS = 0
    LI = 0
    
    # Upper asymptotic: US = 1, UI = -K * exp(-r * tau) (call value approaches S-K*exp(-r*tau) as S gets large)
    US = 1
    UI = -K * np.exp(-r * tau)
    
    return S, call_price, delta, (LS, LI, US, UI)


def generate_black_scholes_regression_data(n_samples=10000, x_min=0, x_max=20, 
                                         K=10, r=0.0, sigma=0.1, t=1.0, T=2.0,
                                         LL=7, UL=13):
    """
    Generate Monte Carlo samples for regression of Black-Scholes call option pricing.
    
    Parameters are same as generate_black_scholes_data.
    """
    # Generate random spot prices at time t
    St = np.random.uniform(x_min, x_max, n_samples)
    
    # Time to maturity
    tau = T - t
    
    # Generate log-normal random increments for simulating ST
    Z = np.random.normal(0, 1, n_samples)
    
    # Simulate ST (exact simulation with log-Forward Euler)
    logSt = np.log(St)
    logST = logSt + (r - 0.5 * sigma**2) * tau + sigma * np.sqrt(tau) * Z
    ST = np.exp(logST)
    
    # Calculate payoffs
    payoff = np.maximum(ST - K, 0)
    
    # Discount payoffs
    discounted_payoff = np.exp(-r * tau) * payoff
    
    # Calculate pathwise derivatives (for DML)
    # Only paths where ST > K contribute to the derivative
    in_the_money = ST > K
    
    # Initialize derivatives
    pathwise_delta = np.zeros_like(St)
    
    # For in-the-money paths, calculate pathwise derivative
    pathwise_delta[in_the_money] = np.exp(-r * tau) * ST[in_the_money] / St[in_the_money]
    
    # Calculate true Black-Scholes prices and deltas for comparison
    d1 = (np.log(St/K) + (r + 0.5 * sigma**2) * tau) / (sigma * np.sqrt(tau))
    d2 = d1 - sigma * np.sqrt(tau)
    
    true_price = St * stats.norm.cdf(d1) - K * np.exp(-r * tau) * stats.norm.cdf(d2)
    true_delta = stats.norm.cdf(d1)
    
    # Calculate asymptotic coefficients
    LS = 0
    LI = 0
    US = 1
    UI = -K * np.exp(-r * tau)
    
    return St, discounted_payoff, pathwise_delta, true_price, true_delta, (LS, LI, US, UI)
