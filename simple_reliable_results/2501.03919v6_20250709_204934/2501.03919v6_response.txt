import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

class PortfolioStructuredEnsembleModel:
    def __init__(self, n_predictors=10, n_neurons=20, learning_rate=0.03, 
                 diversity_param=0.1, pred_reg_param=0.0001, ensemble_reg_param=3, 
                 random_factor=0.01, basis_function='rbf'):
        """
        Initialize PSEM model.
        
        Parameters:
        -----------
        n_predictors: int
            Number of individual predictors (portfolio constituents)
        n_neurons: int
            Number of neurons per layer in individual predictors
        learning_rate: float
            Learning rate for individual predictors
        diversity_param: float
            Diversity parameter (ε) in the range [0, 1]
        pred_reg_param: float
            Regularization parameter for individual predictors
        ensemble_reg_param: float
            Regularization parameter for s-RBFN
        random_factor: float
            Multiplicative factor for random initial weights
        basis_function: str
            Type of basis function ('rbf' or 'gaussian')
        """
        self.M = n_predictors
        self.k = n_neurons
        self.eta = learning_rate
        self.epsilon = diversity_param
        self.lambda_p = pred_reg_param
        self.lambda_s = ensemble_reg_param
        self.chi = random_factor
        self.basis_function = basis_function
        
        # Initialize individual predictors (theta parameters)
        self.theta = [np.random.randn(n_neurons) * random_factor for _ in range(n_predictors)]
        
        # For storing structured dataset
        self.structured_data = None
        self.basis_centers = None
        self.basis_scales = None
        self.ensemble_weights = None
        
    def _predict_individual(self, x, theta):
        """Simple two-layer MLP with tanh activation"""
        # Simplified prediction for demonstration
        return np.tanh(np.dot(x, theta))
    
    def _loss(self, pred, target):
        """MSE loss function"""
        return np.mean((pred - target) ** 2)
    
    def _calculate_delta(self, losses, j):
        """Calculate δ value based on individual predictor's performance"""
        if self.M == 1:
            return 1
        
        if np.all(losses[j] < np.delete(losses, j)):
            return 1 - self.epsilon
        else:
            return self.epsilon / (self.M - 1)
    
    def build_structured_dataset(self, X, y, n_epochs=100):
        """
        Build structured dataset using individual predictors
        
        Parameters:
        -----------
        X: array-like, shape (N, M)
            Input returns time series
        y: array-like, shape (N, M)
            Target returns (with time lag)
        n_epochs: int
            Number of training epochs
        """
        N, M = X.shape
        assert M == self.M, f"Input data has {M} assets but model expects {self.M}"
        
        # For storing predictions
        structured_data = np.zeros((N, M))
        
        # Training loop
        for epoch in range(n_epochs):
            all_losses = np.zeros((N, M))
            
            # Calculate losses for each predictor and sample
            for i in range(N):
                for j in range(M):
                    pred = self._predict_individual(X[i, j], self.theta[j])
                    all_losses[i, j] = self._loss(pred, y[i, j])
            
            # Update parameters with diversity control
            for i in range(N):
                for j in range(M):
                    pred = self._predict_individual(X[i, j], self.theta[j])
                    loss_grad = 2 * (pred - y[i, j])
                    reg_term = self.lambda_p / N * self.theta[j]
                    
                    delta = self._calculate_delta(all_losses[i], j)
                    self.theta[j] -= self.eta * (loss_grad + reg_term) * delta
                    
                    # Store prediction
                    structured_data[i, j] = self._predict_individual(X[i, j], self.theta[j])
        
        self.structured_data = structured_data
        return structured_data
    
    def _apply_basis_function(self, x, mu, sigma):
        """Apply basis function to input"""
        if self.basis_function == 'rbf':
            return 1.0 / (1.0 + np.sum((x - mu) ** 2) / (2 * sigma ** 2))
        elif self.basis_function == 'gaussian':
            return np.exp(-np.sum((x - mu) ** 2) / (2 * sigma ** 2))
        else:
            raise ValueError(f"Unknown basis function: {self.basis_function}")
    
    def transform_basis(self):
        """Transform structured data using basis functions"""
        if self.structured_data is None:
            raise ValueError("Structured data not built yet. Call build_structured_dataset first.")
        
        N, M = self.structured_data.shape
        
        # Calculate centers and scales for basis functions
        self.basis_centers = np.mean(self.structured_data, axis=0)
        self.basis_scales = np.std(self.structured_data, axis=0)
        
        # Apply basis functions
        phi = np.zeros((N, M))
        for i in range(N):
            for j in range(M):
                phi[i, j] = self._apply_basis_function(
                    self.structured_data[i, j], 
                    self.basis_centers[j], 
                    self.basis_scales[j]
                )
        
        return phi
    
    def optimize_weights(self, target_portfolio):
        """
        Optimize portfolio weights using s-RBFN and least squares
        
        Parameters:
        -----------
        target_portfolio: array-like, shape (N,)
            Target portfolio returns (e.g., equal-weighted portfolio)
        
        Returns:
        --------
        weights: array-like, shape (M,)
            Optimized portfolio weights
        """
        phi = self.transform_basis()
        
        # Solve least squares with regularization
        A = phi.T @ phi + self.lambda_s * np.eye(self.M)
        b = phi.T @ target_portfolio
        
        self.ensemble_weights = np.linalg.solve(A, b)
        
        # Normalize weights (ensure they sum to 1 and are non-negative)
        normalized_weights = np.maximum(0, self.ensemble_weights)
        normalized_weights = normalized_weights / np.sum(normalized_weights)
        
        return normalized_weights
    
    def predict_portfolio_returns(self, X_test):
        """Predict portfolio returns using optimized weights"""
        if self.ensemble_weights is None:
            raise ValueError("Model not fitted yet. Call optimize_weights first.")
        
        N_test, M_test = X_test.shape
        assert M_test == self.M, f"Test data has {M_test} assets but model expects {self.M}"
        
        # Generate predictions for test data
        test_predictions = np.zeros((N_test, M_test))
        for i in range(N_test):
            for j in range(M_test):
                test_predictions[i, j] = self._predict_individual(X_test[i, j], self.theta[j])
        
        # Apply basis functions
        phi_test = np.zeros((N_test, M_test))
        for i in range(N_test):
            for j in range(M_test):
                phi_test[i, j] = self._apply_basis_function(
                    test_predictions[i, j], 
                    self.basis_centers[j], 
                    self.basis_scales[j]
                )
        
        # Predict portfolio returns
        portfolio_returns = phi_test @ self.ensemble_weights
        
        return portfolio_returns


def select_assets_with_diversity(predictions, M, gamma, threshold=0.0):
    """
    Select assets using diversity-quality trade-off strategy
    
    Parameters:
    -----------
    predictions: array-like
        Predicted returns for all assets
    M: int
        Number of assets to select
    gamma: float
        Diversity parameter for asset selection
    threshold: float
        Minimum threshold for asset selection
    
    Returns:
    --------
    selected_indices: array-like
        Indices of selected assets
    """
    # Filter assets above threshold
    valid_indices = np.where(predictions >= threshold)[0]
    
    if len(valid_indices) < M:
        print(f"Warning: Only {len(valid_indices)} assets above threshold {threshold}, but {M} requested")
        return valid_indices
    
    # Rank by predicted returns
    ranked_indices = valid_indices[np.argsort(-predictions[valid_indices])]
    
    # Calculate sample size based on diversity parameter
    sample_size = min(int(M * gamma), len(ranked_indices))
    
    # Randomly select M assets from top sample_size
    selected_indices = np.random.choice(ranked_indices[:sample_size], size=M, replace=False)
    
    return selected_indices


def calculate_sharpe_ratio(returns):
    """Calculate Sharpe ratio"""
    if np.std(returns) == 0:
        return 0
    return np.mean(returns) / np.std(returns)


def simulate_stock_returns(n_stocks=500, n_days=252, mean_return=0.0005, 
                          volatility=0.01, correlation=0.3):
    """
    Simulate stock returns with correlation structure
    
    Parameters:
    -----------
    n_stocks: int
        Number of stocks
    n_days: int
        Number of trading days
    mean_return: float
        Mean daily return
    volatility: float
        Daily volatility
    correlation: float
        Correlation between stocks
    
    Returns:
    --------
    returns: array-like, shape (n_days, n_stocks)
        Simulated daily returns
    """
    # Create correlation matrix
    correlation_matrix = np.ones((n_stocks, n_stocks)) * correlation
    np.fill_diagonal(correlation_matrix, 1)
    
    # Create covariance matrix
    std_vector = np.ones(n_stocks) * volatility
    cov_matrix = np.outer(std_vector, std_vector) * correlation_matrix
    
    # Generate correlated returns
    returns = np.random.multivariate_normal(
        mean=np.ones(n_stocks) * mean_return, 
        cov=cov_matrix, 
        size=n_days
    )
    
    return returns

def calculate_max_drawdown(returns):
    """Calculate maximum drawdown from returns"""
    # Convert returns to cumulative returns
    cum_returns = (1 + returns).cumprod()
    
    # Calculate running maximum
    running_max = np.maximum.accumulate(cum_returns)
    
    # Calculate drawdown
    drawdown = (cum_returns - running_max) / running_max
    
    # Return maximum drawdown
    return np.min(drawdown)
