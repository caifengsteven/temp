class BlackScholesMerton:
    """Black-Scholes-Merton model implementation"""
    def __init__(self, S0=100, mu=0, sigma=0.32):
        self.S0 = S0
        self.mu = mu
        self.sigma = sigma
        
    def simulate_path(self, T=1.0, steps=1024):
        """Simulate a BSM price path using Euler-Maruyama method"""
        dt = T / steps
        S = np.zeros(steps + 1)
        S[0] = self.S0
        
        for i in range(steps):
            dW = np.random.normal(0, np.sqrt(dt))
            S[i+1] = S[i] + (self.mu - 0.5 * self.sigma**2) * S[i] * dt + self.sigma * S[i] * dW
            
        return S
    
    def simulate_paths(self, n_paths, T=1.0, steps=1024):
        """Simulate multiple BSM price paths"""
        paths = np.zeros((n_paths, steps + 1))
        for i in range(n_paths):
            paths[i] = self.simulate_path(T, steps)
        return paths

class HestonModel:
    """Heston stochastic volatility model implementation"""
    def __init__(self, S0=100, V0=0.32, mu=0, kappa=3.0, theta=0.25, sigma=0.4, rho=0.3):
        self.S0 = S0
        self.V0 = V0
        self.mu = mu
        self.kappa = kappa
        self.theta = theta
        self.sigma = sigma
        self.rho = rho
        
    def simulate_path(self, T=1.0, steps=1024):
        """Simulate a Heston model price path using Euler-Maruyama method"""
        dt = T / steps
        S = np.zeros(steps + 1)
        V = np.zeros(steps + 1)
        S[0] = self.S0
        V[0] = self.V0
        
        for i in range(steps):
            dW1 = np.random.normal(0, np.sqrt(dt))
            dW2 = self.rho * dW1 + np.sqrt(1 - self.rho**2) * np.random.normal(0, np.sqrt(dt))
            
            # Ensure variance stays positive
            V[i] = max(V[i], 0)
            
            # Update variance
            V[i+1] = V[i] + self.kappa * (self.theta - V[i]) * dt + self.sigma * np.sqrt(V[i]) * dW2
            
            # Update price
            S[i+1] = S[i] + (self.mu - 0.5 * V[i]) * S[i] * dt + np.sqrt(V[i]) * S[i] * dW1
            
        return S, V
    
    def simulate_paths(self, n_paths, T=1.0, steps=1024):
        """Simulate multiple Heston model price paths"""
        price_paths = np.zeros((n_paths, steps + 1))
        variance_paths = np.zeros((n_paths, steps + 1))
        
        for i in range(n_paths):
            price_paths[i], variance_paths[i] = self.simulate_path(T, steps)
            
        return price_paths, variance_paths

class AmericanOption:
    """American option pricing and payoff calculation"""
    def __init__(self, K=100, T=1.0):
        self.K = K
        self.T = T
        
    def put_payoff(self, S):
        """Calculate put option payoff"""
        return np.maximum(self.K - S, 0)
    
    def estimate_price_binomial_tree(self, S0, r, sigma, steps=1000):
        """Price American put option using binomial tree method (for BSM model)"""
        dt = self.T / steps
        u = np.exp(sigma * np.sqrt(dt))
        d = 1 / u
        p = (np.exp(r * dt) - d) / (u - d)
        
        # Initialize asset prices at final nodes
        S = np.zeros(steps + 1)
        for i in range(steps + 1):
            S[i] = S0 * (u ** (steps - i)) * (d ** i)
            
        # Initialize option values at final nodes
        V = self.put_payoff(S)
        
        # Work backwards through the tree
        for j in range(steps-1, -1, -1):
            for i in range(j+1):
                S_node = S0 * (u ** (j - i)) * (d ** i)
                exercise = self.put_payoff(S_node)
                hold = np.exp(-r * dt) * (p * V[i] + (1 - p) * V[i+1])
                V[i] = max(exercise, hold)
                
        return V[0]

class MLP(tf.keras.Model):
    """Multi-Layer Perceptron for vector field representation"""
    def __init__(self, hidden_units=32):
        super(MLP, self).__init__()
        self.hidden1 = tf.keras.layers.Dense(hidden_units, activation='relu')
        self.hidden2 = tf.keras.layers.Dense(hidden_units, activation='relu')
        self.output_layer = tf.keras.layers.Dense(1)
        
    def call(self, inputs):
        x = self.hidden1(inputs)
        x = self.hidden2(x)
        return self.output_layer(x)

class ResNetSDE(tf.keras.Model):
    """First-order deep neural SDE network (ResNet-style)"""
    def __init__(self, n_steps=1024, d=1):
        super(ResNetSDE, self).__init__()
        self.n_steps = n_steps
        self.d = d
        self.dt = 1.0 / n_steps
        
        # Drift (V0) and diffusion (V1...Vd) vector fields
        self.drift = MLP()
        self.diffusions = [MLP() for _ in range(d)]
        
    def call(self, inputs, training=False):
        """
        Forward pass through the network
        inputs: tensor of shape [batch_size, input_dim]
        """
        batch_size = tf.shape(inputs)[0]
        t = 0.0
        X = inputs
        M = tf.zeros([batch_size, 1])
        
        for step in range(self.n_steps):
            t_tensor = tf.ones([batch_size, 1]) * t
            
            # Input for vector fields: time, state, and martingale value
            field_inputs = tf.concat([t_tensor, X, M], axis=1)
            
            # Sample normal random variables for the diffusion terms
            Z = tf.random.normal([batch_size, self.d])
            
            # Euler-Maruyama update for the martingale (without drift)
            diffusion_sum = tf.zeros([batch_size, 1])
            for i in range(self.d):
                diffusion_sum += self.diffusions[i](field_inputs) * Z[:, i:i+1] * tf.sqrt(self.dt)
            
            # Update martingale value
            M_new = M + diffusion_sum
            
            # Center the martingale (as described in section 3.2.2)
            M = M_new - tf.reduce_mean(M_new, axis=0, keepdims=True)
            
            # Update time
            t += self.dt
            
        return M

class NVNetSDE(tf.keras.Model):
    """Second-order deep neural SDE network (NVnet)"""
    def __init__(self, n_steps=4, d=1):
        super(NVNetSDE, self).__init__()
        self.n_steps = n_steps
        self.d = d
        self.dt = 1.0 / n_steps
        
        # Drift (V0) and diffusion (V1...Vd) vector fields
        self.drift = MLP()
        self.diffusions = [MLP() for _ in range(d)]
        
    def call(self, inputs, training=False):
        """
        Forward pass through the NVnet
        inputs: tensor of shape [batch_size, input_dim]
        """
        batch_size = tf.shape(inputs)[0]
        t = 0.0
        X = inputs
        M = tf.zeros([batch_size, 1])
        
        for step in range(self.n_steps):
            t_tensor = tf.ones([batch_size, 1]) * t
            
            # Input for vector fields
            field_inputs = tf.concat([t_tensor, X, M], axis=1)
            
            # Sample random variables for the diffusion terms
            Z = tf.random.normal([batch_size, self.d])
            eta = tf.sign(tf.random.uniform([batch_size, 1]) - 0.5)  # P(eta = ±1) = 1/2
            
            # High-order scheme implementation (simplified version of equation 7)
            # This is a simplified approximation of the NVnet scheme
            
            # First half-step with drift
            drift_val = self.drift(field_inputs) * (self.dt/2)
            M_half = M + drift_val
            
            # Apply diffusion terms (either forward or backward order based on eta)
            M_diff = M_half
            for i in range(self.d):
                idx = i if eta[0, 0] > 0 else self.d - 1 - i
                diff_input = tf.concat([t_tensor, X, M_diff], axis=1)
                diff_val = self.diffusions[idx](diff_input) * Z[:, idx:idx+1] * tf.sqrt(self.dt)
                M_diff = M_diff + diff_val
            
            # Second half-step with drift
            final_input = tf.concat([t_tensor, X, M_diff], axis=1)
            drift_val = self.drift(final_input) * (self.dt/2)
            M_new = M_diff + drift_val
            
            # Center the martingale
            M = M_new - tf.reduce_mean(M_new, axis=0, keepdims=True)
            
            # Update time
            t += self.dt
            
        return M

def train_model(model, asset_model, option, batch_size=5000, iterations=2000):
    """Train the neural SDE model to find optimal hedging strategy"""
    optimizer = tf.keras.optimizers.Adam(learning_rate=0.001)
    loss_history = []
    
    for iteration in tqdm(range(iterations)):
        # Generate asset paths
        if isinstance(asset_model, BlackScholesMerton):
            S_paths = asset_model.simulate_paths(batch_size, steps=model.n_steps)
            S_final = S_paths[:, -1]
        else:  # Heston model
            S_paths, _ = asset_model.simulate_paths(batch_size, steps=model.n_steps)
            S_final = S_paths[:, -1]
        
        # Convert to tensor
        S_tensor = tf.convert_to_tensor(S_final.reshape(-1, 1), dtype=tf.float32)
        
        # Compute loss using Rogers' dual formula
        with tf.GradientTape() as tape:
            # Generate martingale process
            M = model(S_tensor)
            
            # Calculate option payoff
            Z = option.put_payoff(S_final)
            Z_tensor = tf.convert_to_tensor(Z.reshape(-1, 1), dtype=tf.float32)
            
            # Rogers' dual formula: inf_M E[sup_t (Z_t - M_t)]
            # Here we approximate by using only the terminal values
            loss = tf.reduce_mean(tf.maximum(Z_tensor - M, 0))
        
        # Update model parameters
        gradients = tape.gradient(loss, model.trainable_variables)
        optimizer.apply_gradients(zip(gradients, model.trainable_variables))
        
        # Store loss
        loss_history.append(loss.numpy())
        
    return loss_history
