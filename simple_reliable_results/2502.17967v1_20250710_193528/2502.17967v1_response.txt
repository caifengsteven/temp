class LLMAgent:
    def __init__(self, agent: Agent, use_visual: bool = False, use_reflection: bool = False):
        self.agent = agent
        self.use_visual = use_visual
        self.use_reflection = use_reflection
        self.decisions = []
        self.returns = []

    def make_decision(self, stocks: Dict[str, Stock], day: int) -> Dict:
        """Simulate LLM decision making process"""

        # Visual data yields better decisions (as per the paper findings)

            accuracy = np.random.normal(0.5, 0.15)  # Lower mean accuracy for textual
            confidence = np.random.normal(0.6, 0.15)
        
        # Reflection improves performance
        if self.use_reflection:
            accuracy *= 1.2
            confidence *= 1.1
            
        # Cap values between 0 and 1
        accuracy = min(max(accuracy, 0), 1)
        confidence = min(max(confidence, 0), 1)
        
        # Decide which stock to buy based on simulated LLM performance
        stock_potentials = {}
        for name, stock in stocks.items():
            # Calculate a potential return score based on LLM "prediction ability"
            future_performance = (stock.historical_prices[-1] / stock.historical_prices[-5] - 1) * 10
            
            # Add noise based on accuracy (lower accuracy = more noise)
            noise_factor = 1 - accuracy
            noise = np.random.normal(0, noise_factor * 0.5)
            
            # Final stock potential score
            stock_potentials[name] = future_performance + noise
        
        # Find the best stock based on this simulated "LLM analysis"
        best_stock = max(stock_potentials.items(), key=lambda x: x[1])[0]
        
        # Determine amount to invest (influenced by confidence)
        available_cash = self.agent.cash - 2000  # Keep minimum 2000 cash
        if available_cash <= 0:
            return {"action": "hold", "reason": "Insufficient funds"}
        
        invest_ratio = confidence * 0.8  # Higher confidence = more investment
        investment_amount = available_cash * invest_ratio
        
        decision = {
            "action": "buy" if investment_amount > 0 else "hold",
            "stock": best_stock if investment_amount > 0 else None,
            "amount": investment_amount if investment_amount > 0 else 0,
            "day": day,
            "accuracy": accuracy,
            "confidence": confidence
        }
        
        self.decisions.append(decision)
        return decision
    
    def execute_trade(self, decision: Dict, stocks: Dict[str, Stock]):
        """Execute a trade based on the decision"""
        if decision["action"] == "hold":
            return
        
        stock_name = decision["stock"]
        amount = decision["amount"]
        price = stocks[stock_name].current_price
        
        # Calculate shares to buy
        shares = amount / price
        
        # Update agent holdings
        if stock_name not in self.agent.holdings:
            self.agent.holdings[stock_name] = 0
        
        self.agent.holdings[stock_name] += shares
        self.agent.cash -= amount
    
    def calculate_returns(self, stocks: Dict[str, Stock]):

        current_value = self.agent.total_value
        return_rate = (current_value - self.initial_value) / self.initial_value
        self.returns.append(return_rate)
        return return_rate

def calculate_metrics(llm_agents):
    """Calculate performance metrics based on the paper"""
    results = {}
    
    for agent in llm_agents:
        name = agent.agent.name
        returns = agent.returns
        
        # Calculate metrics
        total_return = returns[-1] * 100  # Convert to percentage
        mean_daily_return = np.mean([r * 100 for r in returns])  # Convert to percentage
        std_daily_return = np.std([r * 100 for r in returns])
        
        # Calculate win rate (percentage of positive returns)
        daily_returns = [returns[i] - returns[i-1] for i in range(1, len(returns))]
        win_rate = sum(1 for r in daily_returns if r > 0) / len(daily_returns) * 100
        
        # Calculate Sharpe ratio (assuming risk-free rate = 0)
        sharpe_ratio = mean_daily_return / std_daily_return if std_daily_return > 0 else 0
        
        results[name] = {
            "TR": total_return,
            "Mean": mean_daily_return,
            "Std": std_daily_return,
            "WR": win_rate,
            "SR": sharpe_ratio
        }
    
    return results
