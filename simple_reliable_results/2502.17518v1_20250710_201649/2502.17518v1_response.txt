class StockEnvironment:
    """Simulated stock trading environment similar to FinRL."""
    
    def __init__(self, stock_data: pd.DataFrame, initial_balance: float = 10000.0, 
                 transaction_cost: float = 0.001, window_size: int = 10):
        self.stock_data = stock_data
        self.initial_balance = initial_balance
        self.transaction_cost = transaction_cost
        self.window_size = window_size
        self.reset()
    
    def reset(self):
        """Reset the environment to initial state."""
        self.current_step = self.window_size
        self.balance = self.initial_balance
        self.holdings = np.zeros(len(self.stock_data.columns))
        self.portfolio_value_history = [self.initial_balance]
        return self._get_state()
    
    def _get_state(self):
        """Get current state observation."""
        # Stock prices
        prices = self.stock_data.iloc[self.current_step].values
        # Historical price features (last window_size days)
        hist_prices = self.stock_data.iloc[self.current_step-self.window_size:self.current_step].values
        # Current holdings
        holdings = self.holdings
        # Current balance
        balance = self.balance
        
        # Features for RL models and classifiers
        features = np.concatenate([
            hist_prices.flatten(),  # Historical prices
            holdings,               # Current holdings
            [balance]               # Current balance
        ])
        
        return {
            'prices': prices,
            'holdings': holdings,
            'balance': balance,
            'features': features
        }
    
    def step(self, action):
        """Take action in the environment and return new state, reward, done."""
        prev_portfolio_value = self._get_portfolio_value()
        
        # Apply action (buy/sell stocks)
        prices = self.stock_data.iloc[self.current_step].values
        for i, a in enumerate(action):
            if a > 0:  # Buy
                cost = prices[i] * a * (1 + self.transaction_cost)
                if cost <= self.balance:
                    self.holdings[i] += a
                    self.balance -= cost
            elif a < 0:  # Sell
                a_abs = abs(a)
                if a_abs <= self.holdings[i]:
                    self.holdings[i] -= a_abs
                    self.balance += prices[i] * a_abs * (1 - self.transaction_cost)
        
        # Move to next day
        self.current_step += 1
        done = self.current_step >= len(self.stock_data) - 1
        
        # Calculate reward (portfolio value change)
        new_portfolio_value = self._get_portfolio_value()
        reward = new_portfolio_value - prev_portfolio_value
        self.portfolio_value_history.append(new_portfolio_value)
        
        return self._get_state(), reward, done
    
    def _get_portfolio_value(self):
        """Calculate total portfolio value."""
        if self.current_step < len(self.stock_data):
            stock_values = self.stock_data.iloc[self.current_step].values * self.holdings
            return self.balance + np.sum(stock_values)
        return self.balance

class Classifier:

        """Train the classifier."""
        X_scaled = self.scaler.fit_transform(X)

        """Predict probability distribution."""
        if self.model is None:

        X_scaled = self.scaler.transform(X)
        return self.model.predict_proba(X_scaled)

class EnsembleRL:
    """Ensemble RL method combining multiple agents and classifiers."""
    
    def __init__(self, agent1: BaseAgent, agent2: BaseAgent, classifier_group: ClassifierGroup, 
                 variance_threshold: float = 0.5):
        self.agent1 = agent1
        self.agent2 = agent2
        self.classifier_group = classifier_group
        self.variance_threshold = variance_threshold
        self.name = f"{agent1.name}_{agent2.name}_ensemble"
    
    def act(self, state):
        """Make trading decision using ensemble approach."""
        # Get actions from both agents
        action1 = self.agent1.act(state)
        action2 = self.agent2.act(state)
        
        # Create feature vector for classifiers
        features = state['features'].reshape(1, -1)
        
        # Get probabilities from all classifiers for both agents' actions
        all_probs = []
        for clf in self.classifier_group.classifiers:
            probs = clf.predict_proba(features)[0]  # Get probability distribution
            all_probs.append(probs)
        
        all_probs = np.array(all_probs)
        
        # Create candidate matrix Q
        C = len(self.classifier_group.classifiers)  # Number of classifiers
        Q = np.zeros((C, 2))  # C classifiers x 2 agents
        
        # Fill Q with true probabilities for each agent's actual class
        for i in range(C):
            # Simplified: assume binary classification where class 1 is the positive action
            Q[i, 0] = all_probs[i][1]  # Probability of agent1's action being positive
            Q[i, 1] = all_probs[i][0]  # Probability of agent2's action being positive
        
        # Compute variance for decision
        variance = np.std(Q, axis=0)
        avg_variance = np.mean(variance)
        
        # Make decision based on variance threshold
        if avg_variance < self.variance_threshold:
            # Low variance - select action with highest probability
            if np.mean(Q[:, 0]) > np.mean(Q[:, 1]):
                return action1
            else:
                return action2
        else:
            # High variance - select action with lowest probability (more conservative)
            if np.mean(Q[:, 0]) < np.mean(Q[:, 1]):
                return action1
            else:
                return action2


def generate_stock_data(num_stocks: int = 3, days: int = 252, seed: int = 42):
    """Generate simulated stock price data."""
    np.random.seed(seed)
    
    # Start date
    start_date = datetime(2020, 1, 1)
    dates = [start_date + timedelta(days=i) for i in range(days)]
    
    # Generate stock prices with some correlation
    prices = np.zeros((days, num_stocks))
    
    # Initial prices
    prices[0] = np.random.uniform(50, 200, num_stocks)
    
    # Generate price movements
    for t in range(1, days):
        # Market factor (affects all stocks)
        market_return = np.random.normal(0.0005, 0.01)
        
        for s in range(num_stocks):
            # Stock-specific factor
            stock_return = np.random.normal(0.0003, 0.02)
            # Combined return
            total_return = 0.7 * market_return + 0.3 * stock_return
            # Update price
            prices[t, s] = prices[t-1, s] * (1 + total_return)
    
    # Create DataFrame
    df = pd.DataFrame(prices, index=dates, columns=[f'Stock_{i+1}' for i in range(num_stocks)])
    return df


def calculate_metrics(portfolio_values: List[float], risk_free_rate: float = 0.01/252):
    """Calculate financial metrics: Cumulative Returns, Sharpe Ratio, Calmar Ratio, MDD."""
    if not portfolio_values or len(portfolio_values) < 2:
        return {'cum_return': 0, 'sharpe_ratio': 0, 'calmar_ratio': 0, 'mdd': 0}
    
    # Convert to numpy array for calculations
    values = np.array(portfolio_values)
    
    # Calculate daily returns
    returns = np.diff(values) / values[:-1]
    
    # Cumulative Return
    cum_return = (values[-1] / values[0]) - 1
    
    # Max Drawdown
    peak = np.maximum.accumulate(values)
    drawdown = (values - peak) / peak
    mdd = np.min(drawdown)
    
    # Sharpe Ratio (annualized)
    excess_returns = returns - risk_free_rate
    sharpe_ratio = np.mean(excess_returns) / np.std(excess_returns) * np.sqrt(252)
    
    # Calmar Ratio (annualized)
    annualized_return = (1 + cum_return) ** (252 / len(values)) - 1
    calmar_ratio = annualized_return / abs(mdd) if mdd != 0 else 0
    
    return {
        'cum_return': cum_return,
        'sharpe_ratio': sharpe_ratio,
        'calmar_ratio': calmar_ratio,
        'mdd': mdd
    }


def test_trading_strategy(env: StockEnvironment, agent: Any, episodes: int = 1):
    """Test a trading strategy for multiple episodes."""
    all_portfolio_values = []
    
    for _ in range(episodes):
        state = env.reset()
        done = False
        
        while not done:
            action = agent.act(state)
            state, reward, done = env.step(action)
            
        all_portfolio_values.append(env.portfolio_value_history)
    
    # Average portfolio values across episodes
    avg_portfolio_values = np.mean([values for values in all_portfolio_values], axis=0)
    
    # Calculate metrics
    metrics = calculate_metrics(avg_portfolio_values)
    
    return {
        'portfolio_values': avg_portfolio_values,
        'metrics': metrics
    }

def generate_training_data(env: StockEnvironment, episodes: int = 5):
    """Generate training data for classifiers using random actions."""
    X = []
    y = []
    
    random_agent = RandomAgent(0, len(env.stock_data.columns))
    
    for _ in range(episodes):
        state = env.reset()
        done = False
        
        while not done:
            features = state['features']
            action = random_agent.act(state)
            
            # For binary classification: 1 if action was profitable, 0 otherwise
            X.append(features)
            
            # Take action and see if it was profitable
            next_state, reward, done = env.step(action)
            y.append(1 if reward > 0 else 0)
            
            state = next_state
    
    return np.array(X), np.array(y)
