import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy import stats
from typing import List, Tuple, Dict, Optional

class MSSEstimator:
    """
    Estimator for the Minimum Spanning Set (MSS) as described in the paper.
    """
    def __init__(self, significance_level: float = 0.05, bootstrap_samples: int = 1000):
        """
        Initialize the MSS estimator.
        
        Args:
            significance_level: The significance level for inference (default: 0.05)
            bootstrap_samples: The number of bootstrap samples (default: 1000)
        """
        self.p = significance_level
        self.B = bootstrap_samples
        
    def compute_test_statistics(self, returns: np.ndarray) -> Tuple[np.ndarray, List[Dict]]:
        """
        Compute the test statistics M_i,T for each asset.
        
        Args:
            returns: Array of asset returns with shape (T, d)
            
        Returns:
            M_i_T: Array of test statistics
            regression_results: List of regression results for each asset
        """
        T, d = returns.shape
        M_i_T = np.zeros(d)
        regression_results = []
        
        for i in range(d):
            # Extract the asset i and other assets
            R_i = returns[:, i]
            R_minus_i = np.delete(returns, i, axis=1)
            
            # Add constant for regression
            X = np.column_stack([np.ones(T), R_minus_i])
            
            # Compute LS estimator of (alpha_i, beta_i)
            Q_hat_inv = np.linalg.inv(X.T @ X / T)
            coef = Q_hat_inv @ (X.T @ R_i / T)
            
            alpha_hat = coef[0]
            beta_hat = coef[1:]
            
            # Compute residuals and their variance
            residuals = R_i - X @ coef
            sigma_hat_squared = np.mean(residuals**2)
            
            # Compute s_hat_i,1 and s_hat_i,2
            s_hat_i_1 = np.sqrt(sigma_hat_squared * Q_hat_inv[0, 0])
            
            # For s_hat_i_2, we need to compute 1^T_{d-1} beta_i - 1
            A_2 = np.ones(d-1)
            s_hat_i_2 = np.sqrt(sigma_hat_squared * A_2 @ Q_hat_inv[1:, 1:] @ A_2)
            
            # Compute M_i,T as max of t-statistics
            t_stat_1 = np.abs(alpha_hat) * np.sqrt(T) / s_hat_i_1
            t_stat_2 = np.abs(np.sum(beta_hat) - 1) * np.sqrt(T) / s_hat_i_2
            
            M_i_T[i] = max(t_stat_1, t_stat_2)
            
            # Store regression results
            regression_results.append({
                'alpha': alpha_hat,
                'beta': beta_hat,
                'residuals': residuals,
                's_i_1': s_hat_i_1,
                's_i_2': s_hat_i_2,
                't_stat_1': t_stat_1,
                't_stat_2': t_stat_2
            })
            
        return M_i_T, regression_results
    
    def moving_block_bootstrap(self, returns: np.ndarray, M_i_T: np.ndarray, 
                               regression_results: List[Dict]) -> np.ndarray:
        """
        Implement the moving block bootstrap procedure.
        
        Args:
            returns: Array of asset returns with shape (T, d)
            M_i_T: Array of test statistics
            regression_results: List of regression results for each asset
            
        Returns:
            Bootstrap critical values
        """
        T, d = returns.shape
        
        # Set block length based on T as recommended in the paper
        l = int(1.2 * T**(1/4))
        
        # Number of blocks
        m = T // l
        T_B = m * l
        
        # Initialize bootstrap test statistics
        M_i_T_boot = np.zeros((self.B, d))
        
        for b in range(self.B):
            # Sample blocks
            block_indices = np.random.randint(0, T - l + 1, m)
            boot_indices = np.concatenate([np.arange(idx, idx + l) for idx in block_indices])
            
            # Create bootstrap sample
            boot_returns = returns[boot_indices]
            
            # Compute bootstrap statistics
            for i in range(d):
                R_i_boot = boot_returns[:, i]
                R_minus_i_boot = np.delete(boot_returns, i, axis=1)
                
                # Add constant
                X_boot = np.column_stack([np.ones(T_B), R_minus_i_boot])
                
                # Original coefficients
                alpha_hat = regression_results[i]['alpha']
                beta_hat = regression_results[i]['beta']
                orig_coef = np.concatenate([[alpha_hat], beta_hat])
                
                # Bootstrap LS estimator
                Q_hat_inv_boot = np.linalg.inv(X_boot.T @ X_boot / T_B)
                coef_boot = Q_hat_inv_boot @ (X_boot.T @ R_i_boot / T_B)
                
                # Bootstrap residuals
                residuals_boot = R_i_boot - X_boot @ coef_boot
                sigma_hat_squared_boot = np.mean(residuals_boot**2)
                
                # Compute bootstrap s_hat_i,1 and s_hat_i,2
                s_hat_i_1_boot = np.sqrt(sigma_hat_squared_boot * Q_hat_inv_boot[0, 0])
                
                A_2 = np.ones(d-1)
                s_hat_i_2_boot = np.sqrt(sigma_hat_squared_boot * A_2 @ Q_hat_inv_boot[1:, 1:] @ A_2)
                
                # Compute bootstrap M_i,T
                t_stat_1_boot = np.abs(coef_boot[0] - alpha_hat) * np.sqrt(T) / s_hat_i_1_boot
                t_stat_2_boot = np.abs(np.sum(coef_boot[1:]) - np.sum(beta_hat)) * np.sqrt(T) / s_hat_i_2_boot
                
                M_i_T_boot[b, i] = max(t_stat_1_boot, t_stat_2_boot)
        
        return M_i_T_boot
    
    def step_down_procedure(self, M_i_T: np.ndarray, M_i_T_boot: np.ndarray) -> Tuple[np.ndarray, float]:
        """
        Implement the step-down procedure for MSS estimation.
        
        Args:
            M_i_T: Array of test statistics
            M_i_T_boot: Bootstrap test statistics
            
        Returns:
            MSS_hat: Estimated MSS
            cv: Critical value
        """
        d = M_i_T.shape[0]
        MSS_hat = np.zeros(d, dtype=bool)
        
        # Initialize
        MSS_hat_prev = np.zeros(d, dtype=bool)
        j = 0
        
        while True:
            # Indices not in current MSS estimate
            non_MSS_indices = np.where(~MSS_hat_prev)[0]
            
            if len(non_MSS_indices) == 0:
                cv = 0  # All assets are in MSS
                break
            
            # Compute critical value for current non-MSS
            cv = np.percentile(np.max(M_i_T_boot[:, non_MSS_indices], axis=1), (1 - self.p) * 100)
            
            # Update MSS
            MSS_hat = M_i_T > cv
            
            # Check stopping criterion
            if np.array_equal(MSS_hat, MSS_hat_prev) or np.all(MSS_hat):
                break
            
            MSS_hat_prev = MSS_hat.copy()
            j += 1
        
        # Ensure MSS is not empty (as mentioned in the paper)
        if not np.any(MSS_hat):
            # If empty, include asset with largest M_i_T
            MSS_hat[np.argmax(M_i_T)] = True
            
        return MSS_hat, cv
    
    def fit(self, returns: np.ndarray) -> Tuple[np.ndarray, float, np.ndarray]:
        """
        Estimate the MSS from returns data.
        
        Args:
            returns: Array of asset returns with shape (T, d)
            
        Returns:
            MSS_hat: Boolean array indicating which assets are in the MSS
            cv: Critical value used
            M_i_T: Test statistics for each asset
        """
        # Compute test statistics
        M_i_T, regression_results = self.compute_test_statistics(returns)
        
        # Perform moving block bootstrap
        M_i_T_boot = self.moving_block_bootstrap(returns, M_i_T, regression_results)
        
        # Estimate MSS using step-down procedure
        MSS_hat, cv = self.step_down_procedure(M_i_T, M_i_T_boot)
        
        return MSS_hat, cv, M_i_T


def generate_var_garch_data(T: int, K: int, N: int, rho: float = 0.1) -> Tuple[np.ndarray, np.ndarray]:
    """
    Generate VAR-GARCH data as described in the paper's simulation section.
    
    Args:
        T: Number of time periods
        K: Number of assets in the true MSS
        N: Number of redundant assets
        rho: AR coefficient parameter
        
    Returns:
        returns: Generated returns
        true_MSS: Boolean array indicating the true MSS
    """
    d = K + N
    returns = np.zeros((T, d))
    
    # Initialize GARCH volatility
    d_t = np.ones(d) * np.sqrt(0.1)
    
    # Initialize AR coefficient matrix for MSS assets
    A = np.zeros((K, K))
    for i in range(K):
        for j in range(K):
            A[i, j] = rho**(abs(i-j) + 1)
    
    # Generate data
    for t in range(1, T):
        # Generate MSS asset returns
        if t == 1:
            R_K_prev = np.zeros(K)
        else:
            R_K_prev = returns[t-1, :K]
        
        # Innovation term for all assets
        v_t = np.random.normal(0, 1, d)
        eta_t = d_t * v_t
        
        # MSS assets follow VAR(1)
        returns[t, :K] = 0 + A @ R_K_prev + eta_t[:K]
        
        # Redundant assets
        B = np.ones((N, K)) / K  # Each row sums to 1
        returns[t, K:] = 0 + B @ returns[t, :K] + eta_t[K:]
        
        # Update GARCH volatility
        d_t = np.sqrt(0.1 + 0.1 * eta_t**2 + 0.8 * d_t**2)
    
    # True MSS is the first K assets
    true_MSS = np.zeros(d, dtype=bool)
    true_MSS[:K] = True
    
    return returns, true_MSS


def evaluate_mss_estimator(T_values: List[int], K_values: List[int], 
                           replications: int = 100) -> pd.DataFrame:
    """
    Evaluate the MSS estimator on simulated data with different configurations.
    
    Args:
        T_values: List of sample sizes
        K_values: List of true MSS sizes
        replications: Number of Monte Carlo replications
        
    Returns:
        results: DataFrame with evaluation results
    """
    results = []
    
    for T in T_values:
        for K in K_values:
            N = 8 - K  # Total of 8 assets
            
            contained_count = 0
            exact_count = 0
            
            for rep in range(replications):
                # Generate data
                returns, true_MSS = generate_var_garch_data(T, K, N)
                
                # Estimate MSS
                try:
                    estimator = MSSEstimator(significance_level=0.05, bootstrap_samples=200)
                    MSS_hat, _, _ = estimator.fit(returns)
                    
                    # Check if true MSS is contained in estimated MSS
                    if np.all(true_MSS[MSS_hat]):
                        contained_count += 1
                    
                    # Check if estimated MSS equals true MSS
                    if np.array_equal(MSS_hat, true_MSS):
                        exact_count += 1
                        
                except Exception as e:
                    print(f"Error in simulation T={T}, K={K}, rep={rep}: {e}")
                    continue
            
            results.append({
                'T': T,
                'K': K,
                'N': N,
                'P(MSS ⊂ MSS_hat)': contained_count / replications,
                'P(MSS = MSS_hat)': exact_count / replications
            })
    
    return pd.DataFrame(results)


def plot_mss_example(returns: np.ndarray, MSS_hat: np.ndarray, M_i_T: np.ndarray, 
                    regression_results: List[Dict], asset_names: Optional[List[str]] = None):
    """
    Create a visualization similar to Figure 1 in the paper.
    
    Args:
        returns: Asset returns
        MSS_hat: Estimated MSS
        M_i_T: Test statistics
        regression_results: Regression results
        asset_names: Optional list of asset names
    """
    T, d = returns.shape
    
    if asset_names is None:
        asset_names = [f"Asset {i+1}" for i in range(d)]
    
    # Sort assets by M_i_T
    sorted_indices = np.argsort(-M_i_T)
    sorted_assets = [asset_names[i] for i in sorted_indices]
    
    # Extract alpha and beta values
    alpha_values = np.array([regression_results[i]['alpha'] for i in sorted_indices])
    beta_sum_minus_one = np.array([np.sum(regression_results[i]['beta']) - 1 for i in sorted_indices])
    
    # Extract confidence bands
    alpha_bands = np.array([regression_results[i]['s_i_1'] * stats.norm.ppf(0.975) / np.sqrt(T) 
                            for i in sorted_indices])
    beta_bands = np.array([regression_results[i]['s_i_2'] * stats.norm.ppf(0.975) / np.sqrt(T) 
                           for i in sorted_indices])
    
    plt.figure(figsize=(12, 7))
    
    # Plot alpha values
    plt.scatter(range(d), np.abs(alpha_values), color='blue', label='|α_i|', marker='o', s=50)
    
    # Plot beta values
    plt.scatter(range(d), -np.abs(beta_sum_minus_one), color='green', label='|1^T β_i - 1|', 
                marker='s', s=50)
    
    # Plot confidence bands
    for i in range(d):
        plt.axhspan(0, alpha_bands[i], xmin=(i-0.4)/d, xmax=(i+0.4)/d, alpha=0.2, color='blue')
        plt.axhspan(-beta_bands[i], 0, xmin=(i-0.4)/d, xmax=(i+0.4)/d, alpha=0.2, color='green')
    
    # Highlight MSS
    for i in range(d):
        if MSS_hat[sorted_indices[i]]:
            if np.abs(alpha_values[i]) > alpha_bands[i]:
                plt.plot(i, np.abs(alpha_values[i]), 'ro', markersize=10)
            if np.abs(beta_sum_minus_one[i]) > beta_bands[i]:
                plt.plot(i, -np.abs(beta_sum_minus_one[i]), 'ro', markersize=10)
    
    plt.axhline(y=0, color='black', linestyle='-', alpha=0.5)
    plt.xticks(range(d), sorted_assets, rotation=45)
    plt.ylabel('Magnitude')
    plt.title('MSS Estimation Results')
    plt.legend()
    plt.tight_layout()
    plt.show()
