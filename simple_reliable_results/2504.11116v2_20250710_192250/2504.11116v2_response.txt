def generate_model_params(n_assets, k_factors, seed=42):

        eps = 1e-6 - min_eig
        Psi = Psi + eps * np.eye(n_assets)

    return ModelParams(

class PolicyNetwork(tf.keras.Model):

def simulate_path(params, policy_network, x0, y0, t0, n_steps=20):
    """Simulate a path using the current policy network"""
    dt = (params.T - t0) / n_steps
    x = x0
    y = y0
    
    # Create array to store path information
    times = np.linspace(t0, params.T, n_steps + 1)
    xs = np.zeros(n_steps + 1)
    ys = np.zeros((n_steps + 1, params.k_factors))
    
    xs[0] = x
    ys[0, :] = y.flatten()
    
    # Cholesky decomposition for correlated Brownian motions
    full_cov = np.block([
        [params.Psi * dt, params.rho * dt],
        [params.rho.T * dt, params.Phi * dt]
    ])
    L = np.linalg.cholesky(full_cov)
    
    for i in range(n_steps):
        t = times[i]
        
        # Prepare network input: [t, x, y]
        net_input = np.concatenate([[t, x], y.flatten()]).reshape(1, -1)
        net_input = tf.convert_to_tensor(net_input, dtype=tf.float32)
        
        # Get portfolio weights from the network
        pi = policy_network(net_input).numpy().flatten()
        
        # Generate correlated Brownian increments
        Z = np.random.normal(0, 1, params.n_assets + params.k_factors)
        dW = L @ Z
        dWX = dW[:params.n_assets]
        dWY = dW[params.n_assets:]
        
        # Update state variables (OU process)
        mu_Y = params.kappa_Y @ (params.theta_Y - y)
        y = y + mu_Y * dt + params.sigma_Y @ dWY.reshape(-1, 1)
        
        # Update wealth
        mu = params.r + pi @ (params.sigma * (params.alpha @ y)) - params.r
        x = x + x * (params.r + pi @ mu) * dt + x * pi @ (params.sigma * dWX.reshape(-1, 1))
        
        # Store values
        xs[i+1] = x
        ys[i+1, :] = y.flatten()
        
    return times, xs, ys


def compute_costate_estimates(params, policy_network, t, x, y, n_mc=50):
    """Compute costate estimates using backpropagation-through-time (BPTT)"""
    # Create TensorFlow variables for the state
    t_tf = tf.Variable(t, dtype=tf.float32)
    x_tf = tf.Variable(x, dtype=tf.float32)
    y_tf = tf.Variable(y, dtype=tf.float32)
    
    # Function to compute utility
    def crra_utility(x, gamma=params.gamma):
        return x**(1-gamma) / (1-gamma)
    
    # Accumulate gradients over multiple Monte Carlo paths
    lambda_sum = 0
    lambda_x_sum = 0
    lambda_y_sum = np.zeros(params.k_factors)
    
    for i in range(n_mc):
        with tf.GradientTape(persistent=True) as tape:
            # Simulate a path starting from the given state
            times, xs, ys = simulate_path(params, policy_network, x_tf.numpy(), 
                                         y_tf.numpy().reshape(-1, 1), t_tf.numpy())
            
            # Compute terminal utility
            terminal_utility = crra_utility(xs[-1])
        
        # Compute costate (sensitivity to wealth)
        lambda_val = tape.gradient(terminal_utility, x_tf)
        lambda_sum += lambda_val.numpy()
        
        # Compute derivative of costate w.r.t. wealth
        lambda_x = tape.gradient(lambda_val, x_tf)
        lambda_x_sum += lambda_x.numpy()
        
        # Compute derivative of costate w.r.t. state variables
        lambda_y = tape.gradient(lambda_val, y_tf)
        lambda_y_sum += lambda_y.numpy()
        
        del tape
    
    # Average over Monte Carlo paths
    lambda_avg = lambda_sum / n_mc
    lambda_x_avg = lambda_x_sum / n_mc
    lambda_y_avg = lambda_y_sum / n_mc
    
    return lambda_avg, lambda_x_avg, lambda_y_avg


def compute_pontryagin_control(params, lambda_val, lambda_x, lambda_y, x, y):
    """Compute control using Pontryagin's maximum principle"""
    # Compute asset covariance matrix
    Sigma = np.diag(params.sigma.flatten()) @ params.Psi @ np.diag(params.sigma.flatten())
    Sigma_inv = np.linalg.inv(Sigma)
    
    # Calculate risk premium vector
    risk_premium = params.sigma * (params.alpha @ y) 
    
    # Compute myopic component
    myopic = Sigma_inv @ risk_premium
    
    # Compute hedging component 
    hedging = Sigma_inv @ (params.sigma * (params.rho @ params.sigma_Y.T @ lambda_y.reshape(-1, 1)))
    
    # Scale by lambda ratios as per Eq. (14)-(15) in the paper
    pi_pmp = (-1/(x * lambda_x)) * (lambda_val * myopic + hedging)
    
    return pi_pmp.flatten()
