def simulate_alpha_process(days: int, intervals_per_day: int, 
                          alpha_mean_reversion: float = 5.0, 
                          alpha_vol: float = 1.0, 
                          dt: float = 1.0) -> np.ndarray:
    """Simulate the alpha process according to the Ornstein-Uhlenbeck process"""
    alpha = np.zeros(days * intervals_per_day)
    alpha[0] = 0.0
    
    for i in range(1, len(alpha)):
        dW = np.random.normal(0, np.sqrt(dt))
        alpha[i] = alpha[i-1] - alpha_mean_reversion * alpha[i-1] * dt + alpha_vol * dW
    
    return alpha


def simulate_price_process(alpha: np.ndarray, price_vol: float = 1.0, dt: float = 1.0) -> np.ndarray:
    """Simulate the price process according to dS = alpha * dt + sigma * dW"""
    price = np.zeros(len(alpha))
    price[0] = 100.0
    
    for i in range(1, len(price)):
        dW = np.random.normal(0, np.sqrt(dt))
        price[i] = price[i-1] + alpha[i-1] * dt + price_vol * dW
    
    return price


def simulate_informed_trading(alpha: np.ndarray, 
                             alpha_coefficient: float = 0.5, 
                             inventory_coefficient: float = 0.3) -> Tuple[np.ndarray, np.ndarray]:
    """Simulate informed trader's strategy: v^{I,*} = A*alpha - B*q^I"""
    volume = np.zeros(len(alpha))
    inventory = np.zeros(len(alpha))
    
    for i in range(len(alpha) - 1):
        volume[i] = alpha_coefficient * alpha[i] - inventory_coefficient * inventory[i]
        inventory[i+1] = inventory[i] + volume[i]
    
    # Last period
    volume[-1] = alpha_coefficient * alpha[-1] - inventory_coefficient * inventory[-1]
    
    return volume, inventory


def simulate_uninformed_trading(days: int, intervals_per_day: int,

def execute_broker_strategy(market_data: List[MarketData], 
                           client_data: List[ClientData], 
                           broker_data: List[BrokerData],
                           c: float, g: float, h: float, f: float,

                           a_B: float = 1.0, phi_B: float = 0.01) -> float:
    """
    Execute broker's strategy using the formula:
    V^B_t = -c*Q^B_t - g*Q^I_t + h*V^I_t + f*V^U_t
    
    Returns the performance metric
    """
    total_intervals = len(market_data)
    intervals_per_day = total_intervals // 5  # Assuming 5 days
    
    # Process data day by day
    daily_performance = []

        day_end = (day + 1) * intervals_per_day

        broker_cash = 0.0
        inventory_penalty = 0.0

            external_volume = -c * broker_inventory - g * informed_inventory + h * informed_volume + f * uninformed_volume

        # Calculate day's performance
        day_performance = broker_cash + broker_inventory * market_data[day_end-1].price - a_B * broker_inventory**2 - phi_B * inventory_penalty
        daily_performance.append(day_performance)
    
    # Return average daily performance
    return np.mean(daily_performance)


def optimize_broker_strategy(market_data: List[MarketData], 
                            client_data: List[ClientData], 
                            broker_data: List[BrokerData]) -> Tuple[float, float, float, float, float]:
    """
    Optimize the broker's strategy parameters (c, g, h, f) to maximize performance
    """
    def objective(params):
        c, g, h, f = params
        # Ensure parameters are in valid ranges
        if c < 0 or h < 0 or f < 0:
            return 1e9  # Large penalty for invalid parameters
        
        # Negative because we want to maximize
        return -execute_broker_strategy(market_data, client_data, broker_data, c, g, h, f)
    
    # Initial guess based on typical values from the paper
    initial_guess = [0.5, 0.2, 0.6, 0.4]
    
    # Bounds for parameters
    bounds = [(0.0, 1.0), (-1.0, 1.0), (0.0, 1.0), (0.0, 1.0)]
    
    # Optimize
    result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B')
    
    # Return optimized parameters and performance
    c, g, h, f = result.x
    performance = -result.fun
    
    return c, g, h, f, performance
