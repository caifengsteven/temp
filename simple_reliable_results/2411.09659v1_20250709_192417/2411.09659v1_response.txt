import numpy as np
import matplotlib.pyplot as plt
from collections import deque
import pandas as pd
from scipy.stats import norm
import random


class BlackScholesModel:
    """Simple Black-Scholes model for option pricing and stock price simulation"""
    
    def __init__(self, S0=100, r=0.03/365, sigma=0.2/np.sqrt(252), mu=0.1/252):
        self.S0 = S0
        self.r = r
        self.sigma = sigma
        self.mu = mu
    
    def simulate_path(self, days, n_paths=1, seed=None):
        """Simulate stock price paths"""
        if seed is not None:
            np.random.seed(seed)
        
        dt = 1  # 1 day
        paths = np.zeros((n_paths, days+1))
        paths[:, 0] = self.S0
        
        for t in range(1, days+1):
            z = np.random.standard_normal(n_paths)
            paths[:, t] = paths[:, t-1] * np.exp((self.mu - 0.5 * self.sigma**2) * dt + 
                                               self.sigma * np.sqrt(dt) * z)
        
        return paths
    
    def option_price(self, S, K, T, option_type='call'):
        """Calculate option price using Black-Scholes formula"""
        d1 = (np.log(S/K) + (self.r + 0.5 * self.sigma**2) * T) / (self.sigma * np.sqrt(T))
        d2 = d1 - self.sigma * np.sqrt(T)
        
        if option_type == 'call':
            price = S * norm.cdf(d1) - K * np.exp(-self.r * T) * norm.cdf(d2)
        else:  # put
            price = K * np.exp(-self.r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)
            
        return price
    
    def option_delta(self, S, K, T, option_type='call'):
        """Calculate option delta using Black-Scholes formula"""
        d1 = (np.log(S/K) + (self.r + 0.5 * self.sigma**2) * T) / (self.sigma * np.sqrt(T))
        
        if option_type == 'call':
            delta = norm.cdf(d1)
        else:  # put
            delta = norm.cdf(d1) - 1
            
        return delta


class HedgingEnvironment:
    """Environment for option hedging simulation"""
    
    def __init__(self, model, K=105, T=30, option_type='call', transaction_cost=0.0):
        self.model = model
        self.K = K
        self.T = T
        self.option_type = option_type
        self.transaction_cost = transaction_cost
        self.current_step = 0
        self.stock_price = None
        self.option_price = None
        self.cash = None
        self.stock_position = 0
        self.path = None
        
    def reset(self, path=None, init_price=None):
        """Reset environment to initial state"""
        if path is not None:
            self.path = path
            self.stock_price = path[0]
        elif init_price is not None:
            self.stock_price = init_price
            self.path = None
        else:
            # Simulate a new path if none provided
            self.path = self.model.simulate_path(self.T, n_paths=1)[0]
            self.stock_price = self.path[0]
        
        # Calculate initial option price
        time_to_maturity = self.T / 365  # Convert days to years
        self.option_price = self.model.option_price(
            self.stock_price, self.K, time_to_maturity, self.option_type)
        
        # Initialize cash as option price (we sold the option)
        self.cash = self.option_price
        self.stock_position = 0
        self.current_step = 0
        
        # State: (cash, stock_position, stock_price, time_to_maturity, option_price, K/S0)
        state = np.array([
            self.cash, 
            self.stock_position, 
            self.stock_price, 
            self.T - self.current_step,
            self.option_price,
            self.K / self.stock_price
        ])
        
        return state
    
    def step(self, action):
        """Take a step in the environment by adjusting the hedging position"""
        if self.current_step >= self.T:
            raise Exception("Episode already done")
        
        # Bound action between -1 and 1 (for put and call options)
        action = max(min(action, 1), -1)
        
        # Calculate transaction cost
        transaction_cost = abs(action - self.stock_position) * self.stock_price * self.transaction_cost
        
        # Update cash: pay for stock and transaction cost
        self.cash = self.cash - (action - self.stock_position) * self.stock_price - transaction_cost
        
        # Update stock position
        self.stock_position = action
        
        # Move to next step
        self.current_step += 1
        
        # Get new stock price
        if self.path is not None:
            self.stock_price = self.path[self.current_step]
        else:
            # Simple random walk if no path provided
            self.stock_price = self.stock_price * np.exp(
                (self.model.mu - 0.5 * self.model.sigma**2) + 
                self.model.sigma * np.random.normal())
        
        # Calculate new option price
        time_to_maturity = max(0, (self.T - self.current_step) / 365)
        if time_to_maturity > 0:
            self.option_price = self.model.option_price(
                self.stock_price, self.K, time_to_maturity, self.option_type)
        else:
            # At maturity, option price is intrinsic value
            if self.option_type == 'call':
                self.option_price = max(0, self.stock_price - self.K)
            else:
                self.option_price = max(0, self.K - self.stock_price)
        
        # Calculate portfolio value
        portfolio_value = self.cash + self.stock_position * self.stock_price - self.option_price
        
        # Intermediate reward: penalize negative portfolio value
        reward = -abs(portfolio_value) if portfolio_value < 0 else 0
        
        # Check if done
        done = self.current_step >= self.T
        
        # If done, calculate final P&L
        if done:
            if self.option_type == 'call':
                option_payoff = max(0, self.stock_price - self.K)
            else:
                option_payoff = max(0, self.K - self.stock_price)
            
            # Final portfolio value
            final_pnl = self.cash + self.stock_position * self.stock_price - option_payoff
            reward = final_pnl
        
        # State: (cash, stock_position, stock_price, time_to_maturity, option_price, K/S0)
        state = np.array([
            self.cash, 
            self.stock_position, 
            self.stock_price, 
            self.T - self.current_step,
            self.option_price,
            self.K / self.stock_price
        ])
        
        return state, reward, done, {"pnl": portfolio_value}


class SimpleDeltaHedger:
    """Simple delta hedging strategy using Black-Scholes delta"""
    
    def __init__(self, model, option_type='call'):
        self.model = model
        self.option_type = option_type
    
    def get_action(self, state):
        """Return delta hedging action based on current state"""
        # Extract state variables
        _, _, stock_price, time_to_maturity, _, k_s0 = state
        strike = stock_price * k_s0
        
        # Convert time_to_maturity from days to years
        time_to_maturity = max(time_to_maturity / 365, 1e-10)
        
        # Calculate delta
        delta = self.model.option_delta(stock_price, strike, time_to_maturity, self.option_type)
        
        return delta


class SimpleRLHedger:
    """Simple RL policy that mimics a contract-unified hedging strategy"""
    
    def __init__(self, option_type='call'):
        self.option_type = option_type
    
    def get_action(self, state):
        """Return hedging action based on current state"""
        # Extract state variables
        _, _, stock_price, time_to_maturity, option_price, k_s0 = state
        
        # Simple policy that approximates a learned policy
        # For call options, delta is between 0 and 1
        # For put options, delta is between -1 and 0
        
        moneyness = k_s0
        normalized_time = min(time_to_maturity / 30, 1.0)  # Normalize to [0,1]
        
        if self.option_type == 'call':
            # More aggressive than BS delta for ITM options, more conservative for OTM
            if moneyness < 1.0:  # ITM
                action = 0.5 + 0.5 * (1 - moneyness) * normalized_time
            else:  # OTM
                action = 0.5 * np.exp(-(moneyness - 1.0)) * normalized_time
        else:  # put
            # Reversed logic for puts
            if moneyness > 1.0:  # ITM
                action = -0.5 - 0.5 * (moneyness - 1.0) * normalized_time
            else:  # OTM
                action = -0.5 * np.exp(-(1.0 - moneyness)) * normalized_time
        
        return action


def calculate_var_cvar(returns, alpha=0.95):
    """Calculate VaR and CVaR at given confidence level"""
    sorted_returns = np.sort(returns)
    index = int(np.floor((1-alpha) * len(sorted_returns)))
    var = -sorted_returns[index]
    cvar = -np.mean(sorted_returns[:index+1])
    return var, cvar


def run_hedging_simulation(env, agent, n_episodes=1000, seed=None):
    """Run hedging simulation for multiple episodes"""
    if seed is not None:
        np.random.seed(seed)
        random.seed(seed)
    
    final_pnls = []
    
    for i in range(n_episodes):
        state = env.reset()
        done = False
        
        while not done:
            action = agent.get_action(state)
            state, reward, done, info = env.step(action)
            
            if done:
                final_pnls.append(reward)
    
    return np.array(final_pnls)


def test_multiple_strikes_maturities(model, agent_type, option_type='call', 
                                     transaction_cost=0.0, n_episodes=100, seed=42):
    """Test hedging strategy across multiple strikes and maturities"""
    strikes = [90, 95, 100, 105, 110]
    maturities = [10, 20, 30, 60]
    
    results = []
    
    for K in strikes:
        for T in maturities:
            env = HedgingEnvironment(model, K=K, T=T, option_type=option_type, 
                                    transaction_cost=transaction_cost)
            
            if agent_type == 'delta':
                agent = SimpleDeltaHedger(model, option_type)
            else:  # 'rl'
                agent = SimpleRLHedger(option_type)
            
            final_pnls = run_hedging_simulation(env, agent, n_episodes=n_episodes, seed=seed)
            
            mean_pnl = np.mean(final_pnls)
            std_pnl = np.std(final_pnls)
            var_95, cvar_95 = calculate_var_cvar(final_pnls, alpha=0.95)
            var_975, cvar_975 = calculate_var_cvar(final_pnls, alpha=0.975)
            
            results.append({
                'Strike': K,
                'Maturity': T,
                'Mean': mean_pnl,
                'Std': std_pnl,
                'VaR_95': var_95,
                'CVaR_95': cvar_95,
                'VaR_975': var_975,
                'CVaR_975': cvar_975
            })
    
    return pd.DataFrame(results)
