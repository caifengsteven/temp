import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from arch import arch_model
from statsmodels.tsa.statespace.sarimax import SARIMAX
from scipy.stats import norm
import warnings
warnings.filterwarnings('ignore')

class AgriculturalPriceModel:
    def __init__(self, crop_name, state_name):
        """Initialize the model with crop and state information"""
        self.crop_name = crop_name
        self.state_name = state_name
        self.prices = None
        self.msp = None
        self.log_returns = None
        self.conditional_volatility = None
        self.climate_data = None
        self.volatility_forecast = None
    
    def generate_simulated_data(self, start_date='2001-10-01', end_date='2024-12-31', 
                               random_seed=42):
        """Generate simulated price and climate data"""
        np.random.seed(random_seed)
        
        # Create date range
        date_range = pd.date_range(start=start_date, end=end_date, freq='MS')
        n = len(date_range)
        
        # Generate price data with trend, seasonality, and noise
        trend = np.linspace(0, 1, n)
        seasonality = 0.1 * np.sin(np.linspace(0, 24*np.pi, n))
        
        # Add some price shocks for realism
        noise = np.random.normal(0, 0.05, n)
        price_series = 1000 + 500 * trend + 100 * seasonality + 100 * np.cumsum(noise)
        
        # Generate MSP data (typically increasing over time but below market price)
        msp_series = 800 + 400 * trend
        
        # Generate temperature data with seasonality and warming trend
        tmax_historical = 30 + 5 * np.sin(np.linspace(0, 24*np.pi, n)) + 0.5 * trend + np.random.normal(0, 1, n)
        
        # Generate precipitation data with seasonality
        precip_historical = 100 + 80 * np.sin(np.linspace(0, 24*np.pi, n)) + np.random.exponential(20, n)
        precip_historical = np.maximum(0, precip_historical)  # No negative precipitation
        
        # Store as dataframes
        self.prices = pd.DataFrame({
            'date': date_range,
            'price': price_series
        }).set_index('date')
        
        self.msp = pd.DataFrame({
            'date': date_range,
            'msp': msp_series
        }).set_index('date')
        
        self.climate_data = pd.DataFrame({
            'date': date_range,
            'tmax': tmax_historical,
            'precipitation': precip_historical
        }).set_index('date')
        
        # Generate climate projections for future
        future_dates = pd.date_range(start='2025-01-01', end='2100-12-31', freq='MS')
        n_future = len(future_dates)
        future_trend = np.linspace(1, 3, n_future)
        
        # SSP2-4.5 (moderate) scenario
        tmax_ssp245 = 30 + 5 * np.sin(np.linspace(0, 76*np.pi, n_future)) + 1.0 * future_trend + np.random.normal(0, 1, n_future)
        precip_ssp245 = 100 + 80 * np.sin(np.linspace(0, 76*np.pi, n_future)) + np.random.exponential(20, n_future)
        precip_ssp245 = np.maximum(0, precip_ssp245)
        
        # SSP5-8.5 (severe) scenario
        tmax_ssp585 = 30 + 5 * np.sin(np.linspace(0, 76*np.pi, n_future)) + 2.0 * future_trend + np.random.normal(0, 1, n_future)
        precip_ssp585 = 100 + 80 * np.sin(np.linspace(0, 76*np.pi, n_future)) + 10 * np.random.exponential(20, n_future)
        precip_ssp585 = np.maximum(0, precip_ssp585)
        
        self.climate_projections = {
            'ssp245': pd.DataFrame({
                'date': future_dates,
                'tmax': tmax_ssp245,
                'precipitation': precip_ssp245
            }).set_index('date'),
            'ssp585': pd.DataFrame({
                'date': future_dates,
                'tmax': tmax_ssp585,
                'precipitation': precip_ssp585
            }).set_index('date')
        }
        
        print(f"Generated simulated data for {self.crop_name} in {self.state_name}")
        return self.prices, self.msp, self.climate_data
    
    def calculate_log_returns(self):
        """Calculate log returns from price data"""
        self.log_returns = np.log(self.prices / self.prices.shift(1)).dropna()
        return self.log_returns
    
    def estimate_conditional_volatility(self):
        """Estimate conditional volatility using EGARCH model"""
        if self.log_returns is None:
            self.calculate_log_returns()
            
        # Fit EGARCH(1,1,1) model
        model = arch_model(self.log_returns, p=1, o=1, q=1, vol='EGARCH')
        results = model.fit(disp='off')
        
        # Extract conditional volatility
        self.conditional_volatility = pd.DataFrame(
            results.conditional_volatility,
            index=self.log_returns.index,
            columns=['volatility']
        )
        
        print(f"Estimated conditional volatility for {self.crop_name}")
        return self.conditional_volatility
    
    def fit_sarimax_model(self, train_end_date='2020-12-31'):
        """Fit SARIMAX model with climate variables as exogenous regressors"""
        if self.conditional_volatility is None:
            self.estimate_conditional_volatility()
            
        # Prepare the data
        merged_data = self.conditional_volatility.join(self.climate_data)
        train_data = merged_data.loc[:train_end_date]
        
        # Fit SARIMAX model
        exog_vars = train_data[['tmax', 'precipitation']]
        model = SARIMAX(train_data['volatility'], 
                        exog=exog_vars,
                        order=(1, 0, 1),
                        seasonal_order=(1, 0, 1, 12))
        
        self.sarimax_results = model.fit(disp=False)
        print(f"Fitted SARIMAX model for {self.crop_name}")
        
        # In-sample prediction
        self.volatility_pred = self.sarimax_results.predict(
            exog=merged_data[['tmax', 'precipitation']]
        )
        
        return self.sarimax_results
    
    def forecast_volatility(self, scenario='ssp245'):
        """Forecast volatility using climate projections"""
        if not hasattr(self, 'sarimax_results'):
            self.fit_sarimax_model()
            
        # Forecast using climate projections
        climate_projection = self.climate_projections[scenario]
        exog_future = climate_projection[['tmax', 'precipitation']]
        
        # Generate predictions
        forecast = self.sarimax_results.get_forecast(
            steps=len(exog_future),
            exog=exog_future
        )
        
        forecast_mean = forecast.predicted_mean
        forecast_ci = forecast.conf_int(alpha=0.32)  # ~68% CI
        
        self.volatility_forecast = pd.DataFrame({
            'mean': forecast_mean,
            'lower': forecast_ci.iloc[:, 0],
            'upper': forecast_ci.iloc[:, 1]
        }, index=climate_projection.index)
        
        print(f"Forecasted volatility for {self.crop_name} under {scenario} scenario")
        return self.volatility_forecast
    
    def calculate_option_premium(self, risk_free_rate=0.05, maturity=1.0, scenario='ssp245'):
        """Calculate insurance premium using Black-Scholes model"""
        if self.volatility_forecast is None:
            self.forecast_volatility(scenario)
            
        # Extend prices and MSP to match the forecast period
        last_price = self.prices.iloc[-1, 0]
        last_msp = self.msp.iloc[-1, 0]
        
        # Calculate Black-Scholes put option premium
        premiums = []
        
        for date, row in self.volatility_forecast.iterrows():
            volatility = row['mean']
            S = last_price  # Current price (simplified)
            K = last_msp    # Strike price (MSP)
            r = risk_free_rate
            T = maturity
            
            # Black-Scholes formula for put option
            d1 = (np.log(S/K) + (r + 0.5 * volatility**2) * T) / (volatility * np.sqrt(T))
            d2 = d1 - volatility * np.sqrt(T)
            
            put_premium = K * np.exp(-r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)
            premiums.append(put_premium)
        
        self.option_premiums = pd.DataFrame({
            'premium': premiums
        }, index=self.volatility_forecast.index)
        
        print(f"Calculated option premiums for {self.crop_name} under {scenario} scenario")
        return self.option_premiums
    
    def visualize_results(self, scenario='ssp245'):
        """Visualize key results"""
        if not hasattr(self, 'option_premiums'):
            self.calculate_option_premium(scenario=scenario)
            
        # Create a figure with subplots
        fig, axes = plt.subplots(3, 1, figsize=(12, 15))
        
        # Plot 1: Price and MSP
        self.prices.plot(ax=axes[0], label='Price')
        self.msp.plot(ax=axes[0], color='red', label='MSP')
        axes[0].set_title(f'{self.crop_name} in {self.state_name} - Price and MSP')
        axes[0].legend()
        
        # Plot 2: Conditional Volatility
        self.conditional_volatility.plot(ax=axes[1], label='EGARCH Volatility')
        if hasattr(self, 'volatility_pred'):
            pd.DataFrame(self.volatility_pred).plot(ax=axes[1], color='green', label='SARIMAX Predicted')
        axes[1].set_title('Conditional Volatility')
        axes[1].legend()
        
        # Plot 3: Option Premiums
        self.option_premiums.plot(ax=axes[2])
        axes[2].set_title(f'Insurance Premium (Black-Scholes) - {scenario} scenario')
        
        plt.tight_layout()
        plt.savefig(f"{self.crop_name}_{self.state_name}_{scenario}_analysis.png")
        plt.close()
        
        print(f"Visualizations saved for {self.crop_name} in {self.state_name}")
