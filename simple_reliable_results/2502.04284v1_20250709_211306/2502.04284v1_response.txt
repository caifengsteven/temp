import numpy as np
import matplotlib.pyplot as plt
from typing import Tuple, List, Dict, Callable


class TradingSimulator:
    def __init__(self, rho0: float, rho1: float, transaction_cost: float):
        """
        Initialize the trading simulator.
        
        Args:
            rho0: Correlation coefficient for current signal
            rho1: Correlation coefficient for previous signal
            transaction_cost: Fixed cost for switching positions
        """
        self.rho0 = rho0
        self.rho1 = rho1
        self.transaction_cost = transaction_cost
        
        # Validation
        if not (0 < rho0 < 1 and 0 < rho1 < 1):
            raise ValueError("Correlation coefficients must be between 0 and 1")
        if transaction_cost < 0:
            raise ValueError("Transaction cost must be non-negative")
            
    def generate_signals(self, n_periods: int) -> Tuple[np.ndarray, np.ndarray]:
        """
        Generate random signals and corresponding price movements.
        
        Args:
            n_periods: Number of time periods to simulate
            
        Returns:
            Tuple of (signals, price_movements)
        """
        # Generate random signals
        signals = np.random.normal(0, 1, n_periods)
        
        # Generate noise
        noise = np.random.normal(0, 1, n_periods)
        
        # Generate price movements based on current and previous signals
        price_movements = np.zeros(n_periods)
        price_movements[0] = self.rho0 * signals[0] + noise[0]
        
        for t in range(1, n_periods):
            price_movements[t] = self.rho0 * signals[t] + self.rho1 * signals[t-1] + noise[t]
            
        return signals, price_movements
    
    def naive_policy(self, x0: float, x1: float, current_position: int) -> int:
        """
        Implement the naive trading policy.
        
        Args:
            x0: Current signal
            x1: Previous signal
            current_position: Current position (1 for long, -1 for short)
            
        Returns:
            New position (1 for long, -1 for short)
        """
        # Calculate boundary for the no-trade zone
        g_naive = -self.rho0 / self.rho1 * x0
        
        if current_position == 1:
            # Currently long
            g_naive -= self.transaction_cost / (2 * self.rho1)
            if x1 < g_naive:
                return -1  # Switch to short
            return 1  # Stay long
        else:
            # Currently short
            g_naive += self.transaction_cost / (2 * self.rho1)
            if x1 > g_naive:
                return 1  # Switch to long
            return -1  # Stay short
    
    def optimal_policy_approximation(self, x0: float, x1: float, current_position: int) -> int:
        """
        Implement the first-order approximation of the optimal trading policy.
        
        Args:
            x0: Current signal
            x1: Previous signal
            current_position: Current position (1 for long, -1 for short)
            
        Returns:
            New position (1 for long, -1 for short)
        """
        # Calculate first-order approximation of optimal policy
        g_optimal = -self.rho0 / self.rho1 * x0
        
        # First-order correction term (simplified version based on paper's formula)
        correction = self.transaction_cost * (
            -1/(2*self.rho1) + 
            (self.rho1/self.rho0)**2 * np.exp(-0.5 * (self.rho1/self.rho0 * x0)**2) / np.sqrt(2*np.pi)
        )
        
        if current_position == 1:
            # Currently long
            g_optimal += correction
            if x1 < g_optimal:
                return -1  # Switch to short
            return 1  # Stay long
        else:
            # Currently short
            g_optimal -= correction
            if x1 > g_optimal:
                return 1  # Switch to long
            return -1  # Stay short
    
    def simulate_strategy(self, policy: Callable, n_periods: int = 1000) -> Dict:
        """
        Simulate a trading strategy.
        
        Args:
            policy: Trading policy function
            n_periods: Number of time periods to simulate
            
        Returns:
            Dict with performance metrics
        """
        signals, price_movements = self.generate_signals(n_periods)
        
        # Initialize tracking variables
        position = 1  # Start long
        gross_pnl = 0
        transaction_costs = 0
        position_history = [position]
        
        for t in range(1, n_periods):
            x0 = signals[t]
            x1 = signals[t-1]
            
            # Determine new position
            new_position = policy(x0, x1, position)
            
            # Calculate PnL
            pnl = position * price_movements[t]
            gross_pnl += pnl
            
            # Apply transaction costs if position changed
            if new_position != position:
                transaction_costs += self.transaction_cost
            
            # Update position
            position = new_position
            position_history.append(position)
        
        # Calculate final metrics
        net_pnl = gross_pnl - transaction_costs
        trade_count = sum(1 for i in range(1, len(position_history)) 
                          if position_history[i] != position_history[i-1])
        
        return {
            "gross_pnl": gross_pnl,
            "transaction_costs": transaction_costs,
            "net_pnl": net_pnl,
            "trade_count": trade_count,
            "position_history": position_history
        }

        # Print results
        print("=== Average Results over", n_simulations, "simulations ===")
        print(f"Params: rho0={self.rho0}, rho1={self.rho1}, cost={self.transaction_cost}")

    def visualize_no_trade_zone(self, x_range: List[float] = [-2, 2]) -> None:
        """
        Visualize the no-trade zone for both policies.
        
        Args:
            x_range: Range of x0 values to plot

        x0_values = np.linspace(x_range[0], x_range[1], 100)
        
        # Calculate boundaries for naive policy
        naive_boundary_long = [-self.rho0/self.rho1 * x - self.transaction_cost/(2*self.rho1) for x in x0_values]

        plt.xlabel('Current Signal (x0)')
        plt.ylabel('Previous Signal (x1)')
        plt.title(f'No-Trade Zones (ρ0={self.rho0}, ρ1={self.rho1}, c={self.transaction_cost})')
