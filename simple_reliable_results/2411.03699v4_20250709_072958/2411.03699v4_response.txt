import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

class TreasuryVIXModel:
    """
    Multivariate autoregressive stochastic volatility model for Treasury zero-coupon rates
    using VIX as observed volatility.
    """
    def __init__(self, num_components=3, stochastic_volatility_components=1):
        """
        Initialize model parameters
        
        Parameters:
        -----------
        num_components : int
            Number of principal components to model (typically 1-3)
        stochastic_volatility_components : int
            Number of components affected by stochastic volatility
        """
        self.d = num_components  # dimension (number of PCs)
        self.p = stochastic_volatility_components  # number of components with stochastic volatility
        
        # Model parameters to be estimated
        self.a = None  # constant vector
        self.B = None  # autoregression matrix
        self.c = None  # VIX coefficient vector
        
        # VIX model parameters
        self.alpha = None  # constant in log(VIX) model
        self.beta = None  # autoregression coefficient in log(VIX) model
        
        # Fitted values
        self.fitted_pcs = None
        self.fitted_vix = None
        
    def fit(self, pc_data, vix_data):
        """
        Fit the model using PC data and VIX data
        
        Parameters:
        -----------
        pc_data : DataFrame
            Principal components time series data
        vix_data : Series
            VIX time series data
        """
        # Fit VIX model: ln(V(t)) = alpha + beta*ln(V(t-1)) + Z0(t)
        log_vix = np.log(vix_data)
        X_vix = pd.DataFrame({'const': 1, 'log_vix_lag': log_vix.shift(1)}).dropna()
        y_vix = log_vix.iloc[1:]
        
        # OLS for VIX model
        beta_vix = np.linalg.inv(X_vix.T @ X_vix) @ X_vix.T @ y_vix
        self.alpha = beta_vix.iloc[0]
        self.beta = beta_vix.iloc[1]
        
        # Initialize arrays for PC model
        T = len(pc_data) - 1
        self.a = np.zeros(self.d)
        self.B = np.eye(self.d)
        self.c = np.zeros(self.d)
        
        # Fit each PC equation
        for i in range(self.d):
            # For stochastic volatility components, normalize by VIX
            if i < self.p:
                # X(t) = a + B*X(t-1) + c*V(t) + V(t)*Z(t)
                # Divide by V(t): X(t)/V(t) = a/V(t) + B*X(t-1)/V(t) + c + Z(t)
                X = pd.DataFrame({
                    'const': 1/vix_data.iloc[:-1].values,
                    'pc_lag': pc_data.iloc[:-1, i].values/vix_data.iloc[:-1].values,
                    'vix': np.ones(T)
                })
                y = pc_data.iloc[1:, i].values / vix_data.iloc[1:].values
            else:
                # Regular autoregression for non-stochastic volatility components
                # X(t) = a + B*X(t-1) + c*V(t) + Z(t)
                X = pd.DataFrame({
                    'const': np.ones(T),
                    'pc_lag': pc_data.iloc[:-1, i].values,
                    'vix': vix_data.iloc[:-1].values
                })
                y = pc_data.iloc[1:, i].values
            
            # OLS for PC model
            beta = np.linalg.inv(X.T @ X) @ X.T @ y
            self.a[i] = beta[0]
            self.B[i, i] = beta[1]  # Only diagonal elements for simplicity
            self.c[i] = beta[2]
        
        return self
    
    def simulate(self, n_steps, initial_pcs, initial_vix):
        """
        Simulate the model forward
        
        Parameters:
        -----------
        n_steps : int
            Number of time steps to simulate
        initial_pcs : array-like
            Initial values for principal components
        initial_vix : float
            Initial value for VIX
            
        Returns:
        --------
        simulated_pcs : array
            Simulated principal component values
        simulated_vix : array
            Simulated VIX values
        """
        pcs = np.zeros((n_steps, self.d))
        vix = np.zeros(n_steps)
        
        # Set initial values
        pcs[0] = initial_pcs
        vix[0] = initial_vix
        
        # Generate innovations
        Z0 = np.random.normal(0, 0.1, n_steps)  # VIX innovations
        Z = np.random.normal(0, 0.1, (n_steps, self.d))  # PC innovations
        
        # Simulate
        for t in range(1, n_steps):
            # Simulate VIX
            log_vix_t = self.alpha + self.beta * np.log(vix[t-1]) + Z0[t]
            vix[t] = np.exp(log_vix_t)
            
            # Simulate PCs
            xi_t = np.diag([vix[t] if i < self.p else 1 for i in range(self.d)])
            pcs[t] = self.a + self.B @ pcs[t-1] + self.c * vix[t] + xi_t @ Z[t]
            
        return pcs, vix
    
    def compute_bond_returns(self, pcs, maturities):
        """
        Compute zero-coupon bond returns based on principal components
        
        Parameters:
        -----------
        pcs : array-like
            Principal component values
        maturities : list
            Bond maturities in years
            
        Returns:
        --------
        returns : DataFrame
            Bond returns for each maturity
        """
        n_steps = len(pcs)
        returns = np.zeros((n_steps-1, len(maturities)))
        
        # Simple linear loading factors for demonstration
        # In real application, these would be estimated from data
        gamma = np.zeros((self.d, max(maturities)+1))
        
        # Level factor - almost constant across maturities
        gamma[0] = np.ones(max(maturities)+1)
        
        # Slope factor - decreasing with maturity
        if self.d > 1:
            gamma[1] = np.linspace(1.5, 0.5, max(maturities)+1)
        
        # Curvature factor - U-shaped
        if self.d > 2:
            x = np.linspace(0, max(maturities), max(maturities)+1)
            gamma[2] = 0.5 * ((x - max(maturities)/2)**2) / (max(maturities)/2)**2
        
        # Compute rates for each maturity
        rates = np.zeros((n_steps, max(maturities)+1))
        for t in range(n_steps):
            for l in range(1, max(maturities)+1):
                rates[t, l] = sum(gamma[i, l] * pcs[t, i] for i in range(self.d))
        
        # Compute returns using the approximation from the paper
        for t in range(1, n_steps):
            for i, l in enumerate(maturities):
                returns[t-1, i] = -1 * (l-1) * rates[t, l-1] + l * rates[t-1, l]
        
        return returns
