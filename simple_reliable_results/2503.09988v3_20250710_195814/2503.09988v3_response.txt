class MLPModel(nn.Module):
    def __init__(self, input_dim, hidden_dims, output_dim):
        super(MLPModel, self).__init__()
        layers = []
        prev_dim = input_dim
        
        for dim in hidden_dims:
            layers.append(nn.Linear(prev_dim, dim))
            layers.append(nn.LeakyReLU(negative_slope=0.01))
            prev_dim = dim
            
        layers.append(nn.Linear(prev_dim, output_dim))
        
        self.model = nn.Sequential(*layers)
    
    def forward(self, x):
        return self.model(x)

class LSTMModel(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim, num_layers=1):
        super(LSTMModel, self).__init__()
        self.hidden_dim = hidden_dim
        self.num_layers = num_layers
        
        self.lstm = nn.LSTM(input_dim, hidden_dim, num_layers, batch_first=True)
        self.fc = nn.Linear(hidden_dim, output_dim)
        
    def forward(self, x):
        # Reshape input to (batch_size, seq_length, input_dim)
        # For our case, each sample has 60 time steps with 13 features
        batch_size = x.size(0)
        x = x.view(batch_size, 60, -1)
        
        # Initialize hidden state
        h0 = torch.zeros(self.num_layers, batch_size, self.hidden_dim).to(x.device)
        c0 = torch.zeros(self.num_layers, batch_size, self.hidden_dim).to(x.device)
        
        # Forward propagate LSTM
        out, _ = self.lstm(x, (h0, c0))
        
        # We take the output of the last time step
        out = self.fc(out[:, -1, :])
        return out

def simulate_hft_data(num_samples=10000, seq_length=60, num_features=13):
    """
    Simulate high-frequency trading data based on the paper.
    
    Returns:
        X: Features of shape (num_samples, seq_length * num_features)
        y: Labels of shape (num_samples,)
    """
    # Create features
    X = np.random.randn(num_samples, seq_length, num_features)
    
    # Create some patterns in the data
    for i in range(num_samples):
        # Add some trend patterns
        if random.random() > 0.5:
            trend = np.linspace(0, random.random() * 2 - 1, seq_length)
            X[i, :, 0] += trend  # Add trend to mid price
        
        # Add some volatility patterns
        if random.random() > 0.7:
            vol = np.abs(np.random.randn(seq_length) * 0.5)
            X[i, :, 1] += vol  # Add volatility to price std
        
        # Add some volume patterns
        if random.random() > 0.6:
            vol_pattern = np.random.exponential(1, seq_length)
            X[i, :, 12] += vol_pattern  # Add to log volume
    
    # Reshape to (num_samples, seq_length * num_features)
    X_flat = X.reshape(num_samples, -1)
    
    # Create imbalanced labels: -1, 0, 1 with ratio 1:8:1
    y = np.zeros(num_samples)
    
    # First assign all samples to class 0 (80% of data)
    y[:] = 0
    
    # Assign 10% to class -1
    neg_indices = np.random.choice(num_samples, size=int(num_samples * 0.1), replace=False)
    y[neg_indices] = -1
    
    # Assign 10% to class 1, avoiding overlap with class -1
    remaining_indices = np.setdiff1d(np.arange(num_samples), neg_indices)
    pos_indices = np.random.choice(remaining_indices, size=int(num_samples * 0.1), replace=False)
    y[pos_indices] = 1
    
    return X_flat, y

def normalize_sample(X, seq_length=60, num_features=13):
    """
    Normalize each sample according to the paper's method
    """
    X_normalized = X.copy()
    X_reshaped = X.reshape(-1, seq_length, num_features)
    
    for i in range(X_reshaped.shape[0]):
        for j in range(num_features):
            # Extract the j-th feature for the i-th sample
            feature = X_reshaped[i, :, j]
            
            # Calculate mean and std
            mu = np.mean(feature)
            sigma = np.std(feature)
            
            # Avoid division by zero
            if sigma == 0:
                sigma = 1e-6
                
            # Normalize
            X_reshaped[i, :, j] = (feature - mu) / sigma
    
    # Reshape back to original shape
    X_normalized = X_reshaped.reshape(X.shape)
    
    return X_normalized

def under_sampling(X_train, y_train):
    """
    Implement under-sampling for class 0 to balance the dataset
    """
    # Get indices for each class
    indices_neg = np.where(y_train == -1)[0]
    indices_zero = np.where(y_train == 0)[0]
    indices_pos = np.where(y_train == 1)[0]
    
    # Random sample from the majority class (0)
    # Keep the same number as the sum of the minority classes
    target_size = len(indices_neg) + len(indices_pos)
    sampled_indices_zero = np.random.choice(indices_zero, size=target_size, replace=False)
    
    # Combine indices
    balanced_indices = np.concatenate([indices_neg, sampled_indices_zero, indices_pos])
    np.random.shuffle(balanced_indices)
    
    return X_train[balanced_indices], y_train[balanced_indices]

def weighted_loss(outputs, targets, weights=None):
    """
    Implement weighted cross-entropy loss to handle class imbalance
    """
    if weights is None:
        # Default weights based on class distribution (1:8:1)
        weights = torch.tensor([8.0, 1.0, 8.0], dtype=torch.float32).to(outputs.device)

    targets_idx = (targets + 1).long()
    
    # Cross entropy loss with weight
    criterion = nn.CrossEntropyLoss(weight=weights)
    return criterion(outputs, targets_idx)

def sensitive_loss(outputs, targets, class_counts=None):
    """
    Implement sensitive loss based on equation 10 in the paper
    """
    if class_counts is None:
        # Default counts based on class distribution (1:8:1)
        class_counts = {'neg': 1000, 'zero': 8000, 'pos': 1000}
    
    total_samples = sum(class_counts.values())
    C = 3  # Number of classes
    
    # Convert targets to class indices (0, 1, 2) from (-1, 0, 1)
    targets_idx = (targets + 1).long()
    
    # Calculate weights for each class
    weights = torch.zeros(C, dtype=torch.float32).to(outputs.device)
    class_list = [class_counts['neg'], class_counts['zero'], class_counts['pos']]
    
    for c in range(C):
        N_neg_c = total_samples - class_list[c]
        weights[c] = N_neg_c / ((C - 1) * total_samples)
    
    # Cross entropy loss with calculated weights
    criterion = nn.CrossEntropyLoss(weight=weights)
    return criterion(outputs, targets_idx)

def focal_loss(outputs, targets, gamma=2.0):
    """
    Implement focal loss based on equation 11 in the paper
    """
    # Convert targets to class indices (0, 1, 2) from (-1, 0, 1)
    targets_idx = (targets + 1).long()
    
    # Standard cross entropy
    ce_loss = nn.CrossEntropyLoss(reduction='none')(outputs, targets_idx)
    
    # Get probabilities for the correct class
    pt = torch.exp(-ce_loss)
    
    # Calculate focal loss
    focal_loss = (1 - pt) ** gamma * ce_loss
    
    return focal_loss.mean()
