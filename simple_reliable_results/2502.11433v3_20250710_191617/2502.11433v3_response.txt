@dataclass
class State:
    """Represents the current state of the trading environment"""
    price_history: np.ndarray  # Historical prices
    cash: float                # Available cash
    assets: float              # Number of shares held
    total_value: float         # Total portfolio value
    rewards_memory: List[float]  # Recent rewards
    asset_memory: List[float]    # Recent portfolio values
    action_memory: List[int]     # Recent actions

@dataclass
class Action:
    """Trading action"""
    action_type: int  # -1: Sell, 0: Hold, 1: Buy


class TradingEnvironment:
    """Simulated trading environment"""
    
    def __init__(self, 
                 price_data: np.ndarray, 
                 initial_cash: float = 10000.0,
                 window_size: int = 10):
        self.price_data = price_data
        self.initial_cash = initial_cash
        self.window_size = window_size
        self.current_step = self.window_size
        self.cash = initial_cash
        self.assets = 0.0
        self.rewards_memory = []
        self.asset_memory = [initial_cash]
        self.action_memory = []
        self.done = False
        
    def reset(self) -> State:
        """Reset the environment to initial state"""
        self.current_step = self.window_size
        self.cash = self.initial_cash
        self.assets = 0.0
        self.rewards_memory = []
        self.asset_memory = [self.initial_cash]
        self.action_memory = []
        self.done = False
        return self._get_state()
        
    def step(self, action: Action) -> Tuple[State, float, bool, Dict]:
        """Execute action and return new state, reward, done flag, and info"""
        if self.done:
            return self._get_state(), 0, True, {}
            
        current_price = self.price_data[self.current_step]
        prev_total_value = self.cash + self.assets * self.price_data[self.current_step-1]
        
        # Execute trade
        if action.action_type == -1:  # Sell
            self.cash += self.assets * current_price
            self.assets = 0
        elif action.action_type == 1:  # Buy
            shares_to_buy = self.cash / current_price
            self.assets += shares_to_buy
            self.cash = 0
        # else: Hold (no change)
        
        # Calculate new portfolio value
        new_total_value = self.cash + self.assets * current_price
        
        # Calculate PnL
        pnl = new_total_value - prev_total_value
        
        # Update memories
        self.asset_memory.append(new_total_value)
        self.action_memory.append(action.action_type)
        
        # Calculate reward (using Sharpe ratio approximation)
        if len(self.asset_memory) > 1:
            daily_returns = [(self.asset_memory[i] / self.asset_memory[i-1]) - 1 
                             for i in range(1, len(self.asset_memory))]
            if len(daily_returns) > 1 and np.std(daily_returns) > 0:
                sharpe = np.mean(daily_returns) / np.std(daily_returns)
                reward = sharpe
            else:
                reward = 0
        else:
            reward = 0
            
        self.rewards_memory.append(reward)
        
        # Move to next step
        self.current_step += 1
        self.done = self.current_step >= len(self.price_data) - 1
        
        return self._get_state(), reward, self.done, {"pnl": pnl}
    
    def _get_state(self) -> State:
        """Get current state of the environment"""
        observation_window = self.price_data[self.current_step-self.window_size:self.current_step]
        total_value = self.cash + self.assets * self.price_data[self.current_step-1]
        
        return State(
            price_history=observation_window,
            cash=self.cash,
            assets=self.assets,
            total_value=total_value,

            asset_memory=self.asset_memory[-10:] if len(self.asset_memory) > 0 else [],

        )

class FLAGTrader:
    """Implementation of FLAG-TRADER framework"""
    
    def __init__(self, env: TradingEnvironment, llm_policy: SimpleLLMPolicy):
        self.env = env
        self.policy = llm_policy
        self.replay_buffer = []
        self.max_buffer_size = 1000
        
    def collect_experience(self, n_steps: int = 10) -> List[Tuple]:
        """Collect experience tuples by interacting with environment"""
        experiences = []
        state = self.env.reset()
        
        for _ in range(n_steps):
            action = self.policy.predict(state)
            next_state, reward, done, info = self.env.step(action)
            
            experiences.append((state, action, reward, next_state, done))
            self.replay_buffer.append((state, action, reward, next_state, done))
            
            # Limit buffer size
            if len(self.replay_buffer) > self.max_buffer_size:
                self.replay_buffer.pop(0)
                
            state = next_state
            if done:
                break
                
        return experiences
    
    def compute_advantages(self, experiences: List[Tuple]) -> List[float]:
        """Compute advantage estimates for policy updates (simplified)"""
        rewards = [exp[2] for exp in experiences]
        # Simple advantage calculation (return - baseline)
        baseline = np.mean(rewards) if rewards else 0
        advantages = [r - baseline for r in rewards]
        return advantages
    
    def update_policy(self) -> None:
        """Update policy using sampled experiences"""
        if not self.replay_buffer:
            return
            
        # Sample batch from replay buffer
        batch_size = min(32, len(self.replay_buffer))
        batch = random.sample(self.replay_buffer, batch_size)
        
        states = [exp[0] for exp in batch]
        actions = [exp[1] for exp in batch]
        advantages = self.compute_advantages(batch)
        
        # Update policy
        self.policy.update_policy(states, actions, advantages)
    
    def train(self, n_iterations: int = 100, steps_per_iteration: int = 20) -> Dict:
        """Train the FLAG-TRADER model"""
        metrics = {
            'total_returns': [],
            'sharpe_ratios': [],
            'max_drawdowns': []
        }
        
        for i in range(n_iterations):
            # Collect experience
            self.collect_experience(steps_per_iteration)
            
            # Update policy
            self.update_policy()
            
            # Evaluate current policy
            if i % 10 == 0:
                eval_metrics = self.evaluate()
                metrics['total_returns'].append(eval_metrics['total_return'])
                metrics['sharpe_ratios'].append(eval_metrics['sharpe_ratio'])
                metrics['max_drawdowns'].append(eval_metrics['max_drawdown'])
                
                print(f"Iteration {i}: Return={eval_metrics['total_return']:.2f}, "
                      f"Sharpe={eval_metrics['sharpe_ratio']:.2f}, "
                      f"MDD={eval_metrics['max_drawdown']:.2f}")
        
        return metrics
    
    def evaluate(self) -> Dict:
        """Evaluate current policy performance"""
        state = self.env.reset()
        done = False
        portfolio_values = [state.total_value]
        
        while not done:
            action = self.policy.predict(state)
            state, _, done, _ = self.env.step(action)
            portfolio_values.append(state.total_value)
        
        # Calculate metrics
        total_return = (portfolio_values[-1] / portfolio_values[0] - 1) * 100
        
        # Calculate daily returns
        daily_returns = [(portfolio_values[i] / portfolio_values[i-1]) - 1 
                         for i in range(1, len(portfolio_values))]
        
        # Sharpe ratio (simplified, assuming 0 risk-free rate)
        sharpe_ratio = np.mean(daily_returns) / np.std(daily_returns) if np.std(daily_returns) > 0 else 0
        sharpe_ratio *= np.sqrt(252)  # Annualize
        
        # Maximum drawdown
        max_drawdown = 0
        peak = portfolio_values[0]
        for value in portfolio_values:
            if value > peak:
                peak = value
            drawdown = (peak - value) / peak
            max_drawdown = max(max_drawdown, drawdown)
        
        return {
            'total_return': total_return,
            'sharpe_ratio': sharpe_ratio,
            'max_drawdown': max_drawdown * 100,
            'portfolio_values': portfolio_values
        }
