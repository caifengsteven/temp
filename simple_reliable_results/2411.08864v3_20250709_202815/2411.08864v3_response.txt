class IsotropicCorrelationModel:
    """
    Implementation of the isotropic correlation model described in the paper.
    All pairwise correlations between assets are equal to a constant value.
    """
    def __init__(self, n_assets, rho, volatilities=None):
        """
        Initialize the isotropic correlation model.
        
        Parameters:
        -----------
        n_assets : int
            Number of assets
        rho : float
            Common correlation coefficient (must satisfy rho >= -1/(n_assets-1))
        volatilities : array-like, optional
            Individual asset volatilities. If None, all volatilities are set to 1.0
        """
        self.n_assets = n_assets
        
        # Check if rho is valid (theorem constraint)
        min_rho = -1 / (n_assets - 1) if n_assets > 1 else -1
        if rho < min_rho:
            raise ValueError(f"Invalid rho: {rho}. For {n_assets} assets, rho must be >= {min_rho}")
        
        self.rho = rho
        
        # Set volatilities (homoskedastic by default)
        if volatilities is None:
            self.volatilities = np.ones(n_assets)
        else:
            self.volatilities = np.array(volatilities)
            
        # Create the correlation matrix (G_N in the paper)
        self.corr_matrix = np.ones((n_assets, n_assets)) * rho
        np.fill_diagonal(self.corr_matrix, 1.0)
        
        # Create the volatility matrix (S_N in the paper)
        self.vol_matrix = np.diag(self.volatilities)
        
        # Create the covariance matrix (Σ in the paper)
        self.cov_matrix = self.vol_matrix @ self.corr_matrix @ self.vol_matrix
        
    def generate_returns(self, n_periods=252):
        """
        Generate random returns based on the isotropic correlation model.
        
        Parameters:
        -----------
        n_periods : int
            Number of time periods to generate
            
        Returns:
        --------
        returns : ndarray
            Array of shape (n_periods, n_assets) containing the generated returns
        """
        # Generate correlated standard normal variables
        Z = np.random.multivariate_normal(
            mean=np.zeros(self.n_assets),
            cov=self.corr_matrix,
            size=n_periods
        )
        
        # Scale by volatilities to get returns
        returns = Z * self.volatilities
        
        return returns
    
    def effective_dof(self, n_assets=None):
        """
        Calculate the effective degrees of freedom for a portfolio of the given size.
        
        Parameters:
        -----------
        n_assets : int, optional
            Number of assets to include. If None, use all available assets.
            
        Returns:
        --------
        n_star : float
            Effective degrees of freedom
        """
        if n_assets is None:
            n_assets = self.n_assets
        
        if n_assets > self.n_assets:
            raise ValueError(f"Cannot compute for {n_assets} assets; model only has {self.n_assets}")
            
        # For isotropic model, effective DOF is N/(1 + (N-1)ρ)
        return n_assets / (1 + (n_assets - 1) * self.rho)
    
    def portfolio_variance(self, n_assets=None):
        """
        Calculate the variance of an equal-weighted portfolio.
        
        Parameters:
        -----------
        n_assets : int, optional
            Number of assets to include. If None, use all available assets.
            
        Returns:
        --------
        variance : float
            Portfolio variance
        """
        if n_assets is None:
            n_assets = self.n_assets
            
        if n_assets > self.n_assets:
            raise ValueError(f"Cannot compute for {n_assets} assets; model only has {self.n_assets}")
        
        # Use the first n_assets
        cov_subset = self.cov_matrix[:n_assets, :n_assets]
        
        # Equal weights
        weights = np.ones(n_assets) / n_assets
        
        # Compute portfolio variance
        return weights.T @ cov_subset @ weights
    
    def independent_variance(self, n_assets=None):
        """
        Calculate the variance due to independent returns (trace component).
        
        Parameters:
        -----------
        n_assets : int, optional
            Number of assets to include. If None, use all available assets.
            
        Returns:
        --------
        variance : float
            Independent variance component
        """
        if n_assets is None:
            n_assets = self.n_assets
            
        if n_assets > self.n_assets:
            raise ValueError(f"Cannot compute for {n_assets} assets; model only has {self.n_assets}")
        
        # Use the first n_assets
        cov_subset = self.cov_matrix[:n_assets, :n_assets]
        
        # Equal weights
        weights = np.ones(n_assets) / n_assets
        
        # Compute the trace component of variance (independent component)
        return np.trace(cov_subset) / (n_assets ** 2)


class LinearFactorModel:
    """
    Implementation of a linear factor model as described in the paper.
    Returns are generated from common factors plus idiosyncratic risk.
    """
    def __init__(self, n_assets, n_factors, idiosyncratic_var=0.2, factor_loadings=None):
        """
        Initialize the linear factor model.
        
        Parameters:
        -----------
        n_assets : int
            Number of assets
        n_factors : int
            Number of factors
        idiosyncratic_var : float or array-like
            Idiosyncratic variance for each asset
        factor_loadings : ndarray, optional
            Factor loadings matrix of shape (n_assets, n_factors)
            If None, random loadings are generated
        """
        self.n_assets = n_assets
        self.n_factors = n_factors
        
        # Set idiosyncratic variances
        if np.isscalar(idiosyncratic_var):
            self.idiosyncratic_var = np.ones(n_assets) * idiosyncratic_var
        else:
            self.idiosyncratic_var = np.array(idiosyncratic_var)
            
        # Generate or set factor loadings
        if factor_loadings is None:
            # Generate random factor loadings (with first factor as "market")
            self.factor_loadings = np.random.normal(0, 0.5, size=(n_assets, n_factors))
            self.factor_loadings[:, 0] = abs(self.factor_loadings[:, 0]) + 0.5  # Make first factor positive
        else:
            self.factor_loadings = factor_loadings
            
        # Compute the covariance matrix
        factor_cov = self.factor_loadings @ self.factor_loadings.T
        idio_cov = np.diag(self.idiosyncratic_var)
        self.cov_matrix = factor_cov + idio_cov
        
    def generate_returns(self, n_periods=252):
        """
        Generate random returns based on the linear factor model.
        
        Parameters:
        -----------
        n_periods : int
            Number of time periods to generate
            
        Returns:
        --------
        returns : ndarray
            Array of shape (n_periods, n_assets) containing the generated returns
        """
        # Generate factor returns (standard normal)
        factor_returns = np.random.normal(0, 1, size=(n_periods, self.n_factors))
        
        # Generate idiosyncratic returns
        idio_returns = np.random.normal(
            0, 
            np.sqrt(self.idiosyncratic_var), 
            size=(n_periods, self.n_assets)
        )
        
        # Combine factor and idiosyncratic returns
        returns = factor_returns @ self.factor_loadings.T + idio_returns
        
        return returns
    
    def effective_dof(self, n_assets=None):
        """
        Calculate the effective degrees of freedom for a portfolio of the given size.
        
        Parameters:
        -----------
        n_assets : int, optional
            Number of assets to include. If None, use all available assets.
            
        Returns:
        --------
        n_star : float
            Effective degrees of freedom
        """
        if n_assets is None:
            n_assets = self.n_assets
            
        if n_assets > self.n_assets:
            raise ValueError(f"Cannot compute for {n_assets} assets; model only has {self.n_assets}")
        
        # Use the first n_assets for the calculation
        loadings = self.factor_loadings[:n_assets]
        idio_var = self.idiosyncratic_var[:n_assets]
        
        # Mean of squared loadings
        b_squared = np.sum(loadings**2) / (n_assets**2)
        
        # Mean of idiosyncratic variance
        s_squared = np.mean(idio_var)
        
        # Inner product of mean loadings
        b_vectors = np.mean(loadings, axis=0)
        b_T_b = np.dot(b_vectors, b_vectors)
        
        # Compute effective DOF using equation 36 from the paper
        return (b_squared * n_assets + s_squared) / (b_T_b + s_squared/n_assets)
    
    def portfolio_variance(self, n_assets=None):
        """
        Calculate the variance of an equal-weighted portfolio.
        
        Parameters:
        -----------
        n_assets : int, optional
            Number of assets to include. If None, use all available assets.
            
        Returns:
        --------
        variance : float
            Portfolio variance
        """
        if n_assets is None:
            n_assets = self.n_assets
            
        if n_assets > self.n_assets:
            raise ValueError(f"Cannot compute for {n_assets} assets; model only has {self.n_assets}")
        
        # Use the first n_assets
        cov_subset = self.cov_matrix[:n_assets, :n_assets]
        
        # Equal weights
        weights = np.ones(n_assets) / n_assets
        
        # Compute portfolio variance
        return weights.T @ cov_subset @ weights
    
    def independent_variance(self, n_assets=None):
        """
        Calculate the variance due to independent returns (trace component).
        
        Parameters:
        -----------
        n_assets : int, optional
            Number of assets to include. If None, use all available assets.
            
        Returns:
        --------
        variance : float
            Independent variance component
        """
        if n_assets is None:

        # Use the first n_assets
        cov_subset = self.cov_matrix[:n_assets, :n_assets]
        
        # Compute the trace component of variance (independent component)
        return np.trace(cov_subset) / (n_assets ** 2)


def compare_effective_dof(n_max=500, samples=1000):
    """
    Compare the effective degrees of freedom between isotropic and factor models
    for varying portfolio sizes.
    """
    # Parameters
    rho = 0.15  # Typical correlation in equity markets
    n_factors = 3  # Fama-French type model
    
    # Initialize models
    iso_model = IsotropicCorrelationModel(n_max, rho)
    factor_model = LinearFactorModel(n_max, n_factors)
    
    # Sample sizes to test
    sizes = np.unique(np.round(np.logspace(0, np.log10(n_max), 50)).astype(int))
    
    # Prepare arrays for results
    iso_dof = np.zeros_like(sizes, dtype=float)
    factor_dof = np.zeros_like(sizes, dtype=float)
    
    # Calculate effective DOF for each size
    for i, size in enumerate(sizes):
        iso_dof[i] = iso_model.effective_dof(size)
        factor_dof[i] = factor_model.effective_dof(size)
    
    # Plot results
    plt.figure(figsize=(10, 6))
    plt.plot(sizes, iso_dof, 'b-', label=f'Isotropic (ρ={rho})')
    plt.plot(sizes, factor_dof, 'g-', label=f'Factor Model (K={n_factors})')
    plt.plot(sizes, sizes, 'k--', alpha=0.3, label='N* = N (Independent)')
    
    # Add horizontal line at 1/rho
    plt.axhline(y=1/rho, color='b', linestyle='--', alpha=0.5, 
                label=f'1/ρ = {1/rho:.1f} (Isotropic limit)')
    
    plt.xlabel('Portfolio Size (N)')
    plt.ylabel('Effective Degrees of Freedom (N*)')
    plt.title('Effective Degrees of Freedom vs. Portfolio Size')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.xscale('log')
    plt.tight_layout()
    plt.show()
    
    return sizes, iso_dof, factor_dof


def simulate_empirical_dof(n_max=500, n_samples=1000, n_periods=252):
    """
    Simulate random portfolios to estimate the effective degrees of freedom
    from returns data, similar to the approach in the paper.
    """
    # Parameters
    rho = 0.15  # Typical correlation in equity markets
    
    # Initialize model and generate returns
    model = IsotropicCorrelationModel(n_max, rho)
    returns = model.generate_returns(n_periods)
    
    # Sample sizes to test
    sizes = []
    n_stars = []
    
    # Function to compute N* from returns data
    def compute_n_star(returns_subset):
        cov_matrix = np.cov(returns_subset, rowvar=False)
        
        # Portfolio variance (equal weights)
        weights = np.ones(cov_matrix.shape[0]) / cov_matrix.shape[0]
        portfolio_variance = weights.T @ cov_matrix @ weights
        
        # Independent variance
        independent_variance = np.trace(cov_matrix) / (cov_matrix.shape[0] ** 2)
        
        # Effective DOF
        return cov_matrix.shape[0] * independent_variance / portfolio_variance
    
    # Generate random portfolios of different sizes
    for _ in tqdm(range(n_samples)):
        # Random portfolio size
        size = np.random.randint(5, n_max + 1)
        
        # Random subset of assets
        indices = np.random.choice(n_max, size=size, replace=False)
        returns_subset = returns[:, indices]
        
        # Compute N* for this portfolio
        n_star = compute_n_star(returns_subset)
        
        sizes.append(size)
        n_stars.append(n_star)
    
    # Convert to arrays
    sizes = np.array(sizes)
    n_stars = np.array(n_stars)
    
    # Theoretical curve
    sizes_curve = np.arange(5, n_max + 1)
    n_stars_curve = sizes_curve / (1 + (sizes_curve - 1) * rho)
    
    # Plot results
    plt.figure(figsize=(10, 6))
    plt.scatter(sizes, n_stars, alpha=0.3, label='Simulated Portfolios')
    plt.plot(sizes_curve, n_stars_curve, 'r-', 
             label=f'Theoretical (ρ={rho})')
    
    # Add horizontal line at 1/rho
    plt.axhline(y=1/rho, color='r', linestyle='--', alpha=0.5,

    plt.xlabel('Portfolio Size (N)')
    plt.ylabel('Effective Degrees of Freedom (N*)')
    plt.title('Simulated Effective DOF vs. Portfolio Size')
    plt.legend()

    return sizes, n_stars


def sample_pairwise_correlations(n_assets=500, n_samples=5000, n_periods=252):
    """
    Simulate and analyze the distribution of pairwise correlations in the
    isotropic correlation model, similar to the analysis in the paper.
    """
    # Parameters
    rho = 0.15  # Typical correlation in equity markets
    
    # Initialize model and generate returns
    model = IsotropicCorrelationModel(n_assets, rho)
    returns = model.generate_returns(n_periods)
    
    # Sample random pairs and compute correlations
    correlations = []
    for _ in range(n_samples):
        i, j = np.random.choice(n_assets, size=2, replace=False)
        corr = np.corrcoef(returns[:, i], returns[:, j])[0, 1]
        correlations.append(corr)
    
    correlations = np.array(correlations)
    
    # Compute Fisher Z transformation
    z_scores = np.sqrt(n_periods - 3) * np.arctanh(correlations)
    
    # Plot histograms
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
    
    # Raw correlations
    ax1.hist(correlations, bins=50, alpha=0.7)
    ax1.axvline(x=rho, color='r', linestyle='--', 
                label=f'True ρ = {rho}')
    ax1.axvline(x=np.mean(correlations), color='g', linestyle='--', 
                label=f'Mean = {np.mean(correlations):.3f}')
    ax1.set_xlabel('Correlation Coefficient')
    ax1.set_ylabel('Frequency')
    ax1.set_title('Distribution of Pairwise Correlations')
    ax1.legend()
    
    # Z-scores
    ax2.hist(z_scores, bins=50, alpha=0.7)
    z_mean = np.sqrt(n_periods - 3) * np.arctanh(rho)
    ax2.axvline(x=z_mean, color='r', linestyle='--', 
                label=f'Expected Z = {z_mean:.3f}')
    
    # Add normal distribution curve
    x = np.linspace(min(z_scores), max(z_scores), 100)
    y = stats.norm.pdf(x, z_mean, 1) * len(z_scores) * (max(z_scores) - min(z_scores)) / 50
    ax2.plot(x, y, 'r-', alpha=0.5, label='Normal(μ, 1)')
    
    ax2.set_xlabel('Fisher Z-Score')
    ax2.set_ylabel('Frequency')
    ax2.set_title('Fisher-Transformed Correlations')
    ax2.legend()
    
    plt.tight_layout()
    plt.show()
    
    # Test normality of z-scores
    ks_stat, p_value = stats.kstest(z_scores, 'norm', args=(z_mean, 1))
    print(f"Kolmogorov-Smirnov test for normality of Z-scores:")
    print(f"KS statistic: {ks_stat:.4f}, p-value: {p_value:.4f}")
    
    if p_value < 0.05:
        print("The distribution of Z-scores differs significantly from normal.")
    else:
        print("The distribution of Z-scores is consistent with normality.")
    
    return correlations, z_scores
