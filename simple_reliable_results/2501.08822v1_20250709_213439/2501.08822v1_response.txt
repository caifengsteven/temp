"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from collections import deque
from enum import Enum
import random
import time

# Define event types
class EventType(Enum):
    LIMIT = 0
    CANCEL = 1
    MARKET = 2

# Define queue sides
class Side(Enum):
    BID = -1
    ASK = 1

class QueueState:
    """Representation of a single queue state in the order book"""
    def __init__(self, side, level, size=0):
        self.side = side
        self.level = level  # Level from mid-price
        self.size = size  # Current queue size
        self.last_event_type = None
        self.events_history = deque(maxlen=10)  # Track recent events

    def update(self, event_type, size_change):
        """Update queue state based on event"""
        self.size = max(0, self.size + size_change)
        self.last_event_type = event_type
        self.events_history.append((event_type, size_change))
        
    def __repr__(self):
        side_str = "BID" if self.side == Side.BID else "ASK"
        return f"Queue({side_str}, L{self.level}, Size={self.size})"

class OrderBook:
    """Simplified limit order book representation"""
    def __init__(self, levels=5):
        self.levels = levels
        self.queues = {}
        self.mid_price = 100.0
        self.tick_size = 0.01
        self.timestamp = 0
        self.trade_imbalance = 0.0  # Short-term trade imbalance
        
        # Initialize queues
        for side in [Side.BID, Side.ASK]:
            for level in range(1, levels + 1):
                key = (side, level)
                self.queues[key] = QueueState(side, level, 
                                             size=np.random.gamma(shape=1.3, scale=170))
    
    def get_best_prices(self):
        """Get best bid and ask prices"""
        best_bid = self.mid_price - 0.5 * self.tick_size
        best_ask = self.mid_price + 0.5 * self.tick_size
        return best_bid, best_ask
    
    def get_state_vector(self):
        """Generate state vector for model input"""
        state = []
        
        # Queue sizes (normalized)
        for side in [Side.BID, Side.ASK]:
            for level in range(1, self.levels + 1):
                size = self.queues.get((side, level), QueueState(side, level)).size
                # Apply log transformation as in the paper
                state.append(np.log1p(size))
        
        # Add spread (in ticks)
        state.append(1.0)  # Fixed at 1 tick for simplicity
        
        # Add trade imbalance
        state.append(self.trade_imbalance)
        
        # Add hour of day (simplified)
        hour = (self.timestamp % 86400) // 3600
        state.append(hour / 24.0)  # Normalize to [0,1]
        
        return np.array(state)


class MDQRSimulator:
    """
    Simplified MDQR simulator that generates synthetic order book events
    """
    def __init__(self, n_levels=5):
        self.book = OrderBook(levels=n_levels)
        self.n_levels = n_levels
        # Base intensity parameters (would be learned by neural net in full model)
        self.base_intensities = {
            EventType.LIMIT: 2.0,
            EventType.CANCEL: 1.8,
            EventType.MARKET: 0.2
        }
        
        # Event counter
        self.event_count = {event_type: 0 for event_type in EventType}
        self.event_volume = {event_type: 0 for event_type in EventType}
        
        # Market impact tracking
        self.initial_mid_price = self.book.mid_price
        self.price_history = []
        
    def intensity_function(self, event_type, side, level):
        """
        Simplified intensity function that would be modeled by neural network
        Returns arrival intensity based on order book state
        """
        queue = self.book.queues.get((side, level), QueueState(side, level))
        base = self.base_intensities[event_type]
        
        # Intensity decreases with queue size for market orders (higher liquidity gets consumed)
        if event_type == EventType.MARKET:
            return base * np.exp(-queue.size / 500)
        
        # Intensity decreases with queue size for limit orders (less room for new orders)
        if event_type == EventType.LIMIT:
            return base * np.exp(-queue.size / 1000)
        
        # Intensity increases with queue size for cancellations (more to cancel)
        if event_type == EventType.CANCEL:
            return base * (1 - np.exp(-queue.size / 300))
        
        return base

    def generate_order_size(self, event_type, side, level):
        """
        Generate realistic order sizes based on the event type and queue
        This is a simplified version of the order size model described in the paper
        """
        queue = self.book.queues.get((side, level), QueueState(side, level))
        
        if event_type == EventType.MARKET:
            # Market orders typically smaller but sometimes can match entire queue
            if random.random() < 0.1:  # 10% chance of taking entire queue
                return max(1, min(queue.size, np.random.geometric(0.1)))
            return max(1, min(queue.size, np.random.geometric(0.5)))
            
        if event_type == EventType.LIMIT:
            # Limit orders often small with occasional larger ones
            return max(1, np.random.geometric(0.2))
            
        if event_type == EventType.CANCEL:
            # Cancellations can be partial or full
            if random.random() < 0.3:  # 30% chance of canceling entire queue
                return queue.size
            return max(1, min(queue.size, np.random.geometric(0.3)))
        
        return 1  # Default fallback

    def generate_next_event(self):
        """Generate the next event in the simulation"""
        # Calculate total intensity across all events and queues
        total_intensity = 0
        intensities = {}
        
        for event_type in EventType:
            for side in [Side.BID, Side.ASK]:
                for level in range(1, self.n_levels + 1):
                    intensity = self.intensity_function(event_type, side, level)
                    intensities[(event_type, side, level)] = intensity
                    total_intensity += intensity
        
        # Generate inter-arrival time (exponential distribution)
        inter_arrival_time = np.random.exponential(1.0 / total_intensity)
        
        # Select event type and queue based on intensities
        rand = random.random() * total_intensity
        cumsum = 0
        selected_event = None
        
        for event_info, intensity in intensities.items():
            cumsum += intensity
            if rand < cumsum:
                selected_event = event_info
                break
        
        event_type, side, level = selected_event
        
        # Generate order size
        size = self.generate_order_size(event_type, side, level)
        
        # Update the order book
        self.process_event(event_type, side, level, size)
        
        # Update timestamp
        self.book.timestamp += inter_arrival_time
        
        # Return event details
        return {
            'timestamp': self.book.timestamp,
            'event_type': event_type,
            'side': side,
            'level': level,
            'size': size,
            'mid_price': self.book.mid_price
        }

    def process_event(self, event_type, side, level, size):
        """Process an event and update the order book"""
        queue = self.book.queues.get((side, level), QueueState(side, level))
        
        # Update event counters
        self.event_count[event_type] += 1
        self.event_volume[event_type] += size
        
        if event_type == EventType.LIMIT:
            # Add to the queue
            queue.update(event_type, size)
            
        elif event_type == EventType.CANCEL:
            # Remove from the queue (limited by available size)
            actual_size = min(size, queue.size)
            queue.update(event_type, -actual_size)
            
        elif event_type == EventType.MARKET:
            # Execute against the queue
            actual_size = min(size, queue.size)
            queue.update(event_type, -actual_size)
            
            # Update trade imbalance
            imbalance_contribution = actual_size if side == Side.BID else -actual_size
            self.book.trade_imbalance = 0.8 * self.book.trade_imbalance + 0.2 * (imbalance_contribution / 100)
            
            # Update mid price with market impact
            # Price moves up when buying (consuming ask) and down when selling (consuming bid)
            impact = 0.0001 * np.sqrt(actual_size) * (-1 if side == Side.BID else 1)
            self.book.mid_price += impact
            
        # Save price for tracking
        self.price_history.append(self.book.mid_price)

    def simulate_twap_execution(self, quantity, duration_minutes=5, child_orders=10):
        """
        Simulate a TWAP execution strategy and measure market impact
        
        Args:
            quantity: Total quantity to execute (as % of average volume)
            duration_minutes: Duration of execution in minutes
            child_orders: Number of child orders to split the parent order into
        
        Returns:
            DataFrame with impact measurements
        """
        # Reset simulation
        initial_mid_price = self.book.mid_price
        
        # Prepare child orders
        child_size = quantity / child_orders
        interval = (duration_minutes * 60) / child_orders
        
        # Execution time points
        execution_times = [i * interval for i in range(child_orders)]
        
        # Simulate normal market
        impact_data = []
        current_time = 0
        execution_index = 0
        
        while current_time < duration_minutes * 60 * 2:  # Simulate twice the execution time
            # Check if we need to execute a child order
            if execution_index < len(execution_times) and current_time >= execution_times[execution_index]:
                # Execute child order as market order on ask side (buy order)
                self.process_event(EventType.MARKET, Side.ASK, 1, child_size)
                execution_index += 1
            
            # Generate regular market event
            event = self.generate_next_event()
            current_time = event['timestamp']
            
            # Record market impact
            impact_bps = (self.book.mid_price / initial_mid_price - 1) * 10000  # Convert to basis points
            impact_data.append({
                'time': current_time / 60,  # Convert to minutes
                'mid_price': self.book.mid_price,
                'impact_bps': impact_bps,
                'execution_completed': execution_index == len(execution_times)
            })
        
        return pd.DataFrame(impact_data)

    events = []

        # Update transition matrix
        if prev_event is not None:
            prev_idx = prev_event['event_type'].value
            curr_idx = event['event_type'].value
            transition_matrix[prev_idx][curr_idx] += 1
        
        prev_event = event
    
    # Normalize transition matrix
    for i in range(3):
        row_sum = transition_matrix[i].sum()
        if row_sum > 0:
            transition_matrix[i] = transition_matrix[i] / row_sum
    
    # Print transition matrix
    print("Event Transition Matrix:")
    events_names = ["LIMIT", "CANCEL", "MARKET"]
