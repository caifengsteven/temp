class BoxCoxTransformer:
    """Applies Box-Cox transformation to normalize features"""
    def __init__(self, lambda_values=None):
        self.lambda_values = lambda_values
        self.means = None
        self.stds = None
    
    def fit(self, X):
        """Find optimal lambda values for each feature"""
        n_features = X.shape[1]
        if self.lambda_values is None:
            self.lambda_values = np.ones(n_features)
            # For simplicity, we're using fixed lambdas, but in practice you'd optimize these
            for i in range(n_features):
                self.lambda_values[i] = 0.5 if np.mean(X[:, i]) > 0 else 1.0
        
        # Apply transformation
        X_transformed = np.zeros_like(X, dtype=float)
        for i in range(n_features):
            X_transformed[:, i] = self._transform_feature(X[:, i], self.lambda_values[i])
        
        # Calculate means and stds for standardization
        self.means = np.mean(X_transformed, axis=0)
        self.stds = np.std(X_transformed, axis=0)
        
        return self
    
    def transform(self, X):
        """Apply Box-Cox transformation and standardize"""
        n_features = X.shape[1]
        X_transformed = np.zeros_like(X, dtype=float)
        
        for i in range(n_features):
            X_transformed[:, i] = self._transform_feature(X[:, i], self.lambda_values[i])
        
        # Standardize
        X_standardized = (X_transformed - self.means) / (self.stds + 1e-8)
        
        return X_standardized
    
    def _transform_feature(self, x, lambda_val):
        """Apply Box-Cox transformation to a single feature"""
        if lambda_val == 0:
            return np.log(x + 1e-8)
        else:
            return ((x + 1e-8) ** lambda_val - 1) / lambda_val


class ProbabilisticNN(nn.Module):
    """Neural network that predicts distribution parameters"""
    def __init__(self, input_size, hidden_size=64, output_size=3):
        super(ProbabilisticNN, self).__init__()
        self.layer1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.layer2 = nn.Linear(hidden_size, output_size)
        
    def forward(self, x):
        x = self.relu(self.layer1(x))
        x = self.layer2(x)
        
        # Output parameters for skewed Gaussian: mu, sigma, alpha
        mu = x[:, 0]
        sigma = torch.exp(x[:, 1])  # Ensure sigma > 0
        alpha = x[:, 2]
        
        return torch.stack([mu, sigma, alpha], dim=1)


class OrderFlowVariables:
    """Computes Hawkes-inspired order flow variables"""
    def __init__(self, beta_values=[10, 100, 1000], eta_values=[0.001, 0.1, 1, 10]):
        self.beta_values = beta_values  # Time scales (s^-1)
        self.eta_values = eta_values    # Distance scales (bp^-1)
    
    def compute_limit_order_flow(self, times, volumes, distances, side, current_time):
        """Compute limit order flow variables"""
        features = []
        
        for beta in self.beta_values:
            for eta in self.eta_values:
                # Time decay
                time_weights = np.exp(-beta * (current_time - times))
                
                # Distance decay
                distance_weights = np.exp(-eta * distances)
                
                # Volume weighting (f(v) = v as per paper)
                flow = np.sum(time_weights * volumes * distance_weights)
                features.append(flow)
        
        return features
    
    def compute_market_order_flow(self, times, volumes, side, current_time):
        """Compute market order flow variables"""
        features = []
        
        for beta in self.beta_values:
            # Time decay
            time_weights = np.exp(-beta * (current_time - times))
            
            # Volume weighting (f(v) = v as per paper)
            flow = np.sum(time_weights * volumes)
            features.append(flow)
        
        return features


class LOBSimulator:

            self.limit_orders_bid['distances'].append(distance)

                self.spoof_side = 'bid'
                self.spoof_size = volume
                self.spoof_distance = distance
                self.spoof_start_time = self.current_time

            self.limit_orders_ask['times'].append(self.current_time)
            self.limit_orders_ask['volumes'].append(volume)
            self.limit_orders_ask['distances'].append(distance)

                self.spoofing_active = True
                self.spoof_side = 'ask'

class SpoofingDetector:
    """Detects potential spoofing based on expected cost model"""
    def __init__(self, model, transformer, maker_fee=0, taker_fee=0.0005):
        self.model = model
        self.transformer = transformer
        self.maker_fee = maker_fee
        self.taker_fee = taker_fee
    
    def expected_cost_spoof(self, features, features_no_spoof, mid_price, spread, 
                           delta_b, delta_a, Q, q, side='ask'):
        """Calculate expected cost for spoofing strategy"""
        # Transform features
        features_t = torch.tensor(self.transformer.transform(features.reshape(1, -1)), dtype=torch.float32)
        features_no_spoof_t = torch.tensor(self.transformer.transform(features_no_spoof.reshape(1, -1)), dtype=torch.float32)
        
        # Get distribution parameters
        with torch.no_grad():
            params = self.model(features_t).numpy()[0]
            params_no_spoof = self.model(features_no_spoof_t).numpy()[0]
        
        mu, sigma, alpha = params
        mu_no_spoof, sigma_no_spoof, alpha_no_spoof = params_no_spoof
        
        # For simplicity, we'll use a normal distribution instead of skewed Gaussian
        if side == 'ask':
            # Calculate expected cost components for spoofing
            pb = mid_price * (1 - spread/20000)
            pa = mid_price * (1 + spread/20000)
            
            # Thresholds for order execution
            threshold_bona_fide = delta_a + spread/2
            threshold_non_bona_fide = -(delta_b + spread/2)
            
            # Calculate probabilities and conditional expectations
            prob_bona_fide = 1 - norm.cdf(threshold_bona_fide, mu, sigma)
            prob_non_bona_fide = norm.cdf(threshold_non_bona_fide, mu, sigma)
            
            expected_dp_bona_fide = mu - sigma * norm.pdf(threshold_bona_fide, mu, sigma) / (1 - norm.cdf(threshold_bona_fide, mu, sigma))
            expected_dp_non_bona_fide = mu - sigma * norm.pdf(threshold_non_bona_fide, mu, sigma) / norm.cdf(threshold_non_bona_fide, mu, sigma)
            
            # Calculate cost components
            cost_bona_fide = -prob_bona_fide * (1 - self.maker_fee) * q * (pa + delta_a * mid_price/10000)
            cost_non_bona_fide = prob_non_bona_fide * (1 + self.maker_fee) * Q * (pb - delta_b * mid_price/10000)
            
            cost_liquidation_bona_fide = -(1 - prob_bona_fide) * (1 - self.taker_fee) * q * (pb + expected_dp_bona_fide * mid_price/10000)
            cost_liquidation_non_bona_fide = -prob_non_bona_fide * (1 - self.taker_fee) * Q * (pb + expected_dp_non_bona_fide * mid_price/10000)
            
            # Total expected cost
            cost_spoof = cost_bona_fide + cost_non_bona_fide + cost_liquidation_bona_fide + cost_liquidation_non_bona_fide
            
            # Calculate cost without spoofing (using no_spoof parameters)
            prob_bona_fide_no_spoof = 1 - norm.cdf(threshold_bona_fide, mu_no_spoof, sigma_no_spoof)
            expected_dp_bona_fide_no_spoof = mu_no_spoof - sigma_no_spoof * norm.pdf(threshold_bona_fide, mu_no_spoof, sigma_no_spoof) / (1 - norm.cdf(threshold_bona_fide, mu_no_spoof, sigma_no_spoof))
            
            cost_bona_fide_no_spoof = -prob_bona_fide_no_spoof * (1 - self.maker_fee) * q * (pa + delta_a * mid_price/10000)
            cost_liquidation_bona_fide_no_spoof = -(1 - prob_bona_fide_no_spoof) * (1 - self.taker_fee) * q * (pb + expected_dp_bona_fide_no_spoof * mid_price/10000)
            
            cost_no_spoof = cost_bona_fide_no_spoof + cost_liquidation_bona_fide_no_spoof
            
            return cost_no_spoof - cost_spoof
        else:
            # Similar calculation for bid side (buying intention)
            return 0  # Simplified for brevity
    
    def is_suspicious(self, features, features_no_spoof, mid_price, spread, delta, Q, side='bid', q=100):
        """Determine if an order is suspicious based on expected cost"""
        delta_a = 0  # Bona fide at touch
        delta_b = delta if side == 'bid' else 0
        
        if side == 'bid':
            return False  # Simplified for this example
        else:  # ask
            # Calculate potential gain from spoofing
            gain = self.expected_cost_spoof(
                features, features_no_spoof, mid_price, spread, 
                delta_b, delta_a, Q, q, 'ask'
            )
            return gain > 0


def train_model(X_train, y_train, input_size, epochs=100):
    """Train the probabilistic neural network"""
    # Convert data to tensors
    X_train_tensor = torch.tensor(X_train, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train, dtype=torch.float32)
    
    # Create dataset and loader
    train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
    train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)
    
    # Initialize model
    model = ProbabilisticNN(input_size)
    
    # Loss function and optimizer
    def neg_log_likelihood(predicted, target):
        mu, sigma, alpha = predicted[:, 0], predicted[:, 1], predicted[:, 2]
        # Simplified NLL for normal distribution (without skew)
        return torch.mean(torch.log(sigma) + 0.5 * ((target - mu) / sigma)**2)
    
    optimizer = optim.Adam(model.parameters(), lr=0.001)
    
    # Training loop
    for epoch in range(epochs):
        total_loss = 0
        for X_batch, y_batch in train_loader:
            optimizer.zero_grad()
            params = model(X_batch)
            loss = neg_log_likelihood(params, y_batch)
            loss.backward()
            optimizer.step()
            total_loss += loss.item()
        
        if (epoch + 1) % 10 == 0:
            print(f"Epoch {epoch+1}/{epochs}, Loss: {total_loss/len(train_loader):.6f}")
    
    return model


def generate_synthetic_data(n_samples=1000):

    # Generate price moves with some dependency on features

    bid_limit_sum = np.sum(X[:, 1:13], axis=1)

    # Add noise
    y = 0.1 * (bid_limit_sum - ask_limit_sum) / (bid_limit_sum + ask_limit_sum + 1e-8)
    y += np.random.normal(0, 0.2, n_samples)
    
    return X, y.reshape(-1, 1)
