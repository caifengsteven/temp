class Order:
    """Represents a limit order in the order book."""
    def __init__(self, order_id: int, price: float, size: float, 
                 timestamp: float, side: str, event_type: str):
        self.order_id = order_id
        self.price = price
        self.size = size if side == "sell" else -size  # negative for buy orders

class LOB:
    """Simulates a limit order book."""
    def __init__(self, tick_size=0.01, lot_size=1):
        self.tick_size = tick_size
        self.lot_size = lot_size
        self.buy_orders = {}  # price -> list of orders
        self.sell_orders = {}  # price -> list of orders
        self.order_history = []  # list of all orders
        self.mid_price_history = []  # list of (timestamp, mid_price)
        self.last_mid_price_change = 0  # timestamp of last mid_price change
        self.price_level_history = {}  # price -> timestamp of first order at this price
        self.last_order_at_price = {}  # price -> timestamp of previous order at this price
        
    def add_order(self, order: Order):
        """Add a new order to the order book."""
        if order.side == "buy":
            if order.price not in self.buy_orders:
                self.buy_orders[order.price] = []
            self.buy_orders[order.price].append(order)
        else:
            if order.price not in self.sell_orders:
                self.sell_orders[order.price] = []
            self.sell_orders[order.price].append(order)
            
        # Track order history
        self.order_history.append(order)
        
        # Update price level history
        if order.price not in self.price_level_history:
            self.price_level_history[order.price] = order.timestamp
            
        # Update last order at price
        self.last_order_at_price[order.price] = order.timestamp
        
        # Update mid price
        self._update_mid_price(order.timestamp)
        
    def _update_mid_price(self, timestamp):
        """Update the mid price based on best bid and ask."""
        if not self.buy_orders or not self.sell_orders:
            return
            
        best_bid = max(self.buy_orders.keys())
        best_ask = min(self.sell_orders.keys())
        
        mid_price = (best_bid + best_ask) / 2
        
        if not self.mid_price_history or mid_price != self.mid_price_history[-1][1]:
            self.last_mid_price_change = timestamp
            self.mid_price_history.append((timestamp, mid_price))
    
    def get_available_volume(self, price, side):
        """Get the total volume available at a given price level."""
        if side == "buy":
            return sum(order.size for order in self.buy_orders.get(price, []))
        else:
            return sum(order.size for order in self.sell_orders.get(price, []))
    
    def get_best_bid_ask(self):
        """Get the best bid and ask prices."""
        best_bid = max(self.buy_orders.keys()) if self.buy_orders else 0
        best_ask = min(self.sell_orders.keys()) if self.sell_orders else float('inf')
        return best_bid, best_ask
    
    def get_order_features(self, order: Order) -> Dict:
        """Calculate the six time-dependent features for an order as described in the paper."""
        best_bid, best_ask = self.get_best_bid_ask()
        mid_price = (best_bid + best_ask) / 2
        
        # 1. Available volume at the order price level
        V = self.get_available_volume(order.price, order.side)
        
        # 2. Time elapsed since last mid-price change
        T_m = order.timestamp - self.last_mid_price_change
        
        # 3. Time elapsed since first order at this price level
        T_1 = order.timestamp - self.price_level_history.get(order.price, order.timestamp)
        
        # 4. Time elapsed since previous order at this price level
        T_prime = order.timestamp - self.last_order_at_price.get(order.price, order.timestamp)
        
        # 5. Same Book Side (SBS) - sum of volumes on same side
        if order.side == "buy":
            SBS = sum(self.get_available_volume(p, "buy") for p in self.buy_orders if best_bid >= p >= order.price)
        else:
            SBS = sum(self.get_available_volume(p, "sell") for p in self.sell_orders if order.price >= p >= best_ask)
        
        # 6. Opposite Book Side (OBS) - sum of volumes on opposite side
        symmetric_price = 2 * mid_price - order.price
        if order.side == "buy":
            OBS = sum(self.get_available_volume(p, "sell") for p in self.sell_orders if p >= best_ask and p <= symmetric_price)
        else:
            OBS = sum(self.get_available_volume(p, "buy") for p in self.buy_orders if p <= best_bid and p >= symmetric_price)
        
        return {
            "V": V,
            "T_m": T_m,
            "T_1": T_1,
            "T_prime": T_prime,
            "SBS": SBS,
            "OBS": OBS
        }

class ClusterLOB:
    """Implementation of the ClusterLOB algorithm from the paper."""
    def __init__(self, num_clusters=3, window_size=100):
        self.num_clusters = num_clusters
        self.window_size = window_size
        self.kmeans = None
        self.scaler = StandardScaler()
        self.feature_names = ["V", "T_m", "T_1", "T_prime", "SBS", "OBS"]
        
    def forward_rolling_normalize(self, features: List[Dict]) -> np.ndarray:
        """Apply forward-rolling normalization to features as described in the paper."""
        normalized_features = []
        
        for i in range(len(features)):
            if i < self.window_size:
                # For the first window_size orders, use all available previous orders
                window = features[:i+1]
            else:
                # For subsequent orders, use a sliding window of the previous window_size orders
                window = features[i-self.window_size+1:i+1]
            
            # Extract feature values from the window
            window_values = {feat: [w[feat] for w in window] for feat in self.feature_names}
            
            # Calculate mean and std for each feature
            means = {feat: np.mean(vals) for feat, vals in window_values.items()}
            stds = {feat: np.std(vals) if np.std(vals) > 0 else 1.0 for feat, vals in window_values.items()}
            
            # Normalize current features
            normalized = [(features[i][feat] - means[feat]) / stds[feat] for feat in self.feature_names]
            normalized_features.append(normalized)
            
        return np.array(normalized_features)
    
    def train(self, lob: LOB):
        """Train the clustering model on a limit order book."""
        # Extract features for all orders
        features = [lob.get_order_features(order) for order in lob.order_history]
        
        # Apply forward-rolling normalization
        normalized_features = self.forward_rolling_normalize(features)
        
        # Train KMeans++ (as mentioned in the paper)
        self.kmeans = KMeans(n_clusters=self.num_clusters, init='k-means++', random_state=42, n_init=10)
        self.kmeans.fit(normalized_features)
        
        # Return cluster assignments
        return self.kmeans.labels_
    
    def predict(self, lob: LOB, new_orders):
        """Predict clusters for new orders."""
        if not self.kmeans:
            raise ValueError("Model not trained yet")
        
        # Extract features for new orders
        features = [lob.get_order_features(order) for order in new_orders]
        
        # Apply forward-rolling normalization
        normalized_features = self.forward_rolling_normalize(features)
        
        # Predict clusters
        return self.kmeans.predict(normalized_features)

def calculate_ofi(orders, time_buckets, cluster_labels=None, cluster_id=None, ofi_type="size"):
    """Calculate Order Flow Imbalance for orders, optionally filtered by cluster."""
    buckets = {}
    
    for i, order in enumerate(orders):
        # Skip if we're filtering by cluster and this order is not in the specified cluster
        if cluster_labels is not None and cluster_id is not None and cluster_labels[i] != cluster_id:
            continue
            
        # Find the bucket this order belongs to
        bucket = None
        for start_time, end_time in time_buckets:
            if start_time <= order.timestamp < end_time:
                bucket = (start_time, end_time)
                break
        
        if bucket is None:
            continue
            
        if bucket not in buckets:
            buckets[bucket] = {
                "L_b": 0, "D_b": 0, "M_b": 0,  # buy side
                "L_a": 0, "D_a": 0, "M_a": 0,  # ask side
            }
            
        # Update OFI components based on order type and side
        value = 1 if ofi_type == "count" else abs(order.size)
        
        if order.event_type == "add":
            if order.side == "buy":
                buckets[bucket]["L_b"] += value
            else:
                buckets[bucket]["L_a"] += value
        elif order.event_type == "cancel":
            if order.side == "buy":
                buckets[bucket]["D_b"] += value
            else:
                buckets[bucket]["D_a"] += value
        elif order.event_type == "trade":
            if order.side == "buy":
                buckets[bucket]["M_b"] += value
            else:
                buckets[bucket]["M_a"] += value
    
    # Calculate OFI for each bucket
    ofi_values = {}
    for bucket, components in buckets.items():
        ofi = (components["L_b"] - components["D_b"] + components["M_b"] - 
               components["L_a"] + components["D_a"] - components["M_a"])
        ofi_values[bucket] = ofi
        
    return ofi_values

def calculate_pnl(ofi_values, returns, return_type="FRNB"):
    """Calculate PnL based on OFI signals and returns."""
    pnl = 0
    trades = 0
    
    for bucket, ofi in ofi_values.items():
        if bucket in returns and return_type in returns[bucket]:
            signal = np.sign(ofi)
            ret = returns[bucket][return_type]
            trade_pnl = signal * ret
            pnl += trade_pnl
            trades += 1
            
    # Calculate PnL per trade
    ppt = pnl / trades if trades > 0 else 0
    
    return pnl, ppt, trades

def calculate_sharpe(pnl_series, annualization_factor=252):
    """Calculate annualized Sharpe ratio."""
    if len(pnl_series) <= 1:
        return 0
    
    mean_return = np.mean(pnl_series)
    std_return = np.std(pnl_series, ddof=1)
    
    if std_return == 0:
        return 0
        
    sharpe = mean_return / std_return * np.sqrt(annualization_factor)
    return sharpe

def calculate_correlation(ofi_values, returns, return_type):
    """Calculate correlation between OFI and returns."""
    paired_data = []
    
    for bucket, ofi in ofi_values.items():
        if bucket in returns and return_type in returns[bucket]:
            paired_data.append((ofi, returns[bucket][return_type]))
    
    if len(paired_data) < 2:
        return 0
        
    ofi_arr = np.array([p[0] for p in paired_data])
    ret_arr = np.array([p[1] for p in paired_data])
    
    if np.std(ofi_arr) == 0 or np.std(ret_arr) == 0:
        return 0
        
    correlation = np.corrcoef(ofi_arr, ret_arr)[0, 1]
    return correlation
