    def generate_simulated_data(self, days=252*2):
        """Generate simulated stock, volatility, interest rate, and option price data"""
        dates = [datetime.now() + timedelta(days=i) for i in range(days)]
        
        # Generate stock prices with GBM
        S = [self.S0]
        for i in range(1, days):
            drift = (self.r - 0.5 * self.sigma**2) * self.dt
            diffusion = self.sigma * np.sqrt(self.dt) * np.random.normal()
            S.append(S[-1] * np.exp(drift + diffusion))
        
        # Generate stochastic volatility (mean-reverting process)
        kappa = 2.0  # Mean reversion speed
        theta = 0.2  # Long-term mean
        xi = 0.1  # Volatility of volatility
        vol = [self.sigma]
        for i in range(1, days):
            dvol = kappa * (theta - vol[-1]) * self.dt + xi * np.sqrt(vol[-1]) * np.sqrt(self.dt) * np.random.normal()
            vol.append(max(0.05, min(0.5, vol[-1] + dvol)))  # Bound volatility

        a = 0.1  # Mean reversion speed
        b = 0.05  # Long-term mean
        s = 0.01  # Volatility of interest rate
        rates = [self.r]
        for i in range(1, days):
            dr = a * (b - rates[-1]) * self.dt + s * np.sqrt(self.dt) * np.random.normal()
            rates.append(max(0.01, min(0.1, rates[-1] + dr)))  # Bound interest rates
        
        # Calculate option prices using Black-Scholes for reference
        call_prices = []
        for i in range(days):
            d1 = (np.log(S[i]/self.K) + (rates[i] + 0.5 * vol[i]**2) * self.T) / (vol[i] * np.sqrt(self.T))
            d2 = d1 - vol[i] * np.sqrt(self.T)

            call_prices.append(call)

    def finite_difference_solver(self, S, K, T, r, sigma, M=100, N=100):
        """
        Solve the extended Black-Scholes PDE using finite difference method
        """
        # Set up grid
        S_max = 2 * K
        dS = S_max / M
        dt = T / N
        
        # Initialize grid
        grid = np.zeros((M+1, N+1))
        
        # Set boundary conditions (for a call option)
        for j in range(N+1):
            grid[0, j] = 0  # At S=0, option value is 0
            grid[M, j] = S_max - K * np.exp(-r * (T - j*dt))  # At S=S_max
        
        # Set terminal condition
        for i in range(M+1):
            S_i = i * dS
            grid[i, N] = max(0, S_i - K)  # Payoff at maturity
        
        # Solve PDE using implicit method (backward in time)
        start_time = time.time()
        for j in range(N-1, -1, -1):
            # Set up tridiagonal matrix system
            a = np.zeros(M+1)
            b = np.zeros(M+1)
            c = np.zeros(M+1)
            d = np.zeros(M+1)
            
            for i in range(1, M):
                S_i = i * dS
                a[i] = 0.5 * dt * (sigma**2 * S_i**2 / dS**2 - r * S_i / dS)
                b[i] = 1 + dt * (sigma**2 * S_i**2 / dS**2 + r)
                c[i] = 0.5 * dt * (-(sigma**2) * S_i**2 / dS**2 - r * S_i / dS)
                d[i] = grid[i, j+1]
            
            # Solve tridiagonal system using Thomas algorithm
            for i in range(2, M):
                w = a[i] / b[i-1]
                b[i] = b[i] - w * c[i-1]
                d[i] = d[i] - w * d[i-1]
            
            # Back substitution
            grid[M-1, j] = d[M-1] / b[M-1]
            for i in range(M-2, 0, -1):
                grid[i, j] = (d[i] - c[i] * grid[i+1, j]) / b[i]
        
        # Interpolate to get option price at current stock price
        idx = int(S / dS)
        if idx >= M:
            return grid[M-1, 0]
        elif idx <= 0:
            return grid[1, 0]
        else:
            weight = (S - idx * dS) / dS
            price = (1 - weight) * grid[idx, 0] + weight * grid[idx+1, 0]
        
        computation_time = time.time() - start_time
        return price, computation_time
    
    def create_lstm_model(self, window_size=30, features=4):
        """Create LSTM model for option pricing"""
        model = Sequential([
            LSTM(50, return_sequences=True, input_shape=(window_size, features)),
            Dropout(0.2),
            LSTM(50),
            Dropout(0.2),
            Dense(1)
        ])
        model.compile(optimizer='adam', loss='mean_squared_error')
        return model
    
    def prepare_lstm_data(self, df, window_size=30):
        """Prepare data for LSTM model"""
        # Select features
        features = df[['Stock_Price', 'Volatility', 'Interest_Rate', 'Option_Price']].values
        
        # Normalize data
        mean = features.mean(axis=0)
        std = features.std(axis=0)
        normalized = (features - mean) / std
        
        # Create sequences
        X, y = [], []
        for i in range(window_size, len(normalized)):
            X.append(normalized[i-window_size:i, :-1])  # All features except option price
            y.append(normalized[i, -1])  # Option price
            
        return np.array(X), np.array(y), mean, std
