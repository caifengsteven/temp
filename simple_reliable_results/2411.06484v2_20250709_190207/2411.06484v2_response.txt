def simulate_heston_euler(params, n_steps, dt, n_paths=1):
    """
    Simulate Heston SV model using Euler approximation
    
    Parameters:
    - params: dictionary with keys 'mu', 'k', 'theta', 'sigma_v', 'rho'
    - n_steps: number of time steps
    - dt: time step size
    - n_paths: number of sample paths to generate
    
    Returns:
    - y: returns array of shape (n_paths, n_steps)
    - v: variance array of shape (n_paths, n_steps+1)
    """
    mu = params['mu']
    k = params['k']
    theta = params['theta']
    sigma_v = params['sigma_v']
    rho = params['rho']
    
    # Initialize arrays
    v = np.zeros((n_paths, n_steps+1))
    y = np.zeros((n_paths, n_steps))
    
    # Set initial variance
    v[:, 0] = theta
    
    # Subdivide dt for better approximation
    sub_steps = 10
    sub_dt = dt / sub_steps
    
    for i in range(n_paths):
        for j in range(n_steps):
            v_t = v[i, j]
            p_t = 0  # log price at start of interval
            
            for s in range(sub_steps):
                # Generate correlated random variables
                z1 = np.random.normal(0, 1)
                z2 = rho * z1 + np.sqrt(1 - rho**2) * np.random.normal(0, 1)
                
                # Ensure variance is positive
                v_t = max(v_t, 0)
                
                # Update log price
                p_t += (mu - v_t/2) * sub_dt + np.sqrt(v_t) * z2 * np.sqrt(sub_dt)
                
                # Update variance
                v_t += k * (theta - v_t) * sub_dt + sigma_v * np.sqrt(v_t) * z1 * np.sqrt(sub_dt)
            
            y[i, j] = p_t  # Store return over interval
            v[i, j+1] = v_t  # Store final variance
    
    return y, v

def calculate_sample_moments(returns, max_order=5):
    """Calculate sample moments of the returns up to specified order"""
    moments = []
    for i in range(1, max_order+1):
        moments.append(np.mean(returns**i))
    return moments

def calculate_sample_covariances(returns, max_order=4):
    """Calculate sample covariances between powers of returns and their lags"""
    n = returns.shape[1]
    covariances = []
    
    for i in range(1, max_order+1):
        for j in range(1, max_order+1):
            if i + j <= max_order + 1:  # Limit to reasonable orders
                # Calculate covariance between y_n^i and y_{n+1}^j
                cov_ij = np.mean(returns[:, :-1]**i * returns[:, 1:]**j) - \
                         np.mean(returns[:, :-1]**i) * np.mean(returns[:, 1:]**j)
                covariances.append((i, j, cov_ij))
    
    return covariances

def poly_to_float(poly, params):
    """Convert a Poly object from ajdmom to a float value using parameter values"""
    result = 0
    
    # Map keyfor to parameter names
    param_map = {
        'e^{-kh}': np.exp(-params['k'] * params['h']),
        'h': params['h'],
        'k^{-}': 1/params['k'],
        'mu': params['mu'],
        'theta': params['theta'],
        'sigma_v': params['sigma_v'],
        'rho': params['rho'],
        'sqrt(1-rho^2)': np.sqrt(1 - params['rho']**2),
        'lambda': params.get('lambda', 0),
        'mu_j': params.get('mu_j', 0),
        'sigma_j': params.get('sigma_j', 0)
    }
    
    for key, coef in poly.items():
        term = float(coef)
        for i, param_name in enumerate(poly.keyfor):
            if key[i] != 0:
                term *= param_map[param_name] ** key[i]
        result += term
    
    return result
