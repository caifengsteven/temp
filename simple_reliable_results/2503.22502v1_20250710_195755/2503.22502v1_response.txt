import numpy as np
import matplotlib.pyplot as plt

# Model parameters
@dataclass
class ModelParams:
    S0: float = 2820.0       # Initial exchange rate (ETH/USDC)
    Z0: float = 2820.0       # Initial pool rate
    Y0: float = 50000.0      # Initial pool inventory (ETH)
    sigma: float = 0.0569    # Volatility parameter
    eta: float = 1e-10       # Liquidity provision volatility
    xi: float = 300.0        # Trade size
    a: float = 1e-14         # Temporary price impact in external venue
    r: float = 0.01          # Fee collected by venue (fraction of trade value)
    a0: float = 100.0        # Minimum intensity parameter
    a1: float = 142.7        # Baseline intensity
    a2: float = 0.0          # Depth sensitivity parameter
    a3: float = 13.6         # Arbitrage sensitivity parameter
    gamma: float = 1e-18     # LP risk aversion
    zeta: float = 1e-6       # Venue risk aversion
    nu_inf: float = float('inf')  # Speed bound for LP
    T: float = 1.0           # Time horizon (days)
    dt: float = 1.0/10000    # Time step for simulation


class AutomatedMarketMaker:
    def __init__(self, params: ModelParams):
        self.params = params
        self.X0 = params.Y0 * params.Z0  # Initial X reserves
        self.c0 = self.X0 * params.Y0    # Constant product parameter
        
    def simulate(self, num_steps: int = 10000, with_optimal_strategy: bool = True) -> dict:
        """
        Simulate the AMM dynamics with the optimal LP strategy
        """
        # Initialize parameters
        dt = self.params.T / num_steps
        sqrt_dt = np.sqrt(dt)
        
        # Initialize arrays
        S = np.zeros(num_steps + 1)  # External price
        Z = np.zeros(num_steps + 1)  # Pool price
        Y = np.zeros(num_steps + 1)  # Pool inventory
        X = np.zeros(num_steps + 1)  # Pool reserves X
        nu = np.zeros(num_steps)     # LP's strategy
        N_plus = np.zeros(num_steps + 1, dtype=int)  # Buys counter
        N_minus = np.zeros(num_steps + 1, dtype=int) # Sells counter
        lambda_plus = np.zeros(num_steps)  # Buy intensity
        lambda_minus = np.zeros(num_steps)  # Sell intensity
        
        # Initialize state
        S[0] = self.params.S0
        Z[0] = self.params.Z0
        Y[0] = self.params.Y0
        X[0] = self.X0
        
        # Generate Brownian motions
        dW = np.random.normal(0, sqrt_dt, num_steps)  # Price Brownian motion
        dB = np.random.normal(0, sqrt_dt, num_steps)  # Liquidity Brownian motion
        
        # Time-dependent coefficients for risk-averse venue
        # These would be solutions to the Riccati equation in the paper
        # For simplicity, we use a linear approximation
        def alpha_zeta(t, Z, Y, S):
            # This is an approximation of the function in equation (4.7)
            # For real implementation, solve the Riccati equation system
            remaining_time = self.params.T - t
            return (2 * self.params.a2 * self.params.r * remaining_time - 
                   2 * (S + Z) * self.params.gamma * self.params.a * self.params.eta**2) / (
                   4 * self.params.eta**2 * self.params.a**2 * self.params.gamma + 2 * self.params.a)
        
        for i in range(num_steps):
            t = i * dt
            
            # Calculate intensities based on current state
            lambda_plus[i] = max(self.params.a0, 
                                self.params.a1 + self.params.a2 * Y[i] + 
                                self.params.a3 * (Z[i] - S[i]))
            
            lambda_minus[i] = max(self.params.a0, 
                                 self.params.a1 + self.params.a2 * Y[i] + 
                                 self.params.a3 * (S[i] - Z[i]))
            
            # Generate Poisson jumps
            N_plus_jump = np.random.poisson(lambda_plus[i] * dt)
            N_minus_jump = np.random.poisson(lambda_minus[i] * dt) if Y[i] > self.params.xi else 0
            
            # Compute optimal strategy
            if with_optimal_strategy:
                nu[i] = alpha_zeta(t, Z[i], Y[i], S[i])
                # Bound the strategy if nu_inf is finite
                if self.params.nu_inf < float('inf'):
                    nu[i] = max(min(nu[i], self.params.nu_inf), -self.params.nu_inf)
            else:
                nu[i] = 0  # No LP activity
            
            # Update state variables
            # External price
            S[i+1] = S[i] + self.params.sigma * S[i] * dW[i]
            
            # Update pool variables
            Y[i+1] = Y[i] + nu[i] * dt + self.params.eta * dB[i] + \
                    self.params.xi * N_plus_jump - self.params.xi * N_minus_jump
            
            # Ensure Y doesn't go below xi
            Y[i+1] = max(Y[i+1], self.params.xi)
            
            # Update constant product
            c = Y[i] * X[i]
            
            # Update X reserves based on constant product formula and trades
            X[i+1] = c / Y[i+1]
            
            # Update Z (pool price)
            Z[i+1] = X[i+1] / Y[i+1]
            
            # Update jump counters
            N_plus[i+1] = N_plus[i] + N_plus_jump
            N_minus[i+1] = N_minus[i] + N_minus_jump
        
        # Calculate LP's reward and venue's performance
        # The reward would be calculated according to the optimal contract
        # in the paper. Here we use a simplified version.
        
        # Calculate total fees collected
        total_fees = self.params.r * (N_plus[-1] + N_minus[-1]) * self.params.xi * Z.mean()
        
        # Calculate LP's costs (simplified)
        lp_costs = np.sum(self.params.a * nu**2) * dt
        
        # Split the fees between LP and venue (50/50 for simplicity)
        lp_reward = total_fees / 2 - lp_costs
        venue_performance = total_fees / 2
        
        return {
            'S': S,
            'Z': Z,
            'Y': Y,
            'X': X,
            'nu': nu,
            'N_plus': N_plus,
            'N_minus': N_minus,
            'lambda_plus': lambda_plus,
            'lambda_minus': lambda_minus,
            'lp_reward': lp_reward,
            'venue_performance': venue_performance,
            'total_fees': total_fees
        }


def run_multiple_simulations(amm: AutomatedMarketMaker, num_simulations: int = 1000) -> Tuple[List[float], List[float]]:
    """Run multiple simulations and collect LP rewards and venue performance"""
    lp_rewards = []

def test_price_difference_violations(amm: AutomatedMarketMaker, num_simulations: int = 1000) -> None:
    """Test if there are violations to the non-negative assumption in the intensities"""
    S_Z_diffs = []
    
    for i in range(num_simulations):
        if i % 100 == 0:
            print(f"Running simulation {i}/{num_simulations}")
        
        results = amm.simulate()
        S = results['S']
        Z = results['Z']
        S_Z_diffs.extend(S - Z)
    
    # Calculate violation boundaries
    violation_boundary = amm.params.a1 / amm.params.a3
    
    # Plot histogram
    plt.figure(figsize=(10, 6))
    counts, bins, _ = plt.hist(S_Z_diffs, bins=100, label='simulated data')
    
    # Highlight violation regions
    plt.axvspan(-float('inf'), -violation_boundary, alpha=0.3, color='red', label='violations')
    plt.axvspan(violation_boundary, float('inf'), alpha=0.3, color='red')
    
    plt.xlabel('price difference')
    plt.ylabel('count')
    plt.legend()
    plt.title('Histogram of Simulated Price Differences')
    plt.savefig('price_difference_violations.png')
    
    # Count violations
    violations_count = np.sum((S_Z_diffs < -violation_boundary) | (S_Z_diffs > violation_boundary))
    violation_percentage = violations_count / len(S_Z_diffs) * 100
    
    print(f"Violations: {violations_count} out of {len(S_Z_diffs)} ({violation_percentage:.2f}%)")
