class MMHP:
    """
    Markov-modulated Hawkes Process with δ-piecewise constant kernel as described in
    "High-Frequency Market Manipulation Detection with a Markov-modulated Hawkes process"
    """
    def __init__(self, M: int, delta: float, mu: np.ndarray, alpha: np.ndarray, 
                 beta: np.ndarray, Q: np.ndarray, pi0: np.ndarray):
        """
        Parameters:
        -----------
        M: int
            Number of states in the Markov chain
        delta: float
            Time step for piecewise constant kernel
        mu: np.ndarray
            Baseline intensities for each state (M,)
        alpha: np.ndarray
            Excitation parameters for each state (M,)
        beta: np.ndarray
            Decay parameters for each state (M,)
        Q: np.ndarray
            Transition rate matrix (M,M)
        pi0: np.ndarray
            Initial state distribution (M,)
        """
        self.M = M
        self.delta = delta
        self.mu = mu
        self.alpha = alpha
        self.beta = beta
        self.Q = Q
        self.pi0 = pi0
        
    def kernel(self, t: float, state: int) -> float:
        """Exponential kernel function"""
        return self.alpha[state] * np.exp(-self.beta[state] * t)
    
    def intensity(self, t: float, history: np.ndarray, state: int) -> float:
        """Compute intensity at time t for state"""
        baseline = self.mu[state]
        if len(history) == 0:
            return baseline
        
        # Implement δ-piecewise constant kernel
        last_event = history[history < t].max() if any(history < t) else 0
        l_t = max(int(np.floor((t - last_event) / self.delta)), 0)
        excitation = sum(self.kernel(last_event + l_t * self.delta - s, state) 
                         for s in history[history < t])
        return baseline + excitation
    
    def simulate(self, T: float, seed: int = None) -> Tuple[np.ndarray, np.ndarray]:
        """
        Simulate a trajectory of the MMHP up to time T
        
        Returns:
        --------
        event_times: np.ndarray
            Array of event times
        states: np.ndarray
            Array of states at each event time
        """

            np.random.seed(seed)
            
        event_times = []
        states = []
        current_state = np.random.choice(self.M, p=self.pi0)
        current_time = 0
        
        # Thinning algorithm for point process simulation with Markov modulation
        while current_time < T:
            # Compute maximum possible intensity
            max_intensity = self.mu[current_state]
            if event_times:
                max_intensity += sum(self.kernel(0, current_state) for _ in event_times)
            
            # Sample next potential event time (exponential with rate = max_intensity)
            dt = np.random.exponential(scale=1/max_intensity)
            current_time += dt
            
            # Sample state transitions during dt
            state_transition_time = np.random.exponential(scale=1/(-self.Q[current_state, current_state]))
            if state_transition_time < dt:
                # State transition occurs before potential event
                current_time = current_time - dt + state_transition_time
                next_state_probs = np.copy(self.Q[current_state, :])
                next_state_probs[current_state] = 0
                next_state_probs = next_state_probs / next_state_probs.sum()
                current_state = np.random.choice(self.M, p=next_state_probs)
                continue
            
            # Check if the potential event actually occurs
            if current_time > T:
                break
                
            actual_intensity = self.intensity(current_time, np.array(event_times), current_state)
            acceptance_prob = actual_intensity / max_intensity
            
            if np.random.random() < acceptance_prob:
                # Accept the event
                event_times.append(current_time)
                states.append(current_state)
        
        return np.array(event_times), np.array(states)
