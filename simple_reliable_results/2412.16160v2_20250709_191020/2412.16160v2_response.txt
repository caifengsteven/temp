class FeatureImportance:
    def __init__(self, method='MDI'):
        self.method = method
        self.importance_vector = None
    
    def calculate_importance(self, X, y):
        if self.method == 'MDI':
            # Use Random Forest for MDI-based feature importance
            rf = RandomForestRegressor(n_estimators=100, random_state=42)
            rf.fit(X, y)
            self.importance_vector = rf.feature_importances_
        elif self.method == 'GD':
            # Implement Gradient Descent based feature importance
            self.importance_vector = self._gradient_descent(X, y)
        return self.importance_vector
    
    def _gradient_descent(self, X, y, learning_rate=0.01, iterations=100):
        # Initialize weights
        theta = np.zeros(X.shape[1])
        m = X.shape[0]
        
        for i in range(iterations):
            # Predict
            y_pred = np.dot(X, theta)
            
            # Calculate error
            error = y_pred - y
            
            # Calculate gradient
            gradient = 2/m * np.dot(X.T, error)
            
            # Update parameters
            theta = theta - learning_rate * gradient
        
        # Return absolute values as importance
        return np.abs(theta)

class DistanceMatrix:
    def __init__(self):
        pass
    
    def correlation_distance(self, X):
        # Calculate correlation matrix
        corr_matrix = np.corrcoef(X, rowvar=False)
        
        # Convert correlation to distance
        dist_matrix = np.sqrt(0.5 * (1 - corr_matrix))
        
        return dist_matrix

class ClusterOptimizer:
    def __init__(self, max_clusters=10, n_init=10):
        self.max_clusters = max_clusters
        self.n_init = n_init
        self.best_n_clusters = None
        self.best_silhouette = None
        self.best_q = -np.inf
        
    def find_optimal_clusters(self, dist_matrix):
        for n_clusters in range(2, self.max_clusters + 1):
            # Apply k-means clustering
            kmeans = KMeans(n_clusters=n_clusters, n_init=self.n_init, random_state=42)
            cluster_labels = kmeans.fit_predict(dist_matrix)
            
            # Calculate silhouette scores
            silhouette_avg = silhouette_score(dist_matrix, cluster_labels)
            sample_silhouette_values = silhouette_samples(dist_matrix, cluster_labels)
            
            # Calculate quality metric q
            silhouette_var = np.var(sample_silhouette_values)
            if silhouette_var == 0:  # Avoid division by zero
                q = 0
            else:
                q = silhouette_avg / np.sqrt(silhouette_var)
            
            if q > self.best_q:
                self.best_q = q
                self.best_n_clusters = n_clusters
                self.best_silhouette = silhouette_avg
        
        return self.best_n_clusters

class RBFNN:
    def __init__(self):
        self.centers = None
        self.weights = None
        self.spreads = None
    
    def fit(self, X, y, n_clusters):
        # Apply k-means to find centers
        kmeans = KMeans(n_clusters=n_clusters, n_init=10, random_state=42)
        kmeans.fit(X)
        self.centers = kmeans.cluster_centers_
        
        # Calculate spreads (sigma) using heuristic approach
        self.spreads = self._calculate_spreads()
        
        # Calculate RBF activations
        rbf_activations = self._calculate_rbf_activations(X)
        
        # Calculate weights using pseudo-inverse
        self.weights = np.linalg.pinv(rbf_activations) @ y
    
    def _calculate_spreads(self):
        # Calculate average distance between centers
        k = len(self.centers)
        if k <= 1:
            return np.array([1.0])  # Default for single center
        
        total_distance = 0
        count = 0
        for i in range(k):
            for j in range(k):
                if i != j:
                    total_distance += np.linalg.norm(self.centers[i] - self.centers[j])
                    count += 1
        
        avg_distance = total_distance / count
        spread = avg_distance / (2 * k)  # Heuristic
        
        return np.full(k, spread)
    
    def _calculate_rbf_activations(self, X):
        # Calculate RBF activations for each point and center
        n_samples = X.shape[0]
        n_centers = len(self.centers)
        rbf_activations = np.zeros((n_samples, n_centers))
        
        for i in range(n_samples):
            for j in range(n_centers):
                distance = np.linalg.norm(X[i] - self.centers[j])
                rbf_activations[i, j] = np.exp(-distance**2 / (2 * self.spreads[j]**2))
        
        return rbf_activations
    
    def predict(self, X):
        # Calculate RBF activations
        rbf_activations = self._calculate_rbf_activations(X)
        
        # Predict using weights
        predictions = rbf_activations @ self.weights
        
        return predictions

class HFTForecaster:
    def __init__(self, feature_set='simple'):
        self.feature_set = feature_set
        self.scaler = StandardScaler()
        self.mdi_importance = FeatureImportance(method='MDI')
        self.gd_importance = FeatureImportance(method='GD')
        self.distance_calculator = DistanceMatrix()
        self.cluster_optimizer = ClusterOptimizer()
        self.rbfnn_mdi = RBFNN()
        self.rbfnn_gd = RBFNN()
        
    def prepare_features(self, df, window_size=100):
        # Select features based on feature set
        if self.feature_set == 'simple':
            features = ['ask_price', 'bid_price', 'ask_volume', 'bid_volume']
        else:  # extended
            features = [col for col in df.columns if col != 'mid_price']
        
        # Use mid-price as target
        target = 'mid_price'
        
        return df[features], df[target]
    
    def online_forecast(self, X, y, window_size=100, overlap=99):
        n_samples = len(y)
        predictions_mdi = np.zeros(n_samples)
        predictions_gd = np.zeros(n_samples)
        mse_mdi = []
        mse_gd = []
        n_clusters_history = []
        
        # Time series split for online learning
        for i in range(window_size, n_samples):
            # Create window
            X_window = X.iloc[i-window_size:i]
            y_window = y.iloc[i-window_size:i]
            
            # Scale features
            X_scaled = self.scaler.fit_transform(X_window)
            
            # Calculate feature importance
            mdi_importances = self.mdi_importance.calculate_importance(X_scaled, y_window.values)
            gd_importances = self.gd_importance.calculate_importance(X_scaled, y_window.values)
            
            # Apply feature importance to input
            X_mdi = X_scaled * mdi_importances
            X_gd = X_scaled * gd_importances
            
            # Calculate distance matrices
            dist_matrix_mdi = self.distance_calculator.correlation_distance(X_mdi)
            dist_matrix_gd = self.distance_calculator.correlation_distance(X_gd)
            
            # Find optimal clusters
            n_clusters_mdi = self.cluster_optimizer.find_optimal_clusters(dist_matrix_mdi)
            n_clusters_gd = self.cluster_optimizer.find_optimal_clusters(dist_matrix_gd)
            n_clusters_history.append((n_clusters_mdi, n_clusters_gd))
            
            # Train RBFNN models
            self.rbfnn_mdi.fit(X_mdi, y_window.values, n_clusters_mdi)
            self.rbfnn_gd.fit(X_gd, y_window.values, n_clusters_gd)
            
            # Make predictions for next point
            X_next_scaled = self.scaler.transform(X.iloc[i:i+1])
            X_next_mdi = X_next_scaled * mdi_importances
            X_next_gd = X_next_scaled * gd_importances
            
            predictions_mdi[i] = self.rbfnn_mdi.predict(X_next_mdi)[0]
            predictions_gd[i] = self.rbfnn_gd.predict(X_next_gd)[0]
            
            # Calculate error
            mse_mdi_i = (predictions_mdi[i] - y.iloc[i])**2
            mse_gd_i = (predictions_gd[i] - y.iloc[i])**2
            
            mse_mdi.append(mse_mdi_i)
            mse_gd.append(mse_gd_i)
            
            # Print progress every 500 steps
            if i % 500 == 0:
                print(f"Processed {i}/{n_samples} samples")
        
        # Calculate overall metrics
        valid_indices = ~np.isnan(predictions_mdi) & ~np.isnan(predictions_gd)
        valid_predictions_mdi = predictions_mdi[valid_indices]
        valid_predictions_gd = predictions_gd[valid_indices]
        valid_y = y.iloc[valid_indices].values
        
        mse_mdi_overall = np.mean((valid_predictions_mdi - valid_y)**2)
        mse_gd_overall = np.mean((valid_predictions_gd - valid_y)**2)
        
        rmse_mdi = np.sqrt(mse_mdi_overall)
        rmse_gd = np.sqrt(mse_gd_overall)
        
        return {
            'predictions_mdi': predictions_mdi,
            'predictions_gd': predictions_gd,
            'mse_mdi': mse_mdi_overall,
            'mse_gd': mse_gd_overall,
            'rmse_mdi': rmse_mdi,
            'rmse_gd': rmse_gd,
            'n_clusters_history': n_clusters_history
        }
