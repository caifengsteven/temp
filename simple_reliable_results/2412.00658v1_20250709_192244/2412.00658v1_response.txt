import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

class HestonModel:
    """Heston stochastic volatility model implementation"""
    
    def __init__(self, theta=0.03, kappa=18, sigma_v=0.03, rho=0, lam=-7):
        """Initialize Heston model with parameters"""
        self.theta = theta      # Long-term variance
        self.kappa = kappa      # Mean reversion speed
        self.sigma_v = sigma_v  # Volatility of volatility
        self.rho = rho          # Correlation between spot and vol
        self.lam = lam          # Volatility risk premium
        
        # Risk-neutral parameters
        self.kappa_star = self.kappa + self.lam
        self.theta_star = self.kappa * self.theta / self.kappa_star
    
    def simulate_paths(self, s0, v0, rf, T, dt, n_paths=1):
        """Simulate spot price and volatility paths using Euler discretization"""
        n_steps = int(T/dt)
        
        # Arrays to store paths
        s = np.zeros((n_paths, n_steps+1))
        v = np.zeros((n_paths, n_steps+1))
        
        # Initial values
        s[:, 0] = s0
        v[:, 0] = v0
        
        # Generate correlated random numbers
        z1 = np.random.normal(0, 1, (n_paths, n_steps))
        z2 = self.rho * z1 + np.sqrt(1 - self.rho**2) * np.random.normal(0, 1, (n_paths, n_steps))
        
        # Simulate paths
        for t in range(n_steps):
            # Ensure variance is positive
            v[:, t] = np.maximum(v[:, t], 1e-10)
            
            # Update variance
            v[:, t+1] = v[:, t] + self.kappa * (self.theta - v[:, t]) * dt + \
                        self.sigma_v * np.sqrt(v[:, t] * dt) * z2[:, t]
            
            # Update log-price
            s[:, t+1] = s[:, t] * np.exp((rf - 0.5 * v[:, t]) * dt + 
                                         np.sqrt(v[:, t] * dt) * z1[:, t])
            
        return s, v
    
    def bipower_variation(self, log_returns):
        """Calculate bipower variation from log returns"""
        abs_returns = np.abs(log_returns)
        bv = np.sum(abs_returns[1:] * abs_returns[:-1]) * np.pi / 2
        return bv
    
    def option_price(self, s, v, r, tau, k):
        """Approximate European call option price using Monte Carlo simulation
        
        This is a simplified version, in practice a more accurate method would be used
        """
        # Use risk-neutral parameters for option pricing
        s_paths, _ = self.simulate_paths(s, v, r, tau, tau/100, n_paths=10000)
        
        # Calculate payoffs at maturity
        payoffs = np.maximum(s_paths[:, -1] - k, 0)
        
        # Discount payoffs
        price = np.exp(-r * tau) * np.mean(payoffs)
        
        return price
    
    def black_scholes_implied_vol(self, s, k, r, tau, option_price):
        """Calculate Black-Scholes implied volatility using bisection method"""
        def bs_price(sigma):
            d1 = (np.log(s/k) + (r + 0.5 * sigma**2) * tau) / (sigma * np.sqrt(tau))
            d2 = d1 - sigma * np.sqrt(tau)
            return s * norm.cdf(d1) - k * np.exp(-r * tau) * norm.cdf(d2)
        
        # Initial guesses for volatility
        sigma_low, sigma_high = 0.001, 2.0
        
        # Bisection loop
        for _ in range(50):
            sigma_mid = 0.5 * (sigma_low + sigma_high)
            price_mid = bs_price(sigma_mid)
            
            if abs(price_mid - option_price) < 1e-8:
                return sigma_mid
            
            if price_mid > option_price:
                sigma_high = sigma_mid
            else:
                sigma_low = sigma_mid
                
        return 0.5 * (sigma_low + sigma_high)


class ModularABI:
    """Implementation of Modular Approximate Bayesian Inference for option pricing"""
    
    def __init__(self, prior_samples=10000):
        """Initialize with number of prior samples to generate"""
        self.prior_samples = prior_samples
        self.reference_table = None
        self.gmm_model = None
    
    def generate_priors(self):
        """Generate prior samples for Heston model parameters"""
        # Uniform priors as described in the paper
        theta = np.random.uniform(0, 0.2, self.prior_samples)
        kappa = np.random.uniform(0, 50, self.prior_samples)
        
        # Conditional priors
        lam = np.array([np.random.uniform(-k, 0) for k in kappa])
        sigma_v = np.array([np.random.uniform(0, 2*k*t) for k, t in zip(kappa, theta)])
        
        return np.column_stack((theta, kappa, sigma_v, lam))
    
    def simulate_data(self, params, T=500, s0=100, r=0.01):
        """Simulate data from Heston model with given parameters"""
        theta, kappa, sigma_v, lam = params
        
        # Create Heston model
        model = HestonModel(theta=theta, kappa=kappa, sigma_v=sigma_v, lam=lam)
        
        # Initial variance drawn from stationary distribution
        v0 = theta
        
        # Simulate daily data
        s_path, v_path = model.simulate_paths(s0, v0, r, T, 1/252)
        
        # Calculate daily returns
        daily_returns = np.diff(np.log(s_path[0])) 
        
        # Calculate simulated BV (high-frequency proxy)
        # In real implementation we would simulate intraday data
        # For simplicity, we add noise to true variance
        bv_series = v_path[0, 1:] + 0.2 * np.random.randn(len(v_path[0, 1:]))
        bv_series = np.maximum(bv_series, 1e-10)  # Ensure positive
        
        # Simulate option prices for different maturities and strikes
        option_prices = []
        maturities = [5/252, 20/252, 60/252]  # 5, 20, 60 days in years
        
        for t in range(0, T, 10):  # Sample every 10 days to reduce computation
            s_t = s_path[0, t]
            v_t = v_path[0, t]
            
            for tau in maturities:
                # Range of moneyness
                for k_pct in [0.95, 0.975, 1.0, 1.025, 1.05]:
                    k = k_pct * s_t
                    
                    # Theoretical price
                    price = model.option_price(s_t, v_t, r, tau, k)
                    
                    # Add record
                    option_prices.append({
                        'time': t,
                        'spot': s_t,
                        'vol': v_t,
                        'tau': tau,
                        'strike': k,
                        'price': price,
                        'moneyness': np.log(s_t/k)
                    })
        
        option_df = pd.DataFrame(option_prices)
        
        return {
            'returns': daily_returns,
            'bv': bv_series,
            'options': option_df,
            'spot': s_path[0],
            'vol': v_path[0]
        }
    
    def calculate_summaries(self, data):
        """Calculate summary statistics from data"""
        # Return-based summaries
        ret_mean = np.mean(data['returns'])
        ret_std = np.std(data['returns'])
        ret_skew = np.mean(data['returns']**3) / ret_std**3
        ret_kurt = np.mean(data['returns']**4) / ret_std**4
        
        # BV-based summaries
        bv_mean = np.mean(data['bv'])
        bv_std = np.std(data['bv'])
        bv_skew = np.mean((data['bv'] - bv_mean)**3) / bv_std**3
        bv_kurt = np.mean((data['bv'] - bv_mean)**4) / bv_std**4
        bv_acf1 = np.corrcoef(data['bv'][1:], data['bv'][:-1])[0, 1]
        
        # Option-based summaries 
        # Group by maturity
        option_summaries = []
        for tau in np.unique(data['options']['tau']):
            tau_options = data['options'][data['options']['tau'] == tau]
            
            # Calculate BSIV for each option
            tau_options['bsiv'] = tau_options.apply(
                lambda x: HestonModel().black_scholes_implied_vol(
                    x['spot'], x['strike'], 0.01, x['tau'], x['price']), axis=1)
            
            # Basic summaries of BSIV
            option_summaries.extend([
                np.mean(tau_options['bsiv']),
                np.std(tau_options['bsiv']),
                np.percentile(tau_options['bsiv'], 25),
                np.percentile(tau_options['bsiv'], 75)
            ])
            
            # Add ATM-ITM-OTM spread measures
            atm = tau_options[tau_options['moneyness'].abs() < 0.01]['bsiv'].mean()
            itm = tau_options[tau_options['moneyness'] > 0.02]['bsiv'].mean()
            otm = tau_options[tau_options['moneyness'] < -0.02]['bsiv'].mean()
            
            option_summaries.extend([
                otm - atm,  # Smile left side
                itm - atm   # Smile right side
            ])
        
        # Combine all summaries
        return np.array([
            ret_mean, ret_std, ret_skew, ret_kurt,
            bv_mean, bv_std, bv_skew, bv_kurt, bv_acf1,
            *option_summaries
        ])
    
    def build_reference_table(self):
        """Build reference table for modular ABI"""
        print("Generating prior samples...")
        params = self.generate_priors()
        
        # Storage for summary statistics
        summaries = []
        
        print("Simulating data and calculating summaries...")
        for i, p in enumerate(params):
            if i % 100 == 0:
                print(f"Processing sample {i}/{len(params)}")
            
            try:
                data = self.simulate_data(p)
                summary = self.calculate_summaries(data)
                summaries.append(summary)
            except Exception as e:
                print(f"Error in sample {i}: {e}")
                # Use a placeholder with zeros
                summaries.append(np.zeros(22))
        
        # Convert to numpy array
        summaries = np.array(summaries)
        
        # Store reference table
        self.reference_table = {
            'params': params,
            'summaries': summaries
        }
        
        print("Reference table built")
        return self.reference_table
    
    def fit_gaussian_mixture(self, n_components=10):
        """Fit Gaussian mixture model to joint distribution of parameters and summaries"""
        if self.reference_table is None:
            raise ValueError("Reference table must be built first")
        
        print("Fitting Gaussian mixture model...")
        # Combine parameters and summaries
        joint_data = np.hstack((self.reference_table['params'], self.reference_table['summaries']))
        
        # Fit GMM
        self.gmm_model = GaussianMixture(n_components=n_components, covariance_type='full')
        self.gmm_model.fit(joint_data)
        
        print(f"GMM with {n_components} components fitted")
        return self.gmm_model
    
    def sample_posterior(self, observed_summaries, n_samples=1000):
        """Sample from posterior using fitted GMM"""
        if self.gmm_model is None:
            raise ValueError("GMM model must be fitted first")
        
        # Get dimensions
        n_params = 4  # theta, kappa, sigma_v, lam
        n_summaries = len(observed_summaries)
        
        # Extract GMM parameters
        weights = self.gmm_model.weights_
        means = self.gmm_model.means_
        covariances = self.gmm_model.covariances_
        
        # Allocate storage for posterior samples
        posterior_samples = np.zeros((n_samples, n_params))
        
        # Sample component indices according to weights
        component_indices = np.random.choice(len(weights), size=n_samples, p=weights)
        
        # For each component
        for k in range(len(weights)):
            # Get indices of samples from this component
            k_indices = np.where(component_indices == k)[0]
            if len(k_indices) == 0:
                continue
                
            # Split mean and covariance for conditioning
            mu_x = means[k, :n_params]
            mu_y = means[k, n_params:]
            
            sigma_xx = covariances[k, :n_params, :n_params]
            sigma_xy = covariances[k, :n_params, n_params:]
            sigma_yx = covariances[k, n_params:, :n_params]
            sigma_yy = covariances[k, n_params:, n_params:]
            
            # Conditional mean: mu_x|y = mu_x + sigma_xy * sigma_yy^-1 * (y - mu_y)
            # Conditional covariance: sigma_x|y = sigma_xx - sigma_xy * sigma_yy^-1 * sigma_yx
            
            try:
                sigma_yy_inv = np.linalg.inv(sigma_yy)
                
                conditional_mean = mu_x + sigma_xy.dot(sigma_yy_inv).dot(observed_summaries - mu_y)
                conditional_cov = sigma_xx - sigma_xy.dot(sigma_yy_inv).dot(sigma_yx)
                
                # Sample from conditional distribution
                posterior_samples[k_indices] = np.random.multivariate_normal(
                    conditional_mean, conditional_cov, size=len(k_indices))
            except np.linalg.LinAlgError:
                print(f"Matrix inversion failed for component {k}, using prior samples instead")
                # Fallback to prior samples for this component
                random_indices = np.random.choice(len(self.reference_table['params']), size=len(k_indices))
                posterior_samples[k_indices] = self.reference_table['params'][random_indices]
        
        return posterior_samples
    
    def predict_option_prices(self, posterior_samples, s_t, v_t, r, tau, strikes):
        """Generate predictive distribution for option prices"""
        n_samples = len(posterior_samples)
        n_strikes = len(strikes)
        
        # Storage for option price predictions
        predicted_prices = np.zeros((n_samples, n_strikes))
        
        for i, params in enumerate(posterior_samples):
            theta, kappa, sigma_v, lam = params
            
            # Create model with sampled parameters
            try:
                model = HestonModel(theta=theta, kappa=kappa, sigma_v=sigma_v, lam=lam)
                
                # Calculate option prices for each strike
                for j, k in enumerate(strikes):
                    predicted_prices[i, j] = model.option_price(s_t, v_t, r, tau, k)
            except Exception as e:
                print(f"Error in prediction sample {i}: {e}")
                # Use zeros for this sample
                predicted_prices[i] = 0
        
        return predicted_prices
