```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score, mean_squared_error, mean_absolute_error
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, LSTM, Dropout
import math
import warnings
warnings.filterwarnings('ignore')

class StockReturnModels:
    def __init__(self, seed=42):
        self.seed = seed
        np.random.seed(seed)
        tf.random.set_seed(seed)
        self.models = {
            'F-F3': None,
            'Carhart4': None,
            'F-F5': None,
            'LSTM': None
        }
        self.results = {}
        
    def generate_data(self, n_samples=240, sectors=['Manuf', 'Hitec', 'Other']):
        """
        Generate synthetic data for testing the models based on the paper's factors
        Returns a dictionary with dataframes for each sector
        """
        data = {}
        
        # Factor correlations and volatilities based on realistic market conditions
        # Market factor has highest vol, followed by size, value, momentum, etc.
        for sector in sectors:
            # Different sectors have different factor exposures
            if sector == 'Manuf':
                betas = {
                    'Rmkt_Rf': 0.95, 
                    'SMB': 0.08, 
                    'HML': 0.02, 
                    'RMW': 0.10, 
                    'CMA': 0.01,
                    'MOM': -0.06
                }
                noise_level = 1.5  # Controls unexplained variance
            elif sector == 'Hitec':
                betas = {
                    'Rmkt_Rf': 0.90, 
                    'SMB': 0.02, 
                    'HML': -0.19, 
                    'RMW': -0.08, 
                    'CMA': -0.06,
                    'MOM': -0.02
                }
                noise_level = 1.8  # Higher unexplained variance for tech stocks
            else:  # Other
                betas = {
                    'Rmkt_Rf': 0.83, 
                    'SMB': -0.03, 
                    'HML': 0.30, 
                    'RMW': -0.10, 
                    'CMA': -0.07,
                    'MOM': 0.05
                }
                noise_level = 1.2  # Lower unexplained variance
            
            # Generate factor data with realistic distributions
            Rmkt_Rf = np.random.normal(0.5, 4.5, n_samples)  # Market excess return (%)
            SMB = np.random.normal(0.2, 3.0, n_samples)      # Size factor
            HML = np.random.normal(0.3, 3.2, n_samples)      # Value factor
            RMW = np.random.normal(0.2, 2.5, n_samples)      # Profitability factor
            CMA = np.random.normal(0.1, 2.2, n_samples)      # Investment factor
            MOM = np.random.normal(0.4, 4.0, n_samples)      # Momentum factor
            Rf = np.random.normal(0.2, 0.1, n_samples)       # Risk-free rate
            
            # Add some time-series autocorrelation to factors
            for i in range(1, n_samples):
                Rmkt_Rf[i] = 0.1 * Rmkt_Rf[i-1] + 0.9 * Rmkt_Rf[i]
                SMB[i] = 0.2 * SMB[i-1] + 0.8 * SMB[i]
                HML[i] = 0.15 * HML[i-1] + 0.85 * HML[i]
                MOM[i] = -0.1 * MOM[i-1] + 1.1 * MOM[i]  # Momentum tends to reverse
            
            # Generate returns based on the factor model plus noise
            noise = np.random.normal(0, noise_level, n_samples)
            Ri = Rf + betas['Rmkt_Rf'] * Rmkt_Rf + betas['SMB'] * SMB + betas['HML'] * HML + \
                 betas['RMW'] * RMW + betas['CMA'] * CMA + betas['MOM'] * MOM + noise
            
            # Create dataframe
            df = pd.DataFrame({
                'Ri': Ri,
                'Rf': Rf,
                'Rmkt_Rf': Rmkt_Rf,
                'SMB': SMB,
                'HML': HML,
                'RMW': RMW,
                'CMA': CMA,
                'MOM': MOM,
                'Ri_Rf': Ri - Rf  # Excess return
            })
            
            # Add time component for LSTM
            df['time'] = np.arange(n_samples)
            
            data[sector] = df
            
        return data
    
    def train_factor_models(self, data, sector):
        """
        Train the three factor models mentioned in the paper
        """
        df = data[sector]
        
        # Define X and y for each model
        X_ff3 = df[['Rmkt_Rf', 'SMB', 'HML']]
        X_car4 = df[['Rmkt_Rf', 'SMB', 'HML', 'MOM']]
        X_ff5 = df[['Rmkt_Rf', 'SMB', 'HML', 'RMW', 'CMA']]
        y = df['Ri_Rf']
        
        # Split data
        X_ff3_train, X_ff3_test, y_train, y_test = train_test_split(X_ff3, y, test_size=0.3, random_state=self.seed)
        X_car4_train, X_car4_test, _, _ = train_test_split(X_car4, y, test_size=0.3, random_state=self.seed)
        X_ff5_train, X_ff5_test, _, _ = train_test_split(X_ff5, y, test_size=0.3, random_state=self.seed)
        
        # Train models
        ff3_model = LinearRegression()
        ff3_model.fit(X_ff3_train, y_train)
        
        car4_model = LinearRegression()
        car4_model.fit(X_car4_train, y_train)
        
        ff5_model = LinearRegression()
        ff5_model.fit(X_ff5_train, y_train)
        
        # Store models
        self.models['F-F3'] = {'model': ff3_model, 'X_test': X_ff3_test, 'y_test': y_test}
        self.models['Carhart4'] = {'model': car4_model, 'X_test': X_car4_test, 'y_test': y_test}
        self.models['F-F5'] = {'model': ff5_model, 'X_test': X_ff5_test, 'y_test': y_test}
        
        # Evaluate models
        results = {}
        for name, model_data in self.models.items():
            if name == 'LSTM':
                continue
                
            model = model_data['model']
            X_test = model_data['X_test']
            
            # Make predictions
            y_pred = model.predict(X_test)
            
            # Calculate metrics
            r2 = r2_score(y_test, y_pred)
            rmse = np.sqrt(mean_squared_error(y_test, y_pred))
            mae = mean_absolute_error(y_test, y_pred)
            
            results[name] = {
                'R-squared': r2,
                'RMSE': rmse,
                'MAE': mae,
                'Coefficients': {
                    'Intercept': model.intercept_
                }
            }
            
            # Add coefficients with names
            feature_names = X_test.columns
            for i, feature in enumerate(feature_names):
                results[name]['Coefficients'][feature] = model.coef_[i]
                
        return results
    
    def train_lstm_model(self, data, sector, lookback=6):
        """
        Train LSTM model for time series forecasting
        """
        df = data[sector]
        
        # Create sequences for LSTM
        X_cols = ['Rmkt_Rf', 'SMB', 'HML', 'RMW', 'CMA', 'MOM']
        y_col = 'Ri_Rf'
        
        sequences = []
        targets = []
        
        for i in range(len(df) - lookback):
            seq = df[X_cols].iloc[i:i+lookback].values
            target = df[y_col].iloc[i+lookback]
            sequences.append(seq)
            targets.append(target)
        
        X = np.array(sequences)
        y = np.array(targets)
        
        # Split data
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=self.seed)
        
        # Define LSTM model
        model = Sequential()
        model.add(LSTM(50, activation='relu', input_shape=(X_train.shape[1], X_train.shape[2]), return_sequences=True))
        model.add(Dropout(0.2))
        model.add(LSTM(50, activation='relu'))
        model.add(Dropout(0.2))
        model.add(Dense(1))
        
        # Compile model
        model.compile(optimizer='adam', loss='mse')
        
        # Train model
        history = model.fit(
            X_train, y_train,
            epochs=50,
            batch_size=32,
            validation_split=0.2,
            verbose=0
        )
        
        # Make predictions
        y_pred = model.predict(X_test).flatten()
        
        # Calculate metrics
        r2 = r2_score(y_test, y_pred)
        rmse = np.sqrt(mean_squared_error(y_test, y_pred))
        mae = mean_absolute_error(y_test, y_pred)
        
        # Store model and results
        self.models['LSTM'] = {'model': model, 'X_test': X_test, 'y_test': y_test}
        
        lstm_results = {
            'R-squared': r2,
            'RMSE': rmse,
            'MAE': mae,
            'History': history.history
        }
        
        return lstm_results
    
    def evaluate_sector(self, sector):
        """
        Evaluate all models for a given sector
        """
        data = self.generate_data(sectors=[sector])
        
        # Train factor models
        factor_results = self.train_factor_models(data, sector)
        
        # Train LSTM model
        lstm_results = self.train_lstm_model(data, sector)
        
        # Combine results
        all_results = factor_results.copy()
        all_results['LSTM'] = lstm_results
        
        self.results[sector] = all_results
        return all_results
    
    def print_results(self):
        """
        Print formatted results for all sectors and models
        """
        for sector, results in self.results.items():
            print(f"\n=== {sector} Sector Results ===")
            
            # Create comparison table
            metrics_df = pd.DataFrame({
                'Model': [],
                'R-squared': [],
                'RMSE': [],
                'MAE': []
            })
            
            for model_name, model_results in results.items():
                new_row = {
                    'Model': model_name,
                    'R-squared': round(model_results['R-squared'], 3),
                    'RMSE': round(model_results['RMSE'], 3),
                    'MAE': round(model_results['MAE'], 3)
                }
                metrics_df = pd.concat([metrics_df, pd.DataFrame([new_row])], ignore_index=True)
            
            print(metrics_df)
            
            # Print factor model coefficients
            for model_name in ['F-F3', 'Carhart4', 'F-F5']:
                if model_name in results:
                    print(f"\n{model_name} Coefficients:")
                    coefs = results[model_name]['Coefficients']
                    for factor, value in coefs.items():
                        print(f"  {factor}: {value:.4f}")
    
    def plot_comparison(self):
        """
        Plot comparison of model performance across sectors
        """
        if not self.results:
            print("No results to plot. Run evaluate_sector first.")
            return
            
        metrics = ['R-squared', 'RMSE', 'MAE']
        fig, axes = plt.subplots(1, 3, figsize=(18, 6))
        
        for i, metric in enumerate(metrics):
            data = []
            models = ['F-F3', 'Carhart4', 'F-F5', 'LSTM']
            
            for sector in self.results.keys():
                sector_data = []
                for model in models:
                    sector_data.append(self.results[sector][model][metric])
                data.append(sector_data)
            
            axes[i].set_title(f'{metric} Comparison')
            axes[i].set_xlabel('Model')
            axes[i].set_ylabel(metric)
            
            x = np.arange(len(models))
            width = 0.2
            
            for j, sector in enumerate(self.results.keys()):
                axes[i].bar(x + j*width - width, data[j], width=width, label=sector)
            
            axes[i].set_xticks(x)
            axes[i].set_xticklabels(models)
            axes[i].legend()
        
        plt.tight_layout()
        plt.show()


def main():
```
