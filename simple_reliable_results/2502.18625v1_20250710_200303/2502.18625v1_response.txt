class OrderBookSimulator:
    """Simulates a basic order book with bid/ask queues and prices"""
    
    def __init__(self, initial_price=35000, initial_queue_size=100000):
        self.mid_price = initial_price
        self.tick_size = 0.1
        self.bid_price = initial_price - self.tick_size/2
        self.ask_price = initial_price + self.tick_size/2
        self.bid_queue = initial_queue_size * np.random.uniform(0.8, 1.2)
        self.ask_queue = initial_queue_size * np.random.uniform(0.8, 1.2)
        self.history = []
        self.trades = []
        self.current_time = datetime.now()
        
    def update_queues(self):
        """Update bid and ask queues with some random drift"""
        self.bid_queue = max(10000, self.bid_queue + np.random.normal(0, self.bid_queue * 0.05))
        self.ask_queue = max(10000, self.ask_queue + np.random.normal(0, self.ask_queue * 0.05))
        
    def record_state(self):
        """Record current state of the order book"""
        state = {
            'timestamp': self.current_time,
            'mid_price': self.mid_price,
            'bid_price': self.bid_price,
            'ask_price': self.ask_price,
            'bid_queue': self.bid_queue,
            'ask_queue': self.ask_queue,
            'imbalance': (self.bid_queue - self.ask_queue) / (self.bid_queue + self.ask_queue)
        }
        self.history.append(state)
        return state
    
    def simulate_price_change(self, direction):
        """Simulate a price change in the given direction"""
        self.mid_price += direction * self.tick_size
        self.bid_price = self.mid_price - self.tick_size/2
        self.ask_price = self.mid_price + self.tick_size/2
        
        # Reset the queue that was depleted
        if direction > 0:  # Price went up, ask queue was depleted
            self.ask_queue = np.random.uniform(30000, 70000)
        else:  # Price went down, bid queue was depleted
            self.bid_queue = np.random.uniform(30000, 70000)
    
    def simulate_taker_order(self, is_buy, size):
        """Simulate a taker order hitting the book"""
        if is_buy:
            self.ask_queue -= size
            if self.ask_queue <= 0:
                self.simulate_price_change(1)  # Price goes up
                return True
        else:
            self.bid_queue -= size
            if self.bid_queue <= 0:
                self.simulate_price_change(-1)  # Price goes down
                return True
        return False
    
    def simulate_step(self):
        """Simulate one step of the order book"""
        self.current_time += timedelta(milliseconds=random.randint(10, 100))
        self.update_queues()
        
        # Random taker orders
        if random.random() < 0.3:  # 30% chance of a taker order
            is_buy = random.random() < 0.5
            size = np.random.exponential(10000)  # Random size with mean 10000
            price_changed = self.simulate_taker_order(is_buy, size)
            
            self.trades.append({
                'timestamp': self.current_time,
                'is_buy': is_buy,
                'size': size,
                'price': self.ask_price if is_buy else self.bid_price,
                'price_changed': price_changed
            })
        
        return self.record_state()

class MakerOrder:
    """Represents a maker order in the book"""
    
    def __init__(self, is_buy, price, submission_time, liquidity_ahead):
        self.is_buy = is_buy
        self.price = price
        self.submission_time = submission_time
        self.liquidity_ahead = liquidity_ahead
        self.fill_time = None
        self.queue_position = None
        self.is_filled = False
        self.is_canceled = False
        self.features = {}
    
    def calculate_return(self, reference_price, horizon_seconds=5):
        """Calculate the return from this order"""
        if not self.is_filled:
            return None
        
        direction = 1 if self.is_buy else -1
        return direction * (reference_price / self.price - 1) * 10000  # bp

class ReversalDetector:
    """Detector for reversal opportunities in the order book"""
    
    def __init__(self):
        self.model = LogisticRegression(max_iter=1000)
        self.is_trained = False
    
    def extract_features(self, ob_history, trades, current_time, window_sizes=[0.1, 1, 5, 30]):
        """Extract features from order book history and trades"""
        # This is a simplified version of the feature extraction in the paper
        features = {}
        
        # Convert history to DataFrame for easier manipulation
        history_df = pd.DataFrame(ob_history)
        history_df['timestamp'] = pd.to_datetime(history_df['timestamp'])
        history_df = history_df.set_index('timestamp')
        
        trades_df = pd.DataFrame(trades)
        if not trades_df.empty:
            trades_df['timestamp'] = pd.to_datetime(trades_df['timestamp'])
            trades_df = trades_df.set_index('timestamp')
        
        # Latest snapshot
        latest = history_df.iloc[-1]
        features['imbalance'] = latest['imbalance']
        features['bid_queue'] = latest['bid_queue']
        features['ask_queue'] = latest['ask_queue']
        
        # Calculate features for each window size
        for window in window_sizes:
            window_history = history_df[history_df.index >= (current_time - timedelta(seconds=window))]
            
            if not window_history.empty:
                # Price dynamics
                features[f'volatility_{window}s'] = window_history['mid_price'].std()
                features[f'price_range_{window}s'] = window_history['mid_price'].max() - window_history['mid_price'].min()
                features[f'price_direction_{window}s'] = window_history['mid_price'].iloc[-1] - window_history['mid_price'].iloc[0]
                
                # Queue dynamics
                features[f'bid_queue_change_{window}s'] = window_history['bid_queue'].iloc[-1] - window_history['bid_queue'].iloc[0]
                features[f'ask_queue_change_{window}s'] = window_history['ask_queue'].iloc[-1] - window_history['ask_queue'].iloc[0]
                features[f'imbalance_change_{window}s'] = window_history['imbalance'].iloc[-1] - window_history['imbalance'].iloc[0]
            
            # Trade features
            if not trades_df.empty:
                window_trades = trades_df[trades_df.index >= (current_time - timedelta(seconds=window))]
                
                if not window_trades.empty:
                    buy_trades = window_trades[window_trades['is_buy']]
                    sell_trades = window_trades[~window_trades['is_buy']]
                    
                    features[f'buy_count_{window}s'] = len(buy_trades)
                    features[f'sell_count_{window}s'] = len(sell_trades)
                    features[f'buy_volume_{window}s'] = buy_trades['size'].sum() if not buy_trades.empty else 0
                    features[f'sell_volume_{window}s'] = sell_trades['size'].sum() if not sell_trades.empty else 0
                    features[f'trade_imbalance_{window}s'] = (
                        (features[f'buy_volume_{window}s'] - features[f'sell_volume_{window}s']) / 
                        (features[f'buy_volume_{window}s'] + features[f'sell_volume_{window}s'] + 1e-10)
                    )
        
        return features
    
    def train(self, training_data):
        """Train the model using labeled training data"""
        X = pd.DataFrame([order.features for order in training_data])
        y = np.array([1 if order.is_filled and order.calculate_return(order.features['future_price']) > 0 else 0 
                     for order in training_data])
        
        self.model.fit(X, y)
        self.is_trained = True
        self.feature_names = X.columns
        
        # Print coefficients for interpretation
        coeffs = pd.Series(self.model.coef_[0], index=X.columns)
        print("Top positive coefficients:")
        print(coeffs.sort_values(ascending=False).head(10))
        print("\nTop negative coefficients:")
        print(coeffs.sort_values().head(10))
    
    def predict_reversal_probability(self, features):
        """Predict the probability of a reversal given features"""
        if not self.is_trained:
            raise ValueError("Model is not trained yet")
        
        features_df = pd.DataFrame([features])
        # Ensure we have all required features
        for col in self.feature_names:
            if col not in features_df.columns:
                features_df[col] = 0
        
        # Only use columns the model was trained on
        features_df = features_df[self.feature_names]
        
        return self.model.predict_proba(features_df)[0, 1]
