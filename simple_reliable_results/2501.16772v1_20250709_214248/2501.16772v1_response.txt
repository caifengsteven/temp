    """
    
    def __init__(self, timeframe='daily'):
        """
        Initialize the strategy parameters based on timeframe.

        elif timeframe == 'intraday_long':  # 1 hour to 1 day
            self.b = 0.00132   # Linear coefficient (positive: trends persist)
            self.c = -0.00039  # Cubic coefficient (negative: strong trends revert)
            self.e = -0.00071  # Discontinuity (smaller at longer horizons)
        elif timeframe == 'daily':  # days to 1-2 years
            self.b = 0.0129    # Linear coefficient (positive: trends persist)
            self.c = -0.0062   # Cubic coefficient (negative: strong trends revert)
            self.e = 0         # No discontinuity effect in daily data
        elif timeframe == 'monthly_short':  # Up to 1 year
            self.b = 0.065     # Linear coefficient (positive: trends persist)
            self.c = -0.014    # Cubic coefficient (negative: strong trends revert)
            self.e = 0         # No discontinuity effect in monthly data
        elif timeframe == 'monthly_long':  # 2-16 years
            self.b = -0.027    # Linear coefficient (negative: trends revert)
            self.c = 0.010     # Cubic coefficient (positive: strong trends persist)
            self.e = 0         # No discontinuity effect in monthly data
        
    def calculate_trend_strength(self, returns, horizon):
        """
        Calculate trend strength (phi) as t-statistics for a given horizon.
        
        Args:
            returns: Array of normalized returns
            horizon: Number of periods to consider for trend
            
        Returns:
            Trend strength as t-statistics
        """
        # Using exponentially weighted sum of past returns
        weights = np.zeros(len(returns))
        for i in range(1, len(returns)):
            n = i - 1
            weights[n] = n * np.exp(-2 * n / horizon)
        
        # Normalize weights
        weights = weights / np.sqrt(np.sum(weights**2))
        
        # Calculate trend strength
        trend_strength = np.zeros(len(returns))
        for i in range(horizon, len(returns)):
            weighted_returns = np.sum(weights[:horizon] * returns[i-horizon:i])
            trend_strength[i] = weighted_returns
            
        return trend_strength
    
    def predict_next_return(self, trend_strength):
        """
        Predict the next return based on current trend strength.
        
        Args:
            trend_strength: Current trend strength (phi)
            
        Returns:
            Predicted next return
        """
        # Cubic polynomial model from the paper
        sign_phi = np.sign(trend_strength)
        return self.b * trend_strength + self.c * trend_strength**3 + self.e * sign_phi
    
    def backtest(self, returns, horizon):
        """
        Backtest the strategy on historical returns.
        
        Args:
            returns: Array of normalized returns
            horizon: Number of periods to consider for trend
            
        Returns:
            Strategy returns and metrics
        """
        # Calculate trend strength
        trend_strength = self.calculate_trend_strength(returns, horizon)
        
        # Generate signals (1 for long, -1 for short, 0 for no position)
        signals = np.zeros(len(returns))
        for i in range(horizon, len(returns)-1):
            prediction = self.predict_next_return(trend_strength[i])
            signals[i+1] = np.sign(prediction)
        
        # Calculate strategy returns
        strategy_returns = signals[1:] * returns[1:]
        
        # Calculate metrics
        sharpe_ratio = np.mean(strategy_returns) / np.std(strategy_returns) * np.sqrt(252)
        cumulative_returns = np.cumprod(1 + strategy_returns) - 1
        
        return {
            'signals': signals,
            'strategy_returns': strategy_returns,
            'sharpe_ratio': sharpe_ratio,
            'cumulative_returns': cumulative_returns
        }

    elif timeframe in ['monthly_short', 'monthly_long']:

        autocorrelation = 0.15 if trend_regime else -0.1
    
    # Generate returns with autocorrelation
    returns = np.zeros(n_periods)
    returns[0] = np.random.normal(0, volatility)
    
    for i in range(1, n_periods):
        # Include autocorrelation effect
        returns[i] = autocorrelation * returns[i-1] + np.random.normal(0, volatility)
        
        # Add occasional regime shifts to simulate trend persistence and reversion
        if random.random() < 0.01:  # 1% chance of regime shift
            if trend_regime:
                # Add trend persistence for weak trends, reversion for strong trends
                if abs(returns[i-1]) < volatility:
                    returns[i] += 2 * np.sign(returns[i-1]) * volatility
                else:
                    returns[i] -= np.sign(returns[i-1]) * volatility
            else:
                # Add trend reversion for weak trends, persistence for strong trends
                if abs(returns[i-1]) < volatility:
                    returns[i] -= np.sign(returns[i-1]) * volatility
                else:
                    returns[i] += np.sign(returns[i-1]) * volatility
    
    # Generate prices from returns
    prices = 100 * np.cumprod(1 + returns)
    
    # Create DataFrame
    if timeframe.startswith('intraday'):
        start_date = datetime(2023, 1, 1, 9, 30)
        if timeframe == 'intraday_short':
            dates = [start_date + timedelta(minutes=i) for i in range(n_periods)]
        else:
            dates = [start_date + timedelta(hours=i) for i in range(n_periods)]
    elif timeframe == 'daily':
        start_date = datetime(2023, 1, 1)

    else:
        start_date = datetime(2023, 1, 1)
        dates = [start_date + timedelta(days=30*i) for i in range(n_periods)]
    
    df = pd.DataFrame({
        'date': dates,
        'price': prices,
        'returns': returns
    })
