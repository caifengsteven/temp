class TGNN(nn.Module):
    """Trading Graph Neural Network implementation."""
    
    def __init__(self):
        """Initialize the TGNN model."""
        super(TGNN, self).__init__()
        
        # Parameters to be learned
        self.theta_x = nn.Parameter(torch.randn(1) * 0.1)  # Asset feature parameter
        self.theta_y = nn.Parameter(torch.randn(1) * 0.1)  # Dealer feature parameter
        self.gamma = nn.Parameter(torch.randn(1) * 0.1)    # Relationship feature parameter
        
    def compute_costs(self, X, Y):
        """Compute dealer costs based on asset and dealer features."""
        return torch.exp(X * self.theta_x + Y * self.theta_y)
    
    def compute_bargaining_power(self, E):
        """Compute bargaining power based on relationship features."""
        return torch.sigmoid(E * self.gamma)
    
    def message_passing(self, X, Y, E, u, edge_index, max_iterations=10, tolerance=1e-4):
        """
        Perform message passing to compute equilibrium dealer values.
        
        Args:
            X: Asset features
            Y: Dealer features
            E: Relationship features
            u: Customer values
            edge_index: List of (seller, buyer) edge indices
            max_iterations: Maximum number of message passing iterations
            tolerance: Convergence tolerance
            
        Returns:
            Dealer values and best prices
        """
        # Compute costs for all dealers
        costs = self.compute_costs(X, Y)
        
        # Compute bargaining powers for all edges
        beta = self.compute_bargaining_power(E)
        
        # Initialize dealer values with customer values minus costs
        values = -costs + u
        
        # Create edge mapping for efficient lookup
        edges = {}
        for idx, (i, j) in enumerate(edge_index):
            if i.item() not in edges:
                edges[i.item()] = []
            edges[i.item()].append((j.item(), idx))
        
        # Perform message passing iterations
        for _ in range(max_iterations):
            new_values = -costs + u.clone()
            
            # For each dealer, compute potential prices from neighbors
            for i in range(len(values)):
                if i in edges:
                    max_price = torch.tensor(0.0)
                    for j, edge_idx in edges[i]:
                        # Compute transaction price
                        price = beta[edge_idx] * values[i] + (1 - beta[edge_idx]) * values[j]
                        max_price = torch.max(max_price, price)
                    
                    # Update value based on best price or customer value
                    new_values[i] = -costs[i] + torch.max(max_price, u[i])
            
            # Check for convergence
            if torch.max(torch.abs(new_values - values)) < tolerance:
                break
                
            values = new_values
        
        # Compute best prices for each dealer
        best_prices = torch.zeros_like(values)
        best_counterparties = torch.zeros_like(values, dtype=torch.long)
        
        for i in range(len(values)):
            best_price = u[i]
            best_counterparty = -1  # -1 indicates customer
            
            if i in edges:
                for j, edge_idx in edges[i]:
                    price = beta[edge_idx] * values[i] + (1 - beta[edge_idx]) * values[j]
                    if price > best_price:
                        best_price = price
                        best_counterparty = j
                        
            best_prices[i] = best_price
            best_counterparties[i] = best_counterparty
        
        return values, best_prices, best_counterparties
    
    def forward(self, X, Y, E, u, edge_index, observed_indices=None, observed_prices=None):
        """
        Forward pass of the TGNN model.
        
        Args:
            X: Asset features
            Y: Dealer features
            E: Relationship features
            u: Customer values
            edge_index: List of (seller, buyer) edge indices
            observed_indices: Indices of observed trades
            observed_prices: Observed trade prices
            
        Returns:
            Predicted prices and loss
        """
        # Compute equilibrium values and best prices
        _, best_prices, _ = self.message_passing(X, Y, E, u, edge_index)
        
        # If observed indices are provided, compute loss
        loss = None
        if observed_indices is not None and observed_prices is not None:
            # Extract predicted prices for observed trades
            predicted_prices = best_prices[observed_indices]
            
            # Compute mean squared error loss
            loss = F.mse_loss(predicted_prices, observed_prices)
        
        return best_prices, loss

def ols_baseline(simulator):
    """
    Run OLS baseline models for comparison.
    
    Args:
        simulator: TradingNetworkSimulator with generated data
        
    Returns:
        Dictionary of OLS model results
    """
    # Extract data from simulator
    observed_trades = simulator.observed_df
    
    # Basic OLS model
    X_basic = observed_trades[["asset_feature", "seller_feature", "buyer_feature", "relationship_feature"]]
    y = observed_trades["price"]
    
    model_basic = LinearRegression()
    model_basic.fit(X_basic, y)
    y_pred_basic = model_basic.predict(X_basic)
    
    # Add centrality measures
    # This is a simplified version - in practice, you would compute centrality for each (asset, day) subnetwork
    centrality_data = []
    
    for k in range(simulator.n_assets):
        for t in range(simulator.n_days):
            G = simulator.networks[(k, t)]
            
            # Compute centrality measures
            degree_centrality = nx.degree_centrality(G)
            eigenvector_centrality = nx.eigenvector_centrality(G, max_iter=1000, tol=1e-06)
            try:
                betweenness_centrality = nx.betweenness_centrality(G)
            except:
                # Handle disconnected graphs
                betweenness_centrality = {node: 0 for node in G.nodes()}
            
            for i in range(simulator.n_dealers):
                centrality_data.append({
                    "dealer": i,
                    "asset": k,
                    "day": t,
                    "degree_centrality": degree_centrality.get(i, 0),
                    "eigenvector_centrality": eigenvector_centrality.get(i, 0),
                    "betweenness_centrality": betweenness_centrality.get(i, 0)
                })
    
    centrality_df = pd.DataFrame(centrality_data)
    
    # Merge centrality measures with observed trades
    observed_with_centrality = observed_trades.merge(
        centrality_df, 
        left_on=["seller", "asset", "day"], 
        right_on=["dealer", "asset", "day"],
        suffixes=("", "_seller")
    ).drop(columns=["dealer"])
    
    observed_with_centrality = observed_with_centrality.merge(
        centrality_df, 
        left_on=["buyer", "asset", "day"], 
        right_on=["dealer", "asset", "day"],
        suffixes=("", "_buyer")
    ).drop(columns=["dealer"])
    
    # OLS with centrality measures
    X_centrality = observed_with_centrality[["asset_feature", "seller_feature", "buyer_feature", "relationship_feature", 
                                           "degree_centrality", "eigenvector_centrality", "betweenness_centrality",
                                           "degree_centrality_buyer", "eigenvector_centrality_buyer", "betweenness_centrality_buyer"]]
    
    model_centrality = LinearRegression()
    model_centrality.fit(X_centrality, y)
    y_pred_centrality = model_centrality.predict(X_centrality)
    
    # Compute metrics
    results = {
        "basic": {
            "mse": mean_squared_error(y, y_pred_basic),
            "mae": mean_absolute_error(y, y_pred_basic),
            "r2": r2_score(y, y_pred_basic),
            "coefficients": dict(zip(X_basic.columns, model_basic.coef_)),
            "predicted_prices": y_pred_basic
        },
        "centrality": {
            "mse": mean_squared_error(y, y_pred_centrality),
            "mae": mean_absolute_error(y, y_pred_centrality),
            "r2": r2_score(y, y_pred_centrality),
            "coefficients": dict(zip(X_centrality.columns, model_centrality.coef_)),
            "predicted_prices": y_pred_centrality
        }
    }
    
    return results


def bootstrap_confidence_intervals(model, simulator, n_bootstrap=100, alpha=0.05):
    """
    Compute bootstrap confidence intervals for model parameters.
    
    Args:
        model: Trained TGNN model
        simulator: TradingNetworkSimulator with generated data
        n_bootstrap: Number of bootstrap samples
        alpha: Significance level (e.g., 0.05 for 95% confidence interval)
        
    Returns:
        Dictionary of confidence intervals for each parameter
    """
    # Extract data from simulator
    observed_trades = simulator.observed_df
    nodes = simulator.node_df
    edges = simulator.edge_df
    
    # Create mapping of (dealer, asset, day) to index
    dealer_indices = {}
    for idx, row in nodes.iterrows():
        dealer_indices[(row["dealer"], row["asset"], row["day"])] = idx
    
    # Prepare node features
    X_all = torch.tensor(nodes["asset_feature"].values, dtype=torch.float32)
    Y_all = torch.tensor(nodes["dealer_feature"].values, dtype=torch.float32)
    u_all = torch.tensor(nodes["customer_value"].values, dtype=torch.float32)
    
    # Prepare edge indices and features
    edge_index = []
    E_all = []
    for _, row in edges.iterrows():
        seller_idx = dealer_indices[(row["seller"], row["asset"], row["day"])]
        buyer_idx = dealer_indices[(row["buyer"], row["asset"], row["day"])]
        edge_index.append((seller_idx, buyer_idx))
        E_all.append(row["relationship_feature"])
    
    edge_index = torch.tensor(edge_index, dtype=torch.long)
    E_all = torch.tensor(E_all, dtype=torch.float32)
    
    # Bootstrap parameter estimates
    theta_x_samples = []
    theta_y_samples = []
    gamma_samples = []
    
    for _ in tqdm(range(n_bootstrap), desc="Bootstrap sampling"):
        # Create bootstrap sample by resampling observed trades
        bootstrap_indices = np.random.choice(len(observed_trades), size=len(observed_trades), replace=True)
        bootstrap_trades = observed_trades.iloc[bootstrap_indices].reset_index(drop=True)
        
        # Prepare observed trade indices and prices for bootstrap sample
        observed_indices = []
        for _, row in bootstrap_trades.iterrows():
            seller_idx = dealer_indices[(row["seller"], row["asset"], row["day"])]
            observed_indices.append(seller_idx)
        
        observed_indices = torch.tensor(observed_indices, dtype=torch.long)
        observed_prices = torch.tensor(bootstrap_trades["price"].values, dtype=torch.float32)
        
        # Train model on bootstrap sample (simplified training with fewer epochs)
        bootstrap_model = TGNN()
        optimizer = optim.Adam(bootstrap_model.parameters(), lr=0.01)
        
        for _ in range(50):  # Fewer epochs for bootstrap
            optimizer.zero_grad()
            _, loss = bootstrap_model(X_all, Y_all, E_all, u_all, edge_index, observed_indices, observed_prices)
            loss.backward()
            optimizer.step()
        
        # Collect parameter estimates
        theta_x_samples.append(bootstrap_model.theta_x.item())
        theta_y_samples.append(bootstrap_model.theta_y.item())
        gamma_samples.append(bootstrap_model.gamma.item())
    
    # Compute confidence intervals
    theta_x_ci = np.percentile(theta_x_samples, [alpha/2*100, (1-alpha/2)*100])
    theta_y_ci = np.percentile(theta_y_samples, [alpha/2*100, (1-alpha/2)*100])
    gamma_ci = np.percentile(gamma_samples, [alpha/2*100, (1-alpha/2)*100])
    
    return {
        "theta_x": {"estimate": model.theta_x.item(), "ci": theta_x_ci},
        "theta_y": {"estimate": model.theta_y.item(), "ci": theta_y_ci},
        "gamma": {"estimate": model.gamma.item(), "ci": gamma_ci}
    }


def test_tgnn_random_network():
    """Test TGNN on a random network."""
    print("\n=== Testing TGNN on Random Network ===")
    
    # Set random seed for reproducibility
    np.random.seed(42)
    torch.manual_seed(42)
    random.seed(42)
    
    # Generate random trading network
    print("Generating random trading network...")
    simulator = TradingNetworkSimulator(
        n_dealers=10,
        n_assets=2,
        n_days=5,
        edge_probability=0.7,
        network_type="random"
    )
    simulator.generate_data()
    
    # Print network summary
    summary = simulator.get_network_summary()
    print(f"Network summary: {summary}")
    print(f"Number of observed trades: {len(simulator.observed_df)}")
    
    # Train TGNN model
    print("\nTraining TGNN model...")
    model, history = train_tgnn(simulator, max_epochs=300, learning_rate=0.01)
    
    # Evaluate TGNN model
    print("\nEvaluating TGNN model...")
    evaluation = evaluate_model(model, simulator)
    
    print(f"TGNN MSE: {evaluation['mse']:.6f}")
    print(f"TGNN MAE: {evaluation['mae']:.6f}")
    print(f"TGNN R²: {evaluation['r2']:.6f}")
    
    print("\nParameter estimates:")
    for param, value in evaluation["params"].items():
        true_value = evaluation["true_params"][param]
        error = evaluation["param_errors"][f"{param}_error"]
        print(f"  {param}: {value:.4f} (true: {true_value:.4f}, error: {error:.4f})")
    
    # Run OLS baseline
    print("\nRunning OLS baseline models...")
    ols_results = ols_baseline(simulator)
    
    print(f"OLS Basic MSE: {ols_results['basic']['mse']:.6f}")
    print(f"OLS Basic MAE: {ols_results['basic']['mae']:.6f}")
    print(f"OLS Basic R²: {ols_results['basic']['r2']:.6f}")
    
    print(f"OLS with Centrality MSE: {ols_results['centrality']['mse']:.6f}")
    print(f"OLS with Centrality MAE: {ols_results['centrality']['mae']:.6f}")
    print(f"OLS with Centrality R²: {ols_results['centrality']['r2']:.6f}")
    
    # Compute bootstrap confidence intervals
    print("\nComputing bootstrap confidence intervals...")
    ci_results = bootstrap_confidence_intervals(model, simulator, n_bootstrap=20, alpha=0.05)
    
    print("\nParameter estimates with 95% confidence intervals:")
    for param, result in ci_results.items():
        print(f"  {param}: {result['estimate']:.4f} [{result['ci'][0]:.4f}, {result['ci'][1]:.4f}]")
    
    # Plot predicted vs actual prices
    plt.figure(figsize=(10, 6))
    plt.scatter(evaluation["observed_prices"], evaluation["predicted_prices"], alpha=0.7)
    plt.scatter(evaluation["observed_prices"], ols_results["centrality"]["predicted_prices"], alpha=0.7)
    
    # Add reference line
    min_val = min(evaluation["observed_prices"].min(), evaluation["predicted_prices"].min())
    max_val = max(evaluation["observed_prices"].max(), evaluation["predicted_prices"].max())
    plt.plot([min_val, max_val], [min_val, max_val], 'k--')
    
    plt.xlabel("Observed Prices")
    plt.ylabel("Predicted Prices")
    plt.title("TGNN vs OLS: Predicted vs Observed Prices")
    plt.legend(["Reference", "TGNN", "OLS with Centrality"])
    plt.grid(True, alpha=0.3)
    plt.savefig("tgnn_vs_ols_random.png")
    plt.close()
    
    # Plot training history
    plt.figure(figsize=(12, 8))
    
    plt.subplot(2, 2, 1)
    plt.plot(history["loss"])
    plt.xlabel("Epoch")
    plt.ylabel("Loss")
    plt.title("Training Loss")
    plt.grid(True, alpha=0.3)
    
    plt.subplot(2, 2, 2)
    plt.plot(history["theta_x"])
    plt.axhline(y=simulator.true_params["theta_x"], color='r', linestyle='--')
    plt.xlabel("Epoch")
    plt.ylabel("θₓ")
    plt.title("Asset Feature Parameter")
    plt.grid(True, alpha=0.3)
    
    plt.subplot(2, 2, 3)
    plt.plot(history["theta_y"])
    plt.axhline(y=simulator.true_params["theta_y"], color='r', linestyle='--')
    plt.xlabel("Epoch")
    plt.ylabel("θᵧ")
    plt.title("Dealer Feature Parameter")
    plt.grid(True, alpha=0.3)
    
    plt.subplot(2, 2, 4)
    plt.plot(history["gamma"])
    plt.axhline(y=simulator.true_params["gamma"], color='r', linestyle='--')
    plt.xlabel("Epoch")
    plt.ylabel("γ")
    plt.title("Relationship Feature Parameter")
    plt.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.savefig("tgnn_training_history_random.png")
    plt.close()
    
    # Visualize the network
    print("\nVisualizing the trading network...")
    simulator.plot_network(asset=0, day=0)
    
    return {
        "simulator": simulator,
        "model": model,
        "evaluation": evaluation,
        "ols_results": ols_results,
        "ci_results": ci_results
    }

def compare_network_types():
    """Compare TGNN performance across different network types."""
    print("\n=== Comparing TGNN Performance Across Network Types ===")
    
    # Define network configurations to test
    network_configs = [
        {"name": "Dense Random", "type": "random", "edge_prob": 0.7, "n_dealers": 10},
        {"name": "Sparse Random", "type": "random", "edge_prob": 0.2, "n_dealers": 10},
        {"name": "Core-Periphery", "type": "core_periphery", "n_dealers": 20, "n_core": 4}
    ]
    
    results = {}
    
    for config in network_configs:
        print(f"\nTesting on {config['name']} network...")
        
        # Set random seed for reproducibility
        np.random.seed(42)
        torch.manual_seed(42)
        random.seed(42)
        
        # Generate trading network
        if config["type"] == "random":
            simulator = TradingNetworkSimulator(
                n_dealers=config["n_dealers"],
                n_assets=2,
                n_days=5,
                edge_probability=config["edge_prob"],
                network_type="random"
            )
        else:
            simulator = TradingNetworkSimulator(
                n_dealers=config["n_dealers"],
                n_assets=2,
                n_days=5,
                n_core_dealers=config["n_core"],
                network_type="core_periphery"
            )
        
        simulator.generate_data()
        
        # Train TGNN model
        model, history = train_tgnn(simulator, max_epochs=300, learning_rate=0.01)
        
        # Evaluate TGNN model
        evaluation = evaluate_model(model, simulator)
        
        # Run OLS baseline
        ols_results = ols_baseline(simulator)
        
        # Store results
        results[config["name"]] = {
            "tgnn": {
                "mse": evaluation["mse"],
                "mae": evaluation["mae"],
                "r2": evaluation["r2"],
                "params": evaluation["params"],
                "param_errors": evaluation["param_errors"]
            },
            "ols_basic": {
                "mse": ols_results["basic"]["mse"],
                "mae": ols_results["basic"]["mae"],
                "r2": ols_results["basic"]["r2"]
            },
            "ols_centrality": {
                "mse": ols_results["centrality"]["mse"],
                "mae": ols_results["centrality"]["mae"],
                "r2": ols_results["centrality"]["r2"]
            }
        }
    
    # Print comparison table
    print("\nPerformance Comparison:")
    print("\nMean Squared Error (MSE):")
    print(f"{'Network Type':<15} {'TGNN':<10} {'OLS Basic':<10} {'OLS+Centrality':<15}")
    print("-" * 50)
    for network, result in results.items():
        print(f"{network:<15} {result['tgnn']['mse']:<10.6f} {result['ols_basic']['mse']:<10.6f} {result['ols_centrality']['mse']:<15.6f}")
    
    print("\nR-squared (R²):")
    print(f"{'Network Type':<15} {'TGNN':<10} {'OLS Basic':<10} {'OLS+Centrality':<15}")
