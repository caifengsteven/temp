    def fit_har_model(self, X_train, y_train, X_test):
        """Simple HAR model using only RV_d, RV_w, RV_m"""
        har_cols = ['RV_d', 'RV_w', 'RV_m']
        X_train_har = X_train[har_cols]
        X_test_har = X_test[har_cols]
        
        # Simple OLS (in practice would use weighted least squares as in paper)
        beta = np.linalg.inv(X_train_har.T.dot(X_train_har)).dot(X_train_har.T).dot(y_train)
        return X_test_har.dot(beta)
    
    def fit_har_m_model(self, X_train, y_train, X_test):
        """HAR model with macro dummies"""
        har_cols = ['RV_d', 'RV_w', 'RV_m']
        dummy_cols = [col for col in X_train.columns if col.startswith('dummy_')]
        
        # Create interaction terms between RV_d and macro dummies
        for dummy in dummy_cols:
            X_train[f'RV_d_{dummy}'] = X_train['RV_d'] * X_train[dummy]
            X_test[f'RV_d_{dummy}'] = X_test['RV_d'] * X_test[dummy]
        
        interaction_cols = [f'RV_d_{dummy}' for dummy in dummy_cols]
        X_train_har_m = X_train[har_cols + interaction_cols]
        X_test_har_m = X_test[har_cols + interaction_cols]
        
        # Simple OLS
        beta = np.linalg.inv(X_train_har_m.T.dot(X_train_har_m)).dot(X_train_har_m.T).dot(y_train)
        return X_test_har_m.dot(beta)
    
    def fit_har_a_model(self, X_train, y_train, X_test):
        """HAR model with general attention"""
        har_cols = ['RV_d', 'RV_w', 'RV_m']
        attn_cols = [col for col in X_train.columns if col.startswith('attn_market_')]

    def fit_lasso_model(self, X_train, y_train, X_test, feature_type):
        """Adaptive LASSO model with either attention or sentiment features"""
        har_cols = ['RV_d', 'RV_w', 'RV_m']
        
        if feature_type == 'attention':
            extra_cols = [col for col in X_train.columns if col.startswith('attn_')]
        else:  # sentiment
            extra_cols = [col for col in X_train.columns if col.startswith('sent_')]
        
        # Always include HAR components
        X_train_subset = X_train[har_cols + extra_cols]
        X_test_subset = X_test[har_cols + extra_cols]
        
        # Simple Lasso (in practice would implement adaptive two-stage LASSO)
        model = Lasso(alpha=0.01, max_iter=10000)
        model.fit(X_train_subset, y_train)
        
        return model.predict(X_test_subset)
    
    def fit_rf_model(self, X_train, y_train, X_test, feature_type):
        """Random Forest model with either attention or sentiment features"""
        har_cols = ['RV_d', 'RV_w', 'RV_m']
        
        if feature_type == 'attention':
            extra_cols = [col for col in X_train.columns if col.startswith('attn_')]
        else:  # sentiment
            extra_cols = [col for col in X_train.columns if col.startswith('sent_')]
        
        # Always include HAR components
        X_train_subset = X_train[har_cols + extra_cols]
        X_test_subset = X_test[har_cols + extra_cols]
        
        # Random Forest
        model = RandomForestRegressor(n_estimators=100, max_depth=6, random_state=42)
        model.fit(X_train_subset, y_train)
        
        return model.predict(X_test_subset)
    
    def summarize_results(self, results):
        """Calculate and display summary statistics from model results"""
        model_names = list(results['stock_0'].keys())
        
        # Calculate the proportion of stocks where each model outperforms HAR
        outperformance = {model: 0 for model in model_names if model != 'HAR'}
        
        # Calculate average improvement percentages
        avg_improvements = {model: [] for model in model_names if model != 'HAR'}
        
        for stock, stock_results in results.items():
            har_mse = stock_results['HAR']['MSE']
            
            for model in model_names:
                if model != 'HAR':
                    model_mse = stock_results[model]['MSE']
                    if model_mse < har_mse:
                        outperformance[model] += 1
                    
                    improvement = (har_mse - model_mse) / har_mse * 100
                    avg_improvements[model].append(improvement)
        
        # Convert to percentages
        outperformance = {model: count / self.n_stocks * 100 for model, count in outperformance.items()}
        avg_improvements = {model: np.mean(imps) for model, imps in avg_improvements.items()}
        
        # Create summary table
        summary = pd.DataFrame({
            'Outperformance %': outperformance,
            'Avg Improvement %': avg_improvements
