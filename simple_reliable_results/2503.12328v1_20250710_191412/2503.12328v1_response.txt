        Parameters:

        Parameters:

        covariance_range (tuple): Range for asset covariances

        cov_matrix = np.zeros((n, n))

            cov_matrix[i, i] = np.random.uniform(*variance_range)

        # Interior nodes within the same cluster are correlated
        # Each cluster has 3 interior nodes connected to 3 junction nodes

        n_interior = n - n_junction
        n_clusters = n_interior // 3

            cov_matrix += np.eye(n) * (abs(min_eigenvalue) + 0.01)
            
        self.covariance_matrix = cov_matrix
        return cov_matrix
    
    def compute_schur_complement(self, matrix, n_junction):
        """
        Compute the Schur complement to reduce the hierarchical level.
        
        Parameters:
        matrix (ndarray): The covariance matrix
        n_junction (int): Number of junction nodes
        
        Returns:
        ndarray: The Schur complement matrix
        """
        T = matrix[:n_junction, :n_junction]
        J = matrix[n_junction:, :n_junction]
        X = matrix[n_junction:, n_junction:]
        
        # Compute Schur complement: S = T - J^T * X^-1 * J
        X_inv = linalg.inv(X)
        S = T - J.T @ X_inv @ J
        
        return S

        Returns:
        ndarray: The optimal portfolio weights

        self.w_vectors = {}

        gamma_0 = self.gamma_vectors[0]
        w_0 = linalg.solve(Sigma_0, gamma_0)
        self.w_vectors[0] = w_0
        
        # Recursively compute weights for higher levels
        for k in range(1, self.level + 1):
            n_jun_k = self._calculate_assets(k-1)
            n_k = self._calculate_assets(k)
            
            if k == self.level:
                matrix_k = self.covariance_matrix
            else:
                matrix_k = self.reduced_matrices[k]
            
            # Get the block matrices
            J = matrix_k[n_jun_k:, :n_jun_k]
            X = matrix_k[n_jun_k:, n_jun_k:]
            X_inv = linalg.inv(X)
            
            gamma_k = self.gamma_vectors[k]
            gamma_k_int = gamma_k[n_jun_k:]
            
            # Junction weights are the weights from the previous level
            w_jun_k = self.w_vectors[k-1]
            
            # Compute interior weights
            w_int_k = X_inv @ gamma_k_int - X_inv @ J @ w_jun_k
            
            # Combine junction and interior weights
            w_k = np.zeros(n_k)
            w_k[:n_jun_k] = w_jun_k
            w_k[n_jun_k:] = w_int_k
            
            self.w_vectors[k] = w_k
        
        # Normalize the final weights
        self.optimal_weights = self.w_vectors[self.level] / np.sum(self.w_vectors[self.level])

        """
        Compute the portfolio variance given weights.
        
        Parameters:
        weights (ndarray, optional): Portfolio weights. Uses optimal weights if None.
        
        Returns:
        float: Portfolio variance
        """
        if weights is None:
            if self.optimal_weights is None:
                raise ValueError("Optimal weights not computed. Run compute_weights first.")
            weights = self.optimal_weights
            
        return weights.T @ self.covariance_matrix @ weights

        """
        Compare the hierarchical approach with direct optimization.
        
        Returns:
