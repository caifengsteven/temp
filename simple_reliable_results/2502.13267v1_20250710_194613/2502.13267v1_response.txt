@dataclass
class CentralBank:
    """Central Bank agent in the model"""
    rate: float = 0.01  # Policy interest rate
    
    def taylor_rule(self, inflation: float, output_gap: float, 
                   neutral_rate: float = 0.02, 
                   inflation_target: float = 0.02,
                   inflation_weight: float = 1.5, 
                   output_gap_weight: float = 0.5) -> float:
        """Implement a Taylor rule for monetary policy"""
        return neutral_rate + inflation_weight * (inflation - inflation_target) + output_gap_weight * output_gap


@dataclass
class Bank:
    """Financial sector agent in the model"""
    Pi_k: float = 0.0  # Bank profits
    E_k: float = 1000.0  # Bank equity
    mu: float = 0.02  # Risk premium
    
    def bank_profits(self, loan_volume: float, deposits: float, model) -> float:
        """Calculate bank profits"""
        # Simplified profit calculation
        lending_rate = model.cb.rate + self.mu
        return loan_volume * lending_rate - deposits * model.cb.rate * 0.5
    
    def bank_equity(self, model) -> float:
        """Update bank equity"""
        return self.E_k + self.Pi_k


@dataclass
class Firms:

    num_firms: int = 1000

    L_i: np.ndarray = None  # Labor for each firm

    K_i: np.ndarray = None  # Capital for each firm
    
    def __post_init__(self):

class Model:

    def __init__(self, parameters: Dict, initial_conditions: Dict, T: int):

        self.w_act = Workers(

        self.w_inact = Workers(

            mu=parameters.get("mu", 0.02)

        self.cb = CentralBank(
            rate=initial_conditions.get("policy_rate", 0.01)
        )
        self.gov = Government(
            tau_INC=parameters.get("tau_INC", 0.25)
        )
        self.rotw = RestOfWorld()
        self.agg = Aggregates()
        self.prop = Properties(
            psi=parameters.get("psi", 0.8),
            T=T,
            scale=parameters.get("scale", 0.001)
        )
        
        # Initialize model state
        self.update_variables_with_totals()
    
    def update_variables_with_totals(self):

        self.agg.C = np.sum(self.w_act.C_h) + np.sum(self.w_inact.C_h)
        
        # Calculate aggregate investment
        self.agg.I = np.sum(self.firms.I_i)
        
        # Calculate GDP using expenditure approach
        self.agg.GDP = self.agg.C + self.agg.I + self.gov.G + (self.rotw.X - self.rotw.M)
        
        # Calculate unemployment rate
        total_labor_force = len(self.w_act.Y_h) + len(self.w_inact.Y_h)
        self.agg.unemployment_rate = len(self.w_inact.Y_h) / total_labor_force
        
        # Real GDP (simplified, assuming price_level is updated elsewhere)
        self.agg.real_GDP = self.agg.GDP / self.agg.price_level

        self.policy_rate = np.zeros(T)
    
    def update(self, model: Model, t: int):
        """Update data tracker with current model state"""
        self.nominal_gdp[t] = model.agg.GDP

        self.consumption[t] = model.agg.C
        self.investment[t] = model.agg.I

        self.price_level[t] = model.agg.price_level
        self.policy_rate[t] = model.cb.rate

def step(model: Model) -> None:

    # Update rest of world

    model.update_variables_with_totals()


def firms_profits(firms: Firms, model: Model) -> np.ndarray:
    """Calculate firm profits"""
    # Simple profit function based on labor and capital
    return 0.2 * firms.K_i * np.sqrt(firms.L_i) * np.random.lognormal(0, 0.1, firms.num_firms)


def bank_profits(bank: Bank, model: Model) -> float:
    """Calculate bank profits"""
    # Simple model of bank profits based on lending and deposits
    total_loans = np.sum(model.firms.K_i) * 0.5
    total_deposits = np.sum(model.w_act.S_h) + np.sum(model.w_inact.S_h)
    return bank.bank_profits(total_loans, total_deposits, model)


def bank_equity(bank: Bank, model: Model) -> float:
    """Update bank equity"""
    return bank.bank_equity(model)


def central_bank_rate(cb: CentralBank, model: Model) -> float:
    """Update central bank policy rate"""
    # Use Taylor rule
    inflation = (model.agg.price_level / 
                (model.agg.price_level / (1 + (0.005 + 0.003 * np.random.randn()))) - 1)
    output_gap = 0.01 * np.random.randn()  # Simplified output gap
    return cb.taylor_rule(inflation, output_gap)


def update_worker_consumption(workers: Workers, model: Model) -> None:
    """Update worker consumption and savings"""
    # Simple consumption function based on propensity to consume
    workers.C_h = model.prop.psi * workers.Y_h * (1 + 0.05 * np.random.randn(workers.num_workers))
    workers.S_h = workers.Y_h - workers.C_h
    
    # Ensure non-negative savings (minimum consumption constraint)
    negative_savings = workers.S_h < 0
    if np.any(negative_savings):
        workers.C_h[negative_savings] = workers.Y_h[negative_savings]
        workers.S_h[negative_savings] = 0


def update_firm_investment(firms: Firms, model: Model) -> None:
    """Update firm investment"""
    # Simple investment function based on profits and interest rate
    interest_factor = 1 / (1 + model.cb.rate + model.bank.mu)
    profit_factor = np.clip(firms.Pi_i, 0, None) / (np.mean(firms.Pi_i) + 1e-10)
    firms.I_i = 10 * profit_factor * interest_factor * (1 + 0.1 * np.random.randn(firms.num_firms))


def update_government(gov: Government, model: Model) -> None:
    """Update government accounts"""
    # Update tax revenue based on income tax
    gov.T = gov.tau_INC * (np.sum(model.w_act.Y_h) + np.sum(model.firms.Pi_i))
    
    # Update debt based on deficit
    gov.B += (gov.G - gov.T)


def update_rotw(rotw: RestOfWorld, model: Model) -> None:
    """Update rest of world accounts"""
    # Simple model of exports and imports
    gdp_effect = model.agg.GDP / 10000
    rotw.X = 800 * (1 + 0.02 * np.random.randn())  # Exogenous exports
    rotw.M = 600 + 0.25 * gdp_effect * (1 + 0.02 * np.random.randn())  # Imports depend on GDP
    rotw.NX = rotw.X - rotw.M


def run(model: Model) -> DataTracker:
    """Run a complete simulation"""
    data = DataTracker(model.prop.T)
    
    for t in range(model.prop.T):
        step(model)
        data.update(model, t)
    
    return data


def ensemblerun(model: Model, n_sims: int, shock: Optional[Callable] = None) -> List[DataTracker]:
    """Run multiple simulations"""
    data_vector = []
    
    for _ in range(n_sims):
        # Create a copy of the model for each simulation
        sim_model = init_model(
            {"psi": model.prop.psi, "tau_INC": model.gov.tau_INC, "mu": model.bank.mu},
            {"num_workers": model.w_act.num_workers, "num_inactive": model.w_inact.num_workers, 
             "num_firms": model.firms.num_firms, "policy_rate": model.cb.rate},
            model.prop.T
        )
        
        # Apply shock if provided
        if shock is not None:
            shock(sim_model)
        
        data = run(sim_model)
        data_vector.append(data)
    
    return data_vector


class ConsumptionShock:
    """Shock that increases consumption propensity temporarily"""
    def __init__(self, multiplier: float, final_time: int):
        self.multiplier = multiplier
        self.final_time = final_time
        self.original_psi = None
    
    def __call__(self, model: Model):
        # Store the original value to restore later
        self.original_psi = model.prop.psi
        
        # Apply the shock in the step function
        def apply_shock(m: Model):
            if m.agg.t == 1:
                m.prop.psi *= self.multiplier
            elif m.agg.t == self.final_time:
                m.prop.psi = self.original_psi
        
        # Store the original step function
        original_step = step
        
        # Redefine step to include the shock
        def shocked_step(m: Model):
            apply_shock(m)
            original_step(m)
        
        # Replace the step function
        global step
        step = shocked_step

1. Data models for all the major agent types in the macroeconomic model (households, firms, banks, central bank, government, rest of world)

   - Testing model initialization
   - Verifying accounting identities
   - Testing specific economic functions like the Taylor rule
   - Testing complete simulation runs
   - Benchmarking performance at different scales
