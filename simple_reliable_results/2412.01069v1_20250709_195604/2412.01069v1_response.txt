            qtr_data = self.data.analysis_df[self.data.analysis_df['post_cutoff_qtr'] == qtr]
            if len(qtr_data) > 0:
                gpt_errors.append(qtr_data['gpt_fe'].mean())
                analyst_errors.append(qtr_data['analyst_fe'].mean())
        
        # Create bar chart
        x = np.arange(len(quarters))
        width = 0.35
        
        fig, ax = plt.subplots(figsize=(10, 6))
        ax.bar(x - width/2, gpt_errors, width, label='GPT-4')
        ax.bar(x + width/2, analyst_errors, width, label='Analysts')
        
        ax.set_ylabel('Mean Absolute Forecast Error')
        ax.set_title('Forecast Accuracy: GPT-4 vs. Analysts')
        ax.set_xticks(x)
        ax.set_xticklabels(quarters)
        ax.legend()
        
        # Add value labels
        for i, v in enumerate(gpt_errors):
            ax.text(i - width/2, v + 0.0015, f'{v:.3f}', ha='center')
        
        for i, v in enumerate(analyst_errors):
            ax.text(i + width/2, v + 0.0015, f'{v:.3f}', ha='center')
        
        plt.tight_layout()
        plt.show()
    
    def plot_sentence_selection(self):
        """Plot the characteristics of GPT-selected sentences."""
        # Calculate average characteristics
        selected = self.data.sentence_data[self.data.sentence_data['gpt_selected'] == 1]
        non_selected = self.data.sentence_data[self.data.sentence_data['gpt_selected'] == 0]
        
        chars = ['order', 'length', 'fog', 'pct_num', 'sentiment', 'guidance']
        selected_means = [selected[char].mean() for char in chars]
        non_selected_means = [non_selected[char].mean() for char in chars]
        
        # Normalize for better visualization
        max_vals = [max(s, n) for s, n in zip(selected_means, non_selected_means)]
        selected_norm = [s/m for s, m in zip(selected_means, max_vals)]
        non_selected_norm = [n/m for n, m in zip(non_selected_means, max_vals)]
        
        # Create radar chart
        fig, ax = plt.subplots(figsize=(8, 8), subplot_kw=dict(polar=True))
        
        # Number of variables
        N = len(chars)
        
        # Angle of each axis
        angles = [n / float(N) * 2 * np.pi for n in range(N)]
        angles += angles[:1]  # Close the loop
        
        # Add the first point at the end to close the polygon
        selected_norm += selected_norm[:1]
        non_selected_norm += non_selected_norm[:1]
        
        # Plot
        ax.plot(angles, selected_norm, 'o-', linewidth=2, label='GPT Selected')
        ax.plot(angles, non_selected_norm, 'o-', linewidth=2, label='Not Selected')
        ax.fill(angles, selected_norm, alpha=0.25)
        ax.fill(angles, non_selected_norm, alpha=0.25)
        
        # Labels
        ax.set_xticks(angles[:-1])
        ax.set_xticklabels([c.capitalize() for c in chars])
        
        # Title and legend
        plt.title('Characteristics of GPT-Selected vs. Non-Selected Sentences', size=15)
        plt.legend(loc='upper right')
        
        plt.tight_layout()
        plt.show()
    
    def plot_metric_alignment(self):
        """Plot metric alignment by firm characteristics."""
        # Prepare data
        firm_size_bins = pd.qcut(self.data.analysis_df['firm_size'], 4, labels=False)
        analyst_bins = pd.qcut(self.data.analysis_df['analyst_following'], 4, labels=False)

        self.data.analysis_df['analyst_quartile'] = analyst_bins
        
        # Calculate alignment by quartile

        analyst_alignment = self.data.analysis_df.groupby('analyst_quartile')['metric_overlap_high'].mean()
        
        # Plot
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
        
        size_alignment.plot(kind='bar', ax=ax1, color='skyblue')
        ax1.set_title('Metric Alignment by Firm Size Quartile')
        ax1.set_xlabel('Firm Size Quartile (1=smallest, 4=largest)')
        ax1.set_ylabel('% with High Metric Alignment')
        ax1.set_ylim(0, 0.6)
        
        analyst_alignment.plot(kind='bar', ax=ax2, color='lightgreen')
        ax2.set_title('Metric Alignment by Analyst Following Quartile')
        ax2.set_xlabel('Analyst Following Quartile (1=lowest, 4=highest)')
        ax2.set_ylabel('% with High Metric Alignment')
        ax2.set_ylim(0, 0.6)
        
        plt.tight_layout()
        plt.show()
    
    def plot_metric_frequency(self):
        """Plot frequency of metrics selected by GPT and analysts."""
        # Extract all metrics
        gpt_metrics = []
        for metrics_str in self.data.metrics_data['gpt_metrics']:
            gpt_metrics.extend(metrics_str.split(','))
        
        analyst_metrics = []
        for metrics_str in self.data.metrics_data['analyst_metrics']:
            analyst_metrics.extend(metrics_str.split(','))
        
        # Count frequencies
        gpt_counts = Counter(gpt_metrics)
        analyst_counts = Counter(analyst_metrics)
        
        # Get top 10 metrics for each
        gpt_top10 = dict(gpt_counts.most_common(10))
        analyst_top10 = dict(analyst_counts.most_common(10))
        
        # Plot
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 6))
        
        # GPT metrics
        ax1.barh(list(gpt_top10.keys()), list(gpt_top10.values()), color='skyblue')
        ax1.set_title('Top 10 Metrics Selected by GPT')
        ax1.set_xlabel('Frequency')
        
        # Analyst metrics
        ax2.barh(list(analyst_top10.keys()), list(analyst_top10.values()), color='lightgreen')
        ax2.set_title('Top 10 Metrics Selected by Analysts')
        ax2.set_xlabel('Frequency')
        
        plt.tight_layout()
        plt.show()
