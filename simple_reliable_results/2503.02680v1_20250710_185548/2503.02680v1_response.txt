        Args:
            asset: Asset symbol

        Returns:

        # Calculate quadratic VWAP loss (in millionths)
        quad_vwap_loss = ((execution_price - market_vwap) / market_vwap) ** 2 * 1000000
        
        return execution_price, market_vwap, abs_vwap_loss, quad_vwap_loss
