class BrokerStrategy:

    def __init__(self, params: ModelParams):

def generate_synthetic_data(params: ModelParams, num_days: int = 5, steps_per_day: int = 100, 
                           seed: int = 42) -> Dict[str, np.ndarray]:
    """
    Generate synthetic trading data for testing the strategy.
    
    Args:
        params: Model parameters
        num_days: Number of days to simulate
        steps_per_day: Number of time steps per day
        seed: Random seed
        
    Returns:
        Dictionary containing synthetic data
    """
    np.random.seed(seed)
    
    # Initialize arrays
    dates = np.arange(num_days)
    prices = np.zeros((num_days, steps_per_day))
    alpha = np.zeros((num_days, steps_per_day))
    informed_volumes = np.zeros((num_days, steps_per_day))
    uninformed_volumes = np.zeros((num_days, steps_per_day))
    
    dt = 1/steps_per_day
    
    for day in range(num_days):
        # Initial values
        prices[day, 0] = 100.0
        alpha[day, 0] = 0.0
        nu_u = 0.0
        
        # Generate alpha process and price process
        for t in range(1, steps_per_day):
            # Update alpha (mean-reverting process)
            alpha[day, t] = alpha[day, t-1] * (1 - params.kappa_alpha * dt) + \
                          params.sigma_alpha * np.sqrt(dt) * np.random.randn()
            
            # Update price
            prices[day, t] = prices[day, t-1] + alpha[day, t-1] * dt + \
                           params.sigma_s * np.sqrt(dt) * np.random.randn()
            
            # Generate informed trader's volume (simplified version of the formula in the paper)
            # v_i = A * alpha - B * q_i

                     np.sqrt(4 * params.a_i * params.beta * params.k_i + 
                            (params.beta * params.k_i)**2 + 
                            2 * params.k_i * params.phi_i))
            
            B = np.sqrt(params.a_i * params.beta / params.k_i + 
                       (params.beta/2)**2 + params.phi_i/(2*params.k_i)) - params.beta/2
            
            # Simplified: assuming q_i is close to 0 for demonstration
            informed_volumes[day, t] = A * alpha[day, t] + np.random.normal(0, 0.1)
            
            # Generate uninformed trader's volume (mean-reverting process)
            nu_u = nu_u * (1 - params.kappa_u * dt) + \
                   params.sigma_u * (params.k_b - params.k_u) / (params.k_b - 0.0001) * np.sqrt(dt) * np.random.randn()
            uninformed_volumes[day, t] = nu_u * dt
    
    return {
        'dates': dates,
        'prices': prices,
        'alpha': alpha,
        'informed_volumes': informed_volumes,
        'uninformed_volumes': uninformed_volumes
    }


def run_backtest(strategy: BrokerStrategy, test_data: Dict[str, np.ndarray]) -> Tuple[float, Dict]:
    """
    Backtest the strategy on test data.
    
    Args:
        strategy: BrokerStrategy instance
        test_data: Dictionary containing test data
        
    Returns:
        Total PnL and dictionary with detailed results
    """
    results = {
        'cash': [],

        'pnl': [],

        day_pnl = []

            # Get informed and uninformed trader volumes
            v_i = day_data['informed_volumes'][t]
            v_u = day_data['uninformed_volumes'][t]
            
            try:
                # Calculate broker's trading volume using the strategy
                v_b = strategy.execute_strategy(q_b, q_i, v_i, v_u)
                
                # Update inventories
                q_i += v_i
                q_b = q_b - v_i - v_u + v_b
                
                # Calculate execution prices
                price = day_data['prices'][t]
                price_i = price + strategy.params.k_i * v_i  # Price for informed trader
                price_u = price + strategy.params.k_u * v_u  # Price for uninformed trader
                price_b = price + strategy.params.k_b * v_b  # Price in lit market
                
                # Update broker's cash
                x_b += v_i * price_i + v_u * price_u - v_b * price_b
                
                # Calculate current PnL
                current_pnl = x_b + q_b * price - strategy.params.a_b * q_b**2
                day_pnl.append(current_pnl)
                
                # Store results
                results['cash'].append(x_b)
                results['inventory'].append(q_b)
                results['trades'].append(v_b)
                
            except Exception as e:
                print(f"Error at day {day}, step {t}: {e}")
        
        # Calculate final performance for the day
        final_price = day_data['prices'][-1]
        final_value = x_b + q_b * final_price - strategy.params.a_b * q_b**2
        
        total_pnl += final_value
        results['pnl'].extend(day_pnl)
