def compute_weighted_similarity_contrastive_loss(anchors, augmented, similarity_scores, margin=1.0):
    """Compute the Weighted Similarity Contrastive Loss (WSCL)"""
    batch_size = anchors.shape[0]
    total_loss = 0.0
    
    for i in range(batch_size):
        anchor = anchors[i]
        aug = augmented[i]
        sim_score = similarity_scores[i]
        
        # Compute Euclidean distance
        distance = np.linalg.norm(anchor - aug)
        
        # Compute the pull and push components
        pull_loss = sim_score * (distance ** 2)
        push_loss = (1 - sim_score) * max(0, margin - distance) ** 2
        
        # Sum the components
        total_loss += pull_loss + push_loss
    
    return total_loss / batch_size

def compute_continuously_weighted_contrastive_loss(anchors, augmented, similarity_scores, temperature=0.1):
    """Compute the Continuously Weighted Contrastive Loss (CWCL)"""
    batch_size = anchors.shape[0]
    total_loss = 0.0
    
    for i in range(batch_size):
        anchor = anchors[i]
        aug = augmented[i]
        sim_score = similarity_scores[i]
        
        # Compute cosine similarity
        cos_sim = np.dot(anchor, aug) / (np.linalg.norm(anchor) * np.linalg.norm(aug))
        
        # Scale by temperature
        scaled_sim = -cos_sim / temperature
        
        # Compute softmax (simplified for single pair)
        exp_sim = np.exp(scaled_sim)
        softmax_sim = exp_sim / (exp_sim + np.exp(0))  # Compare to a zero baseline
        
        # Weighted cross-entropy
        loss = -sim_score * np.log(softmax_sim)
        
        total_loss += loss
    
    return total_loss / batch_size

def evaluate_embedding_space_density(embeddings, labels, k=5):
    """Evaluate the embedding space using density metrics from the paper"""
    num_samples = embeddings.shape[0]
    
    # Compute pairwise distances
    distances = np.zeros((num_samples, num_samples))
    for i in range(num_samples):
        for j in range(num_samples):
            if i != j:
                distances[i, j] = np.linalg.norm(embeddings[i] - embeddings[j])
    
    # For each sample, find k nearest neighbors
    neighbors = np.zeros((num_samples, k), dtype=int)
    for i in range(num_samples):
        # Get indices of k nearest neighbors (excluding self)
        row_distances = distances[i].copy()
        row_distances[i] = np.inf  # Exclude self
        neighbors[i] = np.argsort(row_distances)[:k]
    
    # Compute g-KNN metric
    g_knn = 0.0
    for i in range(num_samples):
        neighbor_labels = [labels[n] for n in neighbors[i]]
        # Count labels that match the current point's label
        matching = sum(1 for nl in neighbor_labels if nl == labels[i])
        g_knn += matching / k
    g_knn /= num_samples
    
    # Compute KNN accuracy
    knn_acc = 0.0
    for i in range(num_samples):
        if labels[neighbors[i][0]] == labels[i]:
            knn_acc += 1
    knn_acc /= num_samples
    
    # Compute KL divergence and JSD
    kl_div = 0.0
    jsd = 0.0
    
    # Get global label distribution
    unique_labels = np.unique(labels)
    global_dist = np.zeros(len(unique_labels))
    for i, label in enumerate(unique_labels):
        global_dist[i] = np.mean(labels == label)
    
    for i in range(num_samples):
        # Get local label distribution
        neighbor_labels = [labels[n] for n in neighbors[i]]
        local_dist = np.zeros(len(unique_labels))
        for j, label in enumerate(unique_labels):
            local_dist[j] = np.mean([nl == label for nl in neighbor_labels])
        
        # Add small epsilon to avoid division by zero
        epsilon = 1e-10
        local_dist = local_dist + epsilon
        global_dist_eps = global_dist + epsilon
        
        # Normalize
        local_dist = local_dist / np.sum(local_dist)
        global_dist_eps = global_dist_eps / np.sum(global_dist_eps)
        
        # KL divergence
        kl = np.sum(local_dist * np.log(local_dist / global_dist_eps))
        kl_div += kl
        
        # JS divergence
        m = 0.5 * (local_dist + global_dist_eps)
        js = 0.5 * np.sum(local_dist * np.log(local_dist / m)) + 0.5 * np.sum(global_dist_eps * np.log(global_dist_eps / m))
        jsd += js
    
    kl_div /= num_samples
    jsd /= num_samples
    
    return {
        'g_knn': g_knn,
        'knn_acc': knn_acc,
        'kl_div': kl_div,
        'jsd': jsd
    }

            total_loss_cwcl += cwcl_loss.item()
        
        print(f"Epoch {epoch+1}/{num_epochs}, CWCL Loss: {total_loss_cwcl:.4f}")

    test_texts = [" ".join(dns) for dns in test_headlines]

    projection_wscl.eval()
