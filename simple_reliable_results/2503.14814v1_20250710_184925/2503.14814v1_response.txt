        for event_time in events1:

                lambda2 += alpha21 / ((t - event_time + eps) ** beta21)

        for event_time in events2:
            if event_time < t:
                lambda1 += alpha12 / ((t - event_time + eps) ** beta12)

            sell_count = np.sum((sell_times >= time_points[i-1]) & (sell_times < t))
            
            # Calculate price impact
            price_change = (buy_count - sell_count) * impact_factor
            prices[i] = prices[i-1] + price_change
        
        return prices
