@dataclass
class PortfolioState:
    """State variables for the portfolio allocation problem"""
    t: int  # Time (quarters)
    W: float  # Liquid wealth
    P: float  # Illiquid wealth (PE NAV)
    K: float  # Uncalled commitments
    mu_P: float  # Expected PE return
    s: int  # Macroeconomic state (0=recession, 1=expansion)
    is_defaulted: bool = False  # Whether the LP has defaulted
    
    @property
    def w(self) -> float:
        """Liquid fraction of total wealth"""
        if self.W + self.P == 0:
            return 1.0
        return self.W / (self.W + self.P)
    
    @property
    def k(self) -> float:
        """Uncalled commitments relative to total wealth"""
        if self.W + self.P == 0:
            return 0.0
        return self.K / (self.W + self.P)
    
    @property
    def total_wealth(self) -> float:
        """Total wealth"""
        return self.W + self.P

class PortfolioSimulator:
    """Simulates portfolio evolution over time"""
    
    def __init__(self, params: ModelParameters, allocator, initial_state: PortfolioState):
        self.params = params
        self.allocator = allocator
        self.initial_state = initial_state
        self.history = []
        
    def generate_returns(self, state: PortfolioState) -> Tuple[float, float]:
        """Generate returns for PE and stocks"""
        s = state.s
        mu_P = state.mu_P
        
        # Generate correlated normal random variables
        z = np.random.multivariate_normal(
            mean=[0, 0],
            cov=[[1, self.params.rho[s]], [self.params.rho[s], 1]]
        )
        
        # Transform to get returns
        r_P = np.exp(mu_P + self.params.sigma_P[s] * z[0])
        r_S = np.exp(self.params.mu_S[s] + self.params.sigma_S[s] * z[1])
        
        return r_P, r_S
    
    def next_macro_state(self, current_state: int) -> int:
        """Determine the next macroeconomic state"""
        p = self.params.transition_probs[current_state]
        return np.random.choice([0, 1], p=[p[0], p[1]])
    
    def update_mu_P(self, state: PortfolioState, r_P: float) -> float:
        """Update the expected PE return"""
        return (self.params.rho_P1 * state.mu_P + 
                self.params.rho_P2 * np.log(r_P) + 
                self.params.nu_P[state.s])
    
    def simulate_period(self, state: PortfolioState) -> PortfolioState:
        """Simulate one period"""
        # Record current state
        self.history.append(state)
        
        if state.t >= self.params.T:
            return state
        
        # Get allocations
        N, S, B = self.allocator.allocate(state)
        
        # Generate returns and next macro state
        r_P, r_S = self.generate_returns(state)
        next_s = self.next_macro_state(state.s)
        r_f = np.exp(self.params.log_Rf[state.s])
        
        # Calculate distributions and capital calls
        lambda_D = self.params.lambda_D[next_s]
        lambda_K = self.params.lambda_K[next_s]
        lambda_N = self.params.lambda_N[next_s]
        
        # Calculate risk weight and cost
        theta = self.calculate_risk_weight(state, N, S, B)
        risk_cost = self.calculate_risk_cost(theta)
        
        # Update state variables
        if state.is_defaulted:
            # After default, no PE investments
            W_next = S * r_S + B * r_f - state.W * risk_cost
            P_next = 0.0
            K_next = 0.0
            is_defaulted = True
        else:
            # Update wealth components
            distributions = lambda_D * r_P * state.P
            capital_calls_existing = lambda_K * state.K
            capital_calls_new = lambda_N * N
            total_calls = capital_calls_existing + capital_calls_new
            
            W_next = (distributions + S * r_S + B * r_f - 
                     total_calls - (state.W + state.P) * risk_cost)
            
            P_next = (1 - lambda_D) * r_P * state.P + total_calls
            K_next = (1 - lambda_K) * state.K + (1 - lambda_N) * N
            
            # Check for default
            if W_next < 0:
                # Default: liquidate PE at a discount
                alpha = self.params.alpha[next_s]
                W_next = distributions + alpha * (1 - lambda_D) * r_P * state.P + S * r_S + B * r_f
                P_next = 0.0
                K_next = 0.0
                is_defaulted = True
            else:
                is_defaulted = False
        
        # Update expected PE return
        mu_P_next = self.update_mu_P(state, r_P)
        
        return PortfolioState(
            t=state.t + 1,
            W=W_next,
            P=P_next,
            K=K_next,
            mu_P=mu_P_next,
            s=next_s,
            is_defaulted=is_defaulted
        )
    
    def calculate_risk_weight(self, state: PortfolioState, N: float, S: float, B: float) -> float:
        """Calculate the risk weight of the portfolio"""
        if state.is_defaulted:
            # Risk weight calculation for defaulted state
            numerator = self.params.theta_S * S
            denominator = state.W
        else:
            # Risk weight calculation for non-defaulted state
            total_wealth = state.W + state.P
            n = N / total_wealth if total_wealth > 0 else 0
            
            # Adjustment costs
            commitment_adj_cost = self.params.gamma_N * total_wealth * (n - self.params.n_bar)**2
            stock_adj_cost = self.params.gamma_S * total_wealth * (S / total_wealth)**2 if total_wealth > 0 else 0
            
            numerator = (self.params.theta_B * B + 
                        self.params.theta_S * (S + stock_adj_cost) + 
                        self.params.theta_P * state.P)
            
            denominator = B + S + stock_adj_cost + state.P
        
        return numerator / denominator if denominator > 0 else 0
    
    def calculate_risk_cost(self, theta: float) -> float:
        """Calculate the cost of violating the risk budget"""
        if theta <= self.params.theta:
            return 0.0
        else:
            return self.params.kappa * (theta - self.params.theta)**2
    
    def simulate(self) -> List[PortfolioState]:
        """Simulate the portfolio evolution over the investment horizon"""
        state = self.initial_state
        
        while state.t < self.params.T:
            state = self.simulate_period(state)
        
        return self.history
    
    def get_results(self) -> Dict:
        """Get summary statistics from the simulation"""
        if not self.history:
            return {}
        
        # Extract time series
        times = [s.t for s in self.history]
        total_wealth = [s.total_wealth for s in self.history]
        liquid_wealth_pct = [s.w for s in self.history]
        pe_allocations = [1 - s.w for s in self.history]
        uncalled_commitments = [s.k for s in self.history]
        defaulted = [1 if s.is_defaulted else 0 for s in self.history]
        
        # Calculate returns
        initial_wealth = self.history[0].total_wealth
        final_wealth = self.history[-1].total_wealth
        total_return = final_wealth / initial_wealth - 1
        annualized_return = (final_wealth / initial_wealth)**(4/len(times)) - 1  # Quarterly to annual
        
        # Check if defaulted
        ever_defaulted = any(defaulted)
        default_time = next((t for t, d in enumerate(defaulted) if d), None)
        
        return {
            "initial_wealth": initial_wealth,
            "final_wealth": final_wealth,
            "total_return": total_return,
            "annualized_return": annualized_return,
            "ever_defaulted": ever_defaulted,
            "default_time": default_time,
            "average_pe_allocation": np.mean(pe_allocations),
            "time_series": {
                "times": times,
                "total_wealth": total_wealth,
                "liquid_wealth_pct": liquid_wealth_pct,
                "pe_allocations": pe_allocations,
                "uncalled_commitments": uncalled_commitments,
                "defaulted": defaulted
            }
        }


def run_simulations(num_simulations: int = 100) -> Tuple[Dict, Dict]:

        s=1,    # Start in expansion state
        is_defaulted=False
    )
    
    # Create simulators
    heuristic_results = []
    dynamic_results = []
    
    for i in range(num_simulations):

        heuristic_allocator = HeuristicAllocator(params)
        heuristic_simulator = PortfolioSimulator(params, heuristic_allocator, initial_state)
        heuristic_simulator.simulate()
        heuristic_results.append(heuristic_simulator.get_results())
        
        # Reset random seed
        np.random.seed(i)
        
        # Run dynamic strategy
        dynamic_allocator = DynamicAllocator(params)

        dynamic_simulator.simulate()

    def aggregate_results(results: List[Dict]) -> Dict:
