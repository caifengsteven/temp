class LOBEvent:
    """Represents a single event in the limit order book"""
    def __init__(self, timestamp: float, event_type: int):
        self.timestamp = timestamp
        self.event_type = event_type
    
    def __repr__(self):
        event_name = list(EVENT_TYPES.keys())[list(EVENT_TYPES.values()).index(self.event_type)]
        return f"LOBEvent(t={self.timestamp:.4f}, type={event_name})"

class EventStream:
    """A sequence of LOB events"""
    def __init__(self, events: List[LOBEvent] = None):
        self.events = events or []
    
    def add_event(self, event: LOBEvent):
        self.events.append(event)
    
    def get_substreams(self, window_size: int, step: int = 1) -> List[List[LOBEvent]]:
        """Split the event stream into substreams using a rolling window approach"""
        if len(self.events) < window_size:
            return []
        
        substreams = []
        for i in range(0, len(self.events) - window_size + 1, step):
            substreams.append(self.events[i:i + window_size])
        
        return substreams
    
    def __len__(self):
        return len(self.events)

# Time-Event Encoder
class TimeEventEncoder(nn.Module):
    def __init__(self, embedding_dim: int = 64):
        super().__init__()
        self.embedding_dim = embedding_dim
        self.event_embedding = nn.Embedding(len(EVENT_TYPES), embedding_dim)
        
    def time_encoding(self, t):
        """Positional encoding for time values"""
        encoding = torch.zeros(len(t), self.embedding_dim)
        position = t.unsqueeze(1)
        div_term = torch.exp(torch.arange(0, self.embedding_dim, 2) * -(np.log(10000.0) / self.embedding_dim))
        
        encoding[:, 0::2] = torch.sin(position * div_term)
        encoding[:, 1::2] = torch.cos(position * div_term)
        
        return encoding
    
    def forward(self, times, events):
        """Encode time and event types"""
        time_embed = self.time_encoding(times)
        event_embed = self.event_embedding(events)
        
        # Combined embedding
        combined_embed = time_embed + event_embed
        
        return combined_embed, time_embed, event_embed

# Denoising Network
class DenoisingNetwork(nn.Module):
    def __init__(self, hidden_dim: int = 128):
        super().__init__()
        self.hidden_dim = hidden_dim
        
        # For time and event attention
        self.attention_fc = nn.Linear(hidden_dim * 2, hidden_dim)
        
        # Feed-forward networks
        self.time_ff = nn.Linear(hidden_dim * 2, hidden_dim)
        self.event_ff = nn.Linear(hidden_dim * 2, hidden_dim)
        
        # Output layers
        self.time_out = nn.Linear(hidden_dim, 1)
        self.event_out = nn.Linear(hidden_dim, len(EVENT_TYPES))
        
    def forward(self, x_noisy, history_context, step_embed):
        """
        Forward pass for denoising network
        x_noisy: Tensor containing noisy time and event
        history_context: Encoded history context
        step_embed: Embedding of the denoising step
        """
        # Combine history context with step embedding
        combined = torch.cat([history_context, step_embed], dim=-1)
        
        # Compute attention weights
        time_attention = F.softmax(self.attention_fc(combined), dim=-1)
        event_attention = F.softmax(self.attention_fc(combined), dim=-1)
        
        # Time prediction
        time_input = torch.cat([x_noisy[:, 0].unsqueeze(1), step_embed], dim=-1)
        time_hidden = F.relu(self.time_ff(time_input))
        time_pred = self.time_out(time_hidden)
        
        # Event prediction
        event_input = torch.cat([x_noisy[:, 1].unsqueeze(1), step_embed], dim=-1)
        event_hidden = F.relu(self.event_ff(event_input))
        event_logits = self.event_out(event_hidden)
        
        # Apply attention
        time_noise = torch.sum(time_attention * time_hidden, dim=-1, keepdim=True)
        event_noise = torch.sum(event_attention * event_hidden, dim=-1, keepdim=True)
        
        return time_noise, event_noise

# LOBDIF Model
class LOBDIF(nn.Module):
    def __init__(self, 
                 embedding_dim: int = 64, 
                 hidden_dim: int = 128,
                 diffusion_steps: int = 100,
                 beta_min: float = 1e-4,
                 beta_max: float = 0.02):
        super().__init__()
        
        self.embedding_dim = embedding_dim
        self.hidden_dim = hidden_dim
        self.diffusion_steps = diffusion_steps
        
        # Set up diffusion schedule
        self.betas = torch.linspace(beta_min, beta_max, diffusion_steps)
        self.alphas = 1 - self.betas
        self.alphas_cumprod = torch.cumprod(self.alphas, dim=0)
        
        # Components
        self.encoder = TimeEventEncoder(embedding_dim)
        self.denoiser = DenoisingNetwork(hidden_dim)
        
        # Step embedding
        self.step_embed = nn.Embedding(diffusion_steps, embedding_dim)
        
        # History processor (simplified for demo)
        self.history_processor = nn.Linear(embedding_dim * 3, hidden_dim)
        
    def forward(self, times, events, noise=None):
        """Forward pass for training"""
        batch_size = times.shape[0]
        
        # Get encodings
        combined_embed, time_embed, event_embed = self.encoder(times, events)
        
        # Simplified history context (for the demo)
        history_context = self.history_processor(
            torch.cat([combined_embed, time_embed, event_embed], dim=-1))
        
        # Sample noise if not provided
        if noise is None:
            noise = torch.randn_like(torch.cat([times.unsqueeze(1), 
                                               events.float().unsqueeze(1)], dim=1))
        
        # Sample diffusion step
        t = torch.randint(0, self.diffusion_steps, (batch_size,))
        step_embedding = self.step_embed(t)
        
        # Get noise scaling factors
        a = self.alphas_cumprod[t]
        
        # Add noise to inputs
        x_noisy_time = torch.sqrt(a) * times + torch.sqrt(1 - a) * noise[:, 0]
        x_noisy_event = torch.sqrt(a) * events.float() + torch.sqrt(1 - a) * noise[:, 1]
        x_noisy = torch.cat([x_noisy_time.unsqueeze(1), x_noisy_event.unsqueeze(1)], dim=1)
        
        # Predict noise
        time_noise_pred, event_noise_pred = self.denoiser(x_noisy, history_context, step_embedding)
        
        # Calculate loss
        time_loss = F.mse_loss(time_noise_pred.squeeze(), noise[:, 0])
        event_loss = F.mse_loss(event_noise_pred.squeeze(), noise[:, 1])
        
        return time_loss + event_loss
    
    def sample(self, history_context, skip_steps: int = 10):
        """Sample new time and event using skip-step sampling"""
        batch_size = history_context.shape[0]
        device = history_context.device
        
        # Start from pure noise
        x = torch.randn(batch_size, 2, device=device)
        
        # Simplified sampling schedule with skip steps
        steps = list(range(0, self.diffusion_steps, skip_steps))
        if steps[-1] != self.diffusion_steps - 1:
            steps.append(self.diffusion_steps - 1)
        
        # Reverse process
        for i in range(len(steps) - 1, 0, -1):
            t = steps[i]
            s = steps[i-1]
            
            # Step embedding
            step_embedding = self.step_embed(torch.tensor([t], device=device))
            
            # Predict noise
            time_noise_pred, event_noise_pred = self.denoiser(x, history_context, step_embedding)
            noise_pred = torch.cat([time_noise_pred, event_noise_pred], dim=1)
            
            # Apply skip-step formula
            alpha_t = self.alphas_cumprod[t]
            alpha_s = self.alphas_cumprod[s]
            
            x_0_pred = (x - torch.sqrt(1 - alpha_t) * noise_pred) / torch.sqrt(alpha_t)
            sigma_t = (1 - alpha_s) / (1 - alpha_t) * (1 - alpha_t / alpha_s)
            
            # Skip-step update
            x = torch.sqrt(alpha_s) * x_0_pred + torch.sqrt(1 - alpha_s - sigma_t) * noise_pred
            if s > 0:  # Add noise except for the final step
                x += torch.sqrt(sigma_t) * torch.randn_like(x)
        
        # Extract predictions
        time_pred = x[:, 0]
        event_logits = x[:, 1]
        event_pred = torch.round(torch.clamp(event_logits, 0, len(EVENT_TYPES) - 1)).long()
        
        return time_pred, event_pred

def prepare_data(stream: EventStream, window_size: int = 50):
    """Prepare data for training/testing"""
    substreams = stream.get_substreams(window_size)
    
    X_times = []
    X_events = []
    y_times = []
    y_events = []
    
    for substream in substreams:
        # Input: first window_size events
        X_times.append([event.timestamp for event in substream])
        X_events.append([event.event_type for event in substream])
        
        # Target: next event after window
        idx = substream[-1].timestamp + 1  # This would be the index in the original stream
        if idx < len(stream.events) - 1:
            next_event = stream.events[int(idx)]
            y_times.append(next_event.timestamp)
            y_events.append(next_event.event_type)
    
    # Convert to torch tensors
    X_times = torch.tensor(X_times, dtype=torch.float32)
    X_events = torch.tensor(X_events, dtype=torch.long)
    y_times = torch.tensor(y_times, dtype=torch.float32)
    y_events = torch.tensor(y_events, dtype=torch.long)
    
    return X_times, X_events, y_times, y_events

def train(model, train_data, valid_data, epochs: int = 200, lr: float = 0.001):
    """Train the LOBDIF model"""
    X_train_times, X_train_events, y_train_times, y_train_events = train_data
    X_valid_times, X_valid_events, y_valid_times, y_valid_events = valid_data
    
    optimizer = Adam(model.parameters(), lr=lr)
    
    train_losses = []
    valid_losses = []
    
    for epoch in range(epochs):
        # Training
        model.train()
        optimizer.zero_grad()
        
        loss = model(X_train_times, X_train_events)
        loss.backward()
        optimizer.step()
        
        train_losses.append(loss.item())
        
        # Validation
        model.eval()
        with torch.no_grad():
            valid_loss = model(X_valid_times, X_valid_events)
            valid_losses.append(valid_loss.item())
        
        if (epoch + 1) % 10 == 0:
            print(f"Epoch {epoch+1}/{epochs}, Train Loss: {loss.item():.4f}, Valid Loss: {valid_loss.item():.4f}")
    
    return train_losses, valid_losses
