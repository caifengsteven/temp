class StockEnvironment:
    """Simulated stock trading environment"""
    
    def __init__(self, 
                 stock_data: pd.DataFrame,

                 initial_balance: float = 10000.0,
                 trading_days: int = 252):
        
        self.stock_data = stock_data
        self.news_data = news_data
        self.initial_balance = initial_balance
        self.balance = initial_balance
        self.trading_days = trading_days
        self.current_day = 0
        self.portfolio = {}
        self.portfolio_value_history = []
    
    def reset(self):
        """Reset environment to initial state"""
        self.balance = self.initial_balance
        self.current_day = 0
        self.portfolio = {}
        self.portfolio_value_history = []
        return self._get_state()
    
    def _get_state(self):
        """Get current environment state"""
        current_prices = self.stock_data.iloc[self.current_day]
        current_news = self.news_data.iloc[self.current_day]
        
        portfolio_value = self.balance
        for symbol, shares in self.portfolio.items():
            portfolio_value += shares * current_prices[symbol]
        
        self.portfolio_value_history.append(portfolio_value)
        
        return {
            'day': self.current_day,
            'prices': current_prices,
            'news': current_news,
            'balance': self.balance,
            'portfolio': self.portfolio,
            'portfolio_value': portfolio_value
        }
    
    def step(self, actions: Dict[str, float]):
        """Execute actions and move to next step"""
        current_prices = self.stock_data.iloc[self.current_day]
        
        # Execute trades based on actions
        for symbol, action in actions.items():
            if action > 0:  # Buy
                shares_to_buy = action
                cost = shares_to_buy * current_prices[symbol]
                if cost <= self.balance:
                    self.balance -= cost
                    self.portfolio[symbol] = self.portfolio.get(symbol, 0) + shares_to_buy
            elif action < 0:  # Sell
                shares_to_sell = abs(action)
                if symbol in self.portfolio and self.portfolio[symbol] >= shares_to_sell:
                    self.balance += shares_to_sell * current_prices[symbol]
                    self.portfolio[symbol] -= shares_to_sell
                    if self.portfolio[symbol] == 0:
                        del self.portfolio[symbol]
        
        # Move to next day
        self.current_day += 1
        done = self.current_day >= self.trading_days
        
        new_state = self._get_state() if not done else None
        reward = 0 if not done else (self.portfolio_value_history[-1] - self.initial_balance)
        
        return new_state, reward, done

class CPPOAgent(PPOAgent):
    """Simplified CVaR-PPO agent for risk-sensitive stock trading"""
    
    def __init__(self, stock_symbols: List[str], cvar_alpha: float = 0.05):
        super().__init__(stock_symbols)
        self.cvar_alpha = cvar_alpha  # CVaR confidence level
        self.cvar_threshold = 0.0  # η in the paper
        self.trajectory_returns = []
    
    def update(self, trajectory: List[Tuple]):
        """Update policy based on collected trajectory with CVaR constraint"""
        super().update(trajectory)
        
        # Calculate return for the trajectory
        trajectory_return = sum(reward for _, reward, _ in trajectory)
        self.trajectory_returns.append(trajectory_return)
        
        # Update CVaR threshold (η)
        if len(self.trajectory_returns) > 10:
            # Sort returns in ascending order
            sorted_returns = sorted(self.trajectory_returns)
            # Calculate CVaR threshold as the average of the worst alpha% returns
            worst_returns_count = max(1, int(len(sorted_returns) * self.cvar_alpha))
            worst_returns = sorted_returns[:worst_returns_count]
            self.cvar_threshold = sum(worst_returns) / len(worst_returns)

                sf = 1.0 - self.infusion_strength
            elif recommendation_score >= 3.5 and base_actions[symbol] < 0:
                sf = 1.0 - self.infusion_strength/2
            
            # Apply the factor to the action
            base_actions[symbol] *= sf
        
        # Calculate aggregate risk score (Rf)
        rf = 1.0
        for symbol in self.stock_symbols:

                news_text = state['news'].get(symbol, "")
                risk_score = self.llm.get_risk_score(news_text)
                
                # Calculate symbol risk factor (Rf^i)
                symbol_rf = 1.0  # Default - no change

                elif risk_score <= 2.5:  # Low risk

def evaluate_performance(portfolio_history: List[float], benchmark_history: List[float]) -> Dict:
    """Calculate performance metrics"""
    # Convert to numpy arrays
    portfolio = np.array(portfolio_history)
    benchmark = np.array(benchmark_history)
    
    # Calculate daily returns
    portfolio_returns = np.diff(portfolio) / portfolio[:-1]
    benchmark_returns = np.diff(benchmark) / benchmark[:-1]
    
    # Calculate metrics
    total_return = (portfolio[-1] / portfolio[0]) - 1
    benchmark_return = (benchmark[-1] / benchmark[0]) - 1
    
    # Sharpe ratio (annualized, assuming 252 trading days)
    sharpe_ratio = np.sqrt(252) * np.mean(portfolio_returns) / np.std(portfolio_returns)
    
    # Information ratio
    tracking_error = np.std(portfolio_returns - benchmark_returns)
    information_ratio = np.mean(portfolio_returns - benchmark_returns) / tracking_error if tracking_error > 0 else 0
    
    # CVaR (5%)
    sorted_returns = np.sort(portfolio_returns)
    cvar_cutoff = int(len(sorted_returns) * 0.05)
    cvar = np.mean(sorted_returns[:cvar_cutoff]) if cvar_cutoff > 0 else sorted_returns[0]
    
    # Maximum drawdown
    cumulative_returns = np.cumprod(1 + portfolio_returns)
    peak = np.maximum.accumulate(cumulative_returns)
    drawdown = (cumulative_returns - peak) / peak
    max_drawdown = np.min(drawdown)
    
    # Rachev ratio (simplified)
    sorted_portfolio = np.sort(portfolio_returns)
    sorted_benchmark = np.sort(benchmark_returns)
    portfolio_tail = np.mean(sorted_portfolio[-int(len(sorted_portfolio)*0.05):])
    benchmark_tail = np.mean(sorted_benchmark[-int(len(sorted_benchmark)*0.05):])
    portfolio_loss = np.mean(sorted_portfolio[:int(len(sorted_portfolio)*0.05)])
    benchmark_loss = np.mean(sorted_benchmark[:int(len(sorted_benchmark)*0.05)])
    
    rachev_ratio = (portfolio_tail / benchmark_tail) / (abs(portfolio_loss) / abs(benchmark_loss)) if benchmark_loss != 0 and benchmark_tail != 0 else 1.0
    
    return {
        'total_return': total_return,
        'benchmark_return': benchmark_return,
        'sharpe_ratio': sharpe_ratio,
        'information_ratio': information_ratio,
        'cvar': cvar,
        'max_drawdown': max_drawdown,
        'rachev_ratio': rachev_ratio
    }


def run_backtest(agent, env, epochs: int = 10, steps_per_epoch: int = 20000) -> Tuple[List[float], Dict]:
    """Run backtest simulation for a trading agent"""
    try:
        all_portfolio_values = []
        
        for epoch in range(epochs):
            state = env.reset()
            done = False
            trajectory = []
            
            while not done:
                actions = agent.get_action(state)
                next_state, reward, done = env.step(actions)
                
                if not done:
                    trajectory.append((state, reward, next_state))
                    state = next_state
                    all_portfolio_values.append(state['portfolio_value'])
                
            # Update agent after each epoch
            agent.update(trajectory)
            
            print(f"Epoch {epoch+1}/{epochs} completed. Final portfolio value: ${state['portfolio_value']:.2f}")
        
        # Calculate benchmark (buy and hold index)
        benchmark_values = []
        initial_index_value = env.stock_data.iloc[0].mean()
        for i in range(len(env.stock_data)):
            current_index_value = env.stock_data.iloc[i].mean()
            benchmark_values.append(env.initial_balance * (current_index_value / initial_index_value))
        
        # Evaluate performance
        performance = evaluate_performance(all_portfolio_values, benchmark_values)
        
        return all_portfolio_values, performance
    
    except Exception as e:
        print(f"Error during backtest: {e}")
        return [], {}
