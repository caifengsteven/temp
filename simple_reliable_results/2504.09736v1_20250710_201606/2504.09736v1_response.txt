class AgenticWorkflow:
    def __init__(self, name: str, agents: List[Agent]):
        self.name = name
        self.agents = {agent.name: agent for agent in agents}
        self.execution_logs = []
        
    def execute(self, input_data: Any, checkpoints: List[str] = None) -> Dict[str, Any]:
        """Execute the workflow with human checkpoints if specified"""
        results = {"workflow": self.name, "start_time": datetime.now().isoformat()}
        current_data = input_data
        
        print(f"\n[INFO] Starting {self.name} workflow")
        
        for agent_name, agent in self.agents.items():
            print(f"[INFO] {agent_name} processing...")
            
            # Execute agent
            agent_result = agent.execute(current_data)
            self.execution_logs.append(agent_result)
            
            # Update current data for next agent
            current_data = agent_result["output"]
            
            # If this agent is a checkpoint, simulate human intervention
            if checkpoints and agent_name in checkpoints:
                print(f"[INFO] CHECKPOINT: {agent_name} output ready for human review")
                # Simulate human feedback
                human_feedback = self._simulate_human_feedback(agent_name)
                print(f"[INFO] Human provided feedback: '{human_feedback}'")
                current_data = f"{current_data} with modifications based on '{human_feedback}'"
        
        results["end_time"] = datetime.now().isoformat()
        results["output"] = current_data
        results["execution_logs"] = self.execution_logs
        
        # Calculate total execution time
        start = datetime.fromisoformat(results["start_time"])
        end = datetime.fromisoformat(results["end_time"])
        total_seconds = (end - start).total_seconds()
        results["total_execution_time"] = total_seconds
        
        print(f"[INFO] {self.name} workflow completed in {total_seconds:.2f} seconds")
        return results
