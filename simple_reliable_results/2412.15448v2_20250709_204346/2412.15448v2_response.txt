import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

class TechnicalIndicators:
    """Calculate technical indicators for time series data"""
    
    @staticmethod
    def sma(prices, window=20):
        """Simple Moving Average"""
        return prices.rolling(window=window).mean()
    
    @staticmethod
    def ema(prices, window=20):
        """Exponential Moving Average"""
        return prices.ewm(span=window, adjust=False).mean()
    
    @staticmethod
    def macd(prices, fast=12, slow=26, signal=9):
        """Moving Average Convergence Divergence"""
        fast_ema = TechnicalIndicators.ema(prices, window=fast)
        slow_ema = TechnicalIndicators.ema(prices, window=slow)
        macd_line = fast_ema - slow_ema
        signal_line = macd_line.ewm(span=signal, adjust=False).mean()
        # Normalize MACD
        macd_normalized = (macd_line - signal_line) / (0.5 * (abs(macd_line) + abs(signal_line)))
        return macd_normalized.fillna(0)
    
    @staticmethod
    def rsi(prices, window=14):
        """Relative Strength Index"""
        delta = prices.diff()
        gain = delta.clip(lower=0)
        loss = -delta.clip(upper=0)
        
        avg_gain = gain.rolling(window=window).mean()
        avg_loss = loss.rolling(window=window).mean()
        
        # Handle division by zero
        rs = avg_gain / avg_loss.replace(0, np.finfo(float).eps)
        rsi = 100 - (100 / (1 + rs))
        return rsi
    
    @staticmethod
    def bollinger_bands(prices, window=20, num_std=2):
        """Bollinger Bands"""
        sma = TechnicalIndicators.sma(prices, window)
        std = prices.rolling(window=window).std()
        upper_band = sma + (std * num_std)
        lower_band = sma - (std * num_std)
        
        # Normalized BB percentage
        bb_pct = (prices - lower_band) / (upper_band - lower_band)
        return bb_pct.fillna(0.5)  # Default to middle when undefined

class SimulatedData:
    """Generate simulated minute-level stock price data"""
    
    @staticmethod
    def generate_price_data(start_date, end_date, initial_price=400, volatility=0.0001):
        """
        Generate minute-level OHLCV data for a simulated SPY ETF
        """
        # Create date range for trading hours (9:30 AM to 4:00 PM ET)
        dates = []
        current_date = start_date
        while current_date <= end_date:
            if current_date.weekday() < 5:  # Monday to Friday
                for hour in range(9, 16):
                    if hour == 9:
                        # Only include 9:30 to 9:59
                        for minute in range(30, 60):
                            dates.append(datetime(current_date.year, current_date.month, current_date.day, hour, minute))
                    elif hour == 15:
                        # Only include 15:00 to 15:59
                        for minute in range(0, 60):
                            dates.append(datetime(current_date.year, current_date.month, current_date.day, hour, minute))
                    else:
                        # Include all minutes
                        for minute in range(0, 60):
                            dates.append(datetime(current_date.year, current_date.month, current_date.day, hour, minute))
            current_date += timedelta(days=1)
        
        # Generate price data
        n = len(dates)
        returns = np.random.normal(0, volatility, n)
        # Add small momentum effect
        for i in range(1, len(returns)):
            returns[i] += 0.2 * returns[i-1]
        
        # Calculate log prices
        log_prices = np.cumsum(returns)
        prices = initial_price * np.exp(log_prices)
        
        # Generate OHLCV data
        data = pd.DataFrame(index=dates)
        data['close'] = prices
        
        # Generate open, high, low based on close
        data['open'] = data['close'].shift(1)
        data.loc[data.index[0], 'open'] = initial_price
        
        # Add random variation for high and low
        high_var = np.random.uniform(0, 0.002, n)
        low_var = np.random.uniform(0, 0.002, n)
        
        data['high'] = data['close'] * (1 + high_var)
        data['low'] = data['close'] * (1 - low_var)
        
        # Ensure high >= open, close and low <= open, close
        data['high'] = data[['high', 'open', 'close']].max(axis=1)
        data['low'] = data[['low', 'open', 'close']].min(axis=1)
        
        # Generate volume data
        data['volume'] = np.random.lognormal(10, 1, n)
        
        # Add daily risk-free rate (10-year Treasury yield)
        data['risk_free_rate'] = 0.03 / 252  # Approximate daily rate from annual 3%
        
        return data

class ModelBuilder:
    """Build and train Random Forest models for high-frequency trading"""
    
    def __init__(self, data, test_size=0.2):
        """Initialize with data and test size"""
        self.data = data
        self.test_size = test_size
        self.split_index = int(len(data) * (1 - test_size))
        self.train_data = data.iloc[:self.split_index]
        self.test_data = data.iloc[self.split_index:]
        
    def prepare_features(self, include_indicators=None):
        """Prepare features for the model"""
        # Calculate log returns
        for col in ['open', 'high', 'low', 'close']:
            self.data[f'lr_{col}'] = np.log(self.data[col] / self.data[col].shift(1))
        
        # Normalize volume with Z-score (60-minute rolling window)
        self.data['volz60'] = (self.data['volume'] - self.data['volume'].rolling(60).mean()) / self.data['volume'].rolling(60).std()
        
        # Add technical indicators if requested
        if include_indicators:
            if 'sma' in include_indicators:
                self.data['sma20'] = self.data['close'] / TechnicalIndicators.sma(self.data['close'], 20)
            
            if 'ema' in include_indicators:
                self.data['ema20'] = self.data['close'] / TechnicalIndicators.ema(self.data['close'], 20)
            
            if 'macd' in include_indicators:
                self.data['macd'] = TechnicalIndicators.macd(self.data['close'])
            
            if 'rsi' in include_indicators:
                self.data['rsi'] = TechnicalIndicators.rsi(self.data['close'])
            
            if 'boll' in include_indicators:
                self.data['bollinger_pct'] = TechnicalIndicators.bollinger_bands(self.data['close'])
        
        # Drop NaN values
        self.data = self.data.dropna()
        
        # Update train/test split after adding features
        self.split_index = int(len(self.data) * (1 - self.test_size))
        self.train_data = self.data.iloc[:self.split_index]
        self.test_data = self.data.iloc[self.split_index:]
        
        # Define features and target
        self.feature_cols = [col for col in self.data.columns if col.startswith(('lr_', 'volz', 'sma', 'ema', 'macd', 'rsi', 'boll'))]
        self.target_col = 'lr_close'
        
        # Create X and y for train and test sets
        self.X_train = self.train_data[self.feature_cols].drop(columns=[self.target_col], errors='ignore')
        self.y_train = self.train_data[self.target_col]
        self.X_test = self.test_data[self.feature_cols].drop(columns=[self.target_col], errors='ignore')
        self.y_test = self.test_data[self.target_col]
        
        return self.X_train, self.y_train, self.X_test, self.y_test
        
    def build_model(self):
        """Build a Random Forest model with specified hyperparameters"""
        # Create model with hyperparameters from the paper
        model = RandomForestRegressor(
            n_estimators=100,
            max_depth=60,
            min_samples_split=10,
            max_features='log2',
            min_samples_leaf=1,
            random_state=42,
            n_jobs=-1
        )
        
        # Train the model
        model.fit(self.X_train, self.y_train)
        
        return model
    
    def evaluate_model(self, model):
        """Evaluate model performance"""
        # Make predictions
        y_train_pred = model.predict(self.X_train)
        y_test_pred = model.predict(self.X_test)
        
        # Calculate metrics for training set
        train_rmse = np.sqrt(mean_squared_error(self.y_train, y_train_pred))
        train_mae = mean_absolute_error(self.y_train, y_train_pred)
        train_r2 = r2_score(self.y_train, y_train_pred)
        
        # Calculate metrics for test set
        test_rmse = np.sqrt(mean_squared_error(self.y_test, y_test_pred))
        test_mae = mean_absolute_error(self.y_test, y_test_pred)
        test_r2 = r2_score(self.y_test, y_test_pred)
        
        # Calculate trend accuracy (direction prediction)
        train_trend = np.mean((self.y_train > 0) == (y_train_pred > 0))
        test_trend = np.mean((self.y_test > 0) == (y_test_pred > 0))
        
        # Store predictions in test data
        self.test_data['prediction'] = np.nan
        self.test_data.loc[self.X_test.index, 'prediction'] = y_test_pred
        
        results = {
            'train_rmse': train_rmse,
            'train_mae': train_mae,
            'train_r2': train_r2,
            'train_trend': train_trend,
            'test_rmse': test_rmse,
            'test_mae': test_mae,
            'test_r2': test_r2,
            'test_trend': test_trend
        }
        
        return results
    
    def get_feature_importance(self, model):
        """Extract feature importance from the model"""
        features = self.X_train.columns
        importance = model.feature_importances_
        
        # Sort features by importance
        indices = np.argsort(importance)[::-1]
        
        return {features[i]: importance[i] for i in indices}

class TradingSimulator:
    """Simulate trading based on model predictions"""
    
    def __init__(self, test_data, initial_capital=10000, turnover_limit=0.004):
        """Initialize with test data and parameters"""
        self.data = test_data.copy()
        self.initial_capital = initial_capital
        self.turnover_limit = turnover_limit
        
    def generate_signals(self, quantile_thresholds=(0.33, 0.66)):
        """Generate trading signals based on prediction quantiles"""
        predictions = self.data['prediction'].dropna()
        
        # Calculate quantiles from predictions
        q_low = np.nanquantile(predictions, quantile_thresholds[0])
        q_high = np.nanquantile(predictions, quantile_thresholds[1])
        
        # Generate signals
        self.data['signal'] = np.nan
        self.data.loc[self.data['prediction'] <= q_low, 'signal'] = -1  # Sell
        self.data.loc[(self.data['prediction'] > q_low) & 
                     (self.data['prediction'] < q_high), 'signal'] = 0  # Hold
        self.data.loc[self.data['prediction'] >= q_high, 'signal'] = 1  # Buy
        
        return self.data
    
    def run_simulation(self):
        """Run trading simulation with turnover constraints"""
        # Initialize portfolio variables
        portfolio = {
            'cash': self.initial_capital,
            'shares': 0,
            'value': self.initial_capital
        }
        
        # Create columns for tracking portfolio
        self.data['cash'] = np.nan
        self.data['shares'] = np.nan
        self.data['portfolio_value'] = np.nan
        self.data['returns'] = np.nan
        
        # Initialize first row with signals
        first_idx = self.data.loc[self.data['signal'].notna()].index[0]
        self.data.loc[first_idx, 'cash'] = portfolio['cash']
        self.data.loc[first_idx, 'shares'] = portfolio['shares']
        self.data.loc[first_idx, 'portfolio_value'] = portfolio['value']
        
        # Iterate through each timestamp with a signal
        for i, (idx, row) in enumerate(self.data.loc[self.data['signal'].notna()].iterrows()):
            if i == 0:
                continue  # Skip first row which is already initialized
            
            # Get previous portfolio state
            prev_idx = self.data.loc[self.data['signal'].notna()].index[i-1]
            prev_cash = self.data.loc[prev_idx, 'cash']
            prev_shares = self.data.loc[prev_idx, 'shares']
            prev_value = self.data.loc[prev_idx, 'portfolio_value']
            
            # Calculate current portfolio value before trading
            current_price = row['close']
            current_value = prev_cash + (prev_shares * current_price)
            
            # Determine action based on signal
            signal = row['signal']
            
            if signal == 1:  # Buy signal
                # Calculate how much to buy based on turnover limit
                max_buy_value = current_value * self.turnover_limit
                max_buy_shares = max_buy_value / current_price
                
                # Execute buy
                buy_shares = max_buy_shares
                buy_cost = buy_shares * current_price
                
                # Update portfolio
                new_cash = prev_cash - buy_cost
                new_shares = prev_shares + buy_shares
                
            elif signal == -1:  # Sell signal
                # Calculate how much to sell based on turnover limit
                max_sell_value = current_value * self.turnover_limit
                max_sell_shares = max_sell_value / current_price
                sell_shares = min(max_sell_shares, prev_shares)
                
                # Execute sell
                sell_proceeds = sell_shares * current_price
                
                # Update portfolio
                new_cash = prev_cash + sell_proceeds
                new_shares = prev_shares - sell_shares
                
            else:  # Hold signal
                new_cash = prev_cash
                new_shares = prev_shares
            
            # Calculate new portfolio value
            new_value = new_cash + (new_shares * current_price)
            
            # Calculate return
            if i > 0:
                daily_return = (new_value / prev_value) - 1
            else:
                daily_return = 0
            
            # Update data
            self.data.loc[idx, 'cash'] = new_cash
            self.data.loc[idx, 'shares'] = new_shares
            self.data.loc[idx, 'portfolio_value'] = new_value
            self.data.loc[idx, 'returns'] = daily_return
        
        # Fill forward portfolio values for all timestamps
        self.data[['cash', 'shares', 'portfolio_value']] = self.data[['cash', 'shares', 'portfolio_value']].ffill()
        
        # Calculate buy-and-hold benchmark
        benchmark_shares = self.initial_capital / self.data.iloc[0]['close']
        self.data['benchmark_value'] = benchmark_shares * self.data['close']
        
        return self.data
    
    def calculate_metrics(self):
        """Calculate performance metrics for the trading strategy"""
        # Extract portfolio values
        portfolio_values = self.data.loc[self.data['portfolio_value'].notna(), 'portfolio_value']
        
        # Calculate returns
        returns = portfolio_values.pct_change().dropna()
        
        # Calculate metrics
        sharpe_ratio = np.mean(returns) / np.std(returns) if np.std(returns) > 0 else 0
        
        # Calculate Sortino ratio (downside risk only)
        downside_returns = returns[returns < 0]
        sortino_ratio = np.mean(returns) / np.std(downside_returns) if len(downside_returns) > 0 and np.std(downside_returns) > 0 else 0
        
        # Calculate drawdown
        rolling_max = portfolio_values.cummax()
        drawdown = (portfolio_values - rolling_max) / rolling_max
        max_drawdown = drawdown.min()
        
        # Simplified Rachev ratio (comparing extreme gains to extreme losses)
        gains = returns[returns > 0]
        losses = returns[returns < 0]
        
        if len(gains) > 0 and len(losses) > 0:
            upper_tail = gains.quantile(0.9) if not gains.empty else 0
            lower_tail = losses.quantile(0.1) if not losses.empty else 0
            rachev_ratio = upper_tail / abs(lower_tail) if lower_tail != 0 else 0
        else:
            rachev_ratio = 0
        
        # Calculate final return
        final_return = (portfolio_values.iloc[-1] / portfolio_values.iloc[0]) - 1
        
        # Calculate benchmark return
        benchmark_values = self.data['benchmark_value']
        benchmark_return = (benchmark_values.iloc[-1] / benchmark_values.iloc[0]) - 1
        
        metrics = {
            'final_value': portfolio_values.iloc[-1],
            'total_return': final_return,
            'benchmark_return': benchmark_return,
            'sharpe_ratio': sharpe_ratio,
            'sortino_ratio': sortino_ratio,
            'max_drawdown': max_drawdown,
            'rachev_ratio': rachev_ratio
        }
        
        return metrics
