import random

class TeamMemberType(Enum):
    """Types of team members in a human-AI team"""
    HUMAN = "Human"
    AI = "AI"

class TaskType(Enum):
    """Types of tasks in team collaboration based on the paper's classification"""
    IDEA_GENERATION = "Idea Generation"
    SOLUTION_SELECTION = "Solution Selection"
    NEGOTIATION = "Negotiation"
    EXECUTION = "Execution"

class RoleFluidityLevel(Enum):
    """Levels of role fluidity in the team"""
    RIGID = "Rigid"
    MODERATE = "Moderate"
    FLUID = "Fluid"

class FeedbackType(Enum):
    """Types of feedback in human-AI collaboration"""
    DESCRIPTIVE = "Descriptive"
    NORMATIVE = "Normative"
    CORRECTIVE = "Corrective"
    EVALUATIVE = "Evaluative"

@dataclass
class TeamMember:
    """Represents a team member (human or AI)"""
    id: int
    type: TeamMemberType
    expertise: Dict[TaskType, float]  # Expertise level (0-1) for different task types
    adaptability: float  # Ability to adapt to role changes (0-1)
    trust_level: float  # Trust in other team members (0-1)
    
    def perform_task(self, task_type: TaskType, task_complexity: float) -> float:
        """Simulates task performance based on expertise and task complexity"""
        base_performance = self.expertise.get(task_type, 0.5)
        
        # Add some randomness to performance
        noise = random.uniform(-0.1, 0.1)
        
        # Performance decreases with task complexity
        complexity_factor = max(0, 1 - task_complexity * 0.5)
        
        return min(1.0, max(0.1, base_performance * complexity_factor + noise))

@dataclass
class TeamTask:
    """Represents a task for the team to perform"""
    id: int
    type: TaskType
    complexity: float  # Task complexity (0-1)
    description: str
    requires_collaboration: bool  # Whether the task requires collaboration
    
    def __str__(self):

    @property
    def ai_count(self) -> int:

def simulate_team_performance(team: TeamConfiguration, task: TeamTask) -> TeamPerformanceResult:
    """Simulate team performance on a given task"""
    # Calculate basic performance based on member expertise
    performance_scores = []
    for member in team.members:
        performance_scores.append(member.perform_task(task.type, task.complexity))
    
    # Calculate average performance
    base_performance = np.mean(performance_scores)
    
    # Apply role fluidity factor
    fluidity_factor = {
        RoleFluidityLevel.RIGID: 0.8,
        RoleFluidityLevel.MODERATE: 1.0,
        RoleFluidityLevel.FLUID: 1.2
    }
    
    # Apply collaboration factor if task requires it
    collaboration_factor = 1.0
    if task.requires_collaboration:
        # For collaborative tasks, team composition matters
        human_ratio = team.human_count / len(team.members)
        ai_ratio = team.ai_count / len(team.members)
        
        # Formula based on the paper's insights about human-AI synergy
        # Optimal mix provides bonus, extreme ratios reduce performance
        collaboration_factor = 1.0 + 0.3 * (4 * human_ratio * ai_ratio)
    
    # Apply feedback mechanism factor
    feedback_factor = {
        FeedbackType.DESCRIPTIVE: 0.9,
        FeedbackType.NORMATIVE: 1.0,
        FeedbackType.CORRECTIVE: 1.1,
        FeedbackType.EVALUATIVE: 1.05
    }
    
    # Calculate final performance score
    performance_score = base_performance * fluidity_factor[team.role_fluidity] * collaboration_factor * feedback_factor[team.feedback_mechanism]
    performance_score = min(1.0, max(0.1, performance_score))
    
    # Calculate other metrics
    completion_time = 1.0 + task.complexity - (performance_score * 0.5)
    
    # Conflict level increases with complexity and decreases with role fluidity
    conflict_base = task.complexity * 0.5
    if team.role_fluidity == RoleFluidityLevel.FLUID:
        conflict_adjustment = -0.2
    elif team.role_fluidity == RoleFluidityLevel.MODERATE:
        conflict_adjustment = 0
    else:
        conflict_adjustment = 0.2
    conflict_level = max(0.1, min(1.0, conflict_base + conflict_adjustment))
    
    # Team cohesion based on performance and conflict
    team_cohesion = 0.7 + (performance_score * 0.3) - (conflict_level * 0.3)
    team_cohesion = max(0.1, min(1.0, team_cohesion))
    
    # Role adaptation score
    adaptation_base = np.mean([m.adaptability for m in team.members])
    role_adaptation = adaptation_base * (1.0 if team.role_fluidity == RoleFluidityLevel.FLUID else 0.7)
    
    return TeamPerformanceResult(
        team_config=team,
        task=task,
        performance_score=performance_score,
        completion_time=completion_time,
        conflict_level=conflict_level,
        team_cohesion=team_cohesion,
        role_adaptation=role_adaptation
    )

def run_simulation(teams: List[TeamConfiguration], tasks: List[TeamTask]) -> List[TeamPerformanceResult]:
    """Run simulation for all team configurations across all tasks"""
    results = []
    
    for team in teams:
        for task in tasks:

def analyze_results(results: List[TeamPerformanceResult]) -> Dict:

        avg_cohesion = np.mean([r.team_cohesion for r in team_results])
        
        team_config = team_results[0].team_config
        
        team_summaries[team_id] = {
            "team_config": team_config,
            "avg_performance": avg_performance,
            "avg_completion_time": avg_completion_time,
            "avg_conflict": avg_conflict,
            "avg_cohesion": avg_cohesion,
            "human_count": team_config.human_count,
            "ai_count": team_config.ai_count,
            "role_fluidity": team_config.role_fluidity,
            "feedback_type": team_config.feedback_mechanism
        }
    
    # Analyze performance by task type
    task_type_performance = {}
    for result in results:
        task_type = result.task.type
        if task_type not in task_type_performance:
            task_type_performance[task_type] = []
        task_type_performance[task_type].append(result.performance_score)
    
    task_type_avg = {t: np.mean(scores) for t, scores in task_type_performance.items()}
    
    # Analyze impact of role fluidity
    fluidity_performance = {}
