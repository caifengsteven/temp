import numpy as np

class FBSDE:
    """Class representing Forward-Backward Stochastic Differential Equations"""
    
    def __init__(self, d, T, N, x0, b_func, sigma_func, f_func, g_func, batch_size=4096):
        """
        Initialize FBSDE parameters
        
        Args:
            d: dimension of state space
            T: terminal time
            N: number of time steps
            x0: initial state
            b_func: drift function (t, x, y, z) -> R^d
            sigma_func: diffusion function (t, x) -> R^(d×d)
            f_func: driver function (t, x, y, z) -> R
            g_func: terminal condition function g(x) -> R
            batch_size: size of batches for training
        """
        self.d = d
        self.T = T
        self.N = N
        self.h = T / N
        self.x0 = x0
        self.b_func = b_func
        self.sigma_func = sigma_func
        self.f_func = f_func
        self.g_func = g_func
        self.batch_size = batch_size
        
        # Time grid
        self.time_grid = np.linspace(0, T, N+1)
        
    def generate_brownian_increments(self):
        """Generate Brownian increments for simulation"""
        return np.sqrt(self.h) * np.random.normal(size=(self.batch_size, self.N, self.d))

class DeepBSDESolver:
    """Solver for FBSDE using the deep BSDE method"""
    
    def __init__(self, fbsde, hidden_layers=3, hidden_units=20):
        """
        Initialize the solver
        
        Args:
            fbsde: FBSDE object
            hidden_layers: number of hidden layers in neural networks
            hidden_units: number of units per hidden layer
        """
        self.fbsde = fbsde
        self.hidden_layers = hidden_layers
        self.hidden_units = hidden_units
        
        # Build networks
        self.y0 = tf.Variable(0.0, dtype=tf.float32)
        self.z_nets = []
        
        for _ in range(fbsde.N):
            net = tf.keras.Sequential()
            for _ in range(hidden_layers):
                net.add(Dense(hidden_units, activation='relu'))
            net.add(Dense(fbsde.d))
            self.z_nets.append(net)
            
    def train(self, epochs=100, learning_rate=0.001):
        """Train the deep BSDE method"""
        optimizer = Adam(learning_rate=learning_rate)
        losses = []
        y0_values = []
        
        for epoch in range(epochs):
            dW = self.fbsde.generate_brownian_increments()
            
            with tf.GradientTape() as tape:
                # Forward pass
                x_path = np.zeros((self.fbsde.batch_size, self.fbsde.N+1, self.fbsde.d))
                y_path = np.zeros((self.fbsde.batch_size, self.fbsde.N+1))
                z_path = np.zeros((self.fbsde.batch_size, self.fbsde.N, self.fbsde.d))
                
                # Initial values
                x_path[:, 0, :] = self.fbsde.x0
                y_path[:, 0] = self.y0
                
                # Simulate forward
                for n in range(self.fbsde.N):
                    t = self.fbsde.time_grid[n]
                    x = x_path[:, n, :]
                    y = y_path[:, n]
                    
                    # Get Z from neural network
                    z = self.z_nets[n](x)
                    z_path[:, n, :] = z
                    
                    # Compute drift
                    b = np.array([self.fbsde.b_func(t, x[i], y[i], z[i]) 
                                  for i in range(self.fbsde.batch_size)])
                    
                    # Compute diffusion
                    sigma = np.array([self.fbsde.sigma_func(t, x[i]) 
                                     for i in range(self.fbsde.batch_size)])
                    
                    # Compute driver
                    f = np.array([self.fbsde.f_func(t, x[i], y[i], z[i]) 
                                 for i in range(self.fbsde.batch_size)])
                    
                    # Update X
                    x_path[:, n+1, :] = x + b * self.fbsde.h + np.einsum('ijk,ik->ij', sigma, dW[:, n, :])
                    
                    # Update Y
                    y_path[:, n+1] = y - f * self.fbsde.h + np.sum(z * dW[:, n, :], axis=1)
                
                # Terminal condition
                g_XT = np.array([self.fbsde.g_func(x_path[i, -1, :]) 
                                for i in range(self.fbsde.batch_size)])
                
                # Loss is the squared error at terminal time
                loss = tf.reduce_mean(tf.square(y_path[:, -1] - g_XT))
            
            # Optimization step
            variables = [self.y0] + [p for net in self.z_nets for p in net.trainable_variables]
            gradients = tape.gradient(loss, variables)
            optimizer.apply_gradients(zip(gradients, variables))
            
            # Save for plotting
            losses.append(loss.numpy())
            y0_values.append(self.y0.numpy())
            
            if epoch % 10 == 0:
                print(f"Epoch {epoch}, Loss: {loss.numpy()}, Y0: {self.y0.numpy()}")
        
        return losses, y0_values
