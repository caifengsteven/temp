class AssetFittedDynamicVWAP(VWAPModel):

def generate_synthetic_asset_data(asset_name, n_days=100, bins_per_day=12, seed=None):
    """
    Generate synthetic price and volume data for an asset.
    
    Args:
        asset_name: Name of the asset
        n_days: Number of days of data to generate
        bins_per_day: Number of time bins per day
        seed: Random seed for reproducibility
    
    Returns:
        DataFrame with synthetic price and volume data
    """
    if seed is not None:
        np.random.seed(seed)
        random.seed(seed)
    
    n_bins = n_days * bins_per_day
    
    # Create time index
    start_date = datetime(2024, 1, 1)
    bin_duration = timedelta(hours=24/bins_per_day)
    timestamps = [start_date + i * bin_duration for i in range(n_bins)]
    
    # Generate base volume profile (U-shape for stocks, varies by asset)
    if "BTC" in asset_name or "ETH" in asset_name:
        # High liquidity assets have more stable profiles
        hour_of_day = np.array([i % bins_per_day for i in range(n_bins)])
        base_volume = 1 + 0.5 * np.sin(np.pi * hour_of_day / bins_per_day)**2
    elif "XMR" in asset_name:  
        # Outlier asset with erratic volume
        hour_of_day = np.array([i % bins_per_day for i in range(n_bins)])
        base_volume = 1 + 0.7 * np.sin(np.pi * hour_of_day / bins_per_day)**2
        # Add more noise
        base_volume = base_volume * (1 + 0.5 * np.random.randn(n_bins))
    else:
        # Regular assets
        hour_of_day = np.array([i % bins_per_day for i in range(n_bins)])
        base_volume = 1 + 0.3 * np.sin(np.pi * hour_of_day / bins_per_day)**2
    
    # Add day-of-week effect
    day_of_week = np.array([int(i / bins_per_day) % 7 for i in range(n_bins)])
    day_effect = 1 - 0.2 * (day_of_week >= 5)  # Weekend effect
    
    # Add trend and noise to volume
    trend = np.linspace(0.8, 1.2, n_bins)
    noise = 0.2 * np.random.randn(n_bins)
    volume = base_volume * day_effect * trend + noise
    volume = np.maximum(volume, 0.1)  # Ensure positive volume
    
    # Generate price data with realistic properties
    # Start with random walk
    price_changes = 0.001 * np.random.randn(n_bins)
    
    # Add momentum and mean reversion
    momentum = 0.2 * np.sign(np.convolve(price_changes, np.ones(5)/5, mode='same'))

    price_changes = price_changes + momentum + mean_reversion + vol_price_effect

    base_price = 100 if "BTC" not in asset_name else 30000

def compute_vwap(prices, volumes):
    """Compute the Volume Weighted Average Price."""
    return np.sum(prices * volumes) / np.sum(volumes)

def compute_execution_price(prices, allocation):
    """Compute the execution price given an allocation strategy."""
    return np.sum(prices * allocation)

def compute_vwap_loss(prices, volumes, allocation):
    """
    Compute VWAP execution losses.
    
    Args:
        prices: Array of prices
        volumes: Array of market volumes
        allocation: Array of order allocation fractions (sums to 1)
    
    Returns:
        (absolute_loss, quadratic_loss)
    """
    market_vwap = compute_vwap(prices, volumes)
    execution_price = compute_execution_price(prices, allocation)
    
    # Compute losses in basis points (1bp = 0.01%)
    abs_loss = abs(execution_price - market_vwap) / market_vwap * 10000
    quad_loss = ((execution_price - market_vwap) / market_vwap * 10000) ** 2
    
    return abs_loss, quad_loss

def evaluate_model(model, test_data, horizon=12):
    """
    Evaluate a VWAP model on test data.
    
    Args:
        model: Trained VWAP model
        test_data: Test dataset
        horizon: Number of time bins in each VWAP order
    
    Returns:
        DataFrame with evaluation results
    """
    results = []
    
    # Group by asset
    for asset, asset_data in test_data.groupby('asset'):
        # Sliding window evaluation
        for start_idx in range(0, len(asset_data) - horizon, horizon // 2):
            window = asset_data.iloc[start_idx:start_idx + horizon]
            
            if len(window) < horizon:
                continue
                
            # Naive allocation (equal split)
            naive_allocation = np.ones(horizon) / horizon
            
            # Model allocation
            model_allocation = model.predict_allocation(window)
            
            # Compute VWAP losses
            prices = window['price'].values
            volumes = window['volume'].values
            
            naive_abs_loss, naive_quad_loss = compute_vwap_loss(
                prices, volumes, naive_allocation)
            
            model_abs_loss, model_quad_loss = compute_vwap_loss(
                prices, volumes, model_allocation)
            
            # Compute improvements
            abs_improvement = (naive_abs_loss - model_abs_loss) / naive_abs_loss * 100
            quad_improvement = (naive_quad_loss - model_quad_loss) / naive_quad_loss * 100
            
            # Record results
            results.append({
                'asset': asset,
                'start_time': window['timestamp'].iloc[0],
                'model': model.name,
                'naive_abs_loss': naive_abs_loss,
                'model_abs_loss': model_abs_loss,
                'naive_quad_loss': naive_quad_loss,
                'model_quad_loss': model_quad_loss,
                'abs_improvement': abs_improvement,
                'quad_improvement': quad_improvement
            })
    
    return pd.DataFrame(results)

def plot_results(results_df):

def simulate_real_time_trading(models, assets, duration_minutes=480, num_orders=5):
    """
    Simulate real-time trading with different VWAP models.
    
    Args:
        models: Dictionary of VWAP models
        assets: List of asset data frames
        duration_minutes: Duration of each order in minutes
        num_orders: Number of orders to simulate per asset
    
    Returns:
        DataFrame with trading results
    """
    results = []
    
    for asset_df in assets:
        asset_name = asset_df['asset'].iloc[0]
        print(f"Simulating trades for {asset_name}...")
        
        # Get the bin duration in minutes
        timestamps = asset_df['timestamp']
        bin_duration = (timestamps.iloc[1] - timestamps.iloc[0]).total_seconds() / 60
        
        # Calculate how many bins are needed for the order duration
        bins_per_order = int(duration_minutes / bin_duration)
        
        # Ensure we have enough data
        if len(asset_df) < bins_per_order + 10:
            print(f"  Not enough data for {asset_name}, skipping")
            continue
            
        # Generate random start points for orders
        max_start = len(asset_df) - bins_per_order - 1
        start_indices = random.sample(range(max_start), min(num_orders, max_start))
        
        for start_idx in start_indices:
            # Extract data for this order
            order_data = asset_df.iloc[start_idx:start_idx + bins_per_order].copy()
            
            # Prices and volumes for this order
            prices = order_data['price'].values
            volumes = order_data['volume'].values
            
            # Market VWAP (benchmark)
            market_vwap = compute_vwap(prices, volumes)
            
            # TWAP allocation (equal weights)
            twap_allocation = np.ones(bins_per_order) / bins_per_order
            twap_price = compute_execution_price(prices, twap_allocation)
            
            # Calculate TWAP metrics
            twap_abs_deviation = abs(twap_price - market_vwap) / market_vwap * 10000  # basis points
            twap_quad_deviation = ((twap_price - market_vwap) / market_vwap * 10000) ** 2  # millionths
            
            # Evaluate each model
            for name, model in models.items():
                # Skip TWAP as we've already calculated it
                if name == "TWAP":
                    continue
                    
                # Get model allocation
                model_allocation = model.predict_allocation(order_data)
                
                # Calculate execution price and deviations
                model_price = compute_execution_price(prices, model_allocation)
                model_abs_deviation = abs(model_price - market_vwap) / market_vwap * 10000
                model_quad_deviation = ((model_price - market_vwap) / market_vwap * 10000) ** 2
                
                # Calculate improvements
                abs_improvement = (twap_abs_deviation - model_abs_deviation) / twap_abs_deviation * 100
                quad_improvement = (twap_quad_deviation - model_quad_deviation) / twap_quad_deviation * 100
                
                results.append({
                    'asset': asset_name,
                    'order_start': order_data['timestamp'].iloc[0],
                    'duration_minutes': duration_minutes,
                    'model': name,
                    'twap_abs_deviation': twap_abs_deviation,
                    'model_abs_deviation': model_abs_deviation,
                    'twap_quad_deviation': twap_quad_deviation,
                    'model_quad_deviation': model_quad_deviation,
                    'abs_improvement': abs_improvement,
                    'quad_improvement': quad_improvement
                })
    
    return pd.DataFrame(results)
