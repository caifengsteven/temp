import numpy as np

@dataclass
class Bond:
    isin: str
    notional: float
    fair_value: float
    modified_duration: float
    convexity: float
    maturity_date: datetime
    accounting_type: str  # 'FV' (fair value) or 'AC' (amortized cost)
    
    @property
    def time_to_maturity(self) -> float:
        """Return time to maturity in years"""
        days = (self.maturity_date - datetime.now()).days
        return max(0, days / 365)
    
    @property
    def bucket(self) -> str:
        """Return maturity bucket"""
        ttm = self.time_to_maturity
        if ttm < 1:
            return "<1Y"
        elif ttm < 5:
            return "1Y-5Y"
        elif ttm < 10:
            return "5Y-10Y"
        else:
            return ">10Y"
    
    def price_after_shock(self, rate_shock: float) -> float:
        """Calculate new price after interest rate shock (in basis points)"""
        rate_shock_decimal = rate_shock / 100  # Convert from basis points to decimal
        # Using delta-gamma approximation as described in the paper
        price_change = -self.modified_duration * rate_shock_decimal + 0.5 * self.convexity * rate_shock_decimal**2
        return self.fair_value * (1 + price_change)


@dataclass
class Swap:
    notional: float
    market_value: float
    maturity_date: datetime
    is_long: bool  # True if paying fixed (long), False if receiving fixed (short)
    fixed_rate: float
    dv01: float  # Dollar value of 1bp change
    
    @property
    def time_to_maturity(self) -> float:
        """Return time to maturity in years"""
        days = (self.maturity_date - datetime.now()).days
        return max(0, days / 365)
    
    @property
    def bucket(self) -> str:
        """Return maturity bucket"""
        ttm = self.time_to_maturity
        if ttm < 1:
            return "<1Y"
        elif ttm < 5:
            return "1Y-5Y"
        elif ttm < 10:
            return "5Y-10Y"
        else:
            return ">10Y"
    
    def value_after_shock(self, rate_shock: float) -> float:
        """Calculate new value after interest rate shock (in basis points)"""
        value_change = self.dv01 * rate_shock
        # If long (pay fixed), value increases when rates rise
        # If short (receive fixed), value decreases when rates rise
        sign = 1 if self.is_long else -1
        return self.market_value + sign * value_change


class Bank:
    def __init__(self, name: str, cet1: float):
        self.name = name
        self.cet1 = cet1  # Common Equity Tier 1 capital
        self.bonds: List[Bond] = []
        self.swaps: List[Swap] = []
    
    def add_bond(self, bond: Bond) -> None:
        self.bonds.append(bond)
    
    def add_swap(self, swap: Swap) -> None:
        self.swaps.append(swap)
    
    def total_bond_fair_value(self) -> Dict[str, float]:
        """Return total fair value of bonds by accounting type"""
        fv_bonds = sum(b.fair_value for b in self.bonds if b.accounting_type == 'FV')
        ac_bonds = sum(b.fair_value for b in self.bonds if b.accounting_type == 'AC')
        return {'FV': fv_bonds, 'AC': ac_bonds, 'Total': fv_bonds + ac_bonds}
    
    def total_swap_notional(self) -> Dict[str, float]:
        """Return total notional of swaps by position"""
        long_notional = sum(s.notional for s in self.swaps if s.is_long)
        short_notional = sum(s.notional for s in self.swaps if not s.is_long)
        return {'Long': long_notional, 'Short': short_notional, 'Net': long_notional - short_notional}
    
    def swap_impact(self, rate_shock: float) -> float:
        """Calculate impact of rate shock on swap portfolio"""
        new_values = [s.value_after_shock(rate_shock) for s in self.swaps]
        old_values = [s.market_value for s in self.swaps]
        return sum(new_values) - sum(old_values)
    
    def bond_impact(self, rate_shock: float) -> Dict[str, float]:
        """Calculate impact of rate shock on bond portfolio by accounting type"""
        impact_fv = 0
        impact_ac = 0
        
        for bond in self.bonds:
            new_price = bond.price_after_shock(rate_shock)
            impact = new_price - bond.fair_value
            
            if bond.accounting_type == 'FV':
                impact_fv += impact
            else:
                impact_ac += impact
        
        return {'FV': impact_fv, 'AC': impact_ac, 'Total': impact_fv + impact_ac}
    
    def total_impact(self, rate_shock: float) -> Dict[str, float]:
        """Calculate total impact of rate shock as percentage of CET1"""
        swap_impact = self.swap_impact(rate_shock)
        bond_impact = self.bond_impact(rate_shock)
        
        results = {
            'Swap Impact': swap_impact,
            'FV Bond Impact': bond_impact['FV'],
            'AC Bond Impact': bond_impact['AC'],
            'Total Bond Impact': bond_impact['Total'],
            'FV + Swap Impact': bond_impact['FV'] + swap_impact,
            'Total Impact': bond_impact['Total'] + swap_impact
        }
        
        # Also calculate as percentage of CET1
        results_pct = {k: (v / self.cet1) * 100 for k, v in results.items()}
        
        return {'Absolute': results, 'Percent of CET1': results_pct}


def generate_test_bank(name: str, 
                       size: str,  # 'large' for SI, 'small' for LSI
                       cet1: Optional[float] = None,
                       hedge_ratio: float = 0.5) -> Bank:
    """Generate test bank with realistic bond and swap portfolios"""

        cet1 = random.uniform(10e9, 50e9) if size == 'large' else random.uniform(500e6, 5e9)

        "<1Y": 0.15,
        "1Y-5Y": 0.45,
        "5Y-10Y": 0.30,
        ">10Y": 0.10
    }
    
    # Distribution of accounting types
    ac_ratio = 0.7  # 70% at amortized cost, 30% at fair value
    
    for i in range(bond_count):
        # Determine maturity bucket
        bucket = np.random.choice(list(maturity_weights.keys()), p=list(maturity_weights.values()))
        
        # Set maturity date based on bucket
        if bucket == "<1Y":
            days_to_maturity = random.randint(30, 364)
        elif bucket == "1Y-5Y":
            days_to_maturity = random.randint(365, 5*365)
        elif bucket == "5Y-10Y":
            days_to_maturity = random.randint(5*365+1, 10*365)
        else:  # >10Y
            days_to_maturity = random.randint(10*365+1, 30*365)
            
        maturity_date = datetime.now() + timedelta(days=days_to_maturity)
        
        # Assign accounting type
        accounting_type = 'AC' if random.random() < ac_ratio else 'FV'
        
        # Set modified duration based on maturity
        ttm = days_to_maturity / 365  # time to maturity in years
        modified_duration = min(0.9 * ttm, 15)  # simplification
        
        # Convexity typically scales with duration squared
        convexity = 0.1 * modified_duration**2
        
        # Set fair value
        fair_value = total_bond_value / bond_count
        
        # Create and add bond
        bond = Bond(
            isin=f"BOND{i:06d}",
            notional=fair_value * 1.05,  # slight premium
            fair_value=fair_value,
            modified_duration=modified_duration,
            convexity=convexity,
            maturity_date=maturity_date,
            accounting_type=accounting_type
        )
        bank.add_bond(bond)
    
    # Generate swap portfolio based on bond portfolio and hedge ratio
    # Estimate interest rate risk from bonds to determine swap positions
    bond_dv01 = 0
    for bond in bank.bonds:
        bond_dv01 -= bond.fair_value * bond.modified_duration / 100  # Convert to basis point impact
    
    # Create offsetting swap position based on hedge ratio
    target_swap_dv01 = bond_dv01 * hedge_ratio
    
    # Generate swaps to reach target DV01
    current_dv01 = 0
    swap_count = random.randint(50, 500) if size == 'large' else random.randint(5, 50)
    
    for i in range(swap_count):
        # Determine if long or short to move toward target DV01
        should_be_long = current_dv01 < target_swap_dv01
        
        # Randomize a bit to add noise
        is_long = should_be_long if random.random() < 0.7 else not should_be_long
        
        # Determine maturity bucket
        bucket = np.random.choice(list(maturity_weights.keys()), p=list(maturity_weights.values()))
        
        # Set maturity date based on bucket
        if bucket == "<1Y":
            days_to_maturity = random.randint(30, 364)
        elif bucket == "1Y-5Y":
            days_to_maturity = random.randint(365, 5*365)
        elif bucket == "5Y-10Y":
            days_to_maturity = random.randint(5*365+1, 10*365)
        else:  # >10Y
            days_to_maturity = random.randint(10*365+1, 30*365)
            
        maturity_date = datetime.now() + timedelta(days=days_to_maturity)
        
        # Calculate DV01 based on maturity
        ttm = days_to_maturity / 365
        dv01 = (ttm / 10) * 1e6  # simplified DV01 calculation
        
        notional = random.uniform(5e6, 50e6) if size == 'large' else random.uniform(1e6, 10e6)
        
        # For simplicity, assume initial market value close to zero
        market_value = random.uniform(-1e5, 1e5)  # small initial value
        
        # Fixed rate (approximate market levels)
        fixed_rate = 3.0 + (ttm * 0.2)  # simplified term structure
        
        # Create and add swap
        swap = Swap(

            market_value=market_value,
            maturity_date=maturity_date,
            is_long=is_long,
            fixed_rate=fixed_rate,
            dv01=dv01 * (1 if is_long else -1)
        )
        bank.add_swap(swap)
        
        # Update current dv01
        current_dv01 += swap.dv01
        
        # Stop if we've reached the target
        if abs(current_dv01 - target_swap_dv01) / abs(target_swap_dv01) < 0.1:

    return bank


def analyze_bank_portfolio(bank: Bank, rate_shock: float = 100) -> None:
    """Analyze bank portfolio and interest rate shock impact"""
    # Basic portfolio statistics
    bond_values = bank.total_bond_fair_value()
    swap_notionals = bank.total_swap_notional()
    
    print(f"\n=== {bank.name} Analysis ===")
    print(f"CET1 Capital: €{bank.cet1/1e9:.2f} billion")
    print("\nBond Portfolio:")
    print(f"  FV Bonds: €{bond_values['FV']/1e9:.2f} billion")
    print(f"  AC Bonds: €{bond_values['AC']/1e9:.2f} billion")
    print(f"  Total: €{bond_values['Total']/1e9:.2f} billion")
    
    print("\nSwap Portfolio:")

    print(f"  Short Notional: €{swap_notionals['Short']/1e9:.2f} billion")
    print(f"  Net Notional: €{swap_notionals['Net']/1e9:.2f} billion")
    
    # Interest rate shock impact
    impact = bank.total_impact(rate_shock)
    
    print(f"\nImpact of {rate_shock} bps Interest Rate Shock:")
    print(f"  Swap Impact: €{impact['Absolute']['Swap Impact']/1e9:.2f} billion ({impact['Percent of CET1']['Swap Impact']:.2f}% of CET1)")
    print(f"  FV Bond Impact: €{impact['Absolute']['FV Bond Impact']/1e9:.2f} billion ({impact['Percent of CET1']['FV Bond Impact']:.2f}% of CET1)")
    print(f"  AC Bond Impact: €{impact['Absolute']['AC Bond Impact']/1e9:.2f} billion ({impact['Percent of CET1']['AC Bond Impact']:.2f}% of CET1)")

    print(f"  Total Impact: €{impact['Absolute']['Total Impact']/1e9:.2f} billion ({impact['Percent of CET1']['Total Impact']:.2f}% of CET1)")


def run_banking_system_simulation() -> None:
    """Simulate a banking system with multiple banks and analyze shock impact"""
    try:
        # Create several banks with different characteristics
        banks = [
            generate_test_bank("BigBank Italia", "large", 40e9, 0.7),  # Large bank with high hedge ratio
            generate_test_bank("Banca Medio", "large", 15e9, 0.5),     # Medium bank with average hedge ratio
            generate_test_bank("Piccola Banca", "small", 2e9, 0.3),    # Small bank with low hedge ratio
            generate_test_bank("Micro Financiero", "small", 800e6, 0.1), # Very small bank with minimal hedging
            generate_test_bank("Risk Taker Bank", "large", 20e9, -0.2)  # Bank with negative hedge ratio (amplifying risk)
        ]
        
        # Analyze each bank
        for bank in banks:
            analyze_bank_portfolio(bank)
        
        # Banking system aggregate analysis
        system = Bank("Italian Banking System", sum(b.cet1 for b in banks))
        for bank in banks:
            for bond in bank.bonds:
                system.add_bond(bond)
            for swap in bank.swaps:
                system.add_swap(swap)
        
        analyze_bank_portfolio(system)
        
        # Visualize results
        plot_system_results(banks, system)
        
    except Exception as e:
        print(f"An error occurred during simulation: {e}")


def plot_system_results(banks: List[Bank], system: Bank) -> None:
    """Plot results for the banking system"""
    # Prepare data
    bank_names = [b.name for b in banks] + ["System Total"]
    all_entities = banks + [system]
    
    # Calculate impacts for 100 bps shock
    swap_impacts = [b.total_impact(100)['Percent of CET1']['Swap Impact'] for b in all_entities]
    fv_impacts = [b.total_impact(100)['Percent of CET1']['FV Bond Impact'] for b in all_entities]
    ac_impacts = [b.total_impact(100)['Percent of CET1']['AC Bond Impact'] for b in all_entities]
    total_impacts = [b.total_impact(100)['Percent of CET1']['Total Impact'] for b in all_entities]
    
    # Create stacked bar chart
    fig, ax = plt.subplots(figsize=(12, 8))
    
    x = np.arange(len(bank_names))
    width = 0.6
    
    # Plot bars
    ax.bar(x, swap_impacts, width, label='Swap Impact', color='green')
    ax.bar(x, fv_impacts, width, bottom=swap_impacts, label='FV Bond Impact', color='orange')
    ax.bar(x, ac_impacts, width, bottom=[a+b for a,b in zip(swap_impacts, fv_impacts)], 
           label='AC Bond Impact', color='red')
    
    # Add net impact markers
    ax.scatter(x, total_impacts, color='black', s=100, zorder=3, label='Net Impact')
    
    # Add zero line
    ax.axhline(y=0, color='k', linestyle='-', alpha=0.3)
    
    # Add labels and legend
    ax.set_xlabel('Bank')
    ax.set_ylabel('Impact as % of CET1')
    ax.set_title('Impact of 100 bps Interest Rate Shock on Bank Portfolios')
    ax.set_xticks(x)
    ax.set_xticklabels(bank_names, rotation=45, ha='right')
    ax.legend()
    
    plt.tight_layout()
    plt.show()
