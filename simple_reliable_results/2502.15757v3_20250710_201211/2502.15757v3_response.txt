class LOBRecord:
    """Representation of a single LOB record as described in the paper."""
    def __init__(self, ask_prices: np.ndarray, ask_volumes: np.ndarray, 
                 bid_prices: np.ndarray, bid_volumes: np.ndarray):
        self.ask_prices = ask_prices
        self.ask_volumes = ask_volumes
        self.bid_prices = bid_prices
        self.bid_volumes = bid_volumes
        
    def mid_price(self) -> float:
        """Calculate mid-price as average of best bid and ask."""
        return (self.ask_prices[0] + self.bid_prices[0]) / 2
    
    def spread(self) -> float:
        """Calculate spread as difference between best ask and bid."""
        return self.ask_prices[0] - self.bid_prices[0]
    
    def spread_pct(self) -> float:
        """Calculate spread as percentage of mid-price."""
        return self.spread() / self.mid_price()

class LOBTimeSeries:
    """Time series of LOB records."""
    def __init__(self, records: List[LOBRecord]):
        self.records = records
        
    def get_window(self, start_idx: int, length: int) -> List[LOBRecord]:
        """Get a window of LOB records."""
        end_idx = min(start_idx + length, len(self.records))
        return self.records[start_idx:end_idx]
    
    def calculate_mid_prices(self) -> np.ndarray:
        """Calculate mid-prices for all records."""
        return np.array([record.mid_price() for record in self.records])
    
    def calculate_spreads(self) -> np.ndarray:
        """Calculate spreads for all records."""
        return np.array([record.spread() for record in self.records])

    def calculate_spreads_pct(self) -> np.ndarray:
        """Calculate spreads as percentage of mid-price for all records."""
        return np.array([record.spread_pct() for record in self.records])

class TrendLabeler:
    """Class for labeling price trends using methods described in the paper."""
    
    @staticmethod
    def original_method(mid_prices: np.ndarray, t: int, k: int, theta: float) -> int:
        """Original labeling method from FI-2010 dataset."""
        if t + k >= len(mid_prices):
            return -1  # Cannot label
        
        m_plus = np.mean(mid_prices[t:t+k+1])
        percentage_change = (m_plus - mid_prices[t]) / mid_prices[t]
        
        if percentage_change > theta:
            return 0  # Upward
        elif percentage_change < -theta:
            return 1  # Downward
        else:
            return 2  # Stable
    
    @staticmethod
    def tsantekidis_method(mid_prices: np.ndarray, t: int, k: int, theta: float) -> int:
        """Tsantekidis et al. labeling method."""
        if t - k < 0 or t + k >= len(mid_prices):
            return -1  # Cannot label
        
        m_plus = np.mean(mid_prices[t:t+k+1])
        m_minus = np.mean(mid_prices[t-k:t+1])
        percentage_change = (m_plus - m_minus) / m_minus
        
        if percentage_change > theta:
            return 0  # Upward
        elif percentage_change < -theta:
            return 1  # Downward
        else:
            return 2  # Stable
    
    @staticmethod
    def proposed_method(mid_prices: np.ndarray, t: int, h: int, k: int, theta: float) -> int:
        """Proposed labeling method from the paper."""
        if t - k < 0 or t + h + k >= len(mid_prices):
            return -1  # Cannot label
        
        w_plus = np.mean(mid_prices[t+h-k:t+h+1])
        w_minus = np.mean(mid_prices[t-k:t+1])
        percentage_change = (w_plus - w_minus) / w_minus
        
        if percentage_change > theta:
            return 0  # Upward
        elif percentage_change < -theta:
            return 1  # Downward
        else:
            return 2  # Stable

                errors_needed = int(n_samples * (1 - correct_ratio))
                
                # Choose random indices to introduce errors
                error_indices = np.random.choice(n_samples, errors_needed, replace=False)
                
                for idx in error_indices:
                    true_class = true_labels[h][idx]
                    # Choose a random wrong class
                    other_classes = [c for c in range(n_classes) if c != true_class]
                    predictions[idx] = np.random.choice(other_classes)
            
            # Calculate F1 score
            f1 = f1_score(true_labels[h], predictions, average='weighted')
            
            # Save results
            horizon_results[model_name] = f1
            
            print(f"Horizon {h}, {model_name}: F1={f1:.4f}")
            print(f"  Expected from paper: {ablation_results[h][model_name]:.4f}")
        
        results[h] = horizon_results
    
    return results

def plot_ablation_results(ablation_results):
    """Plot the results of the ablation study."""
    horizons = sorted(ablation_results.keys())
    models = list(ablation_results[horizons[0]].keys())

    for h in horizons:

    plt.ylabel('F1 Score')
    plt.title('Ablation Study Results')
    plt.legend()
