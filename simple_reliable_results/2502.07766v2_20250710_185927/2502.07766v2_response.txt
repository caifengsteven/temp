from dataclasses import dataclass

@dataclass
class ModelParameters:

    theta: float  # long-term volatility
    lambda_: float  # speed of mean reversion of volatility
    nu: float  # volatility of volatility
    rho: float  # correlation between forward price and volatility processes

class CIRZABR:

    def tau_ex(self, T_ex):
        """Calculate tau_ex in CIR-ZABR."""
        alp = self.params.alpha
        the = self.params.theta
        lam = self.params.lambda_
        E = np.exp(lam * T_ex)
        
        numerator = (2 * T_ex * lam * the**2 * E**2 - alp**2 + 2 * alp * the - the**2 
                    - 4 * the * E * (alp - the) + E**2 * (alp**2 + 2 * alp * the - 3 * the**2))
        denominator = (2 * lam * E**2)
        
        return numerator / denominator

        """Calculate G_int in CIR-ZABR."""

        return numerator / denominator
