import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression, Lasso
from sklearn.metrics import mean_squared_error, mean_absolute_error
import warnings
warnings.filterwarnings('ignore')

class VolatilityForecasting:
    def __init__(self, decay_rate=0.5):
        """
        Initialize the volatility forecasting model.
        
        Parameters:
        - decay_rate: The decay parameter for the exponential kernel function
        """
        self.decay_rate = decay_rate
        
    def generate_simulated_data(self, n_days=5000, intraday_periods=78):
        """
        Generate simulated high-frequency price data.
        
        Parameters:
        - n_days: Number of trading days
        - intraday_periods: Number of intraday periods (5-min intervals in a 6.5hr trading day)
        
        Returns:
        - DataFrame with price and return data
        """
        np.random.seed(42)
        
        # Starting price
        initial_price = 1000
        
        # Create empty arrays for timestamps, prices, and returns
        all_timestamps = []
        all_prices = []
        all_returns = []
        
        # Generate day-specific volatility (to create volatility clustering)
        daily_vol = np.zeros(n_days)
        base_vol = 0.02  # Base volatility
        for i in range(n_days):
            if i == 0:
                daily_vol[i] = base_vol
            else:
                # AR(1) process for daily volatility with occasional jumps
                daily_vol[i] = 0.7 * daily_vol[i-1] + 0.3 * base_vol + np.random.normal(0, 0.002)
                # Add occasional jumps to simulate market shocks
                if np.random.random() < 0.01:  # 1% chance of a shock
                    daily_vol[i] += np.random.exponential(0.03)
        
        # Generate intraday price data
        price = initial_price
        day_index = []
        
        for day in range(n_days):
            day_vol = daily_vol[day]
            
            # U-shaped intraday volatility pattern (higher at open and close)
            intraday_vol_pattern = np.ones(intraday_periods)
            for j in range(intraday_periods):
                rel_time = j / (intraday_periods - 1)
                # U-shape: higher volatility at market open and close
                intraday_vol_pattern[j] = 1 + 0.5 * (1 - 4 * (rel_time - 0.5)**2)
            
            for period in range(intraday_periods):
                # Calculate timestamp
                timestamp = day + period / intraday_periods
                
                # Calculate volatility for this period
                period_vol = day_vol * intraday_vol_pattern[period]
                
                # Generate log return with mean reversion
                if len(all_returns) > 0:
                    # Small mean reversion effect
                    prev_return = all_returns[-1]
                    mean_effect = -0.1 * prev_return
                else:
                    mean_effect = 0
                
                # Generate the return with volatility scaling
                log_return = mean_effect + np.random.normal(0, period_vol)
                
                # Add jump component (rare but significant)
                if np.random.random() < 0.005:  # 0.5% chance of a jump
                    jump_size = np.random.normal(0, period_vol * 5)
                    log_return += jump_size
                
                # Update price
                price *= np.exp(log_return)
                
                # Store data
                all_timestamps.append(timestamp)
                all_prices.append(price)
                all_returns.append(log_return)
                day_index.append(day)
        
        # Create DataFrame
        df = pd.DataFrame({
            'timestamp': all_timestamps,
            'day': day_index,
            'price': all_prices,
            'log_return': all_returns,
            'squared_return': np.array(all_returns)**2
        })
        
        # Calculate realized volatility by day
        rv = df.groupby('day')['squared_return'].sum().reset_index()
        rv.columns = ['day', 'RV']
        
        # Create lagged features for HAR model
        rv['RV_lag1'] = rv['RV'].shift(1)
        rv['RV_lag5'] = rv['RV'].shift(1).rolling(window=5).mean()
        rv['RV_lag22'] = rv['RV'].shift(1).rolling(window=22).mean()
        
        # Drop NaN values
        rv = rv.dropna().reset_index(drop=True)
        
        return df, rv
    
    def calculate_exponential_kernel(self, time_diff, decay_rate):
        """Calculate exponential kernel function value."""
        return decay_rate * np.exp(-decay_rate * time_diff)
    
    def calculate_trend_feature(self, returns, days):
        """
        Calculate the trend feature (R1) based on past returns.
        
        Parameters:
        - returns: Array of daily returns
        - days: Number of days to look back
        
        Returns:
        - Trend feature value
        """
        weights = np.zeros(days)
        for i in range(days):
            time_diff = i + 1  # Time difference (in days)
            weights[i] = self.calculate_exponential_kernel(time_diff, self.decay_rate)
        
        # Normalize weights
        weights = weights / weights.sum()
        
        # Calculate weighted sum of past returns
        trend_feature = np.sum(weights * returns[-days:])
        
        return trend_feature
    
    def calculate_volatility_feature(self, squared_returns, days):
        """
        Calculate the volatility feature (R2) based on past squared returns.
        
        Parameters:
        - squared_returns: Array of daily squared returns
        - days: Number of days to look back
        
        Returns:
        - Volatility feature value
        """
        weights = np.zeros(days)
        for i in range(days):
            time_diff = i + 1  # Time difference (in days)
            weights[i] = self.calculate_exponential_kernel(time_diff, self.decay_rate)
        
        # Normalize weights
        weights = weights / weights.sum()
        
        # Calculate weighted sum of past squared returns
        volatility_feature = np.sum(weights * squared_returns[-days:])
        
        return volatility_feature
    
    def prepare_path_dependent_features(self, rv_df, lookback=30):
        """
        Prepare path-dependent features for HAR-PD models.
        
        Parameters:
        - rv_df: DataFrame with realized volatility data
        - lookback: Number of days to look back for path-dependent features
        
        Returns:
        - DataFrame with path-dependent features
        """
        # Get returns and squared returns for feature calculation
        returns = np.diff(np.log(rv_df['RV'].values))
        squared_returns = returns**2
        
        # Initialize features
        R1_features = []
        R2_features = []
        
        # Calculate features for each time step
        for i in range(lookback, len(returns)):
            past_returns = returns[i-lookback:i]
            past_squared_returns = squared_returns[i-lookback:i]
            
            R1 = self.calculate_trend_feature(past_returns, lookback)
            R2 = self.calculate_volatility_feature(past_squared_returns, lookback)
            
            R1_features.append(R1)
            R2_features.append(R2)
        
        # Create a new DataFrame with original and new features
        pd_df = rv_df.iloc[lookback+1:].copy().reset_index(drop=True)
        pd_df['R1'] = R1_features
        pd_df['R2'] = R2_features
        
        # Create lagged path-dependent features
        pd_df['R1_lag1'] = pd_df['R1'].shift(1)
        pd_df['R1_lag5'] = pd_df['R1'].shift(1).rolling(window=5).mean()
        pd_df['R1_lag22'] = pd_df['R1'].shift(1).rolling(window=22).mean()
        
        pd_df['R2_lag1'] = pd_df['R2'].shift(1)
        pd_df['R2_lag5'] = pd_df['R2'].shift(1).rolling(window=5).mean()
        pd_df['R2_lag22'] = pd_df['R2'].shift(1).rolling(window=22).mean()
        
        # Drop NaN values
        pd_df = pd_df.dropna().reset_index(drop=True)
        
        return pd_df
    
    def split_train_test(self, df, test_size=0.2):
        """Split data into training and testing sets."""
        n = len(df)
        train_size = int(n * (1 - test_size))
        train = df.iloc[:train_size]
        test = df.iloc[train_size:]
        return train, test
    
    def build_HAR_model(self, train_data):
        """Build and train HAR-RV model."""
        X = train_data[['RV_lag1', 'RV_lag5', 'RV_lag22']]
        y = train_data['RV']
        model = LinearRegression()
        model.fit(X, y)
        return model
    
    def build_HAR_PD_model(self, train_data):
        """Build and train HAR-PD-RV model."""
        X = train_data[['R2_lag1', 'R2_lag5', 'R2_lag22']]
        y = train_data['RV']
        model = LinearRegression()
        model.fit(X, y)
        return model
    
    def build_LASSO_HAR_PD_model(self, train_data, alpha=0.01):
        """Build and train LASSO-HAR-PD model."""
        X = train_data[['R1_lag1', 'R1_lag5', 'R1_lag22', 
                      'R2_lag1', 'R2_lag5', 'R2_lag22']]
        y = train_data['RV']
        model = Lasso(alpha=alpha)
        model.fit(X, y)
        return model
    
    def evaluate_model(self, model, test_data, feature_cols, target_col='RV'):
        """Evaluate model performance."""
        X_test = test_data[feature_cols]
        y_test = test_data[target_col]
        
        # Make predictions
        y_pred = model.predict(X_test)
        
        # Calculate error metrics
        mse = mean_squared_error(y_test, y_pred)
        mae = mean_absolute_error(y_test, y_pred)
        
        # Calculate HMSE and HMAE as in the paper
        hmse = np.mean((1 - y_pred/y_test)**2)
        hmae = np.mean(np.abs(1 - y_pred/y_test))
        
        # Calculate QLIKE
        qlike = np.mean(np.log(y_pred) + y_test/y_pred)
        
        # Calculate out-of-sample R^2
        y_mean = np.mean(y_test)
        ss_tot = np.sum((y_test - y_mean)**2)
        ss_res = np.sum((y_test - y_pred)**2)
        r2_oos = 1 - ss_res/ss_tot
        
        results = {
            'MSE': mse,
            'MAE': mae,
            'HMSE': hmse,
            'HMAE': hmae,
            'QLIKE': qlike,
            'R2_OOS': r2_oos
        }
        
        return results, y_pred
