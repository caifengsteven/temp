    def generate_metaorders(self, min_metaorders_per_stock: int = 100000):
        """
        Generate metaorders for all traders across all stocks
        
        Args:
            min_metaorders_per_stock: Minimum number of metaorders to generate per stock
        """
        self.metaorders = []
        
        for stock_id in self.stocks:
            metaorders_count = 0
            
            while metaorders_count < min_metaorders_per_stock:
                # Select random day
                day = np.random.randint(0, self.days)
                
                # Select random trader
                trader_id = np.random.randint(0, self.num_traders)

                beta = self.stocks[stock_id]['beta']
                # Use inverse CDF method to generate power law distribution
                u = np.random.uniform(0, 1)
                # Ensure Q is between 0.001% and 10% of daily volume
                min_size = 0.00001
                max_size = 0.1
                q_normalized = ((1 - u)**(1/(beta)) * (max_size**(1-beta) - min_size**(1-beta)) + min_size**(1-beta))**(1/(1-beta))
                Q = q_normalized * daily_volume
                
                # Generate metaorder length (number of child orders)
                alpha = self.stocks[stock_id]['alpha']
                u = np.random.uniform(0, 1)
                # Ensure L is between 2 and 1000 child orders
                min_length = 2
                max_length = 1000
                L = int(((1 - u)**(1/(alpha)) * (max_length**(1-alpha) - min_length**(1-alpha)) + min_length**(1-alpha))**(1/(1-alpha)))
                
                # Generate liquidation time between 60 and 18000 seconds (5 hours)

                sign = np.random.choice([-1, 1])

                c = self.stocks[stock_id]['c']

                # Add noise to the impact
                noise = np.random.normal(0, daily_volatility * 0.05)
                observed_impact = true_impact + noise
                
                # Store the metaorder
                self.metaorders.append({
                    'stock_id': stock_id,
                    'trader_id': trader_id,
                    'day': day,
                    'Q': Q,

                    'L': L,

                    'true_impact': true_impact,
                    'observed_impact': observed_impact,

    def calculate_price_impact_exponent(self, min_bin_size: int = 100) -> Dict:
        """
        Calculate the price impact exponent for each stock
        
        Args:
            min_bin_size: Minimum number of data points required in a bin
            
        Returns:
            Dictionary with measured exponents for each stock
        """
        if not self.metaorders:
            raise ValueError("No metaorders available. Call generate_metaorders() first.")
        
        results = {}
        
        # Process each stock separately
        for stock_id in self.stocks:
            # Filter metaorders for this stock
            stock_metaorders = [mo for mo in self.metaorders if mo['stock_id'] == stock_id]
            
            if len(stock_metaorders) < 1000:
                print(f"Warning: Not enough metaorders for stock {stock_id}")
                continue
                
            # Create logarithmically spaced bins for Q_normalized
            q_min = 0.00001
            q_max = 0.1
            num_bins = 20
            bin_edges = np.logspace(np.log10(q_min), np.log10(q_max), num_bins+1)
            
            # Bin the data
            binned_data = []
            for i in range(num_bins):
                bin_metaorders = [mo for mo in stock_metaorders 
                                 if mo['Q_normalized'] >= bin_edges[i] and mo['Q_normalized'] < bin_edges[i+1]]
                
                if len(bin_metaorders) >= min_bin_size:
                    avg_q = np.mean([mo['Q_normalized'] for mo in bin_metaorders])
                    avg_impact = np.mean([mo['normalized_impact'] for mo in bin_metaorders])
                    binned_data.append((avg_q, avg_impact))
            
            if len(binned_data) < 5:
                print(f"Warning: Not enough bins with sufficient data for stock {stock_id}")
                continue
                
            # Convert to arrays for fitting
            q_values, impact_values = zip(*binned_data)
            q_values = np.array(q_values)
            impact_values = np.array(impact_values)
            
            # Fit power law using least squares
            def power_law(x, c, delta):
                return c * x**delta
            
            try:
                # Use relative least squares as mentioned in the paper
                # This is approximated by taking the log of both sides
                log_q = np.log(q_values)
                log_impact = np.log(np.abs(impact_values))
                
                # Linear fit on log-log scale
                coeffs = np.polyfit(log_q, log_impact, 1)
                delta_measured = coeffs[0]
                c_measured = np.exp(coeffs[1])
                
                # Also try curve_fit for comparison
                popt, _ = curve_fit(power_law, q_values, np.abs(impact_values), 
                                   p0=[self.stocks[stock_id]['c'], self.true_delta],
                                   bounds=([0, 0], [10, 2]))
                
                results[stock_id] = {
                    'delta': delta_measured,
                    'c': c_measured,
                    'delta_curve_fit': popt[1],
                    'c_curve_fit': popt[0],
                    'true_beta': self.stocks[stock_id]['beta'],
                    'true_alpha': self.stocks[stock_id]['alpha'],
                    'binned_data': binned_data
                }
            except Exception as e:
                print(f"Error fitting for stock {stock_id}: {e}")
                
        return results

    def test_universality_hypothesis(self, results: Dict) -> Dict:
        """
        Test the universality hypothesis: delta should be 0.5 for all stocks
        
        Args:
            results: Output from calculate_price_impact_exponent()
            
        Returns:
            Dictionary with test results
        """
        deltas = [results[stock_id]['delta'] for stock_id in results]
        
        mean_delta = np.mean(deltas)
        std_delta = np.std(deltas)
        sem_delta = std_delta / np.sqrt(len(deltas))
        
        # Test if mean delta is statistically different from 0.5
        t_stat = (mean_delta - 0.5) / sem_delta
        p_value = 2 * (1 - np.abs(t_stat))  # Simple approximation for two-sided test
        
        return {
            'mean_delta': mean_delta,
            'std_delta': std_delta,
            'sem_delta': sem_delta,
            't_statistic': t_stat,
            'p_value': p_value,
            'reject_universality': p_value < 0.05,
            'num_stocks_analyzed': len(deltas)
        }
