import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import solve_ivp
from dataclasses import dataclass
import time

@dataclass
class ModelParameters:
    """Parameters for the stochastic trade flow model"""
    T: float  # Trading horizon in days
    dt: float  # Time step
    S0: float  # Initial price
    sigma: float  # Price volatility
    nu: float  # Trade flow standard deviation
    psi: float  # Bid-ask spread
    eta: float  # Running speed penalty
    phi: float  # Execution cost exponent
    gamma: float  # Risk aversion parameter
    K: float  # Terminal inventory penalty
    rho: float = 0.0  # Correlation between price and flow
    k: float = 0.0  # Permanent market impact parameter

class RiccatiSolver:
    """Solves the matrix Riccati differential equation for Model B"""
    
    def __init__(self, params):
        self.params = params
        
    def riccati_rhs(self, t, y):
        """Right-hand side of the Riccati ODE"""
        # Reshape y to 2x2 matrix
        A = y.reshape(2, 2)
        
        # Parameters
        eta = self.params.eta
        gamma = self.params.gamma
        sigma = self.params.sigma
        nu = self.params.nu
        k = self.params.k
        
        # Sigma matrix
        Sigma = np.array([
            [sigma**2, self.params.rho * sigma * nu],
            [self.params.rho * sigma * nu, nu**2]
        ])
        
        # Matrix calculation
        M = np.array([[1, k], [k, k**2]])
        dA = (1/eta) * A @ M @ A - (gamma/2) * Sigma
        
        return dA.flatten()
    
    def solve(self):
        """Solve the Riccati equation backwards from T to 0"""
        # Terminal condition A(T) = [[K, 0], [0, 0]]
        A_T = np.array([[self.params.K, 0], [0, 0]]).flatten()
        
        # Time points
        t_span = (0, self.params.T)
        t_eval = np.linspace(self.params.T, 0, int(self.params.T / self.params.dt) + 1)[::-1]
        
        # Solve ODE backwards in time
        sol = solve_ivp(
            self.riccati_rhs, 
            t_span, 
            A_T, 
            t_eval=t_eval, 
            method='RK45'
        )
        
        # Reshape solution to get 2x2 matrices
        A_matrices = np.zeros((len(t_eval), 2, 2))
        for i, t in enumerate(sol.t):
            A_matrices[i] = sol.y[:, i].reshape(2, 2)
            
        return t_eval, A_matrices

class ModelB:
    """Implementation of Model B from the paper"""
    
    def __init__(self, params):
        self.params = params
        self.time_grid = np.linspace(0, params.T, int(params.T / params.dt) + 1)
        self.solver = RiccatiSolver(params)
        # Solve Riccati equation
        self.t_grid, self.A_matrices = self.solver.solve()
        
    def optimal_control(self, t, q, S):
        """Calculate the optimal control (trading rate) at time t with inventory q and price S"""
        # Find closest time index
        idx = np.abs(self.t_grid - t).argmin()
        A = self.A_matrices[idx]
        
        # v* = -(1/η)([1 k]ᵀA[q S])
        v = -(1/self.params.eta) * np.dot(np.array([1, self.params.k]), A @ np.array([q, S]))
        return v
    
    def simulate_path(self, q0=0):
        """Simulate a path following the optimal strategy"""
        # Initialize arrays
        n_steps = len(self.time_grid)
        q = np.zeros(n_steps)
        S = np.zeros(n_steps)
        v = np.zeros(n_steps)
        X = np.zeros(n_steps)
        
        # Set initial values
        q[0] = q0
        S[0] = self.params.S0
        
        # Generate correlated Brownian motions
        dW = np.random.normal(0, np.sqrt(self.params.dt), n_steps-1)  # Price Brownian
        dB = self.params.rho * dW + np.sqrt(1 - self.params.rho**2) * np.random.normal(0, np.sqrt(self.params.dt), n_steps-1)  # Flow Brownian
        
        # Simulate path
        for i in range(n_steps-1):
            t = self.time_grid[i]
            
            # Calculate optimal control
            v[i] = self.optimal_control(t, q[i], S[i])
            
            # Update state variables
            q[i+1] = q[i] + v[i] * self.params.dt + self.params.nu * dB[i]
            S[i+1] = S[i] + self.params.k * v[i] * self.params.dt + self.params.sigma * dW[i]
            
            # Update cash account
            execution_cost = self.params.psi * abs(v[i])/2 + self.params.eta * (v[i]**2)
            X[i+1] = X[i] - v[i] * S[i] * self.params.dt - execution_cost * self.params.dt
        
        # Calculate final v
        v[-1] = self.optimal_control(self.params.T, q[-1], S[-1])
        
        # Calculate PnL
        terminal_penalty = self.params.K * (q[-1]**2)
        PnL = X[-1] + q[-1] * S[-1] - terminal_penalty
        
        return {
            'time': self.time_grid,
            'inventory': q,
            'price': S,
            'trading_rate': v,
            'cash': X,
            'PnL': PnL
        }

    """Run Monte Carlo simulations to compare strategies"""
