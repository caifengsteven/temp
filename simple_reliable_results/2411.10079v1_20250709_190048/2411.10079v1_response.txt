import numpy as np

@dataclass
class SwapData:
    """Data model for a Receiver Interest Rate Swap"""
    start_time: float  # Ti - start time
    end_time: float    # Te - end time
    strike: float      # K
    
    def price(self, t: float, discount_factors: Dict[float, float], swap_rates: Dict[Tuple[float, float], float]) -> float:
        """Calculate swap price at time t"""
        if t >= self.start_time:
            return 0.0
        
        key = (self.start_time, self.end_time)
        swap_rate = swap_rates.get(key, 0.02)  # Default to 2% if not found
        annuity = self._calc_annuity(t, discount_factors)
        
        return annuity * (self.strike - swap_rate)
    
    def _calc_annuity(self, t: float, discount_factors: Dict[float, float]) -> float:
        """Calculate annuity factor"""
        annuity = 0.0
        current_time = self.start_time
        
        while current_time < self.end_time:
            next_time = min(current_time + 1.0, self.end_time)  # Assuming annual coupons
            df = discount_factors.get(next_time, 0.95)  # Default value
            annuity += (next_time - current_time) * df
            current_time = next_time
            
        return annuity


@dataclass
class SwaptionData:
    """Data model for a European Swaption"""
    exercise_time: float  # Ti - exercise time
    underlying_swap: SwapData
    
    def price(self, t: float, discount_factors: Dict[float, float], 
              swap_rates: Dict[Tuple[float, float], float], 
              volatilities: Dict[Tuple[float, float], float]) -> float:
        """Calculate swaption price at time t using Black's formula"""
        if t >= self.exercise_time:
            return 0.0
        
        key = (self.underlying_swap.start_time, self.underlying_swap.end_time)
        swap_rate = swap_rates.get(key, 0.02)
        vol = volatilities.get(key, 0.2)
        time_to_expiry = self.exercise_time - t
        
        # Black's formula for receiver swaption
        annuity = self.underlying_swap._calc_annuity(t, discount_factors)
        d1 = (np.log(swap_rate / self.underlying_swap.strike) + 0.5 * vol**2 * time_to_expiry) / (vol * np.sqrt(time_to_expiry))
        d2 = d1 - vol * np.sqrt(time_to_expiry)
        
        price = annuity * (self.underlying_swap.strike * norm.cdf(-d2) - swap_rate * norm.cdf(-d1))
        return max(0.0, price)  # Ensure non-negative price


@dataclass
class BermudanSwaption:
    """Data model for a Bermudan Swaption"""
    possible_exercise_times: List[float]  # List of Ti's
    end_time: float  # Te - common maturity for all underlying swaps
    strike: float    # K - strike for all underlying swaps
    
    def __post_init__(self):
        """Initialize underlying swaps and European swaptions"""
        self.underlying_swaps = []
        self.european_swaptions = []
        
        for time in self.possible_exercise_times:
            swap = SwapData(time, self.end_time, self.strike)
            self.underlying_swaps.append(swap)
            
            swaption = SwaptionData(time, swap)
            self.european_swaptions.append(swaption)


class SMaxStrategy:
    """Implementation of the Option Spread Max Strategy (SMax) for Bermudan Swaptions"""
    
    def __init__(self, bermudan_swaption: BermudanSwaption):
        self.bermudan = bermudan_swaption
        
    def optimal_hedge_portfolio(self, t: float, discount_factors: Dict[float, float],
                               swap_rates: Dict[Tuple[float, float], float],
                               volatilities: Dict[Tuple[float, float], float]) -> Dict[int, float]:
        """
        Determine optimal hedge portfolio using SMax strategy
        Returns: Dictionary mapping swaption index to hedge quantity (0 or 1)
        """
        # Calculate all relevant swaption prices
        swaption_prices = []
        for idx, swaption in enumerate(self.bermudan.european_swaptions):
            if swaption.exercise_time > t:  # Only consider unexpired swaptions
                price = swaption.price(t, discount_factors, swap_rates, volatilities)
                swaption_prices.append((idx, price))
        
        if not swaption_prices:
            return {}
            
        # Find swaption with highest value
        max_idx, max_price = max(swaption_prices, key=lambda x: x[1])
        
        # Create hedge portfolio (short the highest-value swaption)
        hedge_portfolio = {idx: 0 for idx, _ in swaption_prices}
        hedge_portfolio[max_idx] = 1
        
        return hedge_portfolio
    
    def exercise_decision(self, t: float, discount_factors: Dict[float, float],
                         swap_rates: Dict[Tuple[float, float], float],
                         volatilities: Dict[Tuple[float, float], float]) -> bool:
        """
        Determine whether to exercise the Bermudan swaption at time t
        """
        # Check if t is an exercise time
        if t not in self.bermudan.possible_exercise_times:
            return False
            
        # Get index of current exercise time
        current_idx = self.bermudan.possible_exercise_times.index(t)
        
        # Calculate all relevant swaption prices
        swaption_prices = []
        for idx, swaption in enumerate(self.bermudan.european_swaptions[current_idx:]):
            actual_idx = current_idx + idx
            price = swaption.price(t, discount_factors, swap_rates, volatilities)
            swaption_prices.append((actual_idx, price))
            
        # Find swaption with highest value
        max_idx, _ = max(swaption_prices, key=lambda x: x[1])
        
        # Exercise if current time's swaption has the highest value
        return max_idx == current_idx

def generate_market_data(times: List[float], possible_exercise_times: List[float], 
                         end_time: float, num_scenarios: int = 1000) -> List[Dict]:
    """
    Generate simulated market data for testing
    Returns a list of dictionaries, each containing market data for one time point across all scenarios
    """
    np.random.seed(42)  # For reproducibility
    
    # Initialize base rates and volatilities
    base_rate = 0.02  # 2%
    base_vol = 0.20   # 20% vol
    
    all_scenarios_data = []
    
    for scenario in range(num_scenarios):
        # Generate random walks for rates
        rate_changes = np.random.normal(0, 0.002, len(times))
        vol_changes = np.random.normal(0, 0.03, len(times))
        
        scenario_data = []
        
        current_rate = base_rate
        current_vol = base_vol
        
        for i, t in enumerate(times):
            # Update rates and vols with random changes
            if i > 0:
                current_rate += rate_changes[i]
                current_vol += vol_changes[i]
                # Keep within reasonable bounds
                current_rate = max(0.001, min(0.05, current_rate))
                current_vol = max(0.05, min(0.50, current_vol))
            
            # Create market data for this time point
            discount_factors = {}
            swap_rates = {}
            volatilities = {}
            
            # Generate discount factors
            for future_time in np.arange(t+1, end_time+1, 1.0):
                df = np.exp(-current_rate * (future_time - t))
                discount_factors[future_time] = df
                
            # Generate swap rates for each potential starting time
            for start_time in possible_exercise_times:
                if start_time >= t:  # Only for future or current exercise times
                    key = (start_time, end_time)
                    # Add some term structure to rates
                    term_adjustment = 0.001 * (end_time - start_time)
                    swap_rates[key] = current_rate + term_adjustment
                    
                    # Add some term structure to vols
                    vol_adjustment = 0.01 * (end_time - start_time)
                    volatilities[key] = current_vol + vol_adjustment
            
            scenario_data.append({
                'time': t,
                'discount_factors': discount_factors,
                'swap_rates': swap_rates,
                'volatilities': volatilities
            })
        
        all_scenarios_data.append(scenario_data)
    
    return all_scenarios_data


def test_smax_strategy(num_scenarios: int = 100) -> Dict:
    """Test the SMax strategy with simulated data"""
    # Define parameters
    start_time = 0.0
    end_time = 9.0
    possible_exercise_times = [4.0, 5.0, 6.0, 7.0, 8.0]  # Based on the paper

    strategy = SMaxStrategy(bermudan)

    print("Generating market scenarios...")
    market_data = generate_market_data(times, possible_exercise_times, end_time, num_scenarios)
    
    # Results storage
    results = {

        'initial_values': [],
        'final_pnl': [],
        'hedge_pnl': [],
        'total_pnl': []
    }

    print(f"Running {num_scenarios} simulations...")
    for scenario_idx, scenario in enumerate(market_data):
        if scenario_idx % 20 == 0:
            print(f"Processing scenario {scenario_idx}...")
            
        initial_time_data = scenario[0]
        
        # Calculate initial value of Bermudan (max of European swaptions at t=0)
        initial_swaption_values = []
        for swaption in bermudan.european_swaptions:
            price = swaption.price(
                initial_time_data['time'],
                initial_time_data['discount_factors'],
                initial_time_data['swap_rates'],
                initial_time_data['volatilities']
            )
            initial_swaption_values.append(price)
        
        initial_value = max(initial_swaption_values)
        results['initial_values'].append(initial_value)
        
        # Initialize tracking variables
        exercised = False
        exercise_time = None
        exercise_value = 0
        
        hedge_portfolio = {}
        hedge_pnl = 0
        
        # Simulate through time
        for i in range(len(scenario) - 1):
            current_data = scenario[i]
            next_data = scenario[i+1]
            
            current_time = current_data['time']
            next_time = next_data['time']
            
            # Check for exercise at current time
            if current_time in possible_exercise_times and not exercised:
                should_exercise = strategy.exercise_decision(
                    current_time,
                    current_data['discount_factors'],
                    current_data['swap_rates'],
                    current_data['volatilities']
                )
                
                if should_exercise:
                    exercised = True
                    exercise_time = current_time
                    
                    # Calculate exercise value
                    exercise_idx = possible_exercise_times.index(current_time)
                    swap = bermudan.underlying_swaps[exercise_idx]
                    exercise_value = swap.price(
                        current_time,
                        current_data['discount_factors'],
                        current_data['swap_rates']
                    )
            
            # Update hedge portfolio if not exercised
            if not exercised:
                # Get new hedge portfolio
                new_hedge_portfolio = strategy.optimal_hedge_portfolio(
                    current_time,
                    current_data['discount_factors'],
                    current_data['swap_rates'],
                    current_data['volatilities']
                )
                
                # Calculate P&L from hedging
                for idx, quantity in new_hedge_portfolio.items():
                    # Calculate change in value of this swaption
                    if idx in hedge_portfolio:  # If we already had a position
                        old_quantity = hedge_portfolio[idx]
                        if old_quantity > 0:  # If we were short before
                            # Calculate value change (positive if swaption value decreases)
                            old_price = bermudan.european_swaptions[idx].price(
                                current_time,
                                current_data['discount_factors'],
                                current_data['swap_rates'],
                                current_data['volatilities']
                            )
                            new_price = bermudan.european_swaptions[idx].price(
                                next_time,
                                next_data['discount_factors'],
                                next_data['swap_rates'],
                                next_data['volatilities']
                            )
                            # P&L is positive if price decreases (since we're short)
                            hedge_pnl += old_quantity * (old_price - new_price)
                
                # Update portfolio
                hedge_portfolio = new_hedge_portfolio
        
        # Calculate final P&L
        if exercised:
            final_pnl = exercise_value
        else:

        total_pnl = final_pnl + hedge_pnl
        
        # Store results
        results['exercised_at'].append(exercise_time)
        results['final_pnl'].append(final_pnl)
        results['hedge_pnl'].append(hedge_pnl)
        results['total_pnl'].append(total_pnl)
    
    return results


def analyze_results(results: Dict):
    """Analyze and display results of the simulation"""
    # Calculate statistics
    initial_values = np.array(results['initial_values'])
    total_pnl = np.array(results['total_pnl'])
    
    # P&L drawdowns (difference between initial model value and realized P&L)
    pnl_drawdowns = initial_values - total_pnl
    
    # Summary statistics
    mean_drawdown = np.mean(pnl_drawdowns)
    median_drawdown = np.median(pnl_drawdowns)
    p25 = np.percentile(pnl_drawdowns, 25)
    p75 = np.percentile(pnl_drawdowns, 75)
    iqr = p75 - p25
    
    # Loss probability
    loss_prob = np.mean(pnl_drawdowns > 0)
    
    # Expected loss
    losses = pnl_drawdowns[pnl_drawdowns > 0]
    expected_loss = np.mean(losses) if len(losses) > 0 else 0
    
    # CVaR
    cvar95 = np.mean(pnl_drawdowns[pnl_drawdowns > np.percentile(pnl_drawdowns, 95)])
    cvar99 = np.mean(pnl_drawdowns[pnl_drawdowns > np.percentile(pnl_drawdowns, 99)])
    
    # Exercise statistics
    exercise_times = [t for t in results['exercised_at'] if t is not None]
    exercise_count = len(exercise_times)
    exercise_rate = exercise_count / len(results['exercised_at']) * 100
    
    # Print results
    print("\n===== SMax Strategy Results =====")
    print(f"Number of scenarios: {len(results['initial_values'])}")
    print(f"Average initial value: {np.mean(initial_values):.2f} basis points")
    print(f"Exercise rate: {exercise_rate:.1f}%")
    
    if exercise_times:
        exercise_distribution = {}
        for t in exercise_times:
            exercise_distribution[t] = exercise_distribution.get(t, 0) + 1
        
        print("\nExercise Distribution:")
        for time, count in sorted(exercise_distribution.items()):
            print(f"  Time {time}: {count} times ({count/exercise_count*100:.1f}%)")
    
    print("\nP&L Drawdown Statistics (in basis points):")
    print(f"  Mean: {mean_drawdown:.2f}")
    print(f"  Median (P50): {median_drawdown:.2f}")
    print(f"  P25: {p25:.2f}")
    print(f"  P75: {p75:.2f}")
    print(f"  IQR: {iqr:.2f}")
    print(f"  Loss Probability: {loss_prob:.3f}")
    print(f"  Expected Loss: {expected_loss:.2f}")
    print(f"  CVaR95: {cvar95:.2f}")
    print(f"  CVaR99: {cvar99:.2f}")
    
    # Plot histogram of P&L drawdowns
    plt.figure(figsize=(10, 6))
    plt.hist(pnl_drawdowns, bins=30, alpha=0.7, color='blue')
    plt.axvline(x=0, color='r', linestyle='--', label='Zero Drawdown')
    plt.axvline(x=mean_drawdown, color='g', linestyle='-', label=f'Mean Drawdown: {mean_drawdown:.2f}')
    plt.axvline(x=cvar95, color='orange', linestyle='-', label=f'CVaR95: {cvar95:.2f}')
    plt.xlabel('P&L Drawdown (basis points)')
    plt.ylabel('Frequency')
    plt.title('Distribution of P&L Drawdowns with SMax Strategy')
    plt.legend()

    # Add vertical lines for percentiles
    plt.axvline(x=p25, color='purple', linestyle=':', label=f'P25: {p25:.2f}')
    plt.axvline(x=median_drawdown, color='cyan', linestyle=':', label=f'P50: {median_drawdown:.2f}')
    plt.axvline(x=p75, color='brown', linestyle=':', label=f'P75: {p75:.2f}')
    
    plt.legend()
    plt.tight_layout()
    plt.show()
