    def __init__(self, 
                 kappa: float = 0.1,  # Diffusion parameter 
                 lambda_: float = 0.1,  # Convection parameter
                 beta: float = 0.05,   # Cancellation parameter
                 alpha: float = 0.7,   # Hurst parameter for volatility (0.5 < alpha < 1)
                 c: float = 0.5,       # Volatility scale
                 L: float = 10.0,      # Order book boundary
                 lambda_interval: float = 0.1,  # Integration interval for imbalance
                 dt: float = 0.001,    # Time step
                 dx: float = 0.1,      # Space step
                 T: float = 1.0,       # Time horizon
                 tick_size: float = 0.01,  # Tick size
                 Ca: float = 0.5,      # Ask price impact parameter
                 Cb: float = 0.5):     # Bid price impact parameter
        
        # Model parameters
        self.kappa = kappa
        self.lambda_ = lambda_
        self.beta = beta
        self.alpha = alpha
        self.c = c
        self.L = L
        self.lambda_interval = lambda_interval
        
        # Discretization parameters
        self.dt = dt
        self.dx = dx
        self.T = T
        self.nt = int(T / dt)
        self.nx = int(2 * L / dx) + 1
        self.x = np.linspace(-L, L, self.nx)
        
        # Price parameters
        self.tick_size = tick_size
        self.Ca = Ca
        self.Cb = Cb
        
        # Initialize arrays
        self.u = np.zeros((self.nt + 1, self.nx))
        self.Y = np.zeros(self.nt + 1)
        self.S = np.zeros(self.nt + 1)
        self.Da = np.zeros(self.nt + 1)
        self.Db = np.zeros(self.nt + 1)
        
        # Initialize random processes
        np.random.seed(42)
        self.dW = np.sqrt(self.dt) * np.random.normal(0, 1, self.nt)
        self.dB = np.sqrt(self.dt) * np.random.normal(0, 1, self.nt)
    
    def initialize_order_book(self, 
                             initial_mid_price: float = 100.0, 
                             initial_shape: str = "linear"):
        """Initialize the order book with a specified shape"""
        # Set initial mid price
        self.S[0] = initial_mid_price
        
        # Create initial order book shape
        if initial_shape == "linear":
            # Linear shape: decreasing volume away from mid-price
            for i, x in enumerate(self.x):
                if x < 0:  # Bid side (negative by convention)
                    self.u[0, i] = -max(0, 1 - abs(x) / 5) 
                elif x > 0:  # Ask side (positive by convention)
                    self.u[0, i] = max(0, 1 - abs(x) / 5)
                # At x=0 (mid-price), u=0
        
        elif initial_shape == "realistic":
            # More realistic shape with highest volume near best bid/ask
            for i, x in enumerate(self.x):
                if x < 0:  # Bid side
                    # Highest volume at best bid, decreasing further away
                    peak_at = -1.0
                    self.u[0, i] = -max(0, 2 * np.exp(-5 * (x - peak_at)**2))
                elif x > 0:  # Ask side
                    # Highest volume at best ask, decreasing further away
                    peak_at = 1.0
                    self.u[0, i] = max(0, 2 * np.exp(-5 * (x - peak_at)**2))

        idx_mid = np.where(self.x == 0)[0][0]
        lambda_idx = int(self.lambda_interval / self.dx)
        
        # Ask depth - sum of volume on the ask side near mid-price
        self.Da[0] = np.sum(self.u[0, idx_mid:idx_mid + lambda_idx + 1])
        
        # Bid depth - sum of volume on the bid side near mid-price
        self.Db[0] = np.sum(self.u[0, idx_mid - lambda_idx:idx_mid + 1])

        self.Y[0] = 0.2  # Starting value for Y
    
    def truncated_derivative(self, u, i):
        """Calculate [u_x]- (truncated spatial derivative)"""
        if i == 0:
            return 0
        elif i == self.nx - 1:
            return 0
        else:
            ux = (u[i+1] - u[i-1]) / (2 * self.dx)
            if ux < 0:
                return ux
            else:
                return 0
    
    def J_term(self, u, x, i, u0=0.5, j_scale=0.1):
        """Calculate J(x, u) term - cancellation with market order replacement"""
        # Simplification of the complex J term from the paper
        j_x = j_scale * np.exp(-abs(x))  # j(x) decreases with distance from mid-price
        
        if x > 0:  # Ask side
            # Find corresponding index on bid side
            bid_idx = np.where(self.x <= -abs(x))[0][-1]
            ask_excess = max(0, u[i] - u0)
            bid_excess = max(0, -u[bid_idx] - u0)
            return -j_x * ask_excess - j_x * bid_excess
        else:  # Bid side
            # Find corresponding index on ask side
            ask_idx = np.where(self.x >= abs(x))[0][0]
            bid_excess = max(0, -u[i] - u0)
            ask_excess = max(0, u[ask_idx] - u0)
            return j_x * bid_excess + j_x * ask_excess
    
    def G_term(self, x, imbalance, g_scale=0.05):
        """Calculate G(x, δ) term - order submission based on imbalance"""
        # G is strictly decreasing in δ with G(x,0) = 0
        return -g_scale * imbalance * np.exp(-0.5 * abs(x))
    
    def simulate(self):
        """Run the simulation"""
        for n in range(self.nt):
            # Calculate imbalance
            idx_mid = np.where(self.x == 0)[0][0]
            lambda_idx = int(self.lambda_interval / self.dx)
            
            # Calculate imbalance (δ) as the difference between ask and bid volumes
            imbalance = np.sum(self.u[n, idx_mid:idx_mid + lambda_idx + 1]) + \
                       np.sum(self.u[n, idx_mid - lambda_idx:idx_mid])
            
            # Update volatility process Y using a simplified version of eq. (4.46)
            # This is a simplified rough volatility process with memory
            memory_term = 0
            for k in range(n):
                memory_kernel = (n*self.dt - k*self.dt)**(self.alpha-1) / gamma(self.alpha)
                memory_term += memory_kernel * (1 - self.Y[k]) * self.dt
            
            stochastic_term = 0
            for k in range(n):
                memory_kernel = (n*self.dt - k*self.dt)**(self.alpha-1) / gamma(self.alpha)
                stochastic_term += memory_kernel * np.sqrt(self.Y[k]) * self.dB[k]
            
            self.Y[n+1] = 0.1 * memory_term + 0.1 * stochastic_term
            self.Y[n+1] = max(0.01, min(self.Y[n+1], 1.0))  # Ensure Y stays positive and bounded
            
            # Update order book density u
            for i in range(1, self.nx-1):
                x = self.x[i]
                
                # Diffusion term
                uxx = (self.u[n, i+1] - 2*self.u[n, i] + self.u[n, i-1]) / self.dx**2
                
                # Sign of x (for convection)
                sgn_x = np.sign(x)
                
                # Truncated derivative
                trunc_ux = self.truncated_derivative(self.u[n], i)
                
                # J and G terms
                j_term = self.J_term(self.u[n], x, i)
                g_term = self.G_term(x, imbalance)
                
                # Update equation
                drift = (self.kappa * uxx - 
                        self.lambda_ * sgn_x * trunc_ux - 
                        self.beta * self.u[n, i] + 
                        j_term + g_term)
                
                diffusion = self.c * self.u[n, i] * np.sqrt(self.Y[n]) * self.dW[n]
                
                self.u[n+1, i] = self.u[n, i] + drift * self.dt + diffusion
            
            # Enforce boundary conditions
            self.u[n+1, 0] = self.u[n+1, -1] = 0
            
            # Enforce conditions at x=0 (mid-price)
            mid_idx = np.where(self.x == 0)[0][0]
            self.u[n+1, mid_idx] = 0
            
            # Ensure ask volumes are non-negative and bid volumes are non-positive
            for i in range(self.nx):
                if self.x[i] > 0:  # Ask side
                    self.u[n+1, i] = max(0, self.u[n+1, i])
                elif self.x[i] < 0:  # Bid side
                    self.u[n+1, i] = min(0, self.u[n+1, i])

            dDa = self.Da[n+1] - self.Da[n]

    def calculate_realized_volatility(self, window_size=50):
        """Calculate realized volatility from price changes"""
        # Calculate log returns
        log_returns = np.diff(np.log(self.S))
        
        # Calculate realized volatility using rolling window
        realized_vol = np.zeros(len(log_returns) - window_size + 1)
