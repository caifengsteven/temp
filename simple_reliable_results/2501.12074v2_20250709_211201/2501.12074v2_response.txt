import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from scipy.optimize import minimize
import datetime
from typing import Dict, List, Tuple
import warnings

class PortfolioOptimizer:
    def __init__(self, data: pd.DataFrame, risk_free_rate: float = 0.0):
        """
        Initialize the portfolio optimizer with historical price data.
        
        Args:
            data: DataFrame with dates as index and tickers as columns
            risk_free_rate: Risk-free rate for Sharpe ratio calculation
        """
        self.data = data
        self.log_returns = np.log(data / data.shift(1)).dropna()
        self.risk_free_rate = risk_free_rate
        self.clusters = None
        self.optimized_weights = {}
        
    def compute_log_returns(self) -> pd.DataFrame:
        """Calculate log returns from price data."""
        return self.log_returns
    
    def perform_clustering(self, n_clusters: int = 3) -> Dict:
        """
        Segment assets into clusters based on return correlation.
        
        Args:
            n_clusters: Number of clusters to form
            
        Returns:
            Dictionary mapping cluster IDs to lists of tickers
        """
        # Calculate correlation matrix
        corr_matrix = self.log_returns.corr()
        
        # Apply KMeans clustering on the correlation matrix
        kmeans = KMeans(n_clusters=n_clusters, random_state=42)
        clusters = kmeans.fit_predict(corr_matrix)
        
        # Create a dictionary mapping cluster IDs to tickers
        cluster_dict = {}
        for i, cluster_id in enumerate(clusters):
            if cluster_id not in cluster_dict:
                cluster_dict[cluster_id] = []
            cluster_dict[cluster_id].append(self.log_returns.columns[i])
        
        self.clusters = cluster_dict
        return cluster_dict
    
    def optimize_portfolio(self, tickers: List[str]) -> np.ndarray:
        """
        Optimize portfolio weights using Sharpe ratio maximization.
        
        Args:
            tickers: List of tickers to include in the portfolio
            
        Returns:
            Array of optimized weights
        """
        subset_returns = self.log_returns[tickers]
        mean_returns = subset_returns.mean()
        cov_matrix = subset_returns.cov()
        
        num_assets = len(tickers)
        initial_weights = np.ones(num_assets) / num_assets
        bounds = tuple((0, 1) for _ in range(num_assets))
        
        # Sum of weights constraint
        constraints = ({'type': 'eq', 'fun': lambda weights: np.sum(weights) - 1})
        
        # Objective function: negative Sharpe ratio (we minimize)
        def neg_sharpe_ratio(weights):
            portfolio_return = np.dot(weights, mean_returns)
            portfolio_volatility = np.sqrt(np.dot(weights.T, np.dot(cov_matrix, weights)))
            if portfolio_volatility == 0:
                return 0  # Avoid division by zero
            return -(portfolio_return - self.risk_free_rate) / portfolio_volatility
        
        # Perform optimization
        result = minimize(
            neg_sharpe_ratio, 
            initial_weights, 
            method='SLSQP',
            bounds=bounds, 
            constraints=constraints
        )
        
        if not result.success:
            raise Exception(f"Optimization failed: {result.message}")
        
        return result.x
    
    def optimize_all_clusters(self) -> Dict:
        """
        Optimize portfolio weights for each cluster.
        
        Returns:
            Dictionary mapping cluster IDs to optimized weights
        """
        if not self.clusters:
            raise Exception("Must perform clustering before optimization")
        
        # For each cluster, optimize the portfolio
        for cluster_id, tickers in self.clusters.items():
            weights = self.optimize_portfolio(tickers)
            self.optimized_weights[cluster_id] = {
                ticker: weight for ticker, weight in zip(tickers, weights)
            }
            
        return self.optimized_weights
    
    def backtest_portfolio(self, 
                          weights: Dict[str, float], 
                          start_date: str, 
                          end_date: str) -> pd.Series:
        """
        Backtest a portfolio with given weights over a specified period.
        
        Args:
            weights: Dictionary mapping tickers to weights
            start_date: Start date for backtesting
            end_date: End date for backtesting
            
        Returns:
            Series of cumulative returns
        """
        # Filter the returns for the backtest period
        backtest_returns = self.log_returns.loc[start_date:end_date][list(weights.keys())]
        
        # Calculate weighted returns
        weighted_returns = pd.Series(0, index=backtest_returns.index)
        for ticker, weight in weights.items():
            weighted_returns += backtest_returns[ticker] * weight
        
        # Calculate cumulative returns
        cumulative_returns = (1 + weighted_returns).cumprod() - 1
        
        return cumulative_returns
    
    def calculate_performance_metrics(self, returns: pd.Series) -> Dict:
        """
        Calculate performance metrics for a returns series.
        
        Args:
            returns: Series of returns
            
        Returns:
            Dictionary of performance metrics
        """
        # Calculate total return
        total_return = returns.iloc[-1] * 100
        
        # Calculate annualized return
        days = (returns.index[-1] - returns.index[0]).days
        years = days / 365
        annualized_return = ((1 + returns.iloc[-1]) ** (1/years) - 1) * 100
        
        # Calculate volatility (annualized)
        daily_returns = returns.pct_change().dropna()
        volatility = daily_returns.std() * np.sqrt(252) * 100
        
        # Calculate Sharpe ratio
        excess_return = annualized_return / 100 - self.risk_free_rate
        sharpe_ratio = excess_return / (volatility / 100) if volatility != 0 else 0
        
        return {
            'total_return': total_return,
            'annualized_return': annualized_return,
            'volatility': volatility,
            'sharpe_ratio': sharpe_ratio
        }
    
    def plot_cumulative_returns(self, returns_dict: Dict[str, pd.Series], title: str) -> None:
        """
        Plot cumulative returns for multiple portfolios.
        
        Args:
            returns_dict: Dictionary mapping portfolio names to returns series
            title: Plot title
        """
        plt.figure(figsize=(10, 6))
        for name, returns in returns_dict.items():
            plt.plot(returns, label=name)
        
        plt.title(title)
        plt.xlabel('Date')
        plt.ylabel('Cumulative Return')

    ])
    
    # Ensure correlation matrix is positive semi-definite
    corr_matrix = np.maximum(corr_matrix, corr_matrix.T) / 2
    
    # Compute standard deviations
    stds = np.array([ticker_params[ticker]['std'] for ticker in tickers])
    
    # Compute covariance matrix
    cov_matrix = np.outer(stds, stds) * corr_matrix
    
    # Generate correlated random returns
    means = np.array([ticker_params[ticker]['mean'] for ticker in tickers])
    log_returns = np.random.multivariate_normal(means, cov_matrix, size=len(date_range))
    
    # Convert log returns to price series
    for i, date in enumerate(date_range):
        if i == 0:
            prices.loc[date] = [initial_prices[ticker] for ticker in tickers]
        else:
            prev_prices = prices.loc[date_range[i-1]].values
            prices.loc[date] = prev_prices * np.exp(log_returns[i-1])
