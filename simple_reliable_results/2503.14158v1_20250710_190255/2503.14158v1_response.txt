class QuinticOUModel:

    def __init__(self, lambda_x, lambda_y, theta, rho, alphas, g0=None):
        """
        Initialize the two-factor Quintic OU model
        
        Parameters:
        - lambda_x, lambda_y: mean reversion speeds for the two OU processes
        - theta: weight parameter for the linear combination
        - rho: correlation between price and volatility
        - alphas: coefficients of the polynomial (should be length 6, with alphas[5] = 1)
        - g0: deterministic function (if None, will be computed to match forward variance)
        """
        self.lambda_x = lambda_x
        self.lambda_y = lambda_y
        self.theta = theta
        self.rho = rho
        
        # Ensure alphas has length 6 with alphas[5] = 1
        if len(alphas) != 6:
            raise ValueError("alphas should contain 6 coefficients")
        
        if alphas[5] != 1:

        self.alphas = np.array(alphas)
        self.g0 = g0
    
    def polynomial(self, z):
        """Evaluate the polynomial p(z) = sum_{k=0}^5 alpha_k * z^k"""
        return sum(alpha * z**k for k, alpha in enumerate(self.alphas))
    
    def simulate_paths(self, T, dt, n_paths=1, seed=None):
        """
        Simulate paths for the model
        
        Parameters:
        - T: time horizon
        - dt: time step
        - n_paths: number of paths to simulate
        - seed: random seed for reproducibility

            np.random.seed(seed)
            
        steps = int(T / dt) + 1
        t = np.linspace(0, T, steps)
        
        # Initialize arrays
        X = np.zeros((n_paths, steps))
        Y = np.zeros((n_paths, steps))
        Z = np.zeros((n_paths, steps))
        sigma = np.zeros((n_paths, steps))
        S = np.ones((n_paths, steps))  # Start with S0 = 1
        
        # Generate correlated Brownian motions
        dW = np.random.normal(0, np.sqrt(dt), (n_paths, steps-1))
        dW_perp = np.random.normal(0, np.sqrt(dt), (n_paths, steps-1))
        
        # Simulate OU processes and derive Z, sigma, and S
        for i in range(1, steps):
            # Update X and Y (OU processes)
            X[:, i] = X[:, i-1] * np.exp(-self.lambda_x * dt) + np.sqrt(
                (1 - np.exp(-2 * self.lambda_x * dt)) / (2 * self.lambda_x)) * dW[:, i-1]
            
            Y[:, i] = Y[:, i-1] * np.exp(-self.lambda_y * dt) + np.sqrt(
                (1 - np.exp(-2 * self.lambda_y * dt)) / (2 * self.lambda_y)) * dW[:, i-1]
            
            # Compute Z as weighted sum of X and Y
            Z[:, i] = self.theta * X[:, i] + (1 - self.theta) * Y[:, i]

            # Update underlying price
            S[:, i] = S[:, i-1] * np.exp(
                -0.5 * sigma[:, i]**2 * dt + 
                sigma[:, i] * (self.rho * dW[:, i-1] + np.sqrt(1 - self.rho**2) * dW_perp[:, i-1])
            )
