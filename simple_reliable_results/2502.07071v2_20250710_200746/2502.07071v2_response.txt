class LimitOrderBook:
    """Simple implementation of a limit order book for simulation purposes"""
    
    def __init__(self, initial_mid_price=200.0, levels=10):
        self.levels = levels
        self.mid_price = initial_mid_price
        self.spread = 0.1
        
        # Initialize the book with some random data
        self.bids = {i: {'price': initial_mid_price - (i+1)*0.05, 
                        'size': np.random.randint(50, 300)} 
                    for i in range(levels)}
        
        self.asks = {i: {'price': initial_mid_price + (i+1)*0.05, 
                        'size': np.random.randint(50, 300)} 
                    for i in range(levels)}
    
    def get_snapshot(self):
        """Returns the current state of the order book"""
        snapshot = []
        
        for i in range(self.levels):
            snapshot.extend([
                self.bids[i]['price'],
                self.bids[i]['size'],
                self.asks[i]['price'],
                self.asks[i]['size']
            ])
            
        # Calculate additional metrics
        self.spread = self.asks[0]['price'] - self.bids[0]['price']
        self.mid_price = (self.asks[0]['price'] + self.bids[0]['price']) / 2
        
        # Add derived metrics
        imbalance = sum(self.bids[i]['size'] for i in range(self.levels)) - \
                    sum(self.asks[i]['size'] for i in range(self.levels))
        
        # Normalize the imbalance
        total_volume = sum(self.bids[i]['size'] for i in range(self.levels)) + \
                      sum(self.asks[i]['size'] for i in range(self.levels))
                      
        imbalance_ratio = imbalance / total_volume if total_volume > 0 else 0
        
        # VWAP calculation (simplified)
        vwap = sum(self.bids[i]['price'] * self.bids[i]['size'] for i in range(self.levels)) + \
               sum(self.asks[i]['price'] * self.asks[i]['size'] for i in range(self.levels))
        vwap /= total_volume if total_volume > 0 else 1
        
        snapshot.extend([self.mid_price, self.spread, imbalance_ratio, vwap])
        
        return snapshot
    
    def process_order(self, order):
        """Process an incoming order and update the book"""
        order_type = order['type']
        direction = order['direction']
        price = order['price']

        if order_type == 'limit':

                        break

                    if self.asks[i]['size'] >= remaining:

                        remaining = 0

                    else:
                        remaining -= self.asks[i]['size']

                        self.bids[i]['size'] -= remaining
                        remaining = 0
                        break
                    else:
                        remaining -= self.bids[i]['size']
                        self.bids[i]['size'] = 0
                        
        elif order_type == 'cancel':
            if direction == 'buy':
                # Find the right level based on price

                    if price == self.asks[i]['price']:

class TRADESSimulator:

                price = mid_price + np.random.exponential(0.2)

            depth = abs(price - self.lob.bids[0]['price'])

            depth = abs(price - self.lob.asks[0]['price'])
            
        # Time offset in milliseconds
        time_offset = int(np.random.exponential(100))
        
        return {
            'type': order_type,

            'time_offset': time_offset

        """Run the simulation for a specified number of steps"""

class SimpleStrategy:
    """A simple trading strategy to test on simulated data"""
    
    def __init__(self, lookback=10):
        self.lookback = lookback
        self.positions = []
        self.cash = 100000
        self.holdings = 0
        self.trades = []
        
    def run(self, data):
        """Run the strategy on the provided data"""
        # Calculate signals - simple momentum and order imbalance strategy
        data['mid_price_change'] = data['mid_price'].pct_change(self.lookback)
        data['imbalance_ma'] = data['imbalance'].rolling(self.lookback).mean()
        
        # Generate signals
        data['signal'] = 0
        # Buy signal: positive momentum and positive imbalance
        data.loc[(data['mid_price_change'] > 0.001) & 
                 (data['imbalance_ma'] > 0.1), 'signal'] = 1
        # Sell signal: negative momentum and negative imbalance
        data.loc[(data['mid_price_change'] < -0.001) & 
                 (data['imbalance_ma'] < -0.1), 'signal'] = -1
        
        # Execute strategy
        position = 0
        entry_price = 0
        
        for idx, row in data.iterrows():
            if row['signal'] == 1 and position == 0:
                # Buy
                position = 1
                shares = int(self.cash / row['mid_price'] * 0.9)  # Use 90% of cash
                cost = shares * row['mid_price']
                self.cash -= cost
                self.holdings += shares
                entry_price = row['mid_price']
                
                self.trades.append({
                    'timestamp': row['timestamp'],
                    'action': 'BUY',
                    'price': row['mid_price'],
                    'shares': shares,
                    'cost': cost,
                    'cash': self.cash,
                    'holdings': self.holdings,
                    'value': self.cash + self.holdings * row['mid_price']
                })
                
            elif row['signal'] == -1 and position == 1:
                # Sell
                position = 0
                revenue = self.holdings * row['mid_price']
                pnl = revenue - (self.holdings * entry_price)
                self.cash += revenue
                
                self.trades.append({
                    'timestamp': row['timestamp'],
                    'action': 'SELL',
                    'price': row['mid_price'],
                    'shares': self.holdings,
                    'revenue': revenue,
                    'pnl': pnl,
                    'cash': self.cash,
                    'holdings': 0,
                    'value': self.cash
                })
                
                self.holdings = 0
                
            # Track current position value
            self.positions.append({
                'timestamp': row['timestamp'],
                'mid_price': row['mid_price'],
                'cash': self.cash,
                'holdings': self.holdings,
                'position_value': self.holdings * row['mid_price'],
                'total_value': self.cash + self.holdings * row['mid_price']
            })
                
        return pd.DataFrame(self.trades), pd.DataFrame(self.positions)
