class ImpliedVolatilitySurface:
    """Simplified representation of the IV surface model from the paper"""
    
    def __init__(self, initial_factors: np.ndarray = None):
        # Initialize IV factors: [long-term ATM IV, time-to-maturity slope, moneyness slope,

            self.factors = np.array([0.25, 0.1, 0.1, 0.01, -0.05])
        else:
            self.factors = initial_factors

        self.Tconv = 0.25
        self.Tmax = 5.0
        self.Tmin = 6/252
    
    def get_iv(self, moneyness: float, time_to_maturity: float) -> float:
        """Calculate implied volatility for given moneyness and time-to-maturity"""
        Mt = moneyness
        tau = time_to_maturity
        
        # f1: Long-term ATM IV
        f1 = self.factors[0]
        
        # f2: Time-to-maturity slope
        f2 = self.factors[1] * np.exp(-np.sqrt(tau/self.Tconv))
        
        # f3: Moneyness slope
        if Mt >= 0:
            f3_term = Mt
        else:
            f3_term = (np.exp(2*Mt) - 1) / (np.exp(2*Mt) + 1)
        f3 = self.factors[2] * f3_term
        
        # f4: Smile attenuation
        f4 = self.factors[3] * (1 - np.exp(-Mt**2)) * np.log(tau/self.Tmax)
        
        # f5: Smirk
        f5 = 0
        if Mt < 0:
            f5 = self.factors[4] * (1 - np.exp((3*Mt)**3)) * np.log(tau/self.Tmax)
        
        # Total implied volatility
        iv = f1 + f2 + f3 + f4 + f5
        return max(0.05, min(0.8, iv))  # Bound the volatility for stability

        """Update IV factors based on simulated market changes"""

class BlackScholes:
    """Black-Scholes option pricing model"""
    
    @staticmethod
    def d1(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        return (np.log(S/K) + (r - q + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    
    @staticmethod
    def d2(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        return BlackScholes.d1(S, K, r, q, sigma, T) - sigma * np.sqrt(T)
    
    @staticmethod
    def call_price(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        if T <= 0:
            return max(0, S - K)
        d1 = BlackScholes.d1(S, K, r, q, sigma, T)
        d2 = BlackScholes.d2(S, K, r, q, sigma, T)
        return S * np.exp(-q * T) * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    
    @staticmethod
    def put_price(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        if T <= 0:
            return max(0, K - S)
        d1 = BlackScholes.d1(S, K, r, q, sigma, T)
        d2 = BlackScholes.d2(S, K, r, q, sigma, T)
        return K * np.exp(-r * T) * norm.cdf(-d2) - S * np.exp(-q * T) * norm.cdf(-d1)
    
    @staticmethod
    def straddle_price(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        return BlackScholes.call_price(S, K, r, q, sigma, T) + BlackScholes.put_price(S, K, r, q, sigma, T)
    
    @staticmethod
    def call_delta(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        if T <= 0:
            return 1.0 if S > K else 0.0
        d1 = BlackScholes.d1(S, K, r, q, sigma, T)
        return np.exp(-q * T) * norm.cdf(d1)
    
    @staticmethod
    def put_delta(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        if T <= 0:
            return -1.0 if S < K else 0.0
        d1 = BlackScholes.d1(S, K, r, q, sigma, T)
        return np.exp(-q * T) * (norm.cdf(d1) - 1)
    
    @staticmethod
    def straddle_delta(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        return BlackScholes.call_delta(S, K, r, q, sigma, T) + BlackScholes.put_delta(S, K, r, q, sigma, T)
    
    @staticmethod
    def gamma(S: float, K: float, r: float, q: float, sigma: float, T: float) -> float:
        if T <= 0:
            return 0.0
        d1 = BlackScholes.d1(S, K, r, q, sigma, T)
        return np.exp(-q * T) * norm.pdf(d1) / (S * sigma * np.sqrt(T))


class MarketSimulator:

    def __init__(self, S0: float = 100.0, r: float = 0.0266, q: float = 0.0177,

        # Correlation parameters
        self.correlations = np.array([
            -0.55,  # between returns and long-term ATM IV
            -0.69,  # between returns and time-to-maturity slope
            0.03,   # between returns and moneyness slope
            -0.22,  # between returns and smile attenuation
            -0.34   # between returns and smirk
        ])
        
    def step(self) -> Tuple[float, np.ndarray]:
        """Simulate one step (day) in the market"""
        # Generate correlated random innovations
        Z = np.random.randn(6)
        Z[1:] = Z[1:] * np.sqrt(1 - self.correlations**2) + self.correlations * Z[0]
        
        # Update S&P 500 price
        daily_vol = self.return_volatility / np.sqrt(252)
        self.S *= np.exp((self.r - self.q - 0.5 * daily_vol**2) * (1/252) + daily_vol * Z[0])
        
        # Update IV surface
        factor_changes = np.array([
            0.02 * Z[1] / np.sqrt(252),  # long-term ATM IV
            0.05 * Z[2] / np.sqrt(252),  # time-to-maturity slope
            0.01 * Z[3] / np.sqrt(252),  # moneyness slope
            0.005 * Z[4] / np.sqrt(252), # smile attenuation
            0.01 * Z[5] / np.sqrt(252)   # smirk
        ])
        self.iv_surface.update_factors(factor_changes)
        
        return self.S, self.iv_surface.factors

    def get_option_price(self, K: float, T: float, option_type: str = 'call') -> float:
        """Calculate option price based on current market state"""
        # Calculate moneyness
        tau = T / 252  # Convert days to years

        elif option_type == 'put':

        elif option_type == 'straddle':
            return BlackScholes.straddle_price(self.S, K, self.r, self.q, sigma, tau)

class DeltaGammaHedging:
    """Implementation of Delta-Gamma hedging strategy"""
    
    def __init__(self, rebalancing_threshold: float = 0.0):
        self.underlying_position = 0.0
        self.option_position = 0.0
        self.cash_position = 0.0
        self.rebalancing_threshold = rebalancing_threshold
    
    def rebalance(self, market: MarketSimulator, straddle_K: float, straddle_T: float, 
                  hedge_option_K: float, hedge_option_T: float, kappa1: float, kappa2: float) -> None:
        """
        Rebalance the portfolio using delta-gamma hedging
        """
        # Calculate deltas and gammas
        straddle_delta = market.get_option_delta(straddle_K, straddle_T, 'straddle')
        straddle_gamma = market.get_option_gamma(straddle_K, straddle_T)
        
        hedge_option_delta = market.get_option_delta(hedge_option_K, hedge_option_T, 'call')
        hedge_option_gamma = market.get_option_gamma(hedge_option_K, hedge_option_T)
        
        # Calculate target positions
        target_option_position = straddle_gamma / hedge_option_gamma
        target_underlying_position = straddle_delta - (target_option_position * hedge_option_delta)
        
        # Check if rebalancing is needed (no-trade region)
        position_change = abs(self.underlying_position - target_underlying_position) + abs(self.option_position - target_option_position)
        
        if position_change > self.rebalancing_threshold:
            # Calculate transaction costs
            S = market.S
            hedge_option_price = market.get_option_price(hedge_option_K, hedge_option_T, 'call')
            
            transaction_cost = (
                kappa1 * S * abs(target_underlying_position - self.underlying_position) +
                kappa2 * hedge_option_price * abs(target_option_position - self.option_position)
            )
            
            # Update positions
            old_portfolio_value = (
                self.cash_position + 
                self.underlying_position * S + 
                self.option_position * hedge_option_price
            )
            
            self.underlying_position = target_underlying_position
            self.option_position = target_option_position
            
            # Update cash to maintain self-financing condition
            self.cash_position = (
                old_portfolio_value - 
                self.underlying_position * S - 
                self.option_position * hedge_option_price - 
                transaction_cost
            )
    
    def get_portfolio_value(self, market: MarketSimulator, hedge_option_K: float, hedge_option_T: float) -> float:
        """Calculate current portfolio value"""
        S = market.S
        hedge_option_price = market.get_option_price(hedge_option_K, hedge_option_T, 'call')
        
        return self.cash_position + self.underlying_position * S + self.option_position * hedge_option_price

def run_simulation(T: int = 63, kappa1: float = 0.0005, kappa2: float = 0.01,

        T: Time horizon in days

        num_paths: Number of price paths to simulate
        
    Returns:

        'rl_hedging_errors': [],

        'rl_rebalance_counts': [],

        'rl_transaction_costs': []
    }
    
    for path_num in range(num_paths):
        # Initialize the market
        market = MarketSimulator()
        S0 = market.S
        
        # Initialize hedging strategies
        dg_strategy = DeltaGammaHedging(rebalancing_threshold=dg_threshold)
        rl_strategy = DeepHedgingSimulation(rebalancing_threshold=rl_threshold)
        
        # Set initial hedging portfolio values
        straddle_K = S0  # ATM straddle
        straddle_T = T
        hedge_option_K = S0  # ATM call option

        rl_strategy.cash_position = initial_straddle_price

        price_path = [market.S]
        dg_rebalance_count = 0
        rl_rebalance_count = 0
        dg_transaction_cost = 0
        rl_transaction_cost = 0
        
        # Run simulation for T days
        for day in range(T):
            # Update market
            market.step()
            price_path.append(market.S)
            
            # Record positions before rebalancing
            dg_prev_underlying = dg_strategy.underlying_position
            dg_prev_option = dg_strategy.option_position
            rl_prev_underlying = rl_strategy.underlying_position
            rl_prev_option = rl_strategy.option_position

            # Count rebalances
            if (dg_prev_underlying != dg_strategy.underlying_position or

                dg_transaction_cost += (
                    kappa1 * market.S * abs(dg_strategy.underlying_position - dg_prev_underlying) +
                    kappa2 * market.get_option_price(hedge_option_K, hedge_option_T - day - 1, 'call') * 
                    abs(dg_strategy.option_position - dg_prev_option)
                )
                
            if (rl_prev_underlying != rl_strategy.underlying_position or 
                rl_prev_option != rl_strategy.option_position):
                rl_rebalance_count += 1
                
                # Calculate transaction costs
                rl_transaction_cost += (

                    abs(rl_strategy.option_position - rl_prev_option)
                )
        
        # Calculate final payoff of the straddle
        final_straddle_payoff = max(market.S - straddle_K, 0) + max(straddle_K - market.S, 0)
        
        # Calculate final portfolio values
        dg_final_value = dg_strategy.get_portfolio_value(market, hedge_option_K, hedge_option_T - T)
        rl_final_value = rl_strategy.get_portfolio_value(market, hedge_option_K, hedge_option_T - T)
        
        # Calculate hedging errors
        dg_hedging_error = final_straddle_payoff - dg_final_value

        # Store results
        results['dg_hedging_errors'].append(dg_hedging_error)

        results['rl_rebalance_counts'].append(rl_rebalance_count)
        results['dg_transaction_costs'].append(dg_transaction_cost)

    return results


def evaluate_results(results: Dict[str, Any]) -> Dict[str, float]:
    """Calculate performance metrics from simulation results"""
    dg_errors = np.array(results['dg_hedging_errors'])
    rl_errors = np.array(results['rl_hedging_errors'])
    
    # Calculate MSE
    dg_mse = np.mean(dg_errors**2)
    rl_mse = np.mean(rl_errors**2)
    
    # Calculate SMSE (semi-mean-square error)
    dg_smse = np.mean((dg_errors * (dg_errors >= 0))**2)
    rl_smse = np.mean((rl_errors * (rl_errors >= 0))**2)
    
    # Calculate CVaR (95%)
    alpha = 0.95
    dg_sorted = np.sort(dg_errors)
    rl_sorted = np.sort(rl_errors)
    cutoff_idx = int(len(dg_errors) * alpha)
    dg_cvar = np.mean(dg_sorted[cutoff_idx:])
    rl_cvar = np.mean(rl_sorted[cutoff_idx:])
    
    # Average rebalance counts and transaction costs
    dg_avg_rebalances = np.mean(results['dg_rebalance_counts'])
    rl_avg_rebalances = np.mean(results['rl_rebalance_counts'])
    dg_avg_tc = np.mean(results['dg_transaction_costs'])
    rl_avg_tc = np.mean(results['rl_transaction_costs'])
    
    return {
        'dg_mse': dg_mse,
        'rl_mse': rl_mse,
        'dg_smse': dg_smse,
        'rl_smse': rl_smse,
        'dg_cvar': dg_cvar,
        'rl_cvar': rl_cvar,
        'dg_avg_rebalances': dg_avg_rebalances,
        'rl_avg_rebalances': rl_avg_rebalances,

    }

This implementation simulates the key concepts from the paper:

1. A market simulator that models both S&P 500 price movements and the implied volatility surface dynamics
2. A delta-gamma hedging benchmark strategy with an option as additional hedging instrument
3. A simplified deep hedging strategy that mimics the reinforcement learning approach described in the paper
4. No-trade regions to optimize rebalancing frequency under transaction costs
5. Evaluation metrics including MSE, SMSE, and CVaR
6. Visualization of hedging performance and sensitivity to transaction costs

Note that this is a simplified simulation. The actual deep hedging framework in the paper uses sophisticated neural networks trained with reinforcement learning, which is beyond the scope of this implementation. Instead, I've simulated the key behaviors observed in the paper: smaller initial positions, more frequent but smaller trades, and adaptability to market conditions.
