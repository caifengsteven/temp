@dataclass
class Order:
    sign: int  # +1 for buy, -1 for sell
    price: float  # in CUP (Cuban Pesos)
    volume: float  # in USD
    timestamp: datetime
    
    def __str__(self):
        order_type = "BUY" if self.sign == 1 else "SELL"
        return f"{order_type} {self.volume} USD at {self.price} CUP ({self.timestamp})"

class LOB:
    """Limit Order Book implementation"""
    
    def __init__(self, expiry_days=7):
        self.buy_orders = []  # bids
        self.sell_orders = []  # asks
        self.expiry_days = expiry_days
        self.executed_orders = []
        
    def add_order(self, order: Order) -> List[Tuple[Order, Order, float, float]]:
        """Add an order to the book and return executed trades"""
        # Clean expired orders
        self._remove_expired_orders(order.timestamp)
        
        executed = []
        
        if order.sign == 1:  # Buy order
            # Check for matching sell orders
            matching_sells = sorted(
                [o for o in self.sell_orders if o.price <= order.price],
                key=lambda x: (x.price, x.timestamp)
            )
            
            remaining_volume = order.volume
            for sell_order in matching_sells:
                if remaining_volume <= 0:
                    break
                    
                execute_volume = min(remaining_volume, sell_order.volume)
                execute_price = sell_order.price
                
                executed.append((order, sell_order, execute_volume, execute_price))
                
                # Update orders
                remaining_volume -= execute_volume
                sell_order.volume -= execute_volume
                
                if sell_order.volume <= 0:
                    self.sell_orders.remove(sell_order)
            
            # Add remaining order to book
            if remaining_volume > 0:
                new_order = Order(
                    sign=order.sign,
                    price=order.price,
                    volume=remaining_volume,
                    timestamp=order.timestamp
                )
                self.buy_orders.append(new_order)
                
        else:  # Sell order
            # Check for matching buy orders
            matching_buys = sorted(
                [o for o in self.buy_orders if o.price >= order.price],
                key=lambda x: (-x.price, x.timestamp)
            )
            
            remaining_volume = order.volume
            for buy_order in matching_buys:
                if remaining_volume <= 0:
                    break
                    
                execute_volume = min(remaining_volume, buy_order.volume)
                execute_price = buy_order.price
                
                executed.append((buy_order, order, execute_volume, execute_price))
                
                # Update orders
                remaining_volume -= execute_volume
                buy_order.volume -= execute_volume
                
                if buy_order.volume <= 0:
                    self.buy_orders.remove(buy_order)
            
            # Add remaining order to book
            if remaining_volume > 0:
                new_order = Order(
                    sign=order.sign,
                    price=order.price,
                    volume=remaining_volume,
                    timestamp=order.timestamp
                )
                self.sell_orders.append(new_order)
        
        # Add executed orders to history
        self.executed_orders.extend(executed)
        return executed
    
    def _remove_expired_orders(self, current_time: datetime):
        """Remove orders that have expired"""
        expiry_time = current_time - timedelta(days=self.expiry_days)
        
        self.buy_orders = [o for o in self.buy_orders if o.timestamp >= expiry_time]
        self.sell_orders = [o for o in self.sell_orders if o.timestamp >= expiry_time]
    
    @property
    def bid_price(self) -> Optional[float]:
        """Return the highest buy price"""
        if not self.buy_orders:
            return None
        return max(o.price for o in self.buy_orders)
    
    @property
    def ask_price(self) -> Optional[float]:
        """Return the lowest sell price"""
        if not self.sell_orders:
            return None
        return min(o.price for o in self.sell_orders)
    
    @property
    def mid_price(self) -> Optional[float]:
        """Return the mid price"""
        if self.bid_price is None or self.ask_price is None:
            return None
        return (self.bid_price + self.ask_price) / 2
    
    @property
    def spread(self) -> Optional[float]:
        """Return the bid-ask spread"""
        if self.bid_price is None or self.ask_price is None:
            return None
        return self.ask_price - self.bid_price
    
    @property
    def relative_spread(self) -> Optional[float]:
        """Return the relative bid-ask spread"""
        if self.mid_price is None or self.mid_price == 0:
            return None
        return self.spread / self.mid_price if self.spread is not None else None
    
    def total_buy_volume(self) -> float:
        """Return the total volume on the buy side"""
        return sum(o.volume for o in self.buy_orders)
    
    def total_sell_volume(self) -> float:
        """Return the total volume on the sell side"""
        return sum(o.volume for o in self.sell_orders)
    
    def executed_volume(self, start_time: datetime, end_time: datetime) -> float:
        """Return the executed volume in the given time period"""
        return sum(vol for _, _, vol, _ in self.executed_orders)
    
    def volume_at_distance(self, distance: float, side: str = 'buy') -> float:
        """Return the volume at a given distance from the best price"""
        if side == 'buy':
            if self.bid_price is None:
                return 0
            return sum(o.volume for o in self.buy_orders if self.bid_price - o.price <= distance)
        else:
            if self.ask_price is None:
                return 0
            return sum(o.volume for o in self.sell_orders if o.price - self.ask_price <= distance)


class MarketMaker:
    """Implementation of the Avellaneda-Stoikov market maker model adapted for informal markets"""
    
    def __init__(self, 
                 initial_cash: float,
                 initial_inventory: float,
                 risk_aversion: float,
                 time_horizon: float,
                 volatility: float,
                 intensity_param_A: float,
                 intensity_param_alpha: float,
                 intensity_param_K: float):
        self.cash = initial_cash
        self.inventory = initial_inventory
        self.gamma = risk_aversion  # risk aversion parameter
        self.T = time_horizon  # time horizon in days
        self.sigma = volatility  # price volatility
        self.A = intensity_param_A  # intensity parameter (λ = A*e^(-α*e^(K*δ)))
        self.alpha = intensity_param_alpha  # order size distribution parameter
        self.K = intensity_param_K  # price impact parameter
        
        # For tracking
        self.cash_history = []
        self.inventory_history = []
        self.bid_prices = []
        self.ask_prices = []
        
    def _solve_transcendental(self, q: float, t: float, side: str) -> float:
        """
        Solve the transcendental equation for optimal spread using fixed-point iteration
        """
        # Initialization
        if side == 'bid':
            delta = q * self.gamma * self.sigma**2 * (self.T - t)
            sign = 1
        else:  # ask
            delta = -q * self.gamma * self.sigma**2 * (self.T - t)
            sign = 1
            
        # Fixed-point iteration
        for _ in range(20):  # usually converges quickly
            next_delta = delta + (1/self.gamma) * np.log(1 + (self.gamma/(self.alpha*self.K)) * np.exp(self.K*delta))
            if abs(next_delta - delta) < 1e-6:
                break
            delta = next_delta
            
        return delta
        
    def get_quotes(self, mid_price: float, current_time: float) -> Tuple[float, float]:
        """
        Get the optimal bid and ask quotes based on the current state
        """
        time_to_go = max(0, self.T - current_time)
        
        # Calculate reservation price (indifference price)
        r = mid_price - self.inventory * self.gamma * self.sigma**2 * time_to_go
        
        # Solve for optimal spreads
        delta_bid = self._solve_transcendental(self.inventory, current_time, 'bid')
        delta_ask = self._solve_transcendental(self.inventory, current_time, 'ask')
        
        # Calculate actual prices
        bid_price = r - delta_bid
        ask_price = r + delta_ask
        
        return bid_price, ask_price
    
    def update(self, mid_price: float, current_time: float, orders: List[Order]) -> List[Order]:
        """
        Update the market maker's state and generate new orders
        """
        # Get optimal quotes
        bid_price, ask_price = self.get_quotes(mid_price, current_time)
        
        # Save for tracking
        self.bid_prices.append(bid_price)
        self.ask_prices.append(ask_price)
        
        # Generate market maker orders
        mm_orders = []
        
        # Create bid order
        mm_bid = Order(
            sign=1,  # buy
            price=bid_price,
            volume=1000,  # fixed volume for simplicity
            timestamp=datetime.now()
        )
        mm_orders.append(mm_bid)
        
        # Create ask order
        mm_ask = Order(
            sign=-1,  # sell
            price=ask_price,
            volume=1000,  # fixed volume for simplicity
            timestamp=datetime.now()
        )
        mm_orders.append(mm_ask)
        
        # Process incoming orders against market maker quotes
        for order in orders:
            if order.sign == 1 and order.price >= ask_price:  # Buy order matching MM's ask
                # Execute at MM's ask price
                volume = min(order.volume, mm_ask.volume)
                self.cash += volume * ask_price
                self.inventory -= volume
                mm_ask.volume -= volume
                
            elif order.sign == -1 and order.price <= bid_price:  # Sell order matching MM's bid
                # Execute at MM's bid price
                volume = min(order.volume, mm_bid.volume)
                self.cash -= volume * bid_price
                self.inventory += volume
                mm_bid.volume -= volume
        
        # Update history
        self.cash_history.append(self.cash)
        self.inventory_history.append(self.inventory)
        
        return mm_orders
