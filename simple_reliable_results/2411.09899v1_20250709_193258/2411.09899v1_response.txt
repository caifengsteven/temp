# Euler-Maruyama discretization for Heston model
def simulate_heston(params, T=1.0, n_steps=2142, n_paths=1):
    """
    Simulate paths using Heston model with Euler-Maruyama discretization
    """
    dt = T / n_steps
    
    # Initialize arrays
    S = np.zeros((n_paths, n_steps + 1))
    V = np.zeros((n_paths, n_steps + 1))
    time = np.linspace(0, T, n_steps + 1)
    
    # Set initial values
    S[:, 0] = params.S0
    V[:, 0] = params.V0
    
    # Generate correlated Brownian motions
    for i in range(n_paths):
        Z1 = np.random.normal(0, np.sqrt(dt), n_steps)
        Z2 = params.rho * Z1 + np.sqrt(1 - params.rho**2) * np.random.normal(0, np.sqrt(dt), n_steps)
        
        for j in range(n_steps):
            # Ensure variance stays positive
            V[i, j] = max(V[i, j], 1e-10)
            
            # Update price
            S[i, j+1] = S[i, j] * (1 + params.mu * dt + np.sqrt(V[i, j]) * Z1[j])
            
            # Update variance
            V[i, j+1] = V[i, j] + params.kappa * (params.theta - V[i, j]) * dt + \
                        params.sigma_v * np.sqrt(V[i, j]) * Z2[j]
            V[i, j+1] = max(V[i, j+1], 1e-10)  # Ensure positive variance
    
    return time, S, V

# Calculate wealth trajectory given a portfolio allocation strategy
def calculate_wealth(params, time, S, V, pi_func, eta=1.0):
    """
    Calculate wealth trajectory given portfolio allocation strategy
    pi_func: function that takes (t, v) and returns allocation weight
    """
    n_paths = S.shape[0]
    n_steps = S.shape[1] - 1
    dt = time[1] - time[0]
    
    W = np.zeros((n_paths, n_steps + 1))
    Pi = np.zeros((n_paths, n_steps + 1))
    W[:, 0] = params.W0
    
    for i in range(n_paths):
        for j in range(n_steps):
            # Get allocation weight
            Pi[i, j] = pi_func(time[j], V[i, j])
            
            # Calculate returns
            stock_return = (S[i, j+1] - S[i, j]) / S[i, j]
            bond_return = params.r * dt
            
            # Update wealth
            portfolio_return = Pi[i, j] * stock_return + (1 - Pi[i, j]) * bond_return
            W[i, j+1] = W[i, j] * (1 + portfolio_return)
    
    # Calculate terminal utility
    if eta == 1:
        utility = np.log(W[:, -1])
    else:
        utility = (W[:, -1]**(1-eta) - 1) / (1 - eta)
    
    return W, Pi, np.mean(utility)
