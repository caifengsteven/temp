import numpy as np
import pandas as pd

class FutureContract:
    """
    Class representing a futures contract with specific maturity
    """
    def __init__(self, name, liquidity_rank, is_lead=False):
        self.name = name
        self.liquidity_rank = liquidity_rank  # F1, F2, F3, or F4
        self.is_lead = is_lead
        self.price_history = []
        self.bid_prices = []
        self.ask_prices = []
        self.volumes = []
        self.spread_history = []  # For tracking bid-ask spread

    def __str__(self):
        return f"{self.name} ({self.liquidity_rank})"


def generate_tick_data(lead_contract, lagging_contracts, n_ticks=10000, volatility=0.15):
    """
    Generate tick-by-tick data for a lead contract and lagging contracts
    """
    # Base parameters
    initial_price = 5000
    tick_size = 0.2  # Minimum price change
    
    # Generate lead contract price series with random walk
    price = initial_price
    lead_prices = [price]
    
    # Random walk for lead contract
    for _ in range(n_ticks-1):
        # Generate random price change
        direction = 1 if random.random() > 0.5 else -1
        # Most changes are 1 tick, but occasionally larger
        tick_change = direction * tick_size * max(1, int(random.expovariate(5)))
        price += tick_change
        lead_prices.append(price)
    
    # Store prices in lead contract
    lead_contract.price_history = lead_prices
    
    # Generate bid-ask for lead contract
    for price in lead_prices:
        spread = tick_size * (1 + int(random.expovariate(3)))  # Typically 1-2 ticks
        lead_contract.bid_prices.append(price - spread/2)
        lead_contract.ask_prices.append(price + spread/2)
        lead_contract.spread_history.append(spread)
        lead_contract.volumes.append(int(random.expovariate(0.001)))  # Higher volumes
    
    # Generate lagging contracts with lead-lag relationship
    for i, contract in enumerate(lagging_contracts):
        lag = i + 1  # Lag increases with liquidity rank
        
        # Base prices on lead prices with lag and some noise
        lagged_prices = [initial_price - (i+1)*5]  # Different starting price
        
        for t in range(1, n_ticks):
            # Follow lead contract with lag and noise
            if t >= lag:
                # Price change is influenced by lead contract's previous move
                lead_change = lead_prices[t-lag] - lead_prices[t-lag-1]
                # Add lag effect and noise
                lag_factor = 0.9 - (i * 0.1)  # Weaker following for less liquid contracts
                noise = np.random.normal(0, volatility * tick_size * (i+1))
                price_change = lead_change * lag_factor + noise
                
                # Round to nearest tick
                price_change = round(price_change / tick_size) * tick_size
                
                new_price = lagged_prices[-1] + price_change
            else:
                # For initial ticks before lag effect kicks in
                direction = 1 if random.random() > 0.5 else -1
                price_change = direction * tick_size
                new_price = lagged_prices[-1] + price_change
                
            lagged_prices.append(new_price)
        
        # Store prices
        contract.price_history = lagged_prices
        
        # Generate bid-ask for lagging contract (wider spreads for less liquid contracts)
        for price in lagged_prices:
            spread = tick_size * (1 + i + int(random.expovariate(2)))
            contract.bid_prices.append(price - spread/2)
            contract.ask_prices.append(price + spread/2)
            contract.spread_history.append(spread)
            contract.volumes.append(int(random.expovariate(0.002 * (i+1))))  # Lower volumes
    
    return lead_contract, lagging_contracts


def calculate_ema(data, period):
    """Calculate exponential moving average"""
    alpha = 2 / (period + 1)
    ema = [data[0]]
    
    for i in range(1, len(data)):
        ema.append(alpha * data[i] + (1 - alpha) * ema[i-1])
    
    return ema


def compute_signals(lead_contract, lag_contract, cycle=50):
    """
    Compute the calendar spread signal as described in the paper
    """
    # Extract midquotes
    lead_midquotes = [(bid + ask) / 2 for bid, ask in zip(lead_contract.bid_prices, lead_contract.ask_prices)]
    lag_midquotes = [(bid + ask) / 2 for bid, ask in zip(lag_contract.bid_prices, lag_contract.ask_prices)]
    
    # Calculate spreads
    spreads = [lead - lag for lead, lag in zip(lead_midquotes, lag_midquotes)]
    
    # Calculate EMA of spreads
    spread_ema = calculate_ema(spreads, cycle)
    
    # Calculate theta (short-term spread deviation)
    theta = [spread - ema for spread, ema in zip(spreads, spread_ema)]
    
    # Calculate lead momentum
    lead_ema = calculate_ema(lead_midquotes, cycle)
    lead_momentum = [lead - ema for lead, ema in zip(lead_midquotes, lead_ema)]
    
    return {
        'midquotes_lead': lead_midquotes,
        'midquotes_lag': lag_midquotes,
        'spreads': spreads,
        'spread_ema': spread_ema,
        'theta': theta,
        'lead_momentum': lead_momentum
    }


def backtest_strategy(lead_contract, lag_contract, signal_data, signal_threshold=1.0, max_spread=0.4):
    """
    Backtest the calendar spread feedback strategy
    """
    # Initialize variables
    position = 0
    trades = []
    pnl = 0
    transaction_fee = 0.1  # Half a tick
    
    for i in range(1, len(signal_data['theta'])):
        theta = signal_data['theta'][i]
        
        # Check if signal exceeds threshold and spread is acceptable
        if abs(theta) >= signal_threshold and lead_contract.spread_history[i] <= max_spread:
            # Determine trade direction
            if theta > 0:  # Sell lead contract (negative feedback)
                price = lead_contract.bid_prices[i]
                direction = -1
            else:  # Buy lead contract
                price = lead_contract.ask_prices[i]
                direction = 1
            
            # Record trade
            trades.append({
                'tick': i,
                'direction': direction,
                'price': price,
                'theta': theta,
                'position': direction  # 1 or -1 (one contract)
            })
            
            # Calculate PnL if closing a position
            if position != 0 and position != direction:
                last_trade = next(t for t in reversed(trades[:-1]) if t['position'] != 0)
                pnl_per_trade = (price - last_trade['price']) * -position - transaction_fee * 2
                pnl += pnl_per_trade
            
            position = direction
    
    # Close any remaining position at the end
    if position != 0:
        if position > 0:
            close_price = lead_contract.bid_prices[-1]
        else:
            close_price = lead_contract.ask_prices[-1]
        
        last_trade = next(t for t in reversed(trades) if t['position'] != 0)
        pnl_per_trade = (close_price - last_trade['price']) * -position - transaction_fee * 2
        pnl += pnl_per_trade
    
    # Calculate performance metrics
    num_trades = len(trades)
    pnl_per_trade = pnl / num_trades if num_trades > 0 else 0
    
    return {
        'trades': trades,
        'num_trades': num_trades,
        'total_pnl': pnl,
        'pnl_per_trade': pnl_per_trade
    }


def optimize_threshold(lead_contract, lag_contract, signal_data, thresholds):
    """
    Test different signal thresholds to find optimal value
    """
    results = []
    
    for threshold in thresholds:
        result = backtest_strategy(lead_contract, lag_contract, signal_data, signal_threshold=threshold)
        
        # Calculate Sharpe ratio (simplified)
        if result['num_trades'] > 0:
            trade_returns = []
            for i in range(1, len(result['trades'])):
                if result['trades'][i-1]['position'] != result['trades'][i]['position']:
                    trade_return = (result['trades'][i]['price'] - result['trades'][i-1]['price']) * -result['trades'][i-1]['position']
                    trade_returns.append(trade_return)
            
            if trade_returns:
                mean_return = np.mean(trade_returns)
                std_return = np.std(trade_returns) if len(trade_returns) > 1 else 1
                sharpe = mean_return / std_return * np.sqrt(252)  # Annualized
            else:
                sharpe = 0
        else:
            sharpe = 0
        
        results.append({
            'threshold': threshold,
            'num_trades': result['num_trades'],
            'total_pnl': result['total_pnl'],
            'pnl_per_trade': result['pnl_per_trade'],
            'sharpe': sharpe
        })
    
    return results

def test_lead_lag_relationship():
    """
    Test function to verify the lead-lag relationship in the simulated data
    """
    # Create contracts

    lag_contracts = [
        FutureContract("IF2408", "F2"),

    ]
    
    # Generate data
    lead_contract, lag_contracts = generate_tick_data(lead_contract, lag_contracts, n_ticks=5000)
    
    # Check lead-lag by analyzing price changes
    lead_changes = np.diff(lead_contract.price_history)
    
    for i, lag_contract in enumerate(lag_contracts):
        lag_changes = np.diff(lag_contract.price_history)
        
        # Shift lag_changes by the expected lag and compute correlation
        lag = i + 1
        corr = np.corrcoef(lead_changes[:-lag], lag_changes[lag:])[0, 1]
        
        print(f"Lead-lag correlation between {lead_contract} and {lag_contract} (lag={lag}): {corr:.4f}")
        
        # Cross-correlation at different lags
        max_lag = 5
        correlations = []
        for l in range(-max_lag, max_lag+1):
            if l <= 0:  # lag_contract lags behind lead_contract
                corr = np.corrcoef(lead_changes[:l] if l != 0 else lead_changes, 
                                   lag_changes[-l:] if l != 0 else lag_changes)[0, 1]
            else:  # lead_contract lags behind lag_contract
                corr = np.corrcoef(lead_changes[l:], lag_changes[:-l])[0, 1]
            correlations.append(corr)
        
        # Determine the lag with maximum correlation
        max_corr_idx = np.argmax(np.abs(correlations))
        max_corr_lag = max_corr_idx - max_lag
        max_corr = correlations[max_corr_idx]
        
        print(f"Maximum correlation at lag {max_corr_lag}: {max_corr:.4f}")
        
    return lead_contract, lag_contracts
