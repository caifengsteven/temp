class Alpha:
    """Represents an algorithmic trading strategy (alpha)"""
    
    def __init__(self, name, positions=None):
        self.name = name
        self.positions = positions if positions is not None else {}
        self.returns = []
        self.pnl = []
        self.cumulative_pnl = []
        self.turnover_history = []
    
    def generate_random_positions(self, tickers, days, seed=None):
        """Generate random positions for simulation purposes"""
        if seed is not None:
            np.random.seed(seed)
        
        self.positions = {}
        for day in days:
            # Generate random positions with some correlation to previous day
            if day == days[0]:
                positions = np.random.normal(0, 1, len(tickers))
            else:
                prev_positions = np.array([self.positions[days[days.index(day)-1]][ticker] for ticker in tickers])
                noise = np.random.normal(0, 0.3, len(tickers))
                positions = 0.7 * prev_positions + noise
            
            # Dollar neutrality constraint
            positions = positions - np.mean(positions)
            
            # Normalize positions
            positions = positions / np.sum(np.abs(positions))
            
            self.positions[day] = {tickers[i]: positions[i] for i in range(len(tickers))}
    
    def calculate_turnover(self, days):
        """Calculate turnover for each day"""
        self.turnover_history = []
        
        for i in range(1, len(days)):
            current_day = days[i]
            prev_day = days[i-1]
            
            turnover = 0
            for ticker in self.positions[current_day]:
                turnover += abs(self.positions[current_day][ticker] - self.positions[prev_day][ticker])
            
            self.turnover_history.append(turnover)
        
        return np.mean(self.turnover_history)
    
    def calculate_returns(self, market_returns, days):
        """Calculate alpha returns based on market returns and positions"""
        self.returns = []
        self.pnl = []
        self.cumulative_pnl = []
        
        for i in range(1, len(days)):
            day = days[i]
            prev_day = days[i-1]
            
            daily_return = 0
            for ticker in self.positions[prev_day]:
                ticker_return = market_returns[day].get(ticker, 0)
                daily_return += self.positions[prev_day][ticker] * ticker_return
            
            self.returns.append(daily_return)
            
            # Calculate PnL
            if not self.pnl:
                self.pnl.append(daily_return)
                self.cumulative_pnl.append(daily_return)
            else:
                self.pnl.append(daily_return)
                self.cumulative_pnl.append(self.cumulative_pnl[-1] + daily_return)
    
    def get_std_returns(self):
        """Get standard deviation of returns"""
        return np.std(self.returns)
    
    def get_sharpe_ratio(self):
        """Calculate Sharpe ratio (assuming zero risk-free rate)"""
        if not self.returns:
            return 0
        
        avg_return = np.mean(self.returns)
        std_return = np.std(self.returns)
        
        if std_return == 0:
            return 0
            
        return np.sqrt(252) * avg_return / std_return
    
    def get_avg_turnover(self):
        """Get average turnover"""
        if not self.turnover_history:
            return 0
        return np.mean(self.turnover_history)
    
    def get_turnover_to_std_ratio(self):
        """Calculate ratio of turnover to standard deviation of returns"""
        std = self.get_std_returns()
        if std == 0:
            return float('inf')
        return self.get_avg_turnover() / std


class Portfolio:
    """Represents a portfolio of alphas"""
    
    def __init__(self, alphas=None, weights=None):
        self.alphas = alphas if alphas is not None else []
        self.weights = weights if weights is not None else []
        
        # Normalize weights to sum to 1
        if self.weights:
            self.weights = np.array(self.weights) / np.sum(self.weights)
            
        self.positions = {}
        self.turnover_history = []
        self.returns = []
    
    def calculate_positions(self, days):
        """Calculate portfolio positions"""
        self.positions = {}
        
        for day in days:
            self.positions[day] = {}
            
            for ticker in set().union(*[alpha.positions[day].keys() for alpha in self.alphas]):
                self.positions[day][ticker] = 0
                
                for i, alpha in enumerate(self.alphas):
                    if ticker in alpha.positions[day]:
                        self.positions[day][ticker] += self.weights[i] * alpha.positions[day][ticker]
    
    def calculate_actual_turnover(self, days):
        """Calculate actual turnover with crossing of trades"""
        self.turnover_history = []
        
        for i in range(1, len(days)):
            current_day = days[i]
            prev_day = days[i-1]
            
            turnover = 0
            for ticker in set(self.positions[current_day].keys()).union(self.positions[prev_day].keys()):
                current_pos = self.positions[current_day].get(ticker, 0)
                prev_pos = self.positions[prev_day].get(ticker, 0)
                turnover += abs(current_pos - prev_pos)
            
            self.turnover_history.append(turnover)
        
        return np.mean(self.turnover_history)
    
    def calculate_max_turnover(self, days):
        """Calculate turnover without crossing of trades"""
        max_turnover_history = []
        
        for i in range(1, len(days)):
            day_turnover = 0
            for j, alpha in enumerate(self.alphas):
                if i <= len(alpha.turnover_history):
                    day_turnover += abs(self.weights[j]) * alpha.turnover_history[i-1]
            
            max_turnover_history.append(day_turnover)
        
        return np.mean(max_turnover_history)
    
    def calculate_returns(self, days):
        """Calculate portfolio returns"""
        self.returns = []
        
        for i in range(min(len(self.alphas[0].returns), len(days)-1)):
            portfolio_return = 0
            for j, alpha in enumerate(self.alphas):
                portfolio_return += self.weights[j] * alpha.returns[i]
            
            self.returns.append(portfolio_return)
    
    def get_covariance_matrix(self):
        """Calculate covariance matrix of alpha returns"""
        returns_matrix = np.array([alpha.returns for alpha in self.alphas])
        return np.cov(returns_matrix)
    
    def get_correlation_matrix(self):
        """Calculate correlation matrix of alpha returns"""
        cov_matrix = self.get_covariance_matrix()
        std_devs = np.sqrt(np.diag(cov_matrix))
        return cov_matrix / np.outer(std_devs, std_devs)
    
    def estimate_turnover_KL(self):
        """Kakushadze and Liew turnover estimation (1.2)"""
        cov_matrix = self.get_covariance_matrix()
        eig_vals, eig_vecs = np.linalg.eigh(cov_matrix)
        
        # Sort eigenvalues and eigenvectors in descending order
        idx = eig_vals.argsort()[::-1]
        eig_vals = eig_vals[idx]
        eig_vecs = eig_vecs[:, idx]
        
        turnover_est = 0
        n = len(self.alphas)
        
        for p in range(n):
            term = 0
            for i in range(n):
                term += eig_vecs[i, p] * self.alphas[i].get_avg_turnover() * abs(self.weights[i])
            
            turnover_est += np.sqrt(eig_vals[p]) * term
        
        return turnover_est / np.sqrt(n)
    
    def estimate_turnover_1(self):
        """T*1 estimation (2.4) - arithmetic mean"""
        sigma = np.sqrt(np.dot(self.weights, np.dot(self.get_covariance_matrix(), self.weights)))
        
        kappa = 0
        for alpha in self.alphas:
            kappa += alpha.get_turnover_to_std_ratio()
        kappa /= len(self.alphas)
        
        return kappa * sigma
    
    def estimate_turnover_2(self):
        """T*2 estimation (2.4) - geometric mean"""
        sigma = np.sqrt(np.dot(self.weights, np.dot(self.get_covariance_matrix(), self.weights)))
        
        kappa = 1
        for alpha in self.alphas:
            kappa *= alpha.get_turnover_to_std_ratio()
        kappa = kappa ** (1/len(self.alphas))
        
        return kappa * sigma
    
    def estimate_turnover_3(self):
        """T*3 estimation (2.4) - weighted average"""
        sigma = np.sqrt(np.dot(self.weights, np.dot(self.get_covariance_matrix(), self.weights)))
        
        num = 0
        denom = 0
        for i, alpha in enumerate(self.alphas):
            num += self.weights[i] * alpha.get_avg_turnover()
            denom += self.weights[i] * alpha.get_std_returns()
        
        if denom == 0:
            return 0
            
        kappa = num / denom
        
        return kappa * sigma
    
    def estimate_turnover_4(self):
        """T*4 estimation (2.4) - overall ratio"""
        sigma = np.sqrt(np.dot(self.weights, np.dot(self.get_covariance_matrix(), self.weights)))
        
        total_turnover = sum(alpha.get_avg_turnover() for alpha in self.alphas)
        total_std = sum(alpha.get_std_returns() for alpha in self.alphas)
        
        if total_std == 0:
            return 0
            
        kappa = total_turnover / total_std
        
        return kappa * sigma


def calculate_metrics(estimation, actual, max_turnover):
    """Calculate the metrics defined in the paper (3.2)"""
    rho1 = np.mean(estimation - actual)
    rho2 = np.mean(np.abs(estimation - actual))
    
    rho1_max = np.mean(max_turnover - actual)
    
    if rho1_max == 0:
        rho3 = float('inf')
        rho4 = float('inf')
    else:
        rho3 = abs(rho1) / rho1_max
        rho4 = rho2 / rho1_max
        
    rho5 = np.mean(np.abs(estimation - actual) / actual)
    
    return {
        'rho1': abs(rho1),
        'rho2': rho2,
        'rho3': rho3,
        'rho4': rho4,
        'rho5': rho5
    }


def generate_market_data(tickers, days, seed=None):
    """Generate simulated market returns"""
    if seed is not None:
        np.random.seed(seed)
        
    market_returns = {}
    
    # Generate random market trend
    market_trend = np.cumsum(np.random.normal(0.0002, 0.01, len(days)))
    
    for i, day in enumerate(days):
        market_returns[day] = {}
        
        # Add market-wide component and idiosyncratic component for each ticker
        for ticker in tickers:
            ticker_beta = np.random.uniform(0.5, 1.5)
            idiosyncratic = np.random.normal(0, 0.02)
            
            if i > 0:
                market_move = market_trend[i] - market_trend[i-1]
                market_returns[day][ticker] = ticker_beta * market_move + idiosyncratic
            else:
                market_returns[day][ticker] = np.random.normal(0, 0.01)
    
    return market_returns

            day = days[day_idx]
            prev_day = days[day_idx-1]
