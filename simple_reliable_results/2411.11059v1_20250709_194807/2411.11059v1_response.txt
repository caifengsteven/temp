class SimplePPOAgent:
    """A simplified PPO agent for demonstration purposes"""
    
    def __init__(self, state_dim, action_dim, is_portfolio=False):
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.is_portfolio = is_portfolio
        
        # Initialize simple policy parameters
        self.policy_params = np.random.randn(state_dim, action_dim) * 0.1
        self.value_params = np.random.randn(state_dim, 1) * 0.1
        
    def select_action(self, state):
        if self.is_portfolio:
            # For portfolio, handle each stock separately
            actions = np.zeros(self.action_dim)
            for i in range(len(state)):
                stock_state = state[i].flatten()
                # Simple linear policy: state @ params + noise
                action = np.clip(stock_state @ self.policy_params + np.random.normal(0, 0.1), 0, 2)
                actions[i] = action
            return actions
        else:
            # Simple linear policy: state @ params + noise
            action = np.clip(state @ self.policy_params + np.random.normal(0, 0.1, self.action_dim), 0, 2)
            if self.action_dim == 2:  # For single stock case
                action[1] = np.clip(action[1], 0, 0.5)  # Clip amount to [0, 0.5]
            return action
        
    def update(self, states, actions, rewards, dones, next_states):
        # In a real PPO implementation, this would update the policy and value networks
        # For simplicity, we just simulate learning by adding a small noise to parameters
        self.policy_params += np.random.normal(0, 0.01, self.policy_params.shape) * np.mean(rewards)
        self.value_params += np.random.normal(0, 0.01, self.value_params.shape) * np.mean(rewards)


def generate_simulated_data(symbols, num_days=252, start_date='2023-11-16'):
    """Generate simulated stock data for testing"""
    
    data_dict = {}
    
    for symbol in symbols:
        # Initialize with a random starting price between 50 and 200
        start_price = np.random.uniform(50, 200)
        
        # Generate dates
        dates = pd.date_range(start=start_date, periods=num_days)
        
        # Generate price data with random walk and some trends
        prices = [start_price]
        for i in range(1, num_days):
            # Add some cyclical component
            cycle = 5 * np.sin(i / 30)
            
            # Random daily return between -2% and +2% plus cycle
            daily_return = np.random.normal(0.0005, 0.015) + cycle * 0.001
            
            # Calculate new price
            new_price = prices[-1] * (1 + daily_return)
            prices.append(new_price)
        
        # Create price data
        price_data = pd.DataFrame({
            'Date': dates,
            'Open': prices,
            'Close': prices,
        })
        
        # Add some variation to High and Low
        price_data['High'] = price_data['Close'] * (1 + abs(np.random.normal(0, 0.01, num_days)))
        price_data['Low'] = price_data['Close'] * (1 - abs(np.random.normal(0, 0.01, num_days)))
        
        # Generate volume data (in millions)
        price_data['Volume'] = np.random.normal(1000000, 500000, num_days)
        
        # Generate sentiment data
        sentiments = []
        for i in range(num_days):
            # Base sentiment on recent price trend with some noise
            if i < 5:
                sentiment = np.random.normal(0, 0.3)  # Random sentiment for first few days
            else:
                # Calculate 5-day price change
                price_change = (price_data['Close'].iloc[i-1] - price_data['Close'].iloc[i-5]) / price_data['Close'].iloc[i-5]
                # Sentiment correlates with recent price trend but with noise
                sentiment = price_change * 5 + np.random.normal(0, 0.3)
            
            # Clip sentiment to [-1, 1]
            sentiment = np.clip(sentiment, -1, 1)
            sentiments.append(sentiment)
            
        price_data['Sentiment'] = sentiments
        
        # Set index
        price_data.set_index('Date', inplace=True)
        
        data_dict[symbol] = price_data
    
    return data_dict
