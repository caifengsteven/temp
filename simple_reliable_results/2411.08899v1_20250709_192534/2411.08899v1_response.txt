class RewardAgent(Agent):
    """Agent that evaluates trading performance"""
    
    def __init__(self, initial_capital: float = 10000.0):
        super().__init__("Reward Agent")
        self.initial_capital = initial_capital
        self.current_capital = initial_capital
        self.positions = {}  # ticker -> {'shares': n, 'entry_price': p}
        self.portfolio_history = []
        self.trade_history = []
    
    def process(self, prediction: Dict, next_day_price: float) -> Dict:
        """Execute trade based on prediction and calculate performance"""
        ticker = prediction['ticker']
        action = prediction['action']
        position_size = prediction['position_size']
        date = prediction['date']
        
        # Calculate position value
        position_value = 0
        if ticker in self.positions:
            position_value = self.positions[ticker]['shares'] * next_day_price
        
        # Calculate capital available for trading
        available_capital = self.current_capital - position_value
        
        # Calculate trade size
        trade_amount = 0
        if action != "HOLD":
            trade_amount = (available_capital * position_size / 10)
        
        # Execute trade
        pnl = 0
        if action == "BUY":
            shares_to_buy = trade_amount / next_day_price
            if ticker not in self.positions:
                self.positions[ticker] = {'shares': shares_to_buy, 'entry_price': next_day_price}
            else:
                # Average down/up
                total_shares = self.positions[ticker]['shares'] + shares_to_buy
                avg_price = (self.positions[ticker]['shares'] * self.positions[ticker]['entry_price'] + 
                             shares_to_buy * next_day_price) / total_shares
                self.positions[ticker] = {'shares': total_shares, 'entry_price': avg_price}
            self.current_capital -= trade_amount
        
        elif action == "SELL" and ticker in self.positions:
            shares_to_sell = min(self.positions[ticker]['shares'], trade_amount / next_day_price)
            if shares_to_sell > 0:
                sell_value = shares_to_sell * next_day_price
                cost_basis = shares_to_sell * self.positions[ticker]['entry_price']
                pnl = sell_value - cost_basis
                
                # Update position
                self.positions[ticker]['shares'] -= shares_to_sell
                if self.positions[ticker]['shares'] <= 0:
                    del self.positions[ticker]
                
                self.current_capital += sell_value
        
        # Calculate portfolio value
        portfolio_value = self.current_capital
        for pos_ticker, pos_data in self.positions.items():
            if pos_ticker == ticker:
                portfolio_value += pos_data['shares'] * next_day_price
            else:
                # Use last known price for other positions (in a real system, we'd use current prices)
                portfolio_value += pos_data['shares'] * pos_data['entry_price']
        
        # Record trade
        trade_record = {
            'ticker': ticker,
            'date': date,
            'action': action,
            'position_size': position_size,
            'execution_price': next_day_price,
            'shares': trade_amount / next_day_price if action != "HOLD" else 0,
            'pnl': pnl,
            'portfolio_value': portfolio_value
        }
        
        self.trade_history.append(trade_record)
        
        # Record portfolio history
        self.portfolio_history.append({
            'date': date,
            'portfolio_value': portfolio_value
        })
        
        return trade_record
    
    def get_portfolio_status(self) -> Dict:
        """Get current portfolio status"""
        return {
            'capital': self.current_capital,
            'positions': self.positions,
            'portfolio_value': sum([pos['shares'] * pos['entry_price'] for pos in self.positions.values()]) + self.current_capital,
            'return_pct': (self.current_capital / self.initial_capital - 1) * 100
        }
    
    def calculate_metrics(self) -> Dict:
        """Calculate performance metrics"""
        if not self.portfolio_history:
            return {
                'ARR': 0,
                'Sharpe': 0,
                'MDD': 0
            }
        
        # Calculate annual return rate
        initial_value = self.initial_capital
        final_value = self.portfolio_history[-1]['portfolio_value']
        days = len(self.portfolio_history)
        
        annual_return = ((final_value / initial_value) ** (365 / days) - 1) * 100
        
        # Calculate daily returns
        portfolio_values = [entry['portfolio_value'] for entry in self.portfolio_history]
        daily_returns = [(portfolio_values[i] / portfolio_values[i-1] - 1) 
                         for i in range(1, len(portfolio_values))]
        
        # Calculate Sharpe ratio (assuming risk-free rate of 0)
        if not daily_returns:
            sharpe = 0
        else:
            mean_return = np.mean(daily_returns)
            std_return = np.std(daily_returns)
            sharpe = (mean_return / std_return) * np.sqrt(252) if std_return > 0 else 0
        
        # Calculate maximum drawdown
        peak = portfolio_values[0]
        mdd = 0
        
        for value in portfolio_values:
            if value > peak:
                peak = value
            dd = (peak - value) / peak
            if dd > mdd:
                mdd = dd
        
        return {
            'ARR': annual_return,
            'Sharpe': sharpe,
            'MDD': mdd * 100
        }


class FinVisionFramework:
    """Main framework class that orchestrates all agents"""

        self.technical_agent = TechnicalAnalystAgent()
        self.reflection_agent = ReflectionAgent()
        self.prediction_agent = PredictionAgent()
        self.reward_agent = RewardAgent(initial_capital)
    
    def run_simulation(
        self, 
        ticker: str, 
        stock_data: StockData, 
        training_end_date: str, 
        testing_end_date: str
    ) -> Dict:
        """Run the trading simulation"""
        # Split data into training and testing periods
        training_end = datetime.strptime(training_end_date, '%Y-%m-%d')
        testing_end = datetime.strptime(testing_end_date, '%Y-%m-%d')
        
        print(f"Starting simulation for {ticker}...")
        print(f"Training period: {stock_data.start_date.strftime('%Y-%m-%d')} to {training_end_date}")
        print(f"Testing period: {(training_end + timedelta(days=1)).strftime('%Y-%m-%d')} to {testing_end_date}")
        
        # Get dataframes for easier handling
        price_df = stock_data.stock_data
        news_data = stock_data.news_data
        
        # First run the training period to build reflection data
        print("Running training period...")
        current_date = stock_data.trading_days[0]
        while current_date <= training_end:
            if current_date in stock_data.trading_days:
                self._process_trading_day(ticker, current_date, price_df, news_data)
            current_date += timedelta(days=1)
        
        # Reset reward agent for testing period
        self.reward_agent = RewardAgent(initial_capital=10000.0)
        
        # Run the testing period
        print("Running testing period...")
        current_date = training_end + timedelta(days=1)
        while current_date <= testing_end:
            if current_date in stock_data.trading_days:
                self._process_trading_day(ticker, current_date, price_df, news_data)
            current_date += timedelta(days=1)
        
        # Calculate final metrics
        metrics = self.reward_agent.calculate_metrics()
        print(f"Simulation complete. Final metrics: {metrics}")
        
        return {
            'ticker': ticker,
            'metrics': metrics,
            'trade_history': self.reward_agent.trade_history,
            'portfolio_history': self.reward_agent.portfolio_history
        }
    
    def _process_trading_day(
        self, 
        ticker: str, 
        date: datetime, 
        price_df: pd.DataFrame, 
        news_data: List[Dict]
    ):
        """Process a single trading day"""
        # Get yesterday's date (for analysis)
        yesterday = date - timedelta(days=1)
        yesterday_data = price_df[price_df['date'] <= yesterday]
        
        # Make sure we have enough data
        if len(yesterday_data) < 2:
            return
        
        # Get news summary
        summary = self.summary_agent.process(ticker, yesterday, news_data)
        
        # Get technical analysis
        technical = self.technical_agent.process(ticker, yesterday, price_df)
        
        # Get reflection
        reflection_short, reflection_visual = self.reflection_agent.process(ticker, yesterday)
        
        # Get portfolio status
        portfolio = self.reward_agent.get_portfolio_status()
        
        # Generate prediction
        prediction = self.prediction_agent.process(
            ticker, 
            date, 
            summary, 
            technical, 
            reflection_short, 
            reflection_visual, 
            portfolio
        )
        
        # Get next day price for reward calculation
        today_data = price_df[price_df['date'] == date]
        if today_data.empty:
            return
        
        next_day_price = today_data.iloc[0]['open']
        
        # Execute trade and calculate reward
        trade_result = self.reward_agent.process(prediction, next_day_price)
        
        # Update reflection data
        self.reflection_agent.add_trade(trade_result)

    final_value = shares * end_price
    
    annual_return = ((final_value / initial_capital) ** (365 / (end - start).days) - 1) * 100
    
    # Calculate daily returns for Sharpe ratio
    daily_data = price_df[(price_df['date'] >= start) & (price_df['date'] <= end)]
    portfolio_values = [row['close'] * shares for _, row in daily_data.iterrows()]
    daily_returns = [(portfolio_values[i] / portfolio_values[i-1] - 1) for i in range(1, len(portfolio_values))]
    
    mean_return = np.mean(daily_returns)
    std_return = np.std(daily_returns)
    sharpe = (mean_return / std_return) * np.sqrt(252) if std_return > 0 else 0
    
    # Calculate MDD
    peak = portfolio_values[0]
    mdd = 0
    
    for value in portfolio_values:
        if value > peak:
            peak = value
        dd = (peak - value) / peak
        if dd > mdd:
            mdd = dd
    
    return {
        'ticker': ticker,
        'strategy': 'Buy and Hold',
        'metrics': {
            'ARR': annual_return,
            'Sharpe': sharpe,
            'MDD': mdd * 100
        },
        'start_price': start_price,
        'end_price': end_price,
        'shares': shares,
        'initial_value': initial_capital,
        'final_value': final_value
    }
