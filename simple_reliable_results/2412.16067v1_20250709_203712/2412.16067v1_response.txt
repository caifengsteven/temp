def fractional_adams_method_mod3(xi, T, M, model_params, n_iterations=2):
    """
    Modification III of the fractional Adams method for solving the fractional Riccati equation.
    
    Args:
        xi: Complex spectral parameter
        T: Maturity
        M: Number of time steps
        model_params: Rough Heston model parameters
        n_iterations: Number of iterations at each time step
        
    Returns:
        Logarithm of the characteristic function
    """
    alpha = model_params.alpha
    gamma = model_params.gamma
    rho = model_params.rho
    theta = model_params.theta
    nu = model_params.nu
    v0 = model_params.v0
    
    # Time grid
    dt = T / M
    t = np.linspace(0, T, M+1)
    
    # Initialize h
    h_tilde = np.zeros(M+1, dtype=complex)
    h_tilde_as = np.zeros(M+1, dtype=complex)
    h_tilde1 = np.zeros(M+1, dtype=complex)
    
    # Factor for scaling
    scale_factor = 1 + abs(xi)
    
    # Asymptotic approximation
    for k in range(1, M+1):
        h_tilde_as[k] = -0.5 * (xi**2 + 1j*xi) / (gamma(alpha+1) * scale_factor) * t[k]**alpha
    
    # Coefficients for the Adams method
    a = np.zeros((M+1, M+1), dtype=float)
    for k in range(M):
        # Calculate a_{k+1,k+1}
        a[k+1, k+1] = dt**alpha / gamma(alpha+2)
        
        # Calculate a_{0,k+1}
        a[0, k+1] = dt**alpha / gamma(alpha+2) * ((k+1)**alpha - (k+1-alpha)*(k+1)**alpha)
        
        # Calculate a_{j,k+1} for j=1,...,k
        for j in range(1, k+1):
            a[j, k+1] = dt**alpha / gamma(alpha+2) * ((k-j+2)**(alpha+1) + (k-j)**(alpha+1) - 2*(k-j+1)**(alpha+1))
    
    # Define modified F function
    def F_tilde_as1(xi, h_as, h1):
        return gamma * (1j*xi*rho*nu - 1) * (h_as + h1) + (1 + abs(xi)) * (gamma*nu)**2 / 2 * (h_as + h1)**2
    
    # Solve the equation
    for k in range(M):
        # Predictor step
        h_tilde1[k+1] = 0
        for j in range(k+1):
            h_tilde1[k+1] += a[j, k+1] * F_tilde_as1(xi, h_tilde_as[j], h_tilde1[j])
        
        # Corrector steps
        for _ in range(n_iterations):
            h_tilde1[k+1] = 0
            for j in range(k):
                h_tilde1[k+1] += a[j, k+1] * F_tilde_as1(xi, h_tilde_as[j], h_tilde1[j])
            h_tilde1[k+1] += a[k+1, k+1] * F_tilde_as1(xi, h_tilde_as[k], h_tilde1[k])
        
        # Update h
        h_tilde[k+1] = h_tilde1[k+1] + h_tilde_as[k+1]
    
    # Calculate the integral in formula (2.8)
    g1 = 0
    g2 = 0
    for k in range(M):
        # Trapezoidal rule
        dt_k = t[k+1] - t[k]
        F_k = gamma * (1j*xi*rho*nu - 1) * h_tilde[k] + (gamma*nu)**2 / 2 * h_tilde[k]**2
        F_k1 = gamma * (1j*xi*rho*nu - 1) * h_tilde[k+1] + (gamma*nu)**2 / 2 * h_tilde[k+1]**2
        
        g1 += gamma * theta * (h_tilde[k] + h_tilde[k+1]) * dt_k / 2
        g2 += v0 * (F_k + F_k1) * dt_k / 2
    
    # Return the logarithm of the characteristic function
    return g1 + g2

def sinh_acceleration(K, S0, T, r, model_params, omega1, b, omega, zeta, N, M, option_type='call'):
    """
    SINH-acceleration method for option pricing in the rough Heston model.
    
    Args:
        K: Strike price
        S0: Spot price
        T: Maturity
        r: Risk-free rate
        model_params: Rough Heston model parameters
        omega1, b, omega: Parameters of the sinh-deformation
        zeta: Step size for the trapezoid rule
        N: Number of terms in the trapezoid rule
        M: Number of time steps for the Adams method
        option_type: 'call' or 'put'
        
    Returns:
        Option price
    """
    # Adjust parameters based on option type
    if option_type == 'put' and K <= S0:
        omega = abs(omega)
    elif option_type == 'call' and K >= S0:
        omega = -abs(omega)
    
    price = 0
    for j in range(N+1):
        y = j * zeta
        
        # Conformal mapping
        xi = 1j * omega1 + b * np.sinh(1j * omega + y)
        
        # Calculate the characteristic function
        phi = np.exp(fractional_adams_method_mod3(xi, T, M, model_params))
        
        # Calculate the integrand
        if option_type == 'call':
            integrand = np.exp(1j * xi * np.log(S0/K)) * phi / (xi * (xi + 1j))
        else:  # put
            integrand = np.exp(1j * xi * np.log(S0/K)) * phi / (xi * (xi - 1j))
        
        # Jacobian of the transformation
        jacobian = b * np.cosh(1j * omega + y)
        
        # Add to the sum with weights
        weight = 1.0 if j > 0 else 0.5
        price += weight * jacobian * integrand
    
    # Multiply by the step size and other factors
    if option_type == 'call':
        price = -K * np.exp(-r * T) * zeta * np.real(price) / np.pi
    else:  # put
        price = S0 * zeta * np.real(price) / np.pi
    
    return price

def black_scholes_implied_vol(price, S, K, T, r, option_type='call'):
    """
    Calculate implied volatility using the Black-Scholes formula.
    """
    def bs_price(sigma):
        d1 = (np.log(S/K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        if option_type == 'call':
            return S * norm_cdf(d1) - K * np.exp(-r * T) * norm_cdf(d2)
        else:  # put
            return K * np.exp(-r * T) * norm_cdf(-d2) - S * norm_cdf(-d1)
    
    def norm_cdf(x):
        return 0.5 * (1 + np.erf(x / np.sqrt(2)))
    
    # Objective function for finding implied volatility
    def objective(sigma):
        return bs_price(sigma) - price
    
    # Initial guess
    sigma_init = 0.2
    
    # Solve for implied volatility
    try:
        implied_vol = fsolve(objective, sigma_init)[0]
        return implied_vol
    except:
        return np.nan
