class LOBMessage:
    """Represents a LOBSTER format message"""
    
    def __init__(self, time, type_id, order_id, size, price, direction):
        self.time = time              # Time in seconds since market open
        self.type_id = type_id        # 1: submission, 2: cancellation, 3: execution
        self.order_id = order_id      # Unique identifier for the order

class OrderBook:
    """Simple order book implementation"""
    
    def __init__(self, tick_size=0.01):
        self.bids = {}  # price -> size
        self.asks = {}  # price -> size
        self.tick_size = tick_size
        self.orders = {}  # order_id -> (price, size, direction)
    
    def process_message(self, message):
        if message.type_id == 1:  # Submission
            if message.direction == 1:  # Buy
                self.bids[message.price] = self.bids.get(message.price, 0) + message.size
            else:  # Sell
                self.asks[message.price] = self.asks.get(message.price, 0) + message.size
            self.orders[message.order_id] = (message.price, message.size, message.direction)
        
        elif message.type_id == 2:  # Cancellation
            if message.order_id in self.orders:
                price, size, direction = self.orders[message.order_id]
                if direction == 1:  # Buy
                    self.bids[price] = max(0, self.bids.get(price, 0) - message.size)
                else:  # Sell
                    self.asks[price] = max(0, self.asks.get(price, 0) - message.size)
                del self.orders[message.order_id]
        
        elif message.type_id == 3:  # Execution
            if message.order_id in self.orders:
                price, size, direction = self.orders[message.order_id]
                if direction == 1:  # Buy
                    self.bids[price] = max(0, self.bids.get(price, 0) - message.size)
                else:  # Sell
                    self.asks[price] = max(0, self.asks.get(price, 0) - message.size)
                # Update order size or remove if fully executed
                remaining_size = size - message.size
                if remaining_size > 0:
                    self.orders[message.order_id] = (price, remaining_size, direction)
                else:
                    del self.orders[message.order_id]
    
    def get_book_state(self, levels=10):
        """Get current book state with top 'levels' price levels"""
        bid_prices = sorted([p for p, s in self.bids.items() if s > 0], reverse=True)
        ask_prices = sorted([p for p, s in self.asks.items() if s > 0])
        
        bid_levels = [(p, self.bids[p]) for p in bid_prices[:levels]] + [(0, 0)] * (levels - len(bid_prices))
        ask_levels = [(p, self.asks[p]) for p in ask_prices[:levels]] + [(0, 0)] * (levels - len(ask_prices))
        
        return bid_levels, ask_levels
    
    def get_mid_price(self):
        """Get current mid price"""
        bid_prices = [p for p, s in self.bids.items() if s > 0]
        ask_prices = [p for p, s in self.asks.items() if s > 0]
        
        if not bid_prices or not ask_prices:
            return None
        
        best_bid = max(bid_prices)
        best_ask = min(ask_prices)
        
        return (best_bid + best_ask) / 2
    
    def get_spread(self):
        """Get current spread"""
        bid_prices = [p for p, s in self.bids.items() if s > 0]
        ask_prices = [p for p, s in self.asks.items() if s > 0]
        
        if not bid_prices or not ask_prices:
            return None
        
        best_bid = max(bid_prices)
        best_ask = min(ask_prices)
        
        return best_ask - best_bid


class LOBScorer:
    """Implements scoring functions for LOB data as described in LOB-Bench"""
    
    @staticmethod
    def compute_score_distribution(data, score_fn):
        """Compute distribution of a scoring function"""
        scores = [score_fn(item) for item in data]
        return np.array(scores)
    
    @staticmethod
    def compute_l1_distance(real_dist, gen_dist, bins=50):
        """Compute L1 distance between two distributions"""
        # Use Freedman-Diaconis rule for bin width
        combined = np.concatenate([real_dist, gen_dist])
        iqr = np.percentile(combined, 75) - np.percentile(combined, 25)
        bin_width = 2 * iqr / (len(combined) ** (1/3))
        
        if bin_width == 0:  # Handle the case when IQR is 0
            bin_width = 0.1
            
        hist_real, bin_edges = np.histogram(real_dist, bins=bins, density=True)
        hist_gen, _ = np.histogram(gen_dist, bins=bin_edges, density=True)
        
        return 0.5 * np.sum(np.abs(hist_real - hist_gen)) * bin_width
    
    @staticmethod
    def compute_wasserstein_distance(real_dist, gen_dist):
        """Compute Wasserstein-1 distance between two distributions"""
        # Normalize distributions
        real_norm = (real_dist - np.mean(real_dist)) / np.std(real_dist)
        gen_norm = (gen_dist - np.mean(gen_dist)) / np.std(gen_dist)
        
        return wasserstein_distance(real_norm, gen_norm)
    
    @staticmethod
    def compute_conditional_score(real_data, gen_data, score_fn, cond_fn, num_bins=10):
        """Compute score distribution conditional on another function"""
        real_cond_scores = np.array([cond_fn(item) for item in real_data])
        gen_cond_scores = np.array([cond_fn(item) for item in gen_data])
        
        # Combine data for binning
        combined_cond = np.concatenate([real_cond_scores, gen_cond_scores])
        bin_edges = np.percentile(combined_cond, np.linspace(0, 100, num_bins + 1))
        
        # Ensure unique bin edges
        bin_edges = np.unique(bin_edges)
        if len(bin_edges) < 3:  # Need at least 2 bins
            return 1.0  # Maximum distance
        
        total_distance = 0
        total_weight = 0
        
        for i in range(len(bin_edges) - 1):
            # Filter data in current bin
            real_bin_mask = (real_cond_scores >= bin_edges[i]) & (real_cond_scores < bin_edges[i+1])
            gen_bin_mask = (gen_cond_scores >= bin_edges[i]) & (gen_cond_scores < bin_edges[i+1])
            
            real_bin_data = [real_data[j] for j in range(len(real_data)) if real_bin_mask[j]]
            gen_bin_data = [gen_data[j] for j in range(len(gen_data)) if gen_bin_mask[j]]
            
            if len(real_bin_data) > 0 and len(gen_bin_data) > 0:
                # Compute distributions of score function within this bin
                real_bin_dist = LOBScorer.compute_score_distribution(real_bin_data, score_fn)
                gen_bin_dist = LOBScorer.compute_score_distribution(gen_bin_data, score_fn)
                
                # Compute distance between distributions
                bin_distance = LOBScorer.compute_l1_distance(real_bin_dist, gen_bin_dist)
                
                # Weight by bin probability
                bin_weight = (len(real_bin_data) / len(real_data) + len(gen_bin_data) / len(gen_data)) / 2
                total_distance += bin_distance * bin_weight
                total_weight += bin_weight
        
        if total_weight > 0:
            return total_distance / total_weight
        else:
            return 1.0  # Maximum distance

def compute_order_book_stats(messages):
    """Compute various order book statistics from a sequence of messages"""
    book = OrderBook()
    
    stats = {
        'spreads': [],
        'mid_prices': [],
        'bid_volumes': [],
        'ask_volumes': [],
        'order_imbalances': [],
        'message_inter_arrival_times': [],
        'time_to_cancel': defaultdict(list),  # order_id -> list of times to cancel
        'limit_depths': [],
        'cancel_depths': [],
        'limit_levels': [],
        'cancel_levels': [],
        'volume_per_min': defaultdict(int),
        'order_flow_imbalance': []
    }
    
    order_submission_times = {}  # order_id -> submission time
    prev_time = None
    
    # Pre-process for OFI calculation
    ofi_window_size = 50  # messages
    ofi_window = []
    
    for msg in messages:
        # Process message and update book
        book.process_message(msg)
        
        # Calculate inter-arrival time
        if prev_time is not None:
            stats['message_inter_arrival_times'].append(msg.time - prev_time)
        prev_time = msg.time
        
        # Get current book state
        mid_price = book.get_mid_price()
        spread = book.get_spread()
        
        if mid_price is not None and spread is not None:
            stats['spreads'].append(spread)
            stats['mid_prices'].append(mid_price)
            
            # Calculate book volumes
            bid_volume = sum(book.bids.values())
            ask_volume = sum(book.asks.values())
            stats['bid_volumes'].append(bid_volume)
            stats['ask_volumes'].append(ask_volume)
            
            # Calculate order imbalance
            if bid_volume + ask_volume > 0:
                imbalance = (bid_volume - ask_volume) / (bid_volume + ask_volume)
                stats['order_imbalances'].append(imbalance)
        
        # Track submission times and cancellations
        if msg.type_id == 1:  # Submission
            order_submission_times[msg.order_id] = msg.time
            
            # Calculate limit order depth
            if mid_price is not None:
                depth = abs(msg.price - mid_price)
                stats['limit_depths'].append(depth)
                stats['limit_levels'].append(int(depth / book.tick_size))
                
        elif msg.type_id == 2:  # Cancellation
            if msg.order_id in order_submission_times:
                time_to_cancel = msg.time - order_submission_times[msg.order_id]
                stats['time_to_cancel'][msg.order_id].append(time_to_cancel)
                
                # Calculate cancellation depth
                if mid_price is not None:
                    depth = abs(msg.price - mid_price)
                    stats['cancel_depths'].append(depth)
                    stats['cancel_levels'].append(int(depth / book.tick_size))
        
        # Calculate volume per minute
        minute_bin = int(msg.time / 60)
        if msg.type_id == 3:  # Execution
            stats['volume_per_min'][minute_bin] += msg.size
            
        # Update OFI window
        if msg.type_id in [1, 2, 3]:
            ofi_event = {
                'type': msg.type_id,
                'direction': msg.direction,
                'size': msg.size,
                'price': msg.price,
                'mid_price': mid_price
            }
            ofi_window.append(ofi_event)
            if len(ofi_window) > ofi_window_size:
                ofi_window.pop(0)
                
            # Calculate OFI if window is full
            if len(ofi_window) == ofi_window_size:
                ofi = 0
                for event in ofi_window:
                    if event['type'] == 1:  # Limit order
                        ofi += event['direction'] * event['size']
                    elif event['type'] == 2:  # Cancel
                        ofi -= event['direction'] * event['size']
                    elif event['type'] == 3:  # Execution
                        ofi -= event['direction'] * event['size']
                
                stats['order_flow_imbalance'].append(ofi)
    
    # Convert time_to_cancel to flat list
    time_to_cancel_flat = []
    for times in stats['time_to_cancel'].values():
        time_to_cancel_flat.extend(times)
    stats['time_to_cancel'] = time_to_cancel_flat
    
    # Convert volume_per_min to list
    volume_per_min_list = []
    for minute, volume in sorted(stats['volume_per_min'].items()):
        volume_per_min_list.append(volume)
    stats['volume_per_min'] = volume_per_min_list
    
    return stats


def compute_impact_response(messages, lag_times=None):
    """Compute impact response functions for different event types"""
    if lag_times is None:
        # Logarithmically spaced lag times between 1 and 200
        lag_times = np.unique(np.logspace(0, 2.3, 20).astype(int))
    
    book = OrderBook()
    event_types = ['MO0', 'MO1', 'LO0', 'LO1', 'CA0', 'CA1']
    
    # Initialize response functions
    response = {et: {lag: [] for lag in lag_times} for et in event_types}
    
    # Process messages
    processed_msgs = []
    for msg in messages:
        # Apply message to book
        book.process_message(msg)
        
        # Store message with current mid price
        mid_price = book.get_mid_price() or 100.0
        processed_msgs.append((msg, mid_price))
    
    # Calculate response functions
    for i, (msg, mid_price) in enumerate(processed_msgs):
        # Classify event
        event_type = None
        if msg.type_id == 3:  # Execution (treat as market order)
            is_price_changing = False
            if i > 0:
                prev_mid = processed_msgs[i-1][1]
                is_price_changing = abs(mid_price - prev_mid) > 1e-9
            event_type = 'MO1' if is_price_changing else 'MO0'
        elif msg.type_id == 1:  # Limit order
            is_price_changing = False
            if i > 0:
                prev_mid = processed_msgs[i-1][1]
                is_price_changing = abs(mid_price - prev_mid) > 1e-9
            event_type = 'LO1' if is_price_changing else 'LO0'
        elif msg.type_id == 2:  # Cancellation
            is_price_changing = False
            if i > 0:
                prev_mid = processed_msgs[i-1][1]
                is_price_changing = abs(mid_price - prev_mid) > 1e-9
            event_type = 'CA1' if is_price_changing else 'CA0'
        
        if event_type:
            # Define epsilon based on direction and event type
            epsilon = msg.direction
            if event_type.startswith('CA'):
                epsilon = -epsilon
            
            # Calculate price changes for each lag
            for lag in lag_times:
                if i + lag < len(processed_msgs):
                    future_mid = processed_msgs[i + lag][1]
                    price_change = (future_mid - mid_price) * epsilon
                    response[event_type][lag].append(price_change)
    
    # Average responses
    avg_response = {et: {lag: np.mean(vals) if vals else 0 
                         for lag, vals in lags.items()} 
                    for et, lags in response.items()}
    
    return avg_response


def compare_impact_curves(real_response, gen_response):
    """Compare impact response curves between real and generated data"""
    event_types = ['MO0', 'MO1', 'LO0', 'LO1', 'CA0', 'CA1']
    
    # Calculate L1 distances
    l1_distances = {}
    for et in event_types:
        if et in real_response and et in gen_response:
            # Get common lags
            common_lags = set(real_response[et].keys()) & set(gen_response[et].keys())
            if common_lags:
                l1_dist = np.mean([abs(real_response[et][lag] - gen_response[et][lag]) 
                                   for lag in common_lags])
                l1_distances[et] = l1_dist
    
    # Calculate aggregate distance
    if l1_distances:
        aggregate_distance = np.mean(list(l1_distances.values()))
    else:
        aggregate_distance = float('inf')
        
    return aggregate_distance, l1_distances
