class SchwarzSmithModel:
    """Implementation of the Schwartz-Smith two-factor model"""
    
    def __init__(self, kappa_chi=1.5, kappa_xi=0.3, mu_xi=0.1, sigma_chi=0.3, 
                 sigma_xi=0.2, rho=-0.3, lambda_chi=0.1, lambda_xi=0.2):
        # Model parameters
        self.kappa_chi = kappa_chi  # Mean-reversion speed for short-term factor
        self.kappa_xi = kappa_xi    # Mean-reversion speed for long-term factor
        self.mu_xi = mu_xi          # Mean level of long-term factor
        self.sigma_chi = sigma_chi  # Volatility of short-term factor
        self.sigma_xi = sigma_xi    # Volatility of long-term factor
        self.rho = rho              # Correlation between factors
        self.lambda_chi = lambda_chi  # Risk premium for short-term factor
        self.lambda_xi = lambda_xi    # Risk premium for long-term factor
        
    def A_function(self, t):
        """Calculate A(t) function from equation (6) in the paper"""
        term1 = -self.lambda_chi / self.kappa_chi * (1 - np.exp(-self.kappa_chi * t))
        term2 = (self.mu_xi - self.lambda_xi) / self.kappa_xi * (1 - np.exp(-self.kappa_xi * t))
        
        var_term1 = (1 - np.exp(-2 * self.kappa_chi * t)) / (2 * self.kappa_chi) * self.sigma_chi**2
        var_term2 = (1 - np.exp(-2 * self.kappa_xi * t)) / (2 * self.kappa_xi) * self.sigma_xi**2
        cov_term = 2 * (1 - np.exp(-(self.kappa_chi + self.kappa_xi) * t)) / (self.kappa_chi + self.kappa_xi) * self.sigma_chi * self.sigma_xi * self.rho
        
        return term1 + term2 + 0.5 * (var_term1 + var_term2 + cov_term)
    
    def simulate_factors(self, T, dt=1/12, random_seed=None):
        """Simulate the short-term and long-term factors over time"""
        if random_seed is not None:
            np.random.seed(random_seed)
            
        steps = int(T / dt)
        times = np.arange(0, T, dt)
        
        # Initialize factors
        chi = np.zeros(steps)
        xi = np.zeros(steps)
        
        # Initial values
        chi[0] = 0
        xi[0] = 0
        
        # Generate correlated Brownian motions
        dW_chi = np.random.normal(0, np.sqrt(dt), steps-1)
        dW_xi = self.rho * dW_chi + np.sqrt(1 - self.rho**2) * np.random.normal(0, np.sqrt(dt), steps-1)
        
        # Simulate factors
        for t in range(1, steps):
            chi[t] = chi[t-1] * np.exp(-self.kappa_chi * dt) + self.sigma_chi * dW_chi[t-1]
            xi[t] = xi[t-1] * np.exp(-self.kappa_xi * dt) + (self.mu_xi / self.kappa_xi) * (1 - np.exp(-self.kappa_xi * dt)) + self.sigma_xi * dW_xi[t-1]
        
        return times, chi, xi
    
    def calculate_futures_price(self, t, T, chi, xi):
        """Calculate futures price at time t for maturity T given factors chi and xi"""
        time_to_maturity = T - t
        log_price = self.A_function(time_to_maturity) + np.exp(-self.kappa_chi * time_to_maturity) * chi + np.exp(-self.kappa_xi * time_to_maturity) * xi
        return np.exp(log_price)
    
    def generate_futures_curve(self, t, maturities, chi, xi, measurement_noise=0.01):
        """Generate futures curve at time t for all maturities given factors chi and xi"""
        prices = np.zeros(len(maturities))
        for i, T in enumerate(maturities):
            prices[i] = self.calculate_futures_price(t, T, chi, xi)
            
        # Add measurement noise
        if measurement_noise > 0:
            prices = prices * np.exp(np.random.normal(0, measurement_noise, len(maturities)))
            
        return prices


class YieldCurveGenerator:
    """Generate simulated yield curves"""
    
    def __init__(self, maturities, mean_level=0.02, slope=0.01, curvature=0.005, 
                 volatility=0.003, inversion_prob=0.2):
        self.maturities = maturities
        self.mean_level = mean_level
        self.slope = slope
        self.curvature = curvature
        self.volatility = volatility
        self.inversion_prob = inversion_prob
        
    def nelson_siegel(self, tau, beta0, beta1, beta2, lambda_param=0.5):
        """Nelson-Siegel yield curve model"""
        term1 = beta0
        term2 = beta1 * (1 - np.exp(-tau / lambda_param)) / (tau / lambda_param)
        term3 = beta2 * ((1 - np.exp(-tau / lambda_param)) / (tau / lambda_param) - np.exp(-tau / lambda_param))
        return term1 + term2 + term3
    
    def generate_curve(self, t, regime_change=None):
        """Generate a yield curve at time t"""
        # Determine if we're in normal or inverted regime
        if regime_change is not None and t >= regime_change:
            # Inverted yield curve (recession)
            beta0 = self.mean_level + np.random.normal(0, self.volatility)
            beta1 = -self.slope + np.random.normal(0, self.volatility/2)
            beta2 = self.curvature + np.random.normal(0, self.volatility/2)
        else:
            # Normal yield curve
            beta0 = self.mean_level + np.random.normal(0, self.volatility)
            beta1 = self.slope + np.random.normal(0, self.volatility/2)
            beta2 = self.curvature + np.random.normal(0, self.volatility/2)
            
        yields = np.array([self.nelson_siegel(tau, beta0, beta1, beta2) for tau in self.maturities])
        return yields
    
    def generate_time_series(self, T, dt=1/12, regime_change=None, random_seed=None):
        """Generate a time series of yield curves"""
        if random_seed is not None:
            np.random.seed(random_seed)
            
        steps = int(T / dt)
        times = np.arange(0, T, dt)
        
        # Initialize yield curves
        yield_curves = np.zeros((steps, len(self.maturities)))
        
        # Generate yield curves over time
        for t in range(steps):
            yield_curves[t] = self.generate_curve(t * dt, regime_change)
            
        return times, yield_curves


class FunctionalRegressionModel:
    """Implementation of the two-factor functional regression model"""
    
    def __init__(self, ss_model, num_factors=2):
        self.ss_model = ss_model  # Base Schwartz-Smith model
        self.num_factors = num_factors  # Number of factors for functional regression
        self.gamma = None  # Will store the functional regression coefficients
        
    def extract_factors(self, yield_curves, n_components=2):
        """Extract factors from yield curves using kernel PCA approximation"""
        # Use RBFSampler as a simple approximation of kernel PCA
        rbf_feature = RBFSampler(gamma=1.0, n_components=n_components, random_state=1)
        yield_factors = rbf_feature.fit_transform(yield_curves)
        return yield_factors
    
    def initialize_gamma(self, num_contracts, num_factors):
        """Initialize functional regression coefficients"""
        # Initialize gamma with small random values
        self.gamma = np.random.normal(0, 0.01, (num_contracts, num_factors))
        
    def calculate_futures_price(self, t, maturities, chi, xi, yield_factors):
        """Calculate futures price using the functional regression model"""
        if self.gamma is None:
            raise ValueError("Model not initialized. Call initialize_gamma first.")
            
        num_contracts = len(maturities)
        log_prices = np.zeros(num_contracts)
        
        for i, T in enumerate(maturities):
            # Calculate base price from SS model
            time_to_maturity = T - t
            ss_term = self.ss_model.A_function(time_to_maturity) + \
                      np.exp(-self.ss_model.kappa_chi * time_to_maturity) * chi + \
                      np.exp(-self.ss_model.kappa_xi * time_to_maturity) * xi
            
            # Add functional regression component
            functional_term = np.sum(self.gamma[i] * yield_factors)
            
            log_prices[i] = ss_term + functional_term
            
        return np.exp(log_prices)
