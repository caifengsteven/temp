def generate_synthetic_data() -> Tuple[np.ndarray, np.ndarray]:

    mu_1 = 0.0005  # Drift for asset 1
    sigma_1 = 0.01  # Volatility for asset 1
    mu_2 = 0.0002  # Drift for asset 2
    sigma_2 = 0.015  # Volatility for asset 2
    T = 390  # Time horizon (minutes)
    dt = 3  # Time step (minutes)
    N = int(T/dt)  # Number of time steps
    
    # Generate price paths
    t = np.linspace(0, T, N+1)
    W1 = np.random.standard_normal(size=N+1)
    W1[0] = 0
    W1 = np.cumsum(W1) * np.sqrt(dt/60)  # Convert to annual units
    
    W2 = np.random.standard_normal(size=N+1)
    W2[0] = 0
    W2 = np.cumsum(W2) * np.sqrt(dt/60)
    
    # Slight correlation between assets
    W2 = 0.7 * W1 + 0.3 * W2
    
    # Calculate price paths
    S1 = S0_1 * np.exp((mu_1 - 0.5 * sigma_1**2) * t/252 + sigma_1 * W1)
    S2 = S0_2 * np.exp((mu_2 - 0.5 * sigma_2**2) * t/252 + sigma_2 * W2)
    
    return S1, S2


def detect_delta_escapes(prices: np.ndarray, delta: float) -> List[int]:
    """Detect delta-escape times in price series (Model B in paper)"""
    escape_times = [0]  # Always include initial time
    
    for i in range(1, len(prices)):
        if i <= escape_times[-1]:
            continue
            
        # Calculate relative price change from last escape time
        last_escape = escape_times[-1]
        rel_change = abs(prices[i] - prices[last_escape]) / abs(prices[last_escape])
        
        if rel_change >= delta:
            escape_times.append(i)
    
    return escape_times


def build_NE_set(S1: np.ndarray, S2: np.ndarray, delta: float, delta_X1: float = 0.01, 
                 delta_X2: float = 0.01) -> List[Tuple]:
    """Build the empirical set NE from price data"""
    NE = []
    
    # Detect escape times using Model B from the paper
    escape_times = detect_delta_escapes(S1, delta)
    if len(escape_times) <= 1:
        # If no escapes detected, use a default move
        NE.append((0.01, 0.01, 1, 1, 1))
        NE.append((-0.01, -0.01, 1, 1, 1))
        NE.append((0.01, -0.01, 1, 1, 1))
        NE.append((-0.01, 0.01, 1, 1, 1))
        return NE
    
    # Calculate increments between escape times
    for i in range(len(escape_times) - 1):
        t0 = escape_times[i]
        t1 = escape_times[i+1]
        
        # Calculate discretized changes
        m1 = round((S1[t1] - S1[t0]) / delta_X1)
        m2 = round((S2[t1] - S2[t0]) / delta_X2)
        q = t1 - t0  # Time difference in steps
        
        # Calculate variation (simple version)
        gamma = abs(m1) + abs(m2)
        
        NE.append((m1 * delta_X1, m2 * delta_X2, 1, q, gamma))
    
    return NE
