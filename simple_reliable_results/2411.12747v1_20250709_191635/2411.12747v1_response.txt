import numpy as np

class MarketEnvironment:
    """Simulated market environment with multiple assets"""
    
    def __init__(self, num_assets: int = 10, time_steps: int = 500, 
                 volatility_base: float = 0.01, trend_strength: float = 0.0002, 
                 correlation_strength: float = 0.3, seed: int = 42):
        """
        Initialize market environment with simulated price data
        
        Args:
            num_assets: Number of assets to simulate
            time_steps: Number of time steps to simulate
            volatility_base: Base volatility level
            trend_strength: Strength of trend component
            correlation_strength: Strength of correlation between assets
            seed: Random seed for reproducibility
        """
        np.random.seed(seed)
        self.num_assets = num_assets
        self.time_steps = time_steps
        
        # Generate market factors (2 factors that influence asset prices)
        market_factor1 = np.cumsum(np.random.normal(0.0005, 0.01, time_steps))
        market_factor2 = np.cumsum(np.random.normal(0.0003, 0.008, time_steps))
        
        # Generate asset prices with correlation to market factors
        self.prices = np.zeros((time_steps, num_assets))
        self.asset_volatilities = np.random.uniform(0.5, 1.5, num_assets) * volatility_base
        self.asset_betas = np.random.uniform(-1, 1, (num_assets, 2))
        
        # Initial prices
        self.prices[0] = np.random.uniform(10, 100, num_assets)
        
        # Generate price series with trends, market correlation, and individual noise
        for t in range(1, time_steps):
            # Market component
            market_influence = (self.asset_betas[:, 0] * market_factor1[t] + 
                               self.asset_betas[:, 1] * market_factor2[t]) * correlation_strength
            
            # Trend component (different for each asset)
            trend = np.random.uniform(-1, 1, num_assets) * trend_strength
            
            # Individual noise component
            noise = np.random.normal(0, self.asset_volatilities, num_assets)
            
            # Combine components to get returns
            returns = market_influence + trend + noise
            
            # Update prices
            self.prices[t] = self.prices[t-1] * (1 + returns)
        
        # Create time index (daily)
        start_date = datetime(2023, 1, 1)
        self.dates = [start_date + timedelta(days=i) for i in range(time_steps)]
        
        # Create DataFrame for easier access
        self.price_df = pd.DataFrame(
            self.prices, 
            index=self.dates,
            columns=[f"Asset_{i}" for i in range(num_assets)]
        )
        
    def get_features(self) -> pd.DataFrame:
        """Extract technical features from price data"""
        features = pd.DataFrame(index=self.price_df.index)
        
        for asset in self.price_df.columns:
            # Returns (1-day)
            features[f"{asset}_return_1d"] = self.price_df[asset].pct_change(1)
            
            # Moving averages
            features[f"{asset}_ma_5d"] = self.price_df[asset].rolling(5).mean()
            features[f"{asset}_ma_20d"] = self.price_df[asset].rolling(20).mean()
            
            # Volatility (20-day rolling std)
            features[f"{asset}_vol_20d"] = self.price_df[asset].pct_change().rolling(20).std()
            
            # RSI (14-day)
            delta = self.price_df[asset].diff()
            gain = delta.clip(lower=0)
            loss = -delta.clip(upper=0)
            avg_gain = gain.rolling(14).mean()
            avg_loss = loss.rolling(14).mean()
            rs = avg_gain / avg_loss
            features[f"{asset}_rsi"] = 100 - (100 / (1 + rs))
        
        # Drop NaN values
        features = features.dropna()
        
        return features


class Agent:
    """Base agent class for portfolio optimization"""
    
    def __init__(self, name: str):
        self.name = name
    
    def act(self, state: Dict) -> np.ndarray:

class TrendObserver(Agent):
    """Agent that observes market trends"""
    
    def __init__(self, num_assets: int, lookback: int = 20):
        super().__init__("TrendObserver")
        self.num_assets = num_assets
        self.lookback = lookback
    
    def act(self, state: Dict) -> np.ndarray:
        """
        Identify trends based on moving averages
        
        Returns:
            Array with trend scores for each asset (positive = uptrend, negative = downtrend)
        """
        features = state["features"]
        assets = [f"Asset_{i}" for i in range(self.num_assets)]
        
        trend_scores = np.zeros(self.num_assets)
        
        for i, asset in enumerate(assets):
            # Short-term vs long-term MA
            short_ma = features[f"{asset}_ma_5d"].iloc[-1]
            long_ma = features[f"{asset}_ma_20d"].iloc[-1]
            
            # Momentum (recent returns)
            recent_returns = features[f"{asset}_return_1d"].iloc[-self.lookback:].mean()
            
            # RSI signal (>70 overbought, <30 oversold)
            rsi = features[f"{asset}_rsi"].iloc[-1]
            rsi_signal = 0
            if rsi > 70:
                rsi_signal = -1  # Overbought
            elif rsi < 30:
                rsi_signal = 1   # Oversold
            
            # Combine signals
            ma_signal = (short_ma / long_ma - 1) * 10  # Normalize
            
            trend_scores[i] = ma_signal + recent_returns * 5 + rsi_signal
        
        return trend_scores


class ReturnOptimizer(Agent):
    """Agent that optimizes for expected returns"""
    
    def __init__(self, num_assets: int):
        super().__init__("ReturnOptimizer")
        self.num_assets = num_assets
    
    def act(self, state: Dict) -> np.ndarray:
        """
        Generate portfolio weights based on expected returns
        
        Returns:
            Array with portfolio weights summing to 1
        """
        # Use trend scores as proxies for expected returns
        trend_scores = state["trend_scores"]
        
        # Convert to weights (only go long on positive trends)
        weights = np.maximum(trend_scores, 0)
        
        # Normalize to sum to 1 (if any positive trends)
        if np.sum(weights) > 0:
            weights = weights / np.sum(weights)
        else:
            # Equal weight if no positive trends
            weights = np.ones(self.num_assets) / self.num_assets
            
        return weights


class RiskManager(Agent):
    """Agent that manages risk in the portfolio"""
    
    def __init__(self, num_assets: int, max_weight: float = 0.3, min_assets: int = 3):
        super().__init__("RiskManager")
        self.num_assets = num_assets
        self.max_weight = max_weight
        self.min_assets = min_assets
    
    def act(self, state: Dict) -> np.ndarray:
        """
        Adjust weights to manage risk
        
        Returns:
            Modified portfolio weights
        """
        proposed_weights = state["proposed_weights"]
        features = state["features"]
        
        # Extract volatility data
        volatilities = np.zeros(self.num_assets)
        for i in range(self.num_assets):
            asset = f"Asset_{i}"
            volatilities[i] = features[f"{asset}_vol_20d"].iloc[-1]
        
        # Penalize high volatility assets
        risk_adjusted_weights = proposed_weights / (volatilities + 0.001)  # Avoid division by zero
        
        # Normalize
        if np.sum(risk_adjusted_weights) > 0:
            risk_adjusted_weights = risk_adjusted_weights / np.sum(risk_adjusted_weights)
        
        # Cap maximum weight
        risk_adjusted_weights = np.minimum(risk_adjusted_weights, self.max_weight)
        
        # Ensure minimum diversification
        nonzero_weights = np.sum(risk_adjusted_weights > 0.01)
        if nonzero_weights < self.min_assets:
            # Add small weights to more assets
            sorted_indices = np.argsort(volatilities)
            for idx in sorted_indices[:self.min_assets - nonzero_weights]:
                if risk_adjusted_weights[idx] < 0.01:
                    risk_adjusted_weights[idx] = 0.01
        
        # Renormalize
        risk_adjusted_weights = risk_adjusted_weights / np.sum(risk_adjusted_weights)
        
        return risk_adjusted_weights


class MultiAgentPortfolioSystem:
    """Multi-agent system for portfolio optimization"""
    
    def __init__(self, num_assets: int):
        self.num_assets = num_assets
        
        # Initialize agents
        self.trend_observer = TrendObserver(num_assets)
        self.return_optimizer = ReturnOptimizer(num_assets)
        self.risk_manager = RiskManager(num_assets)
        
        # Transaction cost parameter
        self.transaction_cost = 0.001
        
        # Current portfolio state
        self.current_weights = np.ones(num_assets) / num_assets
    
    def update(self, features: pd.DataFrame) -> np.ndarray:
        """
        Update portfolio weights based on current market state
        
        Args:
            features: DataFrame with market features
            
        Returns:
            Updated portfolio weights
        """
        # Create state for agents
        state = {
            "features": features,
        }
        
        # Step 1: Observe trends
        trend_scores = self.trend_observer.act(state)
        state["trend_scores"] = trend_scores
        
        # Step 2: Optimize for returns
        proposed_weights = self.return_optimizer.act(state)
        state["proposed_weights"] = proposed_weights
        
        # Step 3: Manage risk
        final_weights = self.risk_manager.act(state)
        
        # Apply transaction cost penalty
        weight_changes = np.abs(final_weights - self.current_weights)
        transaction_costs = np.sum(weight_changes * self.transaction_cost)
        
        # Update current weights
        self.current_weights = final_weights
        
        return final_weights, transaction_costs

    volatility = np.std(daily_returns) * np.sqrt(252)  # Annualized
