class xLSTMCell(nn.Module):
    def __init__(self, input_size, hidden_size):
        super(xLSTMCell, self).__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        
        # Gates similar to LSTM but with exponential gating modifications
        self.input_gate = nn.Linear(input_size + hidden_size, hidden_size)
        self.forget_gate = nn.Linear(input_size + hidden_size, hidden_size)
        self.cell_gate = nn.Linear(input_size + hidden_size, hidden_size)
        self.output_gate = nn.Linear(input_size + hidden_size, hidden_size)
        
        # Additional memory mixing components for xLSTM
        self.memory_mix = nn.Linear(hidden_size, hidden_size)
        
    def forward(self, x, states):
        h_prev, c_prev = states
        
        # Combine input and previous hidden state
        combined = torch.cat((x, h_prev), dim=1)
        
        # Compute gates with exponential gating (simulated)
        i = torch.sigmoid(self.input_gate(combined))
        f = torch.sigmoid(self.forget_gate(combined))
        c_tilde = torch.tanh(self.cell_gate(combined))
        o = torch.sigmoid(self.output_gate(combined))
        
        # Update cell state with memory mixing (simulated xLSTM behavior)
        c_mix = torch.tanh(self.memory_mix(c_prev))
        c_next = f * c_prev + i * c_tilde + 0.1 * c_mix  # Added memory mixing term
        
        # Compute hidden state
        h_next = o * torch.tanh(c_next)
        
        return h_next, c_next

class xLSTM(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers=1):
        super(xLSTM, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        
        self.cells = nn.ModuleList([
            xLSTMCell(input_size if layer == 0 else hidden_size, hidden_size)
            for layer in range(num_layers)
        ])
        
    def forward(self, x, states=None):
        batch_size = x.size(0)
        seq_len = x.size(1)
        
        if states is None:
            h = [torch.zeros(batch_size, self.hidden_size, device=x.device) for _ in range(self.num_layers)]
            c = [torch.zeros(batch_size, self.hidden_size, device=x.device) for _ in range(self.num_layers)]
        else:
            h, c = states
        
        output = []
        
        for t in range(seq_len):
            x_t = x[:, t, :]
            
            for layer in range(self.num_layers):
                if layer == 0:
                    h[layer], c[layer] = self.cells[layer](x_t, (h[layer], c[layer]))
                else:
                    h[layer], c[layer] = self.cells[layer](h[layer-1], (h[layer], c[layer]))
            
            output.append(h[-1])
        
        output = torch.stack(output, dim=1)
        return output, (h, c)

# Actor-Critic model using xLSTM
class xLSTMActorCritic(nn.Module):
    def __init__(self, input_dim, n_actions, hidden_dim=128):
        super(xLSTMActorCritic, self).__init__()
        
        # Shared feature extractor
        self.feature_extractor = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.GELU()
        )
        
        # Actor network with xLSTM
        self.actor_lstm = xLSTM(hidden_dim, hidden_dim)
        self.action_mean = nn.Linear(hidden_dim, n_actions)
        self.action_std = nn.Parameter(torch.ones(n_actions) * 0.1)
        
        # Critic network with xLSTM
        self.critic_lstm = xLSTM(hidden_dim, hidden_dim)
        self.value = nn.Linear(hidden_dim, 1)
        
    def forward(self, state, actor_states=None, critic_states=None):
        batch_size, seq_len, _ = state.size()
        x = state.view(batch_size * seq_len, -1)
        x = self.feature_extractor(x)
        x = x.view(batch_size, seq_len, -1)
        
        # Actor branch
        actor_features, new_actor_states = self.actor_lstm(x, actor_states)
        action_mean = self.action_mean(actor_features[:, -1])
        action_std = self.action_std.expand_as(action_mean)
        
        # Critic branch
        critic_features, new_critic_states = self.critic_lstm(x, critic_states)
        value = self.value(critic_features[:, -1])
        
        return action_mean, action_std, value, new_actor_states, new_critic_states
    
    def act(self, state, actor_states=None):
        with torch.no_grad():
            action_mean, action_std, _, new_actor_states, _ = self(state, actor_states)
            dist = Normal(action_mean, action_std)
            action = dist.sample()
            action = torch.clamp(action, -1, 1)
            
        return action, new_actor_states
    
    def evaluate(self, state, action, actor_states=None, critic_states=None):
        action_mean, action_std, value, new_actor_states, new_critic_states = self(state, actor_states, critic_states)
        dist = Normal(action_mean, action_std)
        
        action_log_probs = dist.log_prob(action).sum(dim=-1, keepdim=True)
        entropy = dist.entropy().mean()
        
        return action_log_probs, value, entropy, new_actor_states, new_critic_states

# Simplified PPO implementation
class PPO:
    def __init__(self, env, model, lr=3e-4, gamma=0.99, eps_clip=0.2, K_epochs=10, batch_size=32):
        self.env = env
        self.model = model
        self.optimizer = optim.Adam(model.parameters(), lr=lr)
        self.gamma = gamma
        self.eps_clip = eps_clip
        self.K_epochs = K_epochs
        self.batch_size = batch_size
        
        self.buffer = deque(maxlen=2000)
        
    def update(self):
        if len(self.buffer) < self.batch_size:
            return 0, 0, 0
        
        # Sample batch
        batch = random.sample(self.buffer, self.batch_size)
        states, actions, rewards, next_states, dones, old_log_probs = zip(*batch)
        
        # Convert to tensors
        states = torch.FloatTensor(np.array(states))
        actions = torch.FloatTensor(np.array(actions))
        rewards = torch.FloatTensor(np.array(rewards)).unsqueeze(-1)
        next_states = torch.FloatTensor(np.array(next_states))
        dones = torch.FloatTensor(np.array(dones)).unsqueeze(-1)
        old_log_probs = torch.FloatTensor(np.array(old_log_probs))
        
        # Compute discounted rewards
        returns = []
        discounted_reward = 0
        for reward, done in zip(reversed(rewards), reversed(dones)):
            if done:
                discounted_reward = 0
            discounted_reward = reward + (self.gamma * discounted_reward)
            returns.insert(0, discounted_reward)
        returns = torch.FloatTensor(returns).unsqueeze(-1)
        
        # Normalize returns
        returns = (returns - returns.mean()) / (returns.std() + 1e-5)
        
        # PPO update
        actor_loss_epoch = 0
        critic_loss_epoch = 0
        entropy_epoch = 0
        
        for _ in range(self.K_epochs):
            # Evaluate actions and calculate ratio
            log_probs, state_values, entropy, _, _ = self.model.evaluate(states, actions)
            ratios = torch.exp(log_probs - old_log_probs)
            
            # Calculate surrogate losses
            advantages = returns - state_values.detach()
            surr1 = ratios * advantages
            surr2 = torch.clamp(ratios, 1-self.eps_clip, 1+self.eps_clip) * advantages
            
            # PPO loss
            actor_loss = -torch.min(surr1, surr2).mean()
            critic_loss = 0.5 * (returns - state_values).pow(2).mean()
            loss = actor_loss + critic_loss - 0.01 * entropy
            
            # Update
            self.optimizer.zero_grad()
            loss.backward()
            self.optimizer.step()
            
            actor_loss_epoch += actor_loss.item()
            critic_loss_epoch += critic_loss.item()
            entropy_epoch += entropy.item()
        
        # Clear buffer
        self.buffer.clear()
        
        return actor_loss_epoch/self.K_epochs, critic_loss_epoch/self.K_epochs, entropy_epoch/self.K_epochs
    
    def train(self, num_episodes):
        best_reward = -float('inf')
        rewards_history = []
        
        for episode in range(num_episodes):
            state = self.env.reset()
            state = torch.FloatTensor(state).unsqueeze(0)  # Add batch dimension
            actor_states = None
            critic_states = None
            
            done = False
            episode_reward = 0
            
            while not done:
                action, actor_states = self.model.act(state, actor_states)
                
                # Calculate log probability
                with torch.no_grad():
                    log_prob, _, _, _, _ = self.model.evaluate(state, action, actor_states, critic_states)
                
                # Take action in environment
                action_np = action.squeeze().numpy()
                next_state, reward, done, _ = self.env.step(action_np)
                next_state = torch.FloatTensor(next_state).unsqueeze(0)
                
                # Store in buffer
                self.buffer.append((
                    state.squeeze().numpy(),
                    action.squeeze().numpy(),
                    reward,
                    next_state.squeeze().numpy(),
                    done,
                    log_prob.squeeze().numpy()
                ))
                
                # Update state
                state = next_state
                episode_reward += reward
                
                # Update policy if buffer is full
                if len(self.buffer) >= self.batch_size:
                    actor_loss, critic_loss, entropy = self.update()
            
            rewards_history.append(episode_reward)
            
            if episode % 10 == 0:
                avg_reward = np.mean(rewards_history[-10:])
                print(f"Episode {episode}, Avg Reward: {avg_reward:.2f}")
                
                if avg_reward > best_reward:
                    best_reward = avg_reward
                    torch.save(self.model.state_dict(), "best_model.pth")
        
        return rewards_history

def generate_simulated_data(num_days=1000, num_stocks=5):
    # Generate price data
    prices = np.zeros((num_days, num_stocks))
    
    # Start with base values
    base_values = np.array([100, 150, 200, 250, 300])
    
    # Generate random walk with drift
    for i in range(num_stocks):
        prices[0, i] = base_values[i]
        drift = np.random.normal(0.0001, 0.0002)
        volatility = np.random.uniform(0.01, 0.02)
        
        for day in range(1, num_days):
            change = drift + volatility * np.random.normal(0, 1)
            prices[day, i] = prices[day-1, i] * (1 + change)
    
    # Generate features (open, high, low, close, volume)
    features = np.zeros((num_days, num_stocks * 6))
    
    for day in range(num_days):
        for i in range(num_stocks):
            price = prices[day, i]
            # Open, high, low, close prices (simulated around the main price)
            features[day, i*6] = price * np.random.uniform(0.99, 1.0)  # Open
            features[day, i*6+1] = price * np.random.uniform(1.0, 1.02)  # High
            features[day, i*6+2] = price * np.random.uniform(0.98, 1.0)  # Low
            features[day, i*6+3] = price * np.random.uniform(0.99, 1.01)  # Close
            features[day, i*6+4] = price  # Adjusted close
            # Volume
            features[day, i*6+5] = np.random.uniform(100000, 1000000)
    
    # Split into train and test
    train_prices = prices[:800]
    test_prices = prices[800:]
    train_features = features[:800]
    test_features = features[800:]
    
    return train_prices, train_features, test_prices, test_features

# Calculate evaluation metrics
def calculate_metrics(env):
    assets_history = env.total_assets_history
    initial_balance = env.initial_balance
    
    # Cumulative Return (CR)
    cr = (assets_history[-1] - initial_balance) / initial_balance * 100
    
    # Max Earning Rate (MER)
    mer = (max(assets_history) - initial_balance) / initial_balance * 100
    
    # Maximum Pullback (MPB)
    peak = assets_history[0]
    max_pullback = 0
    
    for value in assets_history:
        if value > peak:
            peak = value
        else:
            pullback = (peak - value) / peak * 100
            max_pullback = max(max_pullback, pullback)
    
    # Average Profitability Per Trade (APPT)
    # Simplified calculation
    appt = (assets_history[-1] - initial_balance) / 100  # Assuming 100 trades
    
    # Sharpe Ratio (SR)
    returns = np.diff(assets_history) / assets_history[:-1]
    sr = np.mean(returns) / (np.std(returns) + 1e-10) * np.sqrt(252)  # Annualized
    
    return {
        "CR": cr,
        "MER": mer,
        "MPB": max_pullback,
        "APPT": appt,
        "SR": sr
    }
