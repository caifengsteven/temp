class Node:
    """Base class for nodes in the alpha expression tree."""
    def __init__(self, value=None):

class OperatorNode(Node):
    """Node representing mathematical operators or functions."""
    def __init__(self, operator):
        super().__init__(operator)
    
    def evaluate(self, data):
        if self.value == 'rank':
            return self.children[0].evaluate(data).rank(pct=True)
        elif self.value == 'mean':
            return self.children[0].evaluate(data).mean()
        elif self.value == 'std':
            return self.children[0].evaluate(data).std()
        elif self.value == 'max':
            return self.children[0].evaluate(data).max()
        elif self.value == 'min':
            return self.children[0].evaluate(data).min()
        elif self.value == 'ts_mean':
            return self.children[0].evaluate(data).rolling(20).mean()
        elif self.value == 'ts_std':
            return self.children[0].evaluate(data).rolling(20).std()
        elif self.value == 'ts_rank':
            return self.children[0].evaluate(data).rolling(20).apply(lambda x: x.rank(pct=True).iloc[-1])
        elif self.value == 'delay':
            return self.children[0].evaluate(data).shift(5)
        elif self.value == 'delta':
            return self.children[0].evaluate(data).diff()
        elif self.value == 'corr':
            return self.children[0].evaluate(data).rolling(20).corr(self.children[1].evaluate(data))
        elif self.value == '+':
            return self.children[0].evaluate(data) + self.children[1].evaluate(data)
        elif self.value == '-':
            return self.children[0].evaluate(data) - self.children[1].evaluate(data)
        elif self.value == '*':
            return self.children[0].evaluate(data) * self.children[1].evaluate(data)
        elif self.value == '/':
            return self.children[0].evaluate(data) / (self.children[1].evaluate(data) + 1e-6)  # Avoid division by zero
        
    def __str__(self):
        if len(self.children) == 1:
            return f"{self.value}({self.children[0]})"
        elif len(self.children) == 2:
            return f"({self.children[0]} {self.value} {self.children[1]})"

def generate_random_tree(max_depth, terminal_vars, functions, current_depth=0):
    """Generate a random expression tree."""
    if current_depth >= max_depth or (current_depth > 0 and random.random() < 0.5):
        # Terminal node
        return VariableNode(random.choice(terminal_vars))
    else:
        # Operator node
        op = random.choice(functions)
        node = OperatorNode(op)
        
        # Determine number of children
        if op in ['+', '-', '*', '/', 'corr']:
            num_children = 2
        else:
            num_children = 1
        
        # Generate children
        for _ in range(num_children):
            child = generate_random_tree(max_depth, terminal_vars, functions, current_depth + 1)
            node.children.append(child)
        
        return node

def point_mutation(tree, terminal_vars):
    """Perform point mutation on a variable node."""
    offspring = tree.copy()
    
    # Find all variable nodes
    def get_variable_nodes(node, path=None):
        if path is None:
            path = []
            
        if isinstance(node, VariableNode):
            return [path]
        
        result = []
        for i, child in enumerate(node.children):
            result.extend(get_variable_nodes(child, path + [i]))
            
        return result
    
    variable_nodes = get_variable_nodes(offspring.root)
    
    if variable_nodes:
        # Select a random variable node
        point = random.choice(variable_nodes)
        
        # Navigate to the selected node
        node = offspring.root
        for i in range(len(point) - 1):
            node = node.children[point[i]]
        
        # Perform mutation
        if isinstance(node.children[point[-1]], VariableNode):
            node.children[point[-1]].value = random.choice(terminal_vars)
    
    return offspring

def evaluate_alpha(alpha, stock_data, forward_returns):
    """Calculate IC metrics for an alpha."""
    try:
        alpha_values = alpha.evaluate(stock_data)
        
        # Handle NaN values
        alpha_values = alpha_values.fillna(alpha_values.mean())

        ic_values = []
        rank_ic_values = []
        
        for date in forward_returns.index.unique():
            if date in alpha_values.index:
                date_alpha = alpha_values.loc[date]
                date_returns = forward_returns.loc[date]
                
                # Match stocks
                common_stocks = set(date_alpha.index) & set(date_returns.index)
                if common_stocks:
                    a_values = [date_alpha.loc[s] for s in common_stocks]
                    r_values = [date_returns.loc[s] for s in common_stocks]
                    
                    # Calculate Pearson and Spearman correlations
                    if len(a_values) > 5:  # Require at least 5 samples for correlation
                        ic = np.corrcoef(a_values, r_values)[0, 1]
                        rank_ic = spearmanr(a_values, r_values)[0]
                        
                        if not np.isnan(ic):
                            ic_values.append(ic)
                        if not np.isnan(rank_ic):
                            rank_ic_values.append(rank_ic)
        
        # Calculate IC metrics
        if ic_values:
            ic = np.mean(ic_values)
            icir = ic / np.std(ic_values) if np.std(ic_values) > 0 else 0
            rank_ic = np.mean(rank_ic_values)
            rank_icir = rank_ic / np.std(rank_ic_values) if np.std(rank_ic_values) > 0 else 0
            
            return {
                'IC': ic,
                'ICIR': icir,
                'RankIC': rank_ic,
                'RankICIR': rank_icir

def tournament_selection(population, tournament_size=3):
    """Select a parent using tournament selection."""
    tournament = random.sample(population, min(tournament_size, len(population)))
    return max(tournament, key=lambda x: x[1]['IC'])

def warm_start_gp(initial_alpha, stock_data, forward_returns, terminal_vars,
                 pop_size=50, generations=20, crossover_prob=0.7, mutation_prob=0.3):
    """
    Implementation of the Warm Start GP framework as described in the paper.
    """
    # Initialize population with the given alpha
    population = [(initial_alpha, evaluate_alpha(initial_alpha, stock_data, forward_returns))]
    
    best_alpha = None
    best_fitness = 0
    
    for gen in range(generations):
        # Always keep the best individual
        new_population = [max(population, key=lambda x: x[1]['IC'])]
        
        # Generate new individuals
        while len(new_population) < pop_size:
            if gen == 0:
                # First generation: only point mutation
                parent = tournament_selection(population)[0]
                offspring = point_mutation(parent, terminal_vars)
            else:
                # Subsequent generations: crossover or mutation
                r = random.random()
                if r < crossover_prob:
                    parent1 = tournament_selection(population)[0]
                    parent2 = tournament_selection(population)[0]
                    offspring = restricted_crossover(parent1, parent2)
                else:
                    parent = tournament_selection(population)[0]
                    offspring = point_mutation(parent, terminal_vars)
            
            # Avoid duplicates
            if all(str(offspring) != str(alpha) for alpha, _ in new_population):
                fitness = evaluate_alpha(offspring, stock_data, forward_returns)
                new_population.append((offspring, fitness))
        
        # Update population
        population = new_population
        
        # Update best alpha
        current_best = max(population, key=lambda x: x[1]['IC'])
        if current_best[1]['IC'] > best_fitness:
            best_alpha = current_best[0]
            best_fitness = current_best[1]['IC']
        
        print(f"Generation {gen+1}/{generations}, Best IC: {best_fitness:.4f}")
    
    return best_alpha, best_fitness

def generate_synthetic_stock_data(num_stocks=100, num_days=500):

    data['close'] = np.random.normal(data['open'], 5, len(data))

    data['returns'] = (data['close'] / data['open'] - 1)
    
    # Add forward returns (5-day VWAP as mentioned in the paper)
    forward_returns = pd.Series(index=idx)
    
    for date in dates:
        for stock in stock_ids:
            # Simulate 5-day forward return
            if date + pd.Timedelta(days=5) < dates[-1]:
                future_prices = []
                for i in range(1, 6):
                    future_date = date + pd.Timedelta(days=i)
                    if future_date in dates:
                        try:
                            future_price = data.loc[(future_date, stock), 'close']
                            future_volume = data.loc[(future_date, stock), 'volume']
                            future_prices.append(future_price * future_volume)
                        except:
                            pass
                
                if future_prices:
                    vwap_future = sum(future_prices) / (sum(future_prices) / sum(p / v for p, v in zip(future_prices, future_prices)))
                    vwap_current = data.loc[(date, stock), 'close']
                    forward_returns.loc[(date, stock)] = vwap_future / vwap_current - 1
    
    # Handle missing values
    forward_returns = forward_returns.fillna(0)

def backtest_portfolio(alpha, stock_data, forward_returns, top_n=30, holding_period=5):
    """Backtest a portfolio using the alpha signals."""
    # Dates for trading (excluding the first few days needed for alpha calculation)
    unique_dates = stock_data.index.get_level_values('date').unique()
    trading_dates = unique_dates[20:]  # Skip first 20 days to allow for rolling calculations
    
    # Initialize portfolio
    portfolio_values = [1.0]  # Start with $1
    positions = {}
    
    for i, date in enumerate(trading_dates[:-holding_period]):  # Allow room for holding period
        # Calculate alpha values
        try:
            alpha_values = alpha.evaluate(stock_data.loc[date])
            
            # Sort stocks by alpha values
            sorted_stocks = alpha_values.sort_values(ascending=False)
            
            # Select top N stocks
            selected_stocks = sorted_stocks.head(top_n).index.tolist()
            
            # Equal weighting
            weight_per_stock = 1.0 / len(selected_stocks)
            
            # Update positions
            new_positions = {stock: weight_per_stock for stock in selected_stocks}
            
            # Calculate returns for the holding period
            next_date_idx = trading_dates.get_loc(date) + holding_period
            if next_date_idx < len(trading_dates):
                next_date = trading_dates[next_date_idx]
                
                # Calculate portfolio return
                portfolio_return = 0
                for stock, weight in new_positions.items():
                    try:
                        stock_return = forward_returns.loc[(date, stock)]
                        portfolio_return += weight * stock_return
                    except:
                        pass
                
                # Update portfolio value
                portfolio_values.append(portfolio_values[-1] * (1 + portfolio_return))
                
                # Update positions
                positions = new_positions
        except Exception as e:
            print(f"Error in backtesting at date {date}: {e}")
            portfolio_values.append(portfolio_values[-1])
    
    # Calculate performance metrics
    returns = pd.Series(portfolio_values).pct_change().fillna(0)
    
    annualized_return = (portfolio_values[-1] / portfolio_values[0]) ** (252 / len(portfolio_values)) - 1
    annualized_volatility = returns.std() * np.sqrt(252)
    sharpe_ratio = annualized_return / annualized_volatility if annualized_volatility > 0 else 0
    
    return {
        'Portfolio Values': portfolio_values,
        'Annualized Return': annualized_return,
        'Sharpe Ratio': sharpe_ratio
    }

def plot_correlation_matrix(alphas, labels):
    """Plot correlation matrix of alpha values."""
    # Calculate correlation matrix
    corr_matrix = np.zeros((len(alphas), len(alphas)))
    
    for i in range(len(alphas)):
        for j in range(len(alphas)):
            if i == j:
                corr_matrix[i, j] = 1.0
            else:
                alpha_i_values = alphas[i].evaluate(stock_data).values.flatten()
                alpha_j_values = alphas[j].evaluate(stock_data).values.flatten()
                
                # Filter out NaN values
                valid_idx = ~np.isnan(alpha_i_values) & ~np.isnan(alpha_j_values)
                if np.sum(valid_idx) > 10:
                    corr_matrix[i, j] = np.abs(np.corrcoef(
                        alpha_i_values[valid_idx], 
                        alpha_j_values[valid_idx]
                    )[0, 1])
                else:
                    corr_matrix[i, j] = 0
    
    # Plot correlation matrix
    plt.figure(figsize=(10, 8))
    sns.heatmap(corr_matrix, annot=True, fmt=".2f", cmap="YlGnBu", 
                xticklabels=labels, yticklabels=labels)
    plt.title("Alpha Correlation Matrix")
    plt.tight_layout()
    plt.show()
    
    # Return average absolute correlation
    return np.mean(np.abs(corr_matrix - np.eye(len(alphas))))
