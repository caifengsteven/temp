class TechnicalIndicators:

        # Price Range
        df['price_range'] = (df['High'] - df['Low']) / df['Close']
        
        # Moving Averages
        for window in [5, 10, 20, 50]:
            df[f'SMA_{window}'] = df['Close'].rolling(window=window).mean()
            df[f'EMA_{window}'] = df['Close'].ewm(span=window, adjust=False).mean()
        
        # Price Rate of Change
        for window in [5, 10, 20]:
            df[f'ROC_{window}'] = (df['Close'] - df['Close'].shift(window)) / df['Close'].shift(window) * 100
        
        return df
    
    @staticmethod
    def add_momentum_indicators(df):
        """Add momentum indicators"""
        # RSI
        delta = df['Close'].diff()
        gain = (delta.where(delta > 0, 0)).rolling(window=14).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean()
        rs = gain / loss
        df['RSI'] = 100 - (100 / (1 + rs))
        
        # MACD
        ema12 = df['Close'].ewm(span=12, adjust=False).mean()
        ema26 = df['Close'].ewm(span=26, adjust=False).mean()
        df['MACD'] = ema12 - ema26

        df['%K'] = 100 * ((df['Close'] - low_14) / (high_14 - low_14))

class FeatureProcessor:
    """Process features for model input"""
    
    def __init__(self):

        df[feature_columns] = self.scalers['features'].fit_transform(df[feature_columns])

        X, y = self._create_sequences(df, seq_length, price_columns)
        
        # Split data
        train_size = int(len(X) * train_split)
        val_size = int(len(X) * val_split)
        
        X_train, y_train = X[:train_size], y[:train_size]
        X_val, y_val = X[train_size:train_size+val_size], y[train_size:train_size+val_size]
        X_test, y_test = X[train_size+val_size:], y[train_size+val_size:]
        
        return (X_train, y_train), (X_val, y_val), (X_test, y_test)
    
    def _create_sequences(self, df, seq_length, price_columns):
        """Create sequences for time series prediction"""
        # Prepare input features
        feature_cols = [col for col in df.columns if col not in price_columns + ['Volume'] 
                        and not col.endswith('_scaled')]
        price_cols = [f'{col}_scaled' for col in price_columns + ['Volume']]
        
        # Combine all features
        features = df[feature_cols + price_cols].values
        
        # Target is next day's closing price
        target = df['Close_scaled'].values
        
        X, y = [], []
        for i in range(len(df) - seq_length):
            X.append(features[i:i+seq_length])
            y.append(target[i+seq_length])
            
        return np.array(X), np.array(y)
    
    def inverse_transform_predictions(self, predictions):
        """Transform scaled predictions back to original scale"""

    def inverse_transform_actuals(self, actuals):

class VAEModel:
    """Variational Autoencoder model for feature extraction"""
    
    def __init__(self, input_dim, latent_dim=16):
        self.input_dim = input_dim
        self.latent_dim = latent_dim
        self.model = self._build_model()
        
    def _build_model(self):
        # Encoder
        inputs = Input(shape=(self.input_dim,))
        x = Dense(128, activation='relu')(inputs)
        x = BatchNormalization()(x)
        x = Dense(64, activation='relu')(x)
        
        # Latent space
        z_mean = Dense(self.latent_dim)(x)
        z_log_var = Dense(self.latent_dim)(x)
        
        # Sampling function
        def sampling(args):
            z_mean, z_log_var = args
            epsilon = tf.random.normal(shape=(tf.shape(z_mean)[0], self.latent_dim))
            return z_mean + tf.exp(0.5 * z_log_var) * epsilon
        
        z = tf.keras.layers.Lambda(sampling)([z_mean, z_log_var])
        
        # Decoder
        decoder_input = Input(shape=(self.latent_dim,))
        x = Dense(64, activation='relu')(decoder_input)
        x = BatchNormalization()(x)
        x = Dense(128, activation='relu')(x)
        decoder_output = Dense(self.input_dim)(x)
        
        # Define models
        encoder = Model(inputs, [z_mean, z_log_var, z], name='encoder')
        decoder = Model(decoder_input, decoder_output, name='decoder')
        
        # Define VAE model
        outputs = decoder(encoder(inputs)[2])
        vae = Model(inputs, outputs, name='vae')
        
        # Add VAE loss
        reconstruction_loss = tf.reduce_mean(tf.square(inputs - outputs))
        kl_loss = -0.5 * tf.reduce_mean(1 + z_log_var - tf.square(z_mean) - tf.exp(z_log_var))
        vae_loss = reconstruction_loss + 0.001 * kl_loss
        vae.add_loss(vae_loss)
        
        vae.compile(optimizer='adam')
        return vae
    
    def fit(self, X_train, X_val, epochs=50, batch_size=64):
        """Fit the VAE model to the training data"""
        # Reshape data for VAE (sequence to flat)
        X_train_reshaped = X_train.reshape(X_train.shape[0], -1)
        X_val_reshaped = X_val.reshape(X_val.shape[0], -1)
        
        # Train model
        self.model.fit(
            X_train_reshaped, None,
            epochs=epochs,
            batch_size=batch_size,
            validation_data=(X_val_reshaped, None),
            verbose=0
        )
        
    def encode(self, X):
        """Encode the input data to the latent space"""
        X_reshaped = X.reshape(X.shape[0], -1)
        return self.model.get_layer('encoder')(X_reshaped)[2]

class TransformerModel:
    """Transformer model for sequence prediction"""
    
    def __init__(self, input_shape, head_size=256, num_heads=8, ff_dim=256, dropout=0.1):
        self.input_shape = input_shape
        self.head_size = head_size
        self.num_heads = num_heads
        self.ff_dim = ff_dim
        self.dropout = dropout
        self.model = self._build_model()
        
    def _transformer_encoder(self, inputs):
        # Multi-head attention
        attention_output = MultiHeadAttention(
            key_dim=self.head_size, num_heads=self.num_heads, dropout=self.dropout
        )(inputs, inputs)
        attention_output = Dropout(self.dropout)(attention_output)
        
        # Add & Norm
        x1 = LayerNormalization(epsilon=1e-6)(inputs + attention_output)
        
        # Feed Forward
        ff_output = Dense(self.ff_dim, activation='relu')(x1)
        ff_output = Dense(self.input_shape[2])(ff_output)
        ff_output = Dropout(self.dropout)(ff_output)
        
        # Add & Norm
        return LayerNormalization(epsilon=1e-6)(x1 + ff_output)
    
    def _build_model(self):
        inputs = Input(shape=self.input_shape)
        
        # Transformer layers
        x = self._transformer_encoder(inputs)
        x = self._transformer_encoder(x)
        
        # Flatten and predict
        x = tf.keras.layers.GlobalAveragePooling1D()(x)
        x = Dense(128, activation='relu')(x)
        x = Dropout(0.1)(x)
        outputs = Dense(1)(x)
        
        model = Model(inputs=inputs, outputs=outputs)
        model.compile(optimizer='adam', loss='mse')
        return model
    
    def fit(self, X_train, y_train, X_val, y_val, epochs=100, batch_size=64):
        """Fit the Transformer model to the training data"""
        early_stopping = tf.keras.callbacks.EarlyStopping(
            monitor='val_loss', patience=30, restore_best_weights=True
        )
        
        self.model.fit(
            X_train, y_train,
            epochs=epochs,
            batch_size=batch_size,
            validation_data=(X_val, y_val),
            callbacks=[early_stopping],
            verbose=0
        )
        
    def predict(self, X):
        """Make predictions using the Transformer model"""
        return self.model.predict(X, verbose=0)

class LSTMModel:
    """LSTM model for sequence prediction"""
    
    def __init__(self, input_shape, units=128, dropout=0.2):
        self.input_shape = input_shape
        self.units = units
        self.dropout = dropout
        self.model = self._build_model()
        
    def _build_model(self):
        inputs = Input(shape=self.input_shape)
        
        # First LSTM layer
        x = LSTM(self.units, return_sequences=True)(inputs)
        x = Dropout(self.dropout)(x)
        
        # Second LSTM layer
        x = LSTM(self.units)(x)
        x = Dropout(self.dropout)(x)
        
        # Output layer with batch normalization
        x = Dense(32)(x)
        x = BatchNormalization()(x)
        x = tf.keras.layers.Activation('tanh')(x)
        outputs = Dense(1)(x)
        
        model = Model(inputs=inputs, outputs=outputs)
        model.compile(optimizer='adam', loss='mse')
        return model
    
    def fit(self, X_train, y_train, X_val, y_val, epochs=100, batch_size=64):
        """Fit the LSTM model to the training data"""
        early_stopping = tf.keras.callbacks.EarlyStopping(
            monitor='val_loss', patience=30, restore_best_weights=True
        )
        
        self.model.fit(
            X_train, y_train,
            epochs=epochs,
            batch_size=batch_size,
            validation_data=(X_val, y_val),
            callbacks=[early_stopping],
            verbose=0

class EnsembleModel:

        """Fit all models in the ensemble"""
        # Train VAE for feature extraction
        print("Training VAE model...")
        self.vae.fit(X_train, X_val, epochs=min(epochs, 50), batch_size=batch_size)

        print("Training Transformer model...")
        self.transformer.fit(X_train, y_train, X_val, y_val, epochs=epochs, batch_size=batch_size)
        
        # Train LSTM
        print("Training LSTM model...")
        self.lstm.fit(X_train, y_train, X_val, y_val, epochs=epochs, batch_size=batch_size)
        
        # Optimize weights using validation set
        self._optimize_weights(X_val, y_val)

        """Optimize ensemble weights based on validation performance"""
        print("Optimizing ensemble weights...")

        vae_encoded = self.vae.encode(X_val)

        total_error = vae_mse + transformer_mse + lstm_mse
        self.weights = [
            (1/vae_mse) / ((1/vae_mse) + (1/transformer_mse) + (1/lstm_mse)),
            (1/transformer_mse) / ((1/vae_mse) + (1/transformer_mse) + (1/lstm_mse)),
            (1/lstm_mse) / ((1/vae_mse) + (1/transformer_mse) + (1/lstm_mse))
        ]

class ModelEvaluator:
    """Evaluate model performance"""
    
    @staticmethod
    def evaluate_predictions(y_true, y_pred):
        """Evaluate model predictions"""
        # Calculate RMSE
        rmse = math.sqrt(mean_squared_error(y_true, y_pred))
        
        # Calculate R² score
        r2 = r2_score(y_true, y_pred)
        
        # Calculate MAPE
        mape = np.mean(np.abs((y_true - y_pred) / y_true)) * 100
        
        # Calculate directional accuracy
        direction_true = np.diff(y_true) > 0
        direction_pred = np.diff(y_pred) > 0
        directional_accuracy = np.mean(direction_true == direction_pred)
        
        return {
            'RMSE': rmse,
            'R2': r2,
            'MAPE': mape,
            'Directional_Accuracy': directional_accuracy
        }
    
    @staticmethod
    def plot_results(y_true, y_pred, title='Actual vs Predicted'):
        """Plot actual vs predicted values"""
        plt.figure(figsize=(12, 6))
        plt.plot(y_true, label='Actual')
        plt.plot(y_pred, label='Predicted')
        plt.title(title)
        plt.xlabel('Time')
        plt.ylabel('Stock Price')
        plt.legend()
        plt.show()
        
        # Scatter plot
        plt.figure(figsize=(8, 8))
        plt.scatter(y_true, y_pred)
        plt.title('Prediction Scatter Plot')
        plt.xlabel('Actual Price')
        plt.ylabel('Predicted Price')
        
        # Add perfect prediction line
        min_val = min(min(y_true), min(y_pred))
        max_val = max(max(y_true), max(y_pred))
        plt.plot([min_val, max_val], [min_val, max_val], 'r--')
        plt.show()

def test_baseline_models(X_train, y_train, X_test, y_test, processor):
    """Test baseline models for comparison"""
    print("\nTesting baseline models...")
    
    # Prepare data for traditional models
    train_size = len(y_train)
    test_size = len(y_test)
    
    # Extract last close price from each sequence
    train_close = X_train[:, -1, 3]  # Assuming Close is at index 3
    test_close = X_test[:, -1, 3]
    
    # Convert scaled values back to original prices
    train_close_orig = processor.scalers['Close'].inverse_transform(train_close.reshape(-1, 1)).flatten()
    test_close_orig = processor.scalers['Close'].inverse_transform(test_close.reshape(-1, 1)).flatten()
    y_test_orig = processor.inverse_transform_actuals(y_test)
    
    # Simple Moving Average
    sma_preds = np.array([np.mean(train_close_orig[-5:]) for _ in range(test_size)])
    
    # Exponential Moving Average (with alpha=0.3)
    alpha = 0.3
    ema_preds = [train_close_orig[-1]]
    for i in range(1, test_size):
        ema_preds.append(alpha * test_close_orig[i-1] + (1 - alpha) * ema_preds[-1])
    ema_preds = np.array(ema_preds)
    
    # ARIMA-like prediction (AR(1) model)
    # For simplicity, we'll use a simple AR(1) model: y_t = c + φ*y_(t-1) + ε_t
    # Estimate φ from training data
    phi = np.corrcoef(train_close_orig[:-1], train_close_orig[1:])[0, 1]
    c = np.mean(train_close_orig[1:] - phi * train_close_orig[:-1])
    
    arima_preds = [train_close_orig[-1]]
    for i in range(1, test_size):
        arima_preds.append(c + phi * test_close_orig[i-1])
    arima_preds = np.array(arima_preds)
    
    # Evaluate baseline models
    baseline_results = {
        'SMA': ModelEvaluator.evaluate_predictions(y_test_orig, sma_preds),
        'EMA': ModelEvaluator.evaluate_predictions(y_test_orig, ema_preds),
        'ARIMA': ModelEvaluator.evaluate_predictions(y_test_orig, arima_preds)
    }
    
    # Print results
    for model_name, metrics in baseline_results.items():
        print(f"\n{model_name} Results:")
        print(f"  RMSE: {metrics['RMSE']:.4f}")
        print(f"  R²: {metrics['R2']:.4f}")
        print(f"  MAPE: {metrics['MAPE']:.4f}%")
        print(f"  Directional Accuracy: {metrics['Directional_Accuracy']:.4f}")
    
    return baseline_results

    print("\nTesting individual models...")
    
    # Test VAE
    print("Testing VAE model...")
    vae = VAEModel(input_dim=X_train.shape[1] * X_train.shape[2], latent_dim=16)
    vae.fit(X_train, X_val, epochs=30, batch_size=64)
    
    vae_encoded_train = vae.encode(X_train)
    vae_encoded_val = vae.encode(X_val)
    vae_encoded_test = vae.encode(X_test)
    
    vae_predictor = tf.keras.Sequential([
        Dense(64, activation='relu'),
        Dense(1)
    ])
    vae_predictor.compile(optimizer='adam', loss='mse')
    vae_predictor.fit(
        vae_encoded_train, y_train,
        epochs=50,
        batch_size=64,
        validation_data=(vae_encoded_val, y_val),
        verbose=0
    )
    
    vae_preds = vae_predictor.predict(vae_encoded_test, verbose=0).flatten()
    
    # Test Transformer
    print("Testing Transformer model...")
    transformer = TransformerModel(input_shape=X_train.shape[1:])
    transformer.fit(X_train, y_train, X_val, y_val, epochs=100, batch_size=64)
    transformer_preds = transformer.predict(X_test).flatten()
    
    # Test LSTM
    print("Testing LSTM model...")
    lstm = LSTMModel(input_shape=X_train.shape[1:])
    lstm.fit(X_train, y_train, X_val, y_val, epochs=100, batch_size=64)
    lstm_preds = lstm.predict(X_test).flatten()
    
    # Convert scaled predictions back to original scale
    y_test_orig = processor.inverse_transform_actuals(y_test)
    vae_preds_orig = processor.inverse_transform_predictions(vae_preds)
    transformer_preds_orig = processor.inverse_transform_predictions(transformer_preds)
    lstm_preds_orig = processor.inverse_transform_predictions(lstm_preds)
    
    # Evaluate individual models
    individual_results = {
        'VAE': ModelEvaluator.evaluate_predictions(y_test_orig, vae_preds_orig),
        'Transformer': ModelEvaluator.evaluate_predictions(y_test_orig, transformer_preds_orig),
        'LSTM': ModelEvaluator.evaluate_predictions(y_test_orig, lstm_preds_orig)
    }
    
    # Print results

        print(f"\n{model_name} Results:")
        print(f"  RMSE: {metrics['RMSE']:.4f}")
        print(f"  R²: {metrics['R2']:.4f}")
        print(f"  MAPE: {metrics['MAPE']:.4f}%")
        print(f"  Directional Accuracy: {metrics['Directional_Accuracy']:.4f}")
    
    return individual_results, vae, transformer, lstm
