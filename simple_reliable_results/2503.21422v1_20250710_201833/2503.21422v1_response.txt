class FactorGenerator:
    """Generates alpha factors from financial data"""
    
    def __init__(self, price_data: pd.DataFrame, fundamental_data: pd.DataFrame, sentiment_data: pd.DataFrame):
        """
        Initialize with financial data
        
        Args:
            price_data: DataFrame of daily prices
            fundamental_data: DataFrame of fundamental data
            sentiment_data: DataFrame of news sentiment
        """
        self.price_data = price_data
        self.fundamental_data = fundamental_data
        self.sentiment_data = sentiment_data
    
    def calculate_momentum_factor(self, window: int = 20) -> pd.DataFrame:
        """Calculate momentum factor based on past returns"""
        returns = self.price_data.pct_change()
        momentum = returns.rolling(window=window).sum()
        return momentum
    
    def calculate_mean_reversion_factor(self, window: int = 10) -> pd.DataFrame:
        """Calculate mean reversion factor"""
        returns = self.price_data.pct_change()
        z_score = (returns - returns.rolling(window=window).mean()) / returns.rolling(window=window).std()
        mean_reversion = -1 * z_score  # Negative z-score for mean reversion
        return mean_reversion
    
    def calculate_value_factor(self) -> pd.DataFrame:
        """Calculate value factor based on fundamental data"""
        # Resample fundamental data to daily frequency (forward fill)
        assets = self.price_data.columns
        dates = self.price_data.index
        
        # Initialize value factor DataFrame
        value_factor = pd.DataFrame(index=dates, columns=assets)
        
        # For each asset, calculate P/E ratio
        for asset in assets:
            # Get the latest fundamental data for each date
            for date in dates:
                # Find the most recent fundamental data before this date
                prev_quarter = self.fundamental_data.index.get_level_values(0).asof(date)
                if pd.notna(prev_quarter):
                    price = self.price_data.loc[date, asset]
                    earnings = self.fundamental_data.loc[(prev_quarter, asset), 'NetIncome']
                    # P/E ratio (inverted so higher values = more value)
                    if earnings > 0:
                        value_factor.loc[date, asset] = earnings / price
                    else:
                        value_factor.loc[date, asset] = 0
        
        # Normalize by cross-section
        value_factor = value_factor.apply(lambda x: (x - x.mean()) / x.std(), axis=1)
        return value_factor
    
    def calculate_sentiment_factor(self, window: int = 5) -> pd.DataFrame:
        """Calculate sentiment factor based on news sentiment"""
        # Initialize sentiment factor DataFrame
        assets = self.price_data.columns
        dates = self.price_data.index
        sentiment_factor = pd.DataFrame(index=dates, columns=assets, data=0)
        
        # For each news item, add sentiment to the factor
        for _, row in self.sentiment_data.iterrows():
            date = row['Date']
            asset = row['Asset']
            if date in dates and asset in assets:
                # Weight sentiment by volume
                sentiment_factor.loc[date, asset] = row['Sentiment'] * row['Volume']
        
        # Apply exponential decay to spread sentiment effect over window days
        result = sentiment_factor.copy()
        for i in range(1, window):
            shifted = sentiment_factor.shift(i) * (0.8 ** i)  # Decay factor
            result = result.add(shifted, fill_value=0)
        
        # Normalize by cross-section
        result = result.apply(lambda x: (x - x.mean()) / x.std(), axis=1)
        return result
    
    def combine_factors(self) -> pd.DataFrame:
        """Combine all factors with weights"""
        momentum = self.calculate_momentum_factor()
        mean_reversion = self.calculate_mean_reversion_factor()
        value = self.calculate_value_factor()
        sentiment = self.calculate_sentiment_factor()
        
        # Combine factors with weights
        combined = (
            0.3 * momentum.fillna(0) + 
            0.2 * mean_reversion.fillna(0) + 
            0.3 * value.fillna(0) + 
            0.2 * sentiment.fillna(0)
        )
        
        return combined

class PortfolioOptimizer:
    """Portfolio optimizer that converts predictions to positions"""
    
    def __init__(self, predictions: pd.DataFrame, risk_aversion: float = 1.0, max_position: float = 0.2):
        """
        Initialize with return predictions
        
        Args:
            predictions: DataFrame of return predictions
            risk_aversion: Risk aversion parameter (higher = more risk-averse)
            max_position: Maximum allowed position size as percentage
        """
        self.predictions = predictions
        self.risk_aversion = risk_aversion
        self.max_position = max_position
    
    def optimize(self) -> pd.DataFrame:
        """Generate optimized portfolio weights"""
        # Simplified Markowitz-inspired optimization:
        # 1. Rank assets based on predicted returns
        # 2. Apply position limits
        # 3. Ensure dollar neutrality (sum of weights = 0)
        
        # Initialize portfolio weights
        weights = pd.DataFrame(index=self.predictions.index, columns=self.predictions.columns, data=0.0)
        
        for date, row in self.predictions.iterrows():
            # Rank assets by predicted returns
            ranked = row.sort_values(ascending=False)
            
            # Assign weights based on ranks (proportional to prediction)
            n_assets = len(ranked)
            top_assets = ranked.iloc[:n_assets//4]  # Top 25%
            bottom_assets = ranked.iloc[-n_assets//4:]  # Bottom 25%
            
            # Assign long positions to top assets
            for asset, pred in top_assets.items():
                if pred > 0:
                    weights.loc[date, asset] = min(pred * 0.5, self.max_position)
            
            # Assign short positions to bottom assets
            for asset, pred in bottom_assets.items():
                if pred < 0:
                    weights.loc[date, asset] = max(pred * 0.5, -self.max_position)
            
            # Normalize to ensure dollar neutrality
            total_long = weights.loc[date][weights.loc[date] > 0].sum()
            total_short = abs(weights.loc[date][weights.loc[date] < 0].sum())
            
            if total_long > 0 and total_short > 0:
                # Adjust to make dollar neutral
                scale_long = 1.0
                scale_short = 1.0
                
                if total_long > total_short:
                    scale_long = total_short / total_long
                else:
                    scale_short = total_long / total_short
                
                for asset in weights.columns:
                    if weights.loc[date, asset] > 0:
                        weights.loc[date, asset] *= scale_long
                    elif weights.loc[date, asset] < 0:
                        weights.loc[date, asset] *= scale_short
        
        return weights

class OrderExecutor:
    """Simulates order execution for portfolio rebalancing"""
    
    def __init__(self, positions: pd.DataFrame, price_data: pd.DataFrame, transaction_cost: float = 0.001):
        """
        Initialize with target positions and price data
        
        Args:
            positions: DataFrame of target portfolio positions
            price_data: DataFrame of asset prices
            transaction_cost: Cost per transaction as percentage
        """
        self.positions = positions
        self.price_data = price_data
        self.transaction_cost = transaction_cost
    
    def execute(self) -> Tuple[pd.DataFrame, pd.DataFrame]:
        """
        Execute portfolio rebalancing and calculate actual positions and costs
        
        Returns:
            Tuple of (actual positions, transaction costs)
        """
        # Initialize actual positions and costs
        actual_positions = pd.DataFrame(index=self.positions.index, columns=self.positions.columns, data=0.0)
        transaction_costs = pd.DataFrame(index=self.positions.index, columns=['Cost'], data=0.0)
        
        # Start with zero positions
        current_positions = {col: 0.0 for col in self.positions.columns}
        
        # For each date, execute orders and calculate costs
        for i, date in enumerate(self.positions.index):
            if i == 0:
                # First day, just take positions
                for asset in self.positions.columns:
                    current_positions[asset] = self.positions.loc[date, asset]
            else:
                # Calculate changes from previous positions
                for asset in self.positions.columns:
                    target = self.positions.loc[date, asset]
                    current = current_positions[asset]
                    
                    # Calculate trade size
                    trade_size = abs(target - current)
                    
                    # Calculate cost
                    cost = trade_size * self.transaction_cost
                    transaction_costs.loc[date, 'Cost'] += cost
                    
                    # Update position (reduced by transaction cost)
                    if target > current:
                        current_positions[asset] = target - cost
                    elif target < current:
                        current_positions[asset] = target + cost
                    else:
                        current_positions[asset] = target
            
            # Record actual positions
            for asset in self.positions.columns:
                actual_positions.loc[date, asset] = current_positions[asset]
        
        return actual_positions, transaction_costs

class BacktestEngine:
    """Engine for backtesting alpha strategies"""
    
    def __init__(self, positions: pd.DataFrame, price_data: pd.DataFrame, transaction_costs: pd.DataFrame):
        """
        Initialize with positions, price data, and transaction costs
        
        Args:
            positions: DataFrame of portfolio positions
            price_data: DataFrame of asset prices
            transaction_costs: DataFrame of transaction costs
        """
        self.positions = positions
        self.price_data = price_data
        self.transaction_costs = transaction_costs
    
    def calculate_returns(self) -> pd.DataFrame:
        """Calculate daily returns of the strategy"""
        # Calculate asset returns
        asset_returns = self.price_data.pct_change()
        
        # Calculate portfolio returns (dot product of positions and returns)
        portfolio_returns = pd.DataFrame(index=self.positions.index, columns=['Return'], data=0.0)
        
        for date in portfolio_returns.index:
            if date in asset_returns.index and not asset_returns.loc[date].isna().all():
                # Skip first day
                if date == portfolio_returns.index[0]:
                    continue
                
                # Calculate return as dot product of positions and asset returns
                pos = self.positions.loc[date]
                ret = asset_returns.loc[date]
                
                # Handling any NaN values
                valid_assets = ~ret.isna()
                if any(valid_assets):
                    portfolio_returns.loc[date, 'Return'] = np.sum(pos[valid_assets] * ret[valid_assets])
        
        # Subtract transaction costs
        portfolio_returns['Return'] -= self.transaction_costs['Cost']
        
        return portfolio_returns
    
    def calculate_statistics(self, returns: pd.DataFrame) -> Dict[str, float]:
        """Calculate performance statistics"""
        # Skip first 20 days as warm-up period
        returns = returns.iloc[20:]
        
        # Annualization factor (assuming 252 trading days)
        annual_factor = 252
        
        # Calculate statistics
        total_return = (1 + returns['Return']).prod() - 1
        annual_return = (1 + total_return) ** (annual_factor / len(returns)) - 1
        annual_volatility = returns['Return'].std() * np.sqrt(annual_factor)
        sharpe_ratio = annual_return / annual_volatility if annual_volatility > 0 else 0
        
        # Calculate maximum drawdown
        cum_returns = (1 + returns['Return']).cumprod()
        max_drawdown = (cum_returns / cum_returns.cummax() - 1).min()
        
        # Calculate win rate
        win_rate = (returns['Return'] > 0).mean()
        
        return {
            'Total Return': total_return,
            'Annual Return': annual_return,
            'Annual Volatility': annual_volatility,
            'Sharpe Ratio': sharpe_ratio,
            'Maximum Drawdown': max_drawdown,
            'Win Rate': win_rate
        }
    
    def plot_performance(self, returns: pd.DataFrame) -> None:
        """Plot cumulative returns of the strategy"""
        # Calculate cumulative returns
        cum_returns = (1 + returns['Return']).cumprod()
        
        # Plot
        plt.figure(figsize=(12, 6))
        plt.plot(cum_returns.index, cum_returns.values)
        plt.title('Cumulative Returns of Alpha Strategy')
        plt.xlabel('Date')
        plt.ylabel('Cumulative Return')
        plt.grid(True)
        plt.tight_layout()
        plt.show()

class AlphaFactorGenerator:

    def __init__(self, price_data: pd.DataFrame, fundamental_data: pd.DataFrame):

        Returns:
            DataFrame of generated factors
        """
        # Initialize factors DataFrame
        factors = pd.DataFrame(index=self.price_data.index, columns=self.price_data.columns)
        
        # Generate factors
        for i in range(n_factors):
            # Simulate different factor types
            if i == 0:
                # Value factor based on P/E ratio
                factor = self._generate_value_factor()
            elif i == 1:
                # Momentum factor
                factor = self._generate_momentum_factor()
            else:
                # Volatility factor
                factor = self._generate_volatility_factor()
            
            # Add some noise to simulate LLM imperfections
            factor = factor + pd.DataFrame(
                np.random.normal(0, 0.2, factor.shape),
                index=factor.index,
                columns=factor.columns
            )
            
            # Normalize factor
            factor = factor.apply(lambda x: (x - x.mean()) / x.std(), axis=1)
            
            # Add to factors
            factors = factors.add(factor, fill_value=0)
        
        # Normalize combined factors
        factors = factors.apply(lambda x: (x - x.mean()) / x.std(), axis=1)
        
        return factors
    
    def _generate_value_factor(self) -> pd.DataFrame:
        """Generate a value factor based on P/E ratio"""
        # Initialize factor DataFrame
        factor = pd.DataFrame(index=self.price_data.index, columns=self.price_data.columns)
        
        # For each asset, calculate P/E ratio
        for asset in self.price_data.columns:
            for date in self.price_data.index:
                # Find the most recent fundamental data before this date
                prev_quarters = self.fundamental_data.index.get_level_values(0)
                prev_quarter = prev_quarters[prev_quarters <= date].max() if any(prev_quarters <= date) else None
                
                if prev_quarter is not None:
                    try:
                        price = self.price_data.loc[date, asset]
                        earnings = self.fundamental_data.loc[(prev_quarter, asset), 'NetIncome']
                        
                        # P/E ratio (inverted so higher values = more value)
                        if earnings > 0:
                            factor.loc[date, asset] = earnings / price
                        else:
                            factor.loc[date, asset] = 0
                    except KeyError:
                        factor.loc[date, asset] = 0
        
        return factor
    
    def _generate_momentum_factor(self) -> pd.DataFrame:
        """Generate a momentum factor"""
        # Calculate returns
        returns = self.price_data.pct_change()
        
        # Calculate 20-day momentum
        momentum = returns.rolling(window=20).sum()
        
        return momentum
    
    def _generate_volatility_factor(self) -> pd.DataFrame:
        """Generate a volatility factor (low volatility anomaly)"""
        # Calculate returns
        returns = self.price_data.pct_change()
        
        # Calculate 20-day volatility
        volatility = returns.rolling(window=20).std()
        
        # Invert volatility (low vol anomaly)
        volatility = -1 * volatility
        
        return volatility
