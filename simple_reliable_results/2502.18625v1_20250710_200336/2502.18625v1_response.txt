                ]

                    # Trade volumes
                    features.at[i, f'total_buy_{window}s'] = sum(
                        window_trades[window_trades['side'] == 'buy']['size'])

                        window_trades[window_trades['side'] == 'sell']['size'])

                        if len(price_changes) > 1:

                            features.at[i, f'ret_autocov_{window}s'] = 0

    def backtest_strategy(self, model, features, ob_data, reversal_threshold=0.24):
        """Backtest the reversal strategy with the trained model."""
