class TradingEnvironment:
    def __init__(self, prices: np.ndarray, news_data: List[str], window_size: int = 10):
        self.prices = prices
        self.news_data = news_data
        self.window_size = window_size
        self.current_step = window_size

        """Reset environment to initial state"""
        self.current_step = self.window_size
        return self._get_state()
        
    def _get_state(self) -> np.ndarray:
        """Get current state representation"""
        return np.array(self.prices[self.current_step-self.window_size:self.current_step])
        
    def step(self, action: float) -> Tuple[np.ndarray, float, bool, dict]:
        """
        Take a step in the environment given an action
        Returns: (next_state, reward, done, info)
        """
        current_price = self.prices[self.current_step]
        
        # Move to next step
        self.current_step += 1
        done = self.current_step >= len(self.prices)

            next_state = self._get_state()
            reward = 0
        else:
            next_price = self.prices[self.current_step]
            reward = ((next_price / current_price) - 1) * action  # Reward is return * action direction
            next_state = self._get_state()

        info = {
            'price': current_price,
            'news': current_news
        }
        
        return next_state, reward, done, info

def calculate_metrics(returns: List[float]) -> Dict:
    """Calculate performance metrics for a trading strategy"""
    returns = np.array(returns)
    
    # Basic metrics
    total_return = np.prod(1 + returns) - 1
    annualized_return = (1 + total_return) ** (252 / len(returns)) - 1
    volatility = np.std(returns) * np.sqrt(252)
    
    # Risk metrics
    information_ratio = annualized_return / volatility if volatility != 0 else 0
    
    # CVaR calculation
    alpha = 0.05  # 5% worst cases
    sorted_returns = np.sort(returns)
    cvar_threshold = int(np.ceil(alpha * len(returns)))
    cvar = np.mean(sorted_returns[:cvar_threshold])
    
    # Rachev ratio (simplified)
    positive_returns = returns[returns > 0]
    negative_returns = returns[returns < 0]
    rachev_ratio = np.mean(positive_returns) / abs(np.mean(negative_returns)) if len(negative_returns) > 0 else 1
    
    return {
        'total_return': total_return,
        'annualized_return': annualized_return,
        'volatility': volatility,
        'information_ratio': information_ratio,
        'cvar': cvar,
        'rachev_ratio': rachev_ratio
