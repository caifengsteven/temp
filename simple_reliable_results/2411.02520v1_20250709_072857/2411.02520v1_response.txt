@dataclass
class ModelParams:
    S0: float = 1.0        # Initial asset price
    V0: float = 0.1        # Initial variance
    f0: float = 1.0        # Local volatility parameter
    f1: float = -0.1       # Local volatility parameter
    x0: float = 0.0        # Local volatility parameter
    sigma: float = 2.0     # Volatility of volatility
    rho: float = 0.0       # Correlation
    r: float = 0.0         # Risk-free rate
    q: float = 0.0         # Dividend yield
    
    def eta(self, S: float) -> float:
        """Local volatility function"""
        return self.f0 + self.f1 * np.tanh(np.log(S/self.S0) - self.x0)
    
    def eta_prime(self, S: float) -> float:
        """Derivative of local volatility function"""
        x = np.log(S/self.S0) - self.x0
        return self.f1 * (1 - np.tanh(x)**2) / S
    
    def eta_coeffs(self):
        """Get coefficients of the Taylor expansion of eta"""
        eta0 = self.f0 - self.f1 * np.tanh(self.x0)
        eta1 = self.f1 / (np.cosh(self.x0)**2)
        eta2 = self.f1 / (np.cosh(self.x0)**2) * np.tanh(self.x0)
        return eta0, eta1, eta2


def simulate_lsv_paths(
    params: ModelParams, 
    T: float, 
    dt: float, 
    n_paths: int
) -> tuple:
    """
    Simulate paths for local-stochastic volatility model
    
    dS/S = eta(S)*sqrt(V)*(rho*dZ + sqrt(1-rho^2)*dW)
    dV/V = sigma*dZ
    """
    n_steps = int(T / dt)
    
    # Initialize paths
    S = np.ones((n_paths, n_steps+1)) * params.S0
    V = np.ones((n_paths, n_steps+1)) * params.V0
    
    # Generate correlated Brownian motions
    Z = np.random.normal(0, np.sqrt(dt), (n_paths, n_steps))
    W = np.random.normal(0, np.sqrt(dt), (n_paths, n_steps))
    
    # Simulate paths
    for i in range(n_steps):
        # Compute local volatility for each path
        eta = np.array([params.eta(s) for s in S[:, i]])
        
        # Update V first (log-normal process)
        V[:, i+1] = V[:, i] * np.exp(params.sigma * Z[:, i] - 0.5 * params.sigma**2 * dt)
        
        # Update S using Euler-Maruyama scheme
        dW_corr = params.rho * Z[:, i] + np.sqrt(1 - params.rho**2) * W[:, i]
        S[:, i+1] = S[:, i] * np.exp(
            eta * np.sqrt(V[:, i]) * dW_corr - 
            0.5 * (eta * np.sqrt(V[:, i]))**2 * dt
        )
    
    return S, V


def calculate_realized_variance(S: np.ndarray, V: np.ndarray, params: ModelParams, dt: float) -> np.ndarray:
    """Calculate realized variance for each path"""
    n_paths, n_steps = S.shape[0], S.shape[1] - 1
    realized_var = np.zeros(n_paths)
    
    for i in range(n_paths):
        # Calculate quadratic variation of log-price
        eta_squared = np.array([params.eta(s)**2 for s in S[i, :-1]])
        realized_var[i] = np.mean(eta_squared * V[i, :-1])
        
    return realized_var


def price_variance_options(
    realized_var: np.ndarray, 
    K: float, 
    r: float, 
    T: float
) -> tuple:
    """Price call and put options on realized variance"""
    call_payoffs = np.maximum(realized_var - K, 0)
    put_payoffs = np.maximum(K - realized_var, 0)
    
    call_price = np.exp(-r * T) * np.mean(call_payoffs)
    put_price = np.exp(-r * T) * np.mean(put_payoffs)
    
    return call_price, put_price


def implied_volatility(price: float, K: float, forward: float, T: float, option_type: str = 'call') -> float:
    """Calculate implied volatility using bisection method"""
    def bs_price(sigma):
        d1 = (np.log(forward/K) + 0.5 * sigma**2 * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        if option_type == 'call':
            return forward * norm.cdf(d1) - K * norm.cdf(d2)
        else:
            return K * norm.cdf(-d2) - forward * norm.cdf(-d1)
    
    def objective(sigma):
        return bs_price(sigma) - price
    
    # Use bisection to find the implied vol
    a, b = 0.001, 10.0  # Lower and upper bounds
    
    # Check if solution exists within bounds
    if objective(a) * objective(b) > 0:
        return np.nan
    
    while b - a > 1e-6:
        c = (a + b) / 2
        if objective(c) == 0:
            return c
        if objective(c) * objective(a) < 0:
            b = c
        else:
            a = c
    
    return (a + b) / 2


def calculate_asymptotic_vol(params: ModelParams, K: float) -> float:
    """Calculate asymptotic implied vol based on the theory in the paper"""
    eta0, eta1, eta2 = params.eta_coeffs()
    sigma0 = params.sigma
    V0 = params.V0
    rho = params.rho
    
    # ATM forward is eta0^2 * V0
    FV0 = eta0**2 * V0
    x = np.log(K / FV0)  # Log-moneyness
    
    # Calculate ATM vol level
    D = sigma0**2 + 4*eta1*sigma0*rho*np.sqrt(V0) + 4*eta1**2*V0
    vol_atm = 1/np.sqrt(3) * np.sqrt(D)
    
    # Calculate ATM skew
    beta0 = 16 * eta1**2 * V0**2 * (eta1**2 + 6*eta0*eta2)
    beta1 = 8 * eta1 * rho * V0 * (3*eta1*rho*sigma0 + 7*eta1**2*np.sqrt(V0) + 12*eta0*eta2*np.sqrt(V0))
    beta2 = 4 * (6*eta1*rho*sigma0*np.sqrt(V0) + 6*eta0*eta2*rho**2*V0 + eta1**2*(5 + 7*rho**2)*V0)
    
    numerator = (sigma0**4 + 2*sigma0**3*(3*0 + 7*eta1*rho*np.sqrt(V0)) + 
                beta2*sigma0**2 + beta1*sigma0 + beta0)
    
    skew = numerator / (10*np.sqrt(3)*(D)**(3/2))
    
    # Linear approximation
    vol = vol_atm + skew * x
    
    return vol
