import random
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from dataclasses import dataclass
from typing import List, Dict, Tuple, Optional
from enum import Enum

# Define enums and data models based on the paper's concepts
class RoleType(Enum):
    HUMAN = "human"
    AI = "ai"

class TaskType(Enum):
    PERCEPTION = "perception"  # Level 1 SA
    COMPREHENSION = "comprehension"  # Level 2 SA
    PROJECTION = "projection"  # Level 3 SA
    PLANNING = "planning"
    COORDINATION = "coordination"
    EXECUTION = "execution"

class FeedbackType(Enum):
    DESCRIPTIVE = "descriptive"
    NORMATIVE = "normative"
    CORRECTIVE = "corrective"
    EVALUATIVE = "evaluative"

@dataclass
class TeamMember:
    id: int
    role_type: RoleType
    perception_skill: float  # 0-1 scale
    comprehension_skill: float  # 0-1 scale
    projection_skill: float  # 0-1 scale
    task_specialization: List[TaskType]
    role_fluidity: float  # 0-1 scale, ability to adapt to changing roles

@dataclass
class Task:
    id: int
    name: str
    type: TaskType
    complexity: float  # 0-1 scale
    required_skills: Dict[str, float]
    interdependence: float  # 0-1 scale, how much this task depends on others

@dataclass
class TeamInteraction:
    human_member: TeamMember
    ai_member: TeamMember
    task: Task
    feedback_type: FeedbackType
    trust_level: float  # 0-1 scale
    shared_mental_model: float  # 0-1 scale
    performance: Optional[float] = None  # 0-1 scale, computed later

def generate_team_members(num_humans: int, num_ais: int) -> List[TeamMember]:
    """Generate a team with human and AI members with varied skills."""
    members = []
    
    # Generate human members
    for i in range(num_humans):
        members.append(TeamMember(
            id=i,
            role_type=RoleType.HUMAN,
            perception_skill=random.uniform(0.6, 0.9),
            comprehension_skill=random.uniform(0.5, 0.95),
            projection_skill=random.uniform(0.4, 0.9),
            task_specialization=random.sample(list(TaskType), k=random.randint(2, 4)),
            role_fluidity=random.uniform(0.5, 0.9)
        ))
    
    # Generate AI members
    for i in range(num_humans, num_humans + num_ais):
        members.append(TeamMember(
            id=i,
            role_type=RoleType.AI,
            perception_skill=random.uniform(0.7, 1.0),
            comprehension_skill=random.uniform(0.6, 0.9),
            projection_skill=random.uniform(0.5, 0.8),
            task_specialization=random.sample(list(TaskType), k=random.randint(3, 5)),
            role_fluidity=random.uniform(0.7, 1.0)
        ))
    
    return members

def generate_tasks(num_tasks: int) -> List[Task]:
    """Generate a list of tasks with varying complexity and skill requirements."""
    tasks = []
    task_types = list(TaskType)
    
    for i in range(num_tasks):
        # Create required skills with random values
        skills = {
            "perception": random.uniform(0.3, 0.8),
            "comprehension": random.uniform(0.3, 0.9),
            "projection": random.uniform(0.3, 0.7)
        }
        
        tasks.append(Task(
            id=i,
            name=f"Task_{i}",
            type=random.choice(task_types),
            complexity=random.uniform(0.3, 0.9),
            required_skills=skills,
            interdependence=random.uniform(0.2, 0.8)
        ))
    
    return tasks

def calculate_performance(interaction: TeamInteraction) -> float:
    """
    Calculate team performance based on member skills, task requirements,
    trust level, and shared mental model.
    """
    human = interaction.human_member
    ai = interaction.ai_member
    task = interaction.task
    
    # Base performance factors
    human_skill_match = (
        human.perception_skill * task.required_skills["perception"] +
        human.comprehension_skill * task.required_skills["comprehension"] +
        human.projection_skill * task.required_skills["projection"]
    ) / 3
    
    ai_skill_match = (
        ai.perception_skill * task.required_skills["perception"] +
        ai.comprehension_skill * task.required_skills["comprehension"] +
        ai.projection_skill * task.required_skills["projection"]
    ) / 3
    
    # Task type specialization bonus
    human_specialization = 0.2 if task.type in human.task_specialization else 0
    ai_specialization = 0.2 if task.type in ai.task_specialization else 0
    
    # Role fluidity impact (more important for complex tasks)
    fluidity_impact = (human.role_fluidity + ai.role_fluidity) / 2 * task.complexity
    
    # Trust and shared mental model impact
    trust_impact = interaction.trust_level * 0.15
    smm_impact = interaction.shared_mental_model * 0.15
    
    # Task complexity penalty
    complexity_penalty = task.complexity * 0.1
    
    # Calculate final performance (0-1 scale)
    performance = (
        (human_skill_match * 0.3) + 
        (ai_skill_match * 0.3) + 
        human_specialization + 
        ai_specialization + 
        fluidity_impact + 
        trust_impact + 
        smm_impact - 
        complexity_penalty
    )
    
    # Normalize to 0-1 range and add some randomness
    performance = max(0, min(1, performance + random.uniform(-0.05, 0.05)))
    
    return performance

def simulate_team_interactions(
    members: List[TeamMember], 
    tasks: List[Task], 
    num_interactions: int
) -> List[TeamInteraction]:
    """Simulate interactions between team members on tasks."""
    interactions = []
    feedback_types = list(FeedbackType)
    
    humans = [m for m in members if m.role_type == RoleType.HUMAN]
    ais = [m for m in members if m.role_type == RoleType.AI]
    
    for _ in range(num_interactions):
        human = random.choice(humans)
        ai = random.choice(ais)
        task = random.choice(tasks)
        
        # Initial trust and shared mental model levels
        trust_level = random.uniform(0.4, 0.9)
        shared_mental_model = random.uniform(0.4, 0.9)
        
        interaction = TeamInteraction(
            human_member=human,
            ai_member=ai,
            task=task,
            feedback_type=random.choice(feedback_types),
            trust_level=trust_level,
            shared_mental_model=shared_mental_model
        )
        
        # Calculate performance for this interaction
        interaction.performance = calculate_performance(interaction)
        interactions.append(interaction)
    
    return interactions

def analyze_results(interactions: List[TeamInteraction]) -> Dict:
    """Analyze the results of the simulated interactions."""
    results = {}
    
    # Convert to DataFrame for easier analysis
    data = []
    for i in interactions:
        data.append({
            "human_id": i.human_member.id,
            "ai_id": i.ai_member.id,
            "task_id": i.task.id,
            "task_type": i.task.type.value,
            "task_complexity": i.task.complexity,
            "feedback_type": i.feedback_type.value,
            "trust_level": i.trust_level,
            "shared_mental_model": i.shared_mental_model,
            "performance": i.performance
        })
    
    df = pd.DataFrame(data)
    
    # Average performance
    results["avg_performance"] = df["performance"].mean()
    
    # Performance by task type
    results["performance_by_task_type"] = df.groupby("task_type")["performance"].mean().to_dict()
    
    # Performance by feedback type
    results["performance_by_feedback"] = df.groupby("feedback_type")["performance"].mean().to_dict()
    
    # Correlation between trust and performance
    results["trust_performance_corr"] = df[["trust_level", "performance"]].corr().iloc[0, 1]
    
    # Correlation between shared mental model and performance
    results["smm_performance_corr"] = df[["shared_mental_model", "performance"]].corr().iloc[0, 1]
    
    # Correlation between task complexity and performance
    results["complexity_performance_corr"] = df[["task_complexity", "performance"]].corr().iloc[0, 1]
    
    return results, df
