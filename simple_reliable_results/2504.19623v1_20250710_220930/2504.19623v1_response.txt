class EchoStateNetwork:
    def __init__(self, input_dim, state_dim=100, spectral_radius=0.4, leak_rate=0.9, 
                 input_scaling=0.005, state_sparsity=0.15, input_sparsity=0.95):
        """
        Initialize Echo State Network parameters
        
        Parameters:
        - input_dim: dimension of input signals
        - state_dim: dimension of reservoir state
        - spectral_radius: spectral radius of reservoir matrix
        - leak_rate: leaking rate
        - input_scaling: scaling of input weights
        - state_sparsity: sparsity of reservoir matrix
        - input_sparsity: sparsity of input matrix
        """
        self.input_dim = input_dim
        self.state_dim = state_dim
        self.spectral_radius = spectral_radius
        self.leak_rate = leak_rate
        self.input_scaling = input_scaling
        
        # Initialize reservoir matrix with sparse random values
        self.A = np.random.normal(0, 1, (state_dim, state_dim))
        mask = np.random.rand(state_dim, state_dim) > state_sparsity
        self.A = self.A * mask
        
        # Normalize to set spectral radius
        eigvals = np.linalg.eigvals(self.A)
        max_abs_eigval = np.max(np.abs(eigvals))
        self.A = self.A * (spectral_radius / max_abs_eigval)
        
        # Initialize input matrix with sparse random values
        self.C = np.random.uniform(-1, 1, (state_dim, input_dim))
        mask = np.random.rand(state_dim, input_dim) > input_sparsity
        self.C = self.C * mask
        self.C = self.C * input_scaling
        
        # Initialize state
        self.X = np.zeros(state_dim)
        
        # Ridge regression model for output layer
        self.model = Ridge()
        
    def reset_state(self):
        """Reset reservoir state to zeros"""
        self.X = np.zeros(self.state_dim)
        
    def update_state(self, z):
        """
        Update reservoir state with new input
        
        Parameters:
        - z: input signal vector
        """
        if np.isnan(z).any():  # Handle missing data with state decay
            z = np.zeros_like(z)
        
        # Update state according to ESN equation
        self.X = self.leak_rate * self.X + (1 - self.leak_rate) * np.tanh(np.dot(self.A, self.X) + np.dot(self.C, z))
        return self.X
    
    def fit(self, X_states, y, alpha=0.1):
        """
        Fit output layer weights using ridge regression
        
        Parameters:
        - X_states: array of reservoir states
        - y: target values
        - alpha: regularization parameter
        """
        self.model = Ridge(alpha=alpha)
        self.model.fit(X_states, y)
        return self
    
    def predict(self, X_states):
        """
        Make predictions using the trained output weights
        
        Parameters:
        - X_states: array of reservoir states
        """
        return self.model.predict(X_states)


class SignalConstructor:
    def __init__(self, n_factors=15, window_sizes=[10, 20, 30, 60, 100, 150]):
        """
        Initialize Signal Constructor
        
        Parameters:
        - n_factors: number of market factors to extract
        - window_sizes: different time windows for signal construction
        """
        self.n_factors = n_factors
        self.window_sizes = window_sizes
        self.pca = PCA(n_components=n_factors)
        
    def extract_factors(self, returns_df):
        """
        Extract market factors using PCA
        
        Parameters:
        - returns_df: dataframe with stock returns
        """
        # Standardize returns
        z_returns = (returns_df - returns_df.mean()) / returns_df.std()
        
        # Apply PCA to get market factors
        self.pca.fit(z_returns.fillna(0))
        factors = self.pca.transform(z_returns.fillna(0))
        
        return factors
    
    def compute_residuals(self, returns_df, factors):
        """
        Compute residuals after removing market factor effects
        
        Parameters:
        - returns_df: dataframe with stock returns
        - factors: extracted market factors
        """
        residuals = {}
        
        for stock in returns_df.columns:
            # Skip stocks with all NaN values
            if returns_df[stock].isna().all():
                continue
                
            # Regress returns on factors
            valid_idx = ~returns_df[stock].isna()
            if valid_idx.sum() < self.n_factors + 1:
                continue
                
            X = factors[valid_idx]
            y = returns_df[stock][valid_idx]
            
            model = Ridge(alpha=1e-6)
            model.fit(X, y)
            
            # Calculate residuals
            residuals[stock] = y - model.predict(X)
            
        return pd.DataFrame(residuals)
    
    def construct_signals(self, residuals_df):
        """
        Construct signals based on residuals and multiple window sizes
        
        Parameters:
        - residuals_df: dataframe with return residuals
        """
        signals = {}
        
        for stock in residuals_df.columns:
            stock_signals = []
            
            for P in self.window_sizes:
                # Skip if not enough data
                if len(residuals_df) < P + 1:
                    stock_signals.append(np.nan * np.ones(len(residuals_df)))
                    continue
                
                # Create cumulative residuals
                cum_residuals = residuals_df[stock].rolling(P).sum()
                
                # Fit AR(1) model to estimate OU process parameters
                X = cum_residuals[P:-1].values.reshape(-1, 1)
                y = cum_residuals[P+1:].values
                
                if len(X) < 2:  # Skip if not enough data
                    stock_signals.append(np.nan * np.ones(len(residuals_df)))
                    continue
                
                model = Ridge(alpha=1e-6)
                model.fit(X, y)
                
                c0 = model.intercept_
                cu = model.coef_[0]
                
                # Calculate OU process parameters
                kappa = -np.log(cu) if cu > 0 else 0.1
                m = c0 / (1 - cu) if abs(cu) < 1 else 0
                
                # Get residuals from AR(1) model
                residuals = y - (c0 + cu * X.flatten())
                sigma = np.sqrt(np.var(residuals) / (2 * kappa))
                
                # Calculate z-score signal
                z_score = (cum_residuals - m) / sigma if sigma > 0 else np.zeros_like(cum_residuals)
                
                stock_signals.append(z_score)
            
            # Combine signals from different window sizes
            combined_signals = np.vstack([s[:len(residuals_df)] for s in stock_signals]).T
            signals[stock] = combined_signals
            
        return signals


class MultiHorizonForecaster:
    def __init__(self, horizons, signal_constructor, esn_params):
        """
        Initialize multi-horizon forecaster
        
        Parameters:
        - horizons: list of forecast horizons
        - signal_constructor: SignalConstructor instance
        - esn_params: dictionary of ESN parameters for each horizon
        """
        self.horizons = horizons
        self.signal_constructor = signal_constructor
        self.esn_params = esn_params
        self.esn_models = {}
        self.linear_models = {}
        self.baseline_models = {}
        
        # Initialize models for each horizon
        for h in horizons:
            params = esn_params[h]
            self.esn_models[h] = EchoStateNetwork(
                input_dim=len(signal_constructor.window_sizes),
                state_dim=params['state_dim'],
                spectral_radius=params['spectral_radius'],
                leak_rate=params['leak_rate'],
                input_scaling=params['input_scaling'],
                state_sparsity=params['state_sparsity'],
                input_sparsity=params['input_sparsity']
            )
            self.linear_models[h] = Ridge()
            self.baseline_models[h] = Ridge(alpha=0)
            
    def prepare_horizon_data(self, returns_df, signals, horizon, window_size, buffer_size):
        """
        Prepare data for a specific forecast horizon
        
        Parameters:
        - returns_df: dataframe with stock returns
        - signals: dictionary of signals for each stock
        - horizon: forecast horizon
        - window_size: training window size
        - buffer_size: buffer size for training
        """
        X_states = []
        X_signals = []
        y = []
        
        # Get unique timestamps and sort
        timestamps = sorted(returns_df.index.unique())
        
        for t_idx, t in enumerate(timestamps[:-horizon]):
            if t_idx < buffer_size:
                continue
                
            # Get signals at time t
            t_signals = {}
            for stock in signals:
                if t in signals[stock]:
                    t_signals[stock] = signals[stock][t]
            
            # Get future returns at time t + horizon
            t_h = timestamps[t_idx + horizon]
            future_returns = returns_df.loc[t_h]
            
            # For each stock, store data
            for stock in t_signals:
                if stock in future_returns and not np.isnan(future_returns[stock]):
                    # For ESN model
                    self.esn_models[horizon].reset_state()
                    
                    # Process window of signals to build state
                    for w in range(max(0, t_idx - window_size), t_idx + 1):
                        w_t = timestamps[w]
                        if w_t in signals[stock]:
                            self.esn_models[horizon].update_state(signals[stock][w_t])
                    
                    X_states.append(self.esn_models[horizon].X)
                    X_signals.append(t_signals[stock])
                    y.append(future_returns[stock])
        
        return np.array(X_states), np.array(X_signals), np.array(y)
    
    def train_and_evaluate(self, returns_df, test_start_idx=0.7, cv_folds=5):
        """
        Train models and evaluate performance

        # Extract factors and compute residuals

        split_idx = int(len(returns_df) * test_start_idx)

        test_idx = returns_df.index[split_idx:]
        
        results = {}
        
        for h in self.horizons:
            print(f"Training for horizon: {h}")
            
            # Define window and buffer sizes based on horizon
            if h in ['10min', '30min']:
                window_size = 30
                buffer_size = int(h.replace('min', '')) // 10
            elif h in ['60min', '2hr']:
                window_size = 60
                buffer_size = int(h.replace('min', '').replace('hr', '60')) // 10
            else:  # 'eod'
                window_size = 39  # One trading day
                buffer_size = 39
            
            # Prepare data for this horizon
            X_states, X_signals, y = self.prepare_horizon_data(
                returns_df.loc[train_idx],

                h,
                window_size,
                buffer_size
            )
            
            # Skip if not enough data
            if len(X_states) < 10:
                print(f"Not enough data for horizon {h}, skipping...")
                continue
            
            # Cross-validation for hyperparameter tuning
            tscv = TimeSeriesSplit(n_splits=cv_folds)
            best_alpha = 0.1
            best_score = float('-inf')
            
            for alpha in [0.001, 0.01, 0.1, 1.0, 10.0]:
                cv_scores = []
                
                for train, val in tscv.split(X_states):
                    # Train ESN model
                    self.esn_models[h].fit(X_states[train], y[train], alpha=alpha)
                    
                    # Predict and evaluate
                    y_pred = self.esn_models[h].predict(X_states[val])
                    score = -mean_squared_error(y[val], y_pred)
                    cv_scores.append(score)
                
                avg_score = np.mean(cv_scores)
                if avg_score > best_score:
                    best_score = avg_score
                    best_alpha = alpha
            
            # Train final models with best alpha
            self.esn_models[h].fit(X_states, y, alpha=best_alpha)
            self.linear_models[h].set_params(alpha=best_alpha)
            self.linear_models[h].fit(X_signals, y)
            self.baseline_models[h].fit(X_signals, y)
            
            # Prepare test data
            X_test_states, X_test_signals, y_test = self.prepare_horizon_data(

                window_size,
                buffer_size
            )
            
            if len(X_test_states) == 0:

            # Make predictions
            y_pred_esn = self.esn_models[h].predict(X_test_states)

            r2_baseline = r2_score(y_test, y_pred_baseline)
            
            results[h] = {
                'MSFE': {
                    'ESN': msfe_esn,
                    'Linear': msfe_linear,
                    'Baseline': msfe_baseline
                },
                'R2': {
                    'ESN': r2_esn,

                    'Baseline': r2_baseline

def generate_synthetic_data(n_stocks=50, n_timepoints=1000, volatility=0.01, 
                           mean_reversion=0.1, market_influence=0.7):
    """
    Generate synthetic stock price data
    
    Parameters:
    - n_stocks: number of stocks
    - n_timepoints: number of time points
    - volatility: volatility parameter
    - mean_reversion: mean reversion parameter
    - market_influence: influence of market factors
    """
    np.random.seed(42)
    
    # Generate timestamps (10-minute intervals during trading hours)
    start_date = datetime(2013, 1, 1, 9, 30)
    timestamps = []
    current_date = start_date
    
    for _ in range(n_timepoints):
        if current_date.hour < 9 or (current_date.hour == 9 and current_date.minute < 30) or current_date.hour >= 16:
            # Move to next trading day at 9:30 AM
            days_to_add = 1
            if current_date.weekday() >= 4:  # Friday or Saturday
                days_to_add = 7 - current_date.weekday()
            current_date = datetime(current_date.year, current_date.month, current_date.day, 9, 30) + timedelta(days=days_to_add)
        
        timestamps.append(current_date)
        current_date += timedelta(minutes=10)
    
    # Generate market factors (15 as mentioned in the paper)
    n_factors = 15
    market_factors = np.random.randn(n_timepoints, n_factors) * volatility
    
    # Generate stock-specific betas for market factors
    betas = np.random.randn(n_stocks, n_factors) * market_influence
    
    # Generate stock returns
    stock_returns = np.zeros((n_timepoints, n_stocks))
    stock_prices = np.ones((n_timepoints, n_stocks)) * 100  # Start with price of 100
    
    # Generate OU process parameters for each stock
    ou_kappa = np.random.uniform(0.05, 0.2, n_stocks)  # Mean reversion rate
    ou_mean = np.random.uniform(-0.001, 0.001, n_stocks)  # Long-term mean
    ou_sigma = np.random.uniform(0.005, 0.02, n_stocks)  # Volatility
    
    # Initialize residual processes
    residuals = np.zeros(n_stocks)
    
    for t in range(n_timepoints):
        # Update residuals according to OU process
        residuals = residuals + mean_reversion * (ou_mean - residuals) + ou_sigma * np.random.randn(n_stocks)
        
        # Calculate returns (market factors + residuals)
        market_return_component = np.dot(market_factors[t], betas.T)
        stock_returns[t] = market_return_component + residuals
        
        # Update prices
        if t > 0:
            stock_prices[t] = stock_prices[t-1] * (1 + stock_returns[t])
    
    # Convert to DataFrame
    returns_df = pd.DataFrame(stock_returns, index=timestamps, 
                              columns=[f'Stock_{i}' for i in range(n_stocks)])
    
    prices_df = pd.DataFrame(stock_prices, index=timestamps,
                            columns=[f'Stock_{i}' for i in range(n_stocks)])
    
    # Add some missing values to simulate real-world conditions
    mask = np.random.rand(n_timepoints, n_stocks) > 0.95
    returns_df = returns_df.mask(mask)
    
    return returns_df, prices_df


def calculate_multi_horizon_returns(prices_df, horizons):
    """
    Calculate returns for multiple horizons
    
    Parameters:
    - prices_df: dataframe with stock prices
    - horizons: list of horizons in minutes
    """
    horizon_returns = {}
    
    for h in horizons:
        if h == 'eod':
            # For end-of-day, we need to group by date
            eod_returns = {}
            for date, group in prices_df.groupby(prices_df.index.date):
                last_price = group.iloc[-1]
                for t, row in group.iterrows():
                    eod_return = (last_price / row - 1)
                    eod_returns[t] = eod_return
            horizon_returns[h] = pd.DataFrame(eod_returns).T
        else:
            # For intraday horizons
            if 'min' in h:
                periods = int(h.replace('min', '')) // 10
            elif 'hr' in h:
                periods = int(h.replace('hr', '')) * 6  # 6 ten-minute periods per hour
            
            horizon_returns[h] = prices_df.pct_change(periods=periods).shift(-periods)
    
    return horizon_returns
