import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import torch
import torch.nn as nn
import torch.optim as optim

from sklearn.metrics import mean_squared_error, mean_absolute_error
import math

# Set random seeds for reproducibility
np.random.seed(42)
torch.manual_seed(42)

class StockDataset(Dataset):
    def __init__(self, data, seq_length=24):
        """
        Dataset for stock price prediction
        Args:
            data: Pandas DataFrame with multivariate time series data
            seq_length: Number of time steps to look back
        """
        self.data = data
        self.seq_length = seq_length
        
    def __len__(self):
        return len(self.data) - self.seq_length
        
    def __getitem__(self, idx):
        # Get sequence of inputs (multiple stocks)
        x = self.data[idx:idx+self.seq_length].values
        # Get target (only the target stock's next value)
        y = self.data.iloc[idx+self.seq_length, -1]  # Last column is target stock
        return torch.FloatTensor(x), torch.FloatTensor([y])

class PositionalEncoding(nn.Module):
    def __init__(self, d_model, max_len=5000):
        super(PositionalEncoding, self).__init__()
        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, d_model, 2).float() * (-math.log(10000.0) / d_model))
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        pe = pe.unsqueeze(0)
        self.register_buffer('pe', pe)
        
    def forward(self, x):
        return x + self.pe[:, :x.size(1), :]

class Stockformer(nn.Module):
    def __init__(self, input_dim, hidden_dim, num_layers, num_heads, dropout=0.1):
        super(Stockformer, self).__init__()
        
        # 1D CNN for temporal feature extraction (as per paper's token embedding)
        self.conv1d = nn.Conv1d(
            in_channels=input_dim, 
            out_channels=hidden_dim,
            kernel_size=3, 
            padding=1, 
            padding_mode='circular',
            stride=1
        )
        
        self.positional_encoding = PositionalEncoding(hidden_dim)
        
        # Transformer encoder
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim,
            nhead=num_heads,
            dim_feedforward=4*hidden_dim,
            dropout=dropout,
            batch_first=True
        )
        self.transformer_encoder = nn.TransformerEncoder(
            encoder_layer,
            num_layers=num_layers
        )
        
        # Output layer
        self.output_layer = nn.Linear(hidden_dim, 1)
        
    def forward(self, x):
        # x shape: [batch_size, seq_len, input_dim]
        
        # Transpose for 1D CNN: [batch_size, input_dim, seq_len]
        x = x.transpose(1, 2)
        
        # Apply 1D CNN: [batch_size, hidden_dim, seq_len]
        x = self.conv1d(x)
        
        # Transpose back: [batch_size, seq_len, hidden_dim]
        x = x.transpose(1, 2)
        
        # Add positional encoding
        x = self.positional_encoding(x)
        
        # Apply transformer encoder
        x = self.transformer_encoder(x)
        
        # Use last sequence element for prediction
        x = x[:, -1, :]
        
        # Apply output layer
        x = self.output_layer(x)
        
        return x

def generate_simulated_stock_data(num_stocks=7, num_samples=1000):
    """
    Generate simulated stock price percent changes for multiple stocks
    with some correlation between them.
    """
    # Create a covariance matrix to simulate correlations between stocks
    # Higher values in correlation_factor means stocks are more correlated
    correlation_factor = 0.7
    cov_matrix = np.eye(num_stocks) * 1.0
    for i in range(num_stocks):
        for j in range(num_stocks):
            if i != j:
                # Introduce some correlation between stocks
                cov_matrix[i, j] = correlation_factor if np.random.random() > 0.3 else 0.2
    
    # Generate multivariate normal data for log percent changes
    log_percent_changes = np.random.multivariate_normal(
        mean=np.zeros(num_stocks) + 0.0003,  # Slight upward bias
        cov=cov_matrix * 0.0001,  # Scale down for realistic volatility
        size=num_samples
    )
    
    # Create a pandas DataFrame with the generated data
    stock_names = [f"Stock_{chr(65+i)}" for i in range(num_stocks-1)] + ["Target_Stock"]
    df = pd.DataFrame(log_percent_changes, columns=stock_names)
    
    # Add some seasonal patterns and trends
    for i, stock in enumerate(stock_names):
        # Add a trend component
        trend = np.linspace(0, 0.1, num_samples) * np.random.uniform(0.5, 1.5)
        # Add a seasonal component (e.g., a sine wave)
        seasonality = np.sin(np.linspace(0, 10*np.pi, num_samples)) * 0.002
        df[stock] += trend + seasonality
    
    # Add some lagged effects between stocks (causal relationships)
    for i in range(1, num_stocks):
        # Stock i is influenced by previous values of other stocks
        for j in range(i):
            # With some random lag between 1 and 5 time steps
            lag = np.random.randint(1, 6)
            if lag < num_samples:
                weight = np.random.uniform(0.1, 0.3) * (-1 if np.random.random() > 0.5 else 1)
                df.iloc[lag:, i] += weight * df.iloc[:-lag, j].values
    
    return df

def evaluate_model(model, test_loader, device):
    """Evaluate the model on test data and calculate metrics"""
    model.eval()
    predictions = []
    actuals = []
    
    with torch.no_grad():
        for inputs, targets in test_loader:
            inputs, targets = inputs.to(device), targets.to(device)
            outputs = model(inputs)
            predictions.extend(outputs.cpu().numpy())
            actuals.extend(targets.cpu().numpy())
    
    # Convert to numpy arrays
    predictions = np.array(predictions).flatten()
    actuals = np.array(actuals).flatten()
    
    # Calculate metrics
    mse = mean_squared_error(actuals, predictions)
    mae = mean_absolute_error(actuals, predictions)
    
    # Calculate ROI metrics
    roi_actual = np.exp(np.cumsum(actuals))[-1]
    
    # Stock Direction ROI
    direction_predictions = np.sign(predictions)
    roi_direction = np.exp(np.sum(direction_predictions * actuals))
    
    # Stock Tanh ROI
    tanh_predictions = np.tanh(predictions)
    roi_tanh = np.exp(np.sum(tanh_predictions * actuals))
    
    return {
        'MSE': mse,
        'MAE': mae,
        'ROI_Actual': roi_actual,
        'ROI_Direction': roi_direction,
        'ROI_Tanh': roi_tanh
    }

def train_and_evaluate():
    # Generate simulated data
    print("Generating simulated stock data...")
    data = generate_simulated_stock_data(num_stocks=7, num_samples=1000)
    
    # Split into train and test sets
    train_size = int(len(data) * 0.8)
    train_data = data[:train_size]
    test_data = data[train_size:]

    seq_length = 24  # Hours to look back
    train_dataset = StockDataset(train_data, seq_length)
    test_dataset = StockDataset(test_data, seq_length)
    
    # Create data loaders
    batch_size = 32
    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
    test_loader = DataLoader(test_dataset, batch_size=batch_size, shuffle=False)
    
    # Set device
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    print(f"Using device: {device}")
    
    # Model parameters
    input_dim = data.shape[1]  # Number of stocks
    hidden_dim = 256
    num_layers = 3
    num_heads = 8
    
    # Initialize model
    model = Stockformer(
        input_dim=input_dim,
        hidden_dim=hidden_dim,
        num_layers=num_layers,
        num_heads=num_heads
    ).to(device)
    
    # Loss function and optimizer
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=1e-4)
    
    # Learning rate scheduler
    scheduler = optim.lr_scheduler.ReduceLROnPlateau(
        optimizer, mode='min', factor=0.5, patience=5, verbose=True
    )
    
    # Training loop
    num_epochs = 20
    train_losses = []
    test_metrics = []
    
    print("Starting training...")
    for epoch in range(num_epochs):
        model.train()
        epoch_loss = 0
        
        for inputs, targets in train_loader:
            inputs, targets = inputs.to(device), targets.to(device)
            
            # Forward pass
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            
            # Backward pass and optimize
            optimizer.zero_grad()
            loss.backward()
            # Clip gradients to prevent exploding gradients
            torch.nn.utils.clip_grad_norm_(model.parameters(), 0.5)
            optimizer.step()
            
            epoch_loss += loss.item()
        
        # Average loss for the epoch
        epoch_loss /= len(train_loader)
        train_losses.append(epoch_loss)
        
        # Evaluate on test set
        metrics = evaluate_model(model, test_loader, device)
        test_metrics.append(metrics)
        
        # Update learning rate based on validation loss
        scheduler.step(metrics['MSE'])
        
        print(f"Epoch {epoch+1}/{num_epochs}, Loss: {epoch_loss:.6f}, Test MSE: {metrics['MSE']:.6f}, ROI Tanh: {metrics['ROI_Tanh']:.4f}")
    
    # Final evaluation
    final_metrics = evaluate_model(model, test_loader, device)
    print("\nFinal Test Metrics:")
    for metric, value in final_metrics.items():

    # Baseline comparison - always predict zero
    zeros_predictions = np.zeros(len(test_dataset))

    zeros_mse = mean_squared_error(actuals, zeros_predictions)
    zeros_roi = np.exp(np.sum(np.zeros_like(actuals) * actuals))
    
    print("\nBaseline (Zeros) Metrics:")
    print(f"MSE: {zeros_mse:.6f}")
    print(f"ROI: {zeros_roi:.6f}")
    
    # Plot results
    plt.figure(figsize=(12, 8))
    
    # Plot training loss
    plt.subplot(2, 2, 1)

    # Plot test MSE
    plt.subplot(2, 2, 2)
    plt.plot([m['MSE'] for m in test_metrics])
    plt.axhline(y=zeros_mse, color='r', linestyle='--', label='Baseline (Zeros)')
    plt.title('Test MSE')

    # Plot ROI comparison
    plt.subplot(2, 1, 2)
    
    # Extract actual returns
    actual_returns = np.exp(np.cumsum(actuals))
    # Extract returns using Tanh strategy
    model.eval()
    with torch.no_grad():
        all_inputs = torch.stack([test_dataset[i][0] for i in range(len(test_dataset))])
        all_inputs = all_inputs.to(device)
        all_outputs = model(all_inputs).cpu().numpy().flatten()
    
    tanh_predictions = np.tanh(all_outputs)
    tanh_returns = np.exp(np.cumsum(tanh_predictions * actuals))
    
    plt.plot(actual_returns, label='Buy & Hold')
    plt.plot(tanh_returns, label='Stockformer (Tanh)')
    plt.axhline(y=1.0, color='r', linestyle='--', label='No Change')
    plt.title('Cumulative Returns')

    plt.ylabel('Cumulative Return')
