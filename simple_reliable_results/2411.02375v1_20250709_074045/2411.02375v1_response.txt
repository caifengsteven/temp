        activation = x
        
        # Hidden layers with tanh activation
        for i in range(self.hidden_layers):
            z = np.dot(activation, self.weights[i]) + self.biases[i]
            activation = self.tanh(z)
            
        # Output layer with softplus activation (ensuring positive IV)
        z = np.dot(activation, self.weights[-1]) + self.biases[-1]

        return output
    
    def black_scholes_price(self, F, K, tau, sigma, phi=1):
        """
        Calculate Black-Scholes option price
        phi: 1 for call, -1 for put

        if tau <= 0 or sigma <= 0:
            return max(0, phi * (F - K))
        
        d1 = (np.log(F/K) + (sigma**2/2)*tau) / (sigma * np.sqrt(tau))
        d2 = d1 - sigma * np.sqrt(tau)
        
        return phi * (F * norm.cdf(phi * d1) - K * norm.cdf(phi * d2))

def generate_simulated_option_data(num_points=100, t=0.0):

    time_to_maturity = np.random.exponential(0.2, num_points) + 0.01
    time_to_maturity = np.minimum(time_to_maturity, 2.0)
    
    # Current time (constant for a batch)
    current_time = np.ones(num_points) * t

    # Generate realistic implied volatilities
    base_vol = 0.2
    skew = -0.4  # negative skew
    term_structure = 0.1  # upward sloping term structure
    
    # Volatility surface with skew and term structure
    impl_vol = base_vol + skew * log_moneyness + term_structure * np.sqrt(time_to_maturity)
    impl_vol = np.maximum(impl_vol, 0.05)  # Ensure positive volatility
    
    # Add some noise
    impl_vol += np.random.normal(0, 0.01, num_points)
    
    # Calculate option prices
    F = 100  # Arbitrary forward price
    K = F * np.exp(log_moneyness)
    
    # Black-Scholes price calculation
    d1 = (np.log(F/K) + (impl_vol**2/2)*time_to_maturity) / (impl_vol * np.sqrt(time_to_maturity))
    d2 = d1 - impl_vol * np.sqrt(time_to_maturity)
    
    # For simplicity, calculate call option prices
    prices = F * norm.cdf(d1) - K * norm.cdf(d2)
    
    return x_data, prices, impl_vol

            price_minus = model.black_scholes_price(F, K[i-1], tau, iv_minus)
            
            delta[i] = (price_plus - price_minus) / (K[i+1] - K[i-1])

        gamma = np.zeros_like(log_moneyness)

            gamma[i] = (price_plus - 2*price_current + price_minus) / ((K[i+1] - K[i-1])/2)**2
