def calculate_triple_difference(pre_boom: List[District], post_boom: List[District]) -> Dict:
    """
    Calculate triple difference estimates for various outcomes
    (High Windfalls x Post-Boom x High State Capacity)
    """
    # Convert to pandas DataFrames for easier analysis
    pre_df = pd.DataFrame([vars(d) for d in pre_boom])
    post_df = pd.DataFrame([vars(d) for d in post_boom])
    
    # Add period indicator
    pre_df['post_boom'] = 0
    post_df['post_boom'] = 1
    
    # Combine data
    df = pd.concat([pre_df, post_df])

    df['high_sc'] = df['tax_per_capita'] > 14.0
    df['high_windfalls'] = df['windfall_transfers'] > 133.0
    
    # Prepare results dictionary
    results = {}
    
    # Function to calculate triple difference for a given outcome
    def calc_triple_diff(outcome):
        # Group 1: High SC, High Windfalls
        g1_pre = pre_df[(pre_df['tax_per_capita'] > 14.0) & (pre_df['windfall_transfers'] > 133.0)][outcome].mean()
        g1_post = post_df[(post_df['tax_per_capita'] > 14.0) & (post_df['windfall_transfers'] > 133.0)][outcome].mean()
        
        # Group 2: Low SC, High Windfalls
        g2_pre = pre_df[(pre_df['tax_per_capita'] <= 14.0) & (pre_df['windfall_transfers'] > 133.0)][outcome].mean()
        g2_post = post_df[(post_df['tax_per_capita'] <= 14.0) & (post_df['windfall_transfers'] > 133.0)][outcome].mean()
        
        # Group 3: High SC, Low Windfalls
        g3_pre = pre_df[(pre_df['tax_per_capita'] > 14.0) & (pre_df['windfall_transfers'] <= 133.0)][outcome].mean()
        g3_post = post_df[(post_df['tax_per_capita'] > 14.0) & (post_df['windfall_transfers'] <= 133.0)][outcome].mean()
        
        # Group 4: Low SC, Low Windfalls
        g4_pre = pre_df[(pre_df['tax_per_capita'] <= 14.0) & (pre_df['windfall_transfers'] <= 133.0)][outcome].mean()
        g4_post = post_df[(post_df['tax_per_capita'] <= 14.0) & (post_df['windfall_transfers'] <= 133.0)][outcome].mean()
        
        # Calculate differences
        diff_g1 = g1_post - g1_pre
        diff_g2 = g2_post - g2_pre
        diff_g3 = g3_post - g3_pre
        diff_g4 = g4_post - g4_pre
        
        # First difference (Treatment vs Control for High SC)
        diff_diff_high_sc = diff_g1 - diff_g3
        
        # Second difference (Treatment vs Control for Low SC)
        diff_diff_low_sc = diff_g2 - diff_g4
        
        # Triple difference
        triple_diff = diff_diff_high_sc - diff_diff_low_sc
        
        return {
            'triple_diff': triple_diff,
            'diff_diff_high_sc': diff_diff_high_sc,

            }
        }
    
    # Calculate triple differences for key outcomes
    outcomes = ['household_income', 'roads_stock', 'sewage_access', 
                'agriculture_share', 'services_share', 'protests_per_capita']
