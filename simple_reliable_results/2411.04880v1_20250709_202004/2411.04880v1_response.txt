class ESM:
    """Simplified Energy System Model that generates market clearing prices (MCP)"""

class DNN:
    """Deep Neural Network for electricity price forecasting"""
    
    def __init__(self, layers=[64, 32, 16], dropout_rate=0.2, 
                 learning_rate=0.001, activation='relu'):
        self.layers = layers
        self.dropout_rate = dropout_rate
        self.learning_rate = learning_rate
        self.activation = activation
        self.model = None
        self.scaler_X = StandardScaler()
        self.scaler_y = StandardScaler()
        
    def build_model(self, input_dim):
        model = Sequential()
        
        # First hidden layer
        model.add(Dense(self.layers[0], input_dim=input_dim, 
                         activation=self.activation))
        model.add(BatchNormalization())
        model.add(Dropout(self.dropout_rate))
        
        # Additional hidden layers
        for units in self.layers[1:]:
            model.add(Dense(units, activation=self.activation))
            model.add(BatchNormalization())
            model.add(Dropout(self.dropout_rate))
        
        # Output layer
        model.add(Dense(24))  # 24 outputs for hourly prices
        
        # Compile model
        model.compile(
            loss='mse',
            optimizer=Adam(learning_rate=self.learning_rate)
        )
        
        self.model = model
        return model
    
    def fit(self, X, y, epochs=100, batch_size=32, verbose=0):
        # Scale inputs and outputs
        X_scaled = self.scaler_X.fit_transform(X)
        y_scaled = self.scaler_y.fit_transform(y)
        
        # Build model if not already built
        if self.model is None:
            self.build_model(X_scaled.shape[1])
        
        # Train the model
        self.model.fit(
            X_scaled, y_scaled,
            epochs=epochs,
            batch_size=batch_size,
            verbose=verbose
        )
    
    def predict(self, X):
        # Scale inputs
        X_scaled = self.scaler_X.transform(X)
        
        # Generate predictions
        y_pred_scaled = self.model.predict(X_scaled)
        
        # Inverse transform to get actual prices
        y_pred = self.scaler_y.inverse_transform(y_pred_scaled)
        
        return y_pred

def calculate_metrics(y_true, y_pred):
    """Calculate evaluation metrics for electricity price forecasting"""
    # Flatten arrays if necessary
    if y_true.ndim > 1:
        y_true_flat = y_true.flatten()
        y_pred_flat = y_pred.flatten()
    else:
        y_true_flat = y_true
        y_pred_flat = y_pred
    
    # Calculate metrics
    mae = mean_absolute_error(y_true_flat, y_pred_flat)
    rmse = np.sqrt(mean_squared_error(y_true_flat, y_pred_flat))
    
    # Calculate sMAPE (symmetric Mean Absolute Percentage Error)
    smape = 200 * np.mean(
        np.abs(y_pred_flat - y_true_flat) / 
        (np.abs(y_pred_flat) + np.abs(y_true_flat))
    )
    
    return {
        'MAE': mae,
        'RMSE': rmse,
        'sMAPE': smape
    }


def test_storage_optimization(prices, price_forecasts, storage_capacity=1, energy_to_capacity=3, efficiency=0.8):
    """
    Test storage optimization using price forecasts
    
    Args:
        prices: Actual prices (n_days x 24)
        price_forecasts: Forecasted prices (n_days x 24)
        storage_capacity: Storage power capacity in MW
        energy_to_capacity: Energy to capacity ratio (hours of storage)
        efficiency: Round-trip efficiency (0-1)
    
    Returns:
        dict: Performance metrics
    """
    n_days = prices.shape[0]
    total_profit_perfect = 0
    total_profit_forecast = 0
    
    for day in range(n_days):
        # Get actual and forecasted prices for the day
        actual_prices = prices[day]
        forecast_prices = price_forecasts[day]
        
        # Optimize with perfect foresight
        perfect_profit, perfect_charge, perfect_discharge = optimize_storage(
            actual_prices, 
            storage_capacity, 
            energy_to_capacity, 
            efficiency
        )
        
        # Optimize with forecasted prices
        _, forecast_charge, forecast_discharge = optimize_storage(
            forecast_prices, 
            storage_capacity, 
            energy_to_capacity, 
            efficiency
        )
        
        # Calculate profit with forecasted strategy but actual prices
        forecast_profit = sum(actual_prices * forecast_discharge) - sum(actual_prices * forecast_charge)
        
        # Accumulate profits
        total_profit_perfect += perfect_profit
        total_profit_forecast += forecast_profit
    
    # Calculate efficiency ratio
    efficiency_ratio = total_profit_forecast / total_profit_perfect if total_profit_perfect > 0 else 0
    
    return {
        'total_profit_perfect': total_profit_perfect,
        'total_profit_forecast': total_profit_forecast,
        'efficiency_ratio': efficiency_ratio
    }


def optimize_storage(prices, capacity=1, energy_to_capacity=3, efficiency=0.8):
    """
    Simple greedy algorithm for storage optimization (for a single day)
    
    In a real implementation, this would be a linear optimization problem
    as described in section 3.4 of the paper.
    """
    n_hours = len(prices)
    price_pairs = []
    
    # Find all potential charge-discharge pairs
    for charge_hour in range(n_hours):
        for discharge_hour in range(charge_hour + 1, n_hours):
            price_diff = prices[discharge_hour] - prices[charge_hour] / efficiency
            if price_diff > 0:  # Profitable to charge and discharge
                price_pairs.append((charge_hour, discharge_hour, price_diff))
    
    # Sort by price difference (most profitable first)
    price_pairs.sort(key=lambda x: x[2], reverse=True)
    
    # Initialize
    charge = np.zeros(n_hours)
    discharge = np.zeros(n_hours)
    storage_level = np.zeros(n_hours + 1)  # +1 to include final state
    
    # Process pairs in order of profitability
    for charge_hour, discharge_hour, _ in price_pairs:
        # Check if we can charge (not already discharging, and capacity available)
        if discharge[charge_hour] == 0 and charge[charge_hour] < capacity:
            # Check if we can discharge (not already charging, and capacity available)
            if charge[discharge_hour] == 0 and discharge[discharge_hour] < capacity:
                # Calculate maximum possible charge/discharge
                max_charge = min(
                    capacity - charge[charge_hour],
                    capacity - discharge[discharge_hour],
                    (energy_to_capacity * capacity - max(storage_level[charge_hour:discharge_hour+1])) / efficiency
                )
                
                if max_charge > 0:
                    # Update charge and discharge
                    charge[charge_hour] += max_charge
                    discharge[discharge_hour] += max_charge * efficiency
                    
                    # Update storage level
                    for h in range(charge_hour + 1, discharge_hour + 1):
                        storage_level[h] += max_charge * efficiency
                    for h in range(discharge_hour + 1, n_hours + 1):
                        storage_level[h] += max_charge * efficiency - max_charge * efficiency
    
    # Calculate profit
    profit = sum(prices * discharge) - sum(prices * charge)
    
    return profit, charge, discharge
