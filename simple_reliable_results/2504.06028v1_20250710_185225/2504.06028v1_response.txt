import numpy as np
import pandas as pd

import statsmodels.api as sm

class OUExchangeRateModel:
    def __init__(self, theta=0.1, mu=0, sigma_k=0.02, sigma_s=0.01):
        """
        Initialize the Ornstein-Uhlenbeck Exchange Rate Model
        
        Parameters:
        -----------
        theta : float
            Mean reversion speed parameter
        mu : float
            Long-term mean of the risk premium
        sigma_k : float
            Volatility of the risk premium process
        sigma_s : float
            Base volatility of the exchange rate
        """
        self.theta = theta
        self.mu = mu
        self.sigma_k = sigma_k
        self.sigma_s = sigma_s
        self.params_fitted = False
    
    def fit(self, K_series):
        """
        Fit OU parameters to observed risk premium data
        
        Parameters:
        -----------
        K_series : pandas.Series
            Time series of realized risk premiums
        """
        # Fit OU process using maximum likelihood estimation
        dt = 1  # Assuming daily data
        y = K_series.values[1:]
        x = K_series.values[:-1]
        
        # Linear regression for AR(1) process
        model = sm.OLS(y, sm.add_constant(x))
        results = model.fit()
        
        # Extract parameters
        a = results.params[1]
        b = results.params[0]
        sigma = np.sqrt(np.var(results.resid))
        
        # Convert to OU parameters
        self.theta = -np.log(a) / dt
        self.mu = b / (1 - a)
        self.sigma_k = sigma * np.sqrt(2 * self.theta / (1 - a**2))
        
        self.params_fitted = True
        
        return {
            'theta': self.theta,
            'mu': self.mu,
            'sigma_k': self.sigma_k
        }
    
    def forecast(self, S_current, K_current, int_diff, horizon):
        """
        Generate forecast distribution for future exchange rate
        
        Parameters:
        -----------
        S_current : float
            Current exchange rate
        K_current : float
            Current risk premium
        int_diff : float
            Interest rate differential (iUS - iKR)
        horizon : int
            Forecast horizon in days
        
        Returns:
        --------
        dict
            Mean and variance of forecast distribution
        """
        if not self.params_fitted:
            print("Warning: Model not fitted. Using initialization parameters.")
        
        # Mean of log exchange rate
        log_s_mean = np.log(S_current) + int_diff * horizon + \
                     (K_current / self.theta) * (1 - np.exp(-self.theta * horizon)) + \
                     self.mu * (horizon - (1 - np.exp(-self.theta * horizon)) / self.theta)
        
        # Variance of log exchange rate
        log_s_var = self.sigma_s**2 * horizon + \
                    (self.sigma_k**2 / (2 * self.theta)) * (1 - np.exp(-2 * self.theta * horizon))
        
        return {
            'mean': log_s_mean,
            'variance': log_s_var
        }
    
    def confidence_interval(self, forecast, confidence=0.95):
        """
        Generate confidence interval for exchange rate forecast
        
        Parameters:
        -----------
        forecast : dict
            Forecast mean and variance
        confidence : float
            Confidence level (default: 0.95)
        
        Returns:
        --------
        tuple
            Lower and upper bounds of the confidence interval
        """
        z_score = norm.ppf((1 + confidence) / 2)
        log_s_mean = forecast['mean']
        log_s_std = np.sqrt(forecast['variance'])
        
        log_lower = log_s_mean - z_score * log_s_std
        log_upper = log_s_mean + z_score * log_s_std
        
        return np.exp(log_lower), np.exp(log_upper)


def simulate_ou_process(theta, mu, sigma, S0, K0, int_diff, days, dt=1):
    """
    Simulate exchange rate with OU risk premium
    
    Parameters:
    -----------
    theta : float
        Mean reversion speed
    mu : float
        Long-term mean of risk premium
    sigma : float
        Volatility of risk premium
    S0 : float
        Initial exchange rate
    K0 : float
        Initial risk premium
    int_diff : float
        Interest rate differential
    days : int
        Number of days to simulate
    dt : float
        Time step in days
        
    Returns:
    --------
    tuple
        Arrays of dates, exchange rates, and risk premiums
    """
    # Initialize arrays
    steps = int(days / dt)
    times = np.linspace(0, days, steps+1)
    
    # Create date range
    start_date = datetime(2010, 1, 1)
    dates = [start_date + timedelta(days=t) for t in times]
    
    # Initialize arrays for S and K
    S = np.zeros(steps+1)
    K = np.zeros(steps+1)
    
    # Set initial values
    S[0] = S0
    K[0] = K0
    
    # Generate Brownian motion increments
    dW_S = np.random.normal(0, np.sqrt(dt), steps)
    dW_K = np.random.normal(0, np.sqrt(dt), steps)
    
    # Simulate processes
    for i in range(steps):
        # Update K (risk premium) using Euler-Maruyama method
        K[i+1] = K[i] + theta * (mu - K[i]) * dt + sigma * dW_K[i]
        
        # Update S (exchange rate)
        S[i+1] = S[i] * np.exp((int_diff + K[i]) * dt + 0.01 * dW_S[i])
    
    return dates, S, K


def calculate_realized_premium(exchange_rates, int_diff, horizon):
    """
    Calculate realized risk premium from exchange rate data
    
    Parameters:
    -----------
    exchange_rates : pandas.Series
        Time series of exchange rates
    int_diff : float or pandas.Series
        Interest rate differential
    horizon : int
        Forecast horizon in days
    
    Returns:
    --------
    pandas.Series
        Realized risk premium series
    """
    # Calculate log returns over horizon
    log_returns = np.log(exchange_rates.shift(-horizon) / exchange_rates)
    
    # Handle int_diff as either constant or time series
    if isinstance(int_diff, (int, float)):
        int_diff_h = int_diff * horizon
    else:
        int_diff_h = int_diff * horizon  # Simplified assumption
    
    # Calculate realized premium
    K = log_returns - int_diff_h
    
    return K


def evaluate_coverage(model, data, int_diff, horizons, confidence_levels):
    """
    Evaluate forecast coverage rates across different horizons and confidence levels
    
    Parameters:
    -----------
    model : OUExchangeRateModel
        Fitted model
    data : pandas.DataFrame
        DataFrame with 'S' (exchange rate) and 'K' (risk premium) columns
    int_diff : float
        Interest rate differential
    horizons : list
        List of forecast horizons to evaluate
    confidence_levels : list
        List of confidence levels to evaluate
    
    Returns:
    --------
    pandas.DataFrame
        Coverage rates by horizon and confidence level
    """
    results = {}
    
    for horizon in horizons:
        coverage_by_level = {}
        
        # Extract testing period data
        test_data = data.iloc[:-horizon].copy()  # Avoid lookahead bias
        
        for level in confidence_levels:
            hits = 0
            total = len(test_data)
            
            for i in range(total):
                # Current values
                S_current = test_data.iloc[i]['S']
                K_current = test_data.iloc[i]['K']
                
                # Generate forecast
                forecast = model.forecast(S_current, K_current, int_diff, horizon)
                lower, upper = model.confidence_interval(forecast, level)
                
                # Actual future value
                if i + horizon < len(data):
                    S_future = data.iloc[i + horizon]['S']
                    
                    # Check if actual value falls within interval
                    if lower <= S_future <= upper:
                        hits += 1
            
            # Calculate coverage rate
            coverage_by_level[level] = hits / total * 100
        
        results[horizon] = coverage_by_level
    
    # Convert to DataFrame
    return pd.DataFrame(results)
