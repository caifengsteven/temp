class Parameters:

class MarketState:
    """Represents the current state of the market"""
    time: float
    inventory: float
    price: float
    price_deviation: float

class HawkesProcessSimulator:
    """Simulates a Hawkes process for order arrivals"""
    
    def __init__(self, params: Parameters):
        self.params = params
        self.base_intensity = params.lambda_base
        self.kappa = params.kappa
        self.epsilon = params.epsilon
        
    def simulate_intensities(self, time_points: np.ndarray, regime: int) -> Tuple[np.ndarray, np.ndarray]:
        """Simulate the buy and sell intensities over time for a given regime"""
        dt = time_points[1] - time_points[0]
        n_steps = len(time_points)
        
        # Initialize intensities
        buy_intensity = np.zeros(n_steps)
        sell_intensity = np.zeros(n_steps)
        
        # Set initial values based on regime
        if regime == 0:  # High buy, low sell regime
            buy_intensity[0] = self.base_intensity * 1.5
            sell_intensity[0] = self.base_intensity * 0.5
        else:  # Low buy, high sell regime
            buy_intensity[0] = self.base_intensity * 0.5
            sell_intensity[0] = self.base_intensity * 1.5
        
        # Generate jumps for the processes
        for i in range(1, n_steps):
            # Mean-reverting drift
            buy_drift = -self.kappa * (buy_intensity[i-1] - self.base_intensity) * dt
            sell_drift = -self.kappa * (sell_intensity[i-1] - self.base_intensity) * dt
            
            # Random jumps based on current intensities
            buy_jump = np.random.poisson(buy_intensity[i-1] * dt)
            sell_jump = np.random.poisson(sell_intensity[i-1] * dt)
            
            # Self-excitation effect
            buy_self_excite = buy_jump * self.epsilon
            sell_self_excite = sell_jump * self.epsilon
            
            # Cross-excitation effect
            buy_cross_excite = sell_jump * self.epsilon * 0.5
            sell_cross_excite = buy_jump * self.epsilon * 0.5
            
            # Update intensities
            buy_intensity[i] = buy_intensity[i-1] + buy_drift + buy_self_excite + buy_cross_excite
            sell_intensity[i] = sell_intensity[i-1] + sell_drift + sell_self_excite + sell_cross_excite
            
            # Ensure intensities stay positive
            buy_intensity[i] = max(buy_intensity[i], 0.1)
            sell_intensity[i] = max(sell_intensity[i], 0.1)
        
        return buy_intensity, sell_intensity


class MarkovChainSimulator:
    """Simulates a hidden Markov chain for liquidity regimes"""
    
    def __init__(self, params: Parameters):
        self.transition_matrix = params.transition_matrix
        self.n_states = len(self.transition_matrix)
        
    def simulate_path(self, time_points: np.ndarray, initial_state: int = 0) -> np.ndarray:
        """Simulate a path of the Markov chain"""
        dt = time_points[1] - time_points[0]
        n_steps = len(time_points)
        states = np.zeros(n_steps, dtype=int)
        states[0] = initial_state
        
        for i in range(1, n_steps):
            current_state = states[i-1]
            transition_probs = np.zeros(self.n_states)
            
            # Compute transition probabilities
            for j in range(self.n_states):
                if j != current_state:
                    transition_probs[j] = self.transition_matrix[current_state, j] * dt
            
            # Probability of staying in the same state
            transition_probs[current_state] = 1 - sum(transition_probs)
            
            # Sample next state
            states[i] = np.random.choice(self.n_states, p=transition_probs)
        
        return states


class PriceImpactModel:
    """Implements the price impact model from the paper"""
    
    def __init__(self, params: Parameters):
        self.c = params.c
        self.e = params.e
        self.c0 = params.c0
        self.omega = params.omega
        self.alpha = params.alpha
        
    def market_impact(self, order_size: float) -> float:
        """Calculate the market impact Q(v) for an order of size v"""
        # Using the power-shaped density function from the paper
        # Q(v) = (e*v/c)^(1/e)
        return ((self.e * order_size) / self.c) ** (1/self.e)
    
    def transaction_cost(self, price: float, order_size: float) -> float:
        """Calculate the transaction cost C(p, v) for an order at price p with size v"""
        # C(p, v) = p*v - c'*v^((e+1)/e) - c0
        c_prime = (self.c / (self.e + 1)) * (self.e / self.c) ** ((self.e + 1) / self.e)
        return price * order_size - c_prime * (order_size ** ((self.e + 1) / self.e)) - self.c0
    
    def update_price(self, current_price: float, price_deviation: float, order_size: float) -> Tuple[float, float]:
        """Update the price after a trade of size order_size"""
        impact = self.market_impact(order_size)
        
        # Update the fundamental price component (permanent impact)
        new_price = current_price - self.omega * impact
        
        # Update the deviation component (transient impact)
        new_deviation = price_deviation - (1 - self.omega) * impact
        
        return new_price, new_deviation


class FilterEstimator:
    """Implements the filter for estimating the hidden state"""
    
    def __init__(self, params: Parameters):
        self.transition_matrix = params.transition_matrix
        self.n_states = len(self.transition_matrix)
        
    def update_filter(self, current_filter: np.ndarray, dt: float, 
                      observed_buy_intensity: float, observed_sell_intensity: float,
                      buy_jump: bool, sell_jump: bool) -> np.ndarray:
        """
        Update the filter based on new observations using a simplified version of 
        the Kushner-Stratonovich equations from Theorem 3.1
        """
        new_filter = current_filter.copy()
        
        # Prediction step (ODE between jumps)
        for i in range(self.n_states):
            drift = 0
            for j in range(self.n_states):
                if j != i:
                    drift += current_filter[j] * self.transition_matrix[j, i]
            drift += current_filter[i] * self.transition_matrix[i, i]
            new_filter[i] += drift * dt
        
        # Update step (at jumps)
        if buy_jump or sell_jump:
            # Simple update based on which intensity is higher in each regime
            if buy_jump:
                # Regime 0 has higher buy intensity
                likelihood_ratio = 1.5 / 0.5  # Approximation based on the regime characteristics
                new_filter[0] *= likelihood_ratio
            if sell_jump:
                # Regime 1 has higher sell intensity
                likelihood_ratio = 1.5 / 0.5  # Approximation based on the regime characteristics
                new_filter[1] *= likelihood_ratio
            
            # Normalize
            new_filter = new_filter / np.sum(new_filter)
        
        return new_filter


class OptimalLiquidationStrategy:
    """Implements a simplified version of the optimal liquidation strategy"""
    
    def __init__(self, params: Parameters, price_impact_model: PriceImpactModel):
        self.params = params
        self.price_impact = price_impact_model
        
    def determine_action(self, state: MarketState) -> Tuple[bool, float]:
        """
        Determine whether to trade and how much based on the current state
        Returns: (should_trade, trade_size)
        """
        remaining_time = self.params.T - state.time
        
        # Simplified logic based on the findings from the paper:
        # 1. Trade when the filter probability suggests a downward trend
        # 2. Trade more aggressively when inventory is high
        # 3. Trade more aggressively when time is running out
        
        # Estimate the current regime (0 = upward trend, 1 = downward trend)
        estimated_regime = 1 if state.filter_proba[1] > 0.5 else 0
        
        # Base trade probability on time remaining and regime
        base_trade_prob = min(1.0, (1.0 - remaining_time / self.params.T) + 0.3 * estimated_regime)
        
        # Adjust based on inventory level
        inventory_factor = state.inventory / self.params.initial_inventory
        trade_probability = base_trade_prob * (0.3 + 0.7 * inventory_factor)
        
        # Determine whether to trade
        should_trade = random.random() < trade_probability
        
        if not should_trade:
            return False, 0.0
        
        # Determine trade size based on findings from Figure 6 in the paper
        # Order sizes are initially proportional to inventory, then stabilize
        max_order_size = state.inventory
        
        # The concave impact function leads to smaller trades
        if state.inventory > 1.0:
            # For larger inventories, use a concave function for order size
            order_size = min(max_order_size, 0.5 * state.inventory ** 0.7)
        else:
            # For small inventories, liquidate more aggressively
            order_size = max_order_size
        
        return True, order_size


class MarketSimulator:

                observed_buy_intensities[i] = buy_intensities_regime1[i]
                observed_sell_intensities[i] = sell_intensities_regime1[i]

        for i in range(1, n_steps):

            buy_jump = np.random.poisson(observed_buy_intensities[i] * dt) > 0

            current_deviation *= np.exp(-self.params.alpha * dt)
            
            # Update filter based on observations
            current_filter = self.filter_estimator.update_filter(
