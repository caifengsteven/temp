import numpy as np

class LinearAutoencoder:
    """Linear autoencoder implementation for low-rank covariance approximation"""
    
    def __init__(self, input_dim, latent_dim, learning_rate=0.01):
        self.input_dim = input_dim
        self.latent_dim = latent_dim
        self.learning_rate = learning_rate
        
        # Initialize encoder and decoder weights
        self.A = np.random.randn(input_dim, latent_dim) * 0.1  # Decoder
        self.B = np.random.randn(latent_dim, input_dim) * 0.1  # Encoder
        
    def encode(self, X):
        """Encode input data to latent space"""
        return np.dot(X, self.B.T)
    
    def decode(self, Z):
        """Decode latent representation back to input space"""
        return np.dot(Z, self.A.T)
    
    def reconstruct(self, X):
        """Full reconstruction (encode then decode)"""
        return self.decode(self.encode(X))
    
    def train_step(self, X):
        """Single training step using simplified backpropagation"""
        # Forward pass
        Z = self.encode(X)
        X_hat = self.decode(Z)
        
        # Compute loss
        loss = np.mean(np.square(X - X_hat))
        
        # Compute gradients
        dX_hat = 2 * (X_hat - X) / X.shape[0]
        dA = np.dot(dX_hat.T, Z)
        dZ = np.dot(dX_hat, self.A)
        dB = np.dot(dZ.T, X)
        
        # Update weights
        self.A -= self.learning_rate * dA
        self.B -= self.learning_rate * dB
        
        return loss
    
    def train(self, X, epochs=100, verbose=True):
        """Train the autoencoder"""
        losses = []
        for epoch in range(epochs):
            loss = self.train_step(X)
            losses.append(loss)
            if verbose and (epoch % 10 == 0 or epoch == epochs - 1):
                print(f"Epoch {epoch+1}/{epochs}, Loss: {loss:.6f}")
        return losses
    
    def get_covariance_approximation(self, X):
        """Get low-rank approximation of covariance matrix"""
        # Encode data to get latent variables
        Z = self.encode(X)
        # Calculate covariance of latent variables
        P = np.cov(Z.T)
        # Low-rank approximation of covariance
        low_rank_cov = np.dot(np.dot(self.A, P), self.A.T)
        return low_rank_cov


class ContinuousHopfieldNetwork:
    """Continuous Hopfield Network for portfolio optimization"""
    
    def __init__(self, n_assets, cov_matrix, returns, annealing_rate=0.01):
        self.n_assets = n_assets
        self.cov_matrix = cov_matrix
        self.returns = returns
        self.annealing_rate = annealing_rate
        
    def initialize_weights(self):
        """Initialize weights randomly"""
        return np.random.uniform(0, 1, self.n_assets)
    
    def normalize_weights(self, weights):
        """Normalize weights to sum to 1"""
        return weights / np.sum(weights)
    
    def calculate_variance(self, weights):
        """Calculate portfolio variance"""
        return np.dot(np.dot(weights, self.cov_matrix), weights)
    
    def calculate_expected_return(self, weights):
        """Calculate expected portfolio return"""
        return np.dot(weights, self.returns)
    
    def energy_function(self, weights, desired_return, lambda1=1.0, lambda2=1.0):
        """Energy function to minimize (Eq. 4 in the paper)"""
        weights = self.normalize_weights(weights)
        variance = self.calculate_variance(weights)
        return_constraint = lambda1 * (self.calculate_expected_return(weights) - desired_return) ** 2
        sum_constraint = lambda2 * (np.sum(weights) - 1) ** 2
        
        return variance + return_constraint + sum_constraint
    
    def update_rule(self, weights, desired_return, t, max_t, lambda1=1.0, lambda2=1.0):
        """Update rule for weights based on Eq. 2 in the paper"""
        # Create the matrix J = -2Σ - 2λ1μμT - 2λ211T
        J = -2 * self.cov_matrix
        J -= 2 * lambda1 * np.outer(self.returns, self.returns)
        J -= 2 * lambda2 * np.ones((self.n_assets, self.n_assets))
        
        # Create vector m = 2Rλ1μ + 2λ21
        m = 2 * desired_return * lambda1 * self.returns + 2 * lambda2 * np.ones(self.n_assets)
        
        # Annealing parameter
        p = self.annealing_rate * (1 - t / max_t)
        
        # Compute update: dx/dt = -p(t)x + Jv + m
        # Here v = logistic(x), but we simplify and use v = x with constraints
        dx = -p * weights + np.dot(J, weights) + m
        
        # Apply update
        new_weights = weights + 0.1 * dx  # Step size 0.1
        
        # Ensure non-negativity
        new_weights = np.maximum(0, new_weights)
        
        # Normalize to sum to 1
        new_weights = self.normalize_weights(new_weights)
        
        return new_weights
    
    def optimize(self, desired_return, max_iterations=100):
        """Find optimal portfolio weights for a desired return"""
        weights = self.initialize_weights()
        weights = self.normalize_weights(weights)
        
        energy_values = [self.energy_function(weights, desired_return)]
        
        for t in range(max_iterations):
            weights = self.update_rule(weights, desired_return, t, max_iterations)
            energy = self.energy_function(weights, desired_return)
            energy_values.append(energy)
            
            # Convergence check
            if t > 0 and abs(energy_values[-1] - energy_values[-2]) < 1e-6:
                break
                
        return weights, energy_values


def generate_synthetic_data(n_assets=25, n_samples=50, mean=0.1, std=0.2):

    # Ensure diagonal is 1
    np.fill_diagonal(correlation, 1.0)
    
    # Create variances
    variances = np.random.uniform(0.1, 0.5, n_assets)
    
    # Create covariance matrix
    std_dev = np.sqrt(variances)
    covariance = np.outer(std_dev, std_dev) * correlation
    
    # Generate returns data
    returns_data = np.random.multivariate_normal(
        mean=np.ones(n_assets) * mean, 
        cov=covariance, 
        size=n_samples
    )
    
    # Expected returns (average of historical returns)
    expected_returns = np.mean(returns_data, axis=0)
    
    # Sample covariance
    sample_covariance = np.cov(returns_data, rowvar=False)
    
    return returns_data, expected_returns, sample_covariance, covariance

    # Compare with classical Markowitz solution
    print("Comparing with classical Markowitz solution...")
    
    def objective(weights):
        return np.sqrt(np.dot(np.dot(weights, cov_matrix), weights))
    
    def constraint_return(weights, target_return):
        return np.dot(weights, expected_returns) - target_return
    
    def constraint_sum(weights):
        return np.sum(weights) - 1.0
    
    markowitz_weights = []
    markowitz_returns = []
    markowitz_risks = []
    
    for target_return in returns_range:
        constraints = [
