        for i, name in enumerate(feature_names):

        df['short_term_target'] = short_term_y

        """Preprocess data by handling missing values and standardizing features."""
        # Forward fill missing values as mentioned in the paper
        df = df.ffill()
        
        # Select features (alphas)
        feature_cols = [col for col in df.columns if col.startswith('alpha')]

        X = df[feature_cols].values
        y = df[target_column].values

    def handle_imbalance(self, X_train, y_train, method='smote'):
        """Handle class imbalance using SMOTE or other techniques."""
        if method == 'smote':
            smote = SMOTE(random_state=self.random_state)
            X_resampled, y_resampled = smote.fit_resample(X_train, y_train)
            return X_resampled, y_resampled
        elif method == 'oversample':
            # Identify the minority and majority classes
            X_train_df = pd.DataFrame(X_train)
            X_train_df['target'] = y_train
            
            minority_class = X_train_df[X_train_df['target'] == 1]
            majority_class = X_train_df[X_train_df['target'] == 0]
            
            # Oversample the minority class
            minority_oversampled = resample(

                random_state=self.random_state
            )
            
            # Combine the oversampled minority class with the majority class
            oversampled_df = pd.concat([majority_class, minority_oversampled])
            
            y_resampled = oversampled_df['target'].values
            X_resampled = oversampled_df.drop('target', axis=1).values
            
            return X_resampled, y_resampled
        else:
            return X_train, y_train

        # Neural Network

        """Evaluate all models and return results."""
        results = {}
        
        for name, model in self.models.items():
            y_pred = model.predict(X_test)
            
            results[name] = {
