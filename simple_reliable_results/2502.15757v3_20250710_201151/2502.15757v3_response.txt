class BilinearNormalization(nn.Module):
    """
    Bilinear Normalization layer as described in the paper to handle non-stationarity
    and magnitude disparities in financial time series.
    """
    def __init__(self, input_dim):
        super(BilinearNormalization, self).__init__()
        self.alpha = nn.Parameter(torch.ones(input_dim))
        self.beta = nn.Parameter(torch.zeros(input_dim))
        
    def forward(self, x):
        # Compute batch statistics
        mean = x.mean(dim=0, keepdim=True)
        std = x.std(dim=0, keepdim=True) + 1e-5
        
        # Apply normalization
        x_norm = (x - mean) / std
        
        # Apply learnable parameters
        return self.alpha * x_norm + self.beta


class MLPBlock(nn.Module):
    """
    MLP block for feature and temporal mixing as described in the paper.
    """
    def __init__(self, dim):
        super(MLPBlock, self).__init__()
        self.fc1 = nn.Linear(dim, 4 * dim)
        self.fc2 = nn.Linear(4 * dim, dim)
        self.layer_norm = nn.LayerNorm(dim)
        
    def forward(self, x):
        residual = x
        x = self.layer_norm(x)
        x = F.gelu(self.fc1(x))
        x = self.fc2(x)
        return x + residual


class SelfAttention(nn.Module):
    """
    Self-attention mechanism used in the TLOB model.
    """
    def __init__(self, dim, heads=1):
        super(SelfAttention, self).__init__()
        self.dim = dim
        self.heads = heads
        self.head_dim = dim // heads
        
        self.q = nn.Linear(dim, dim)
        self.k = nn.Linear(dim, dim)
        self.v = nn.Linear(dim, dim)
        self.out = nn.Linear(dim, dim)
        
    def forward(self, x):
        batch_size, seq_len, _ = x.shape
        
        q = self.q(x).reshape(batch_size, seq_len, self.heads, self.head_dim).permute(0, 2, 1, 3)
        k = self.k(x).reshape(batch_size, seq_len, self.heads, self.head_dim).permute(0, 2, 1, 3)
        v = self.v(x).reshape(batch_size, seq_len, self.heads, self.head_dim).permute(0, 2, 1, 3)
        
        attention = torch.matmul(q, k.transpose(-1, -2)) / (self.head_dim ** 0.5)
        attention = F.softmax(attention, dim=-1)
        
        out = torch.matmul(attention, v).permute(0, 2, 1, 3).reshape(batch_size, seq_len, self.dim)
        out = self.out(out)
        
        return out


class TLOBBlock(nn.Module):
    """
    Dual-attention block as described in the TLOB paper.
    """
    def __init__(self, dim):
        super(TLOBBlock, self).__init__()
        self.layer_norm1 = nn.LayerNorm(dim)
        self.layer_norm2 = nn.LayerNorm(dim)
        
        self.temporal_attention = SelfAttention(dim)
        self.spatial_attention = SelfAttention(dim)
        self.mlp_block = MLPBlock(dim)
        
    def forward(self, x):
        # Temporal attention (across LOB snapshots)
        residual = x
        x = self.layer_norm1(x)
        x = self.temporal_attention(x) + residual
        
        # Spatial attention (across LOB features)
        residual = x
        x = self.layer_norm2(x)
        x = self.spatial_attention(x) + residual
        
        # MLP block
        x = self.mlp_block(x)
        
        return x


class TLOB(nn.Module):
    """
    TLOB model as described in the paper.
    """
    def __init__(self, input_dim, hidden_dim, num_layers, num_classes):
        super(TLOB, self).__init__()
        
        # Bilinear normalization layer
        self.bilinear_norm = BilinearNormalization(input_dim)
        
        # Initial projection
        self.projection = nn.Linear(input_dim, hidden_dim)
        
        # Positional encoding for temporal information
        self.register_buffer("pos_encoding", self._create_positional_encoding(512, hidden_dim))
        
        # TLOB blocks
        self.tlob_blocks = nn.ModuleList([TLOBBlock(hidden_dim) for _ in range(num_layers)])
        
        # Final prediction layers
        self.fc1 = nn.Linear(hidden_dim, hidden_dim // 2)
        self.fc2 = nn.Linear(hidden_dim // 2, hidden_dim // 4)
        self.fc3 = nn.Linear(hidden_dim // 4, num_classes)
        
    def _create_positional_encoding(self, max_seq_len, dim):
        pos_encoding = torch.zeros(max_seq_len, dim)
        position = torch.arange(0, max_seq_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, dim, 2).float() * (-np.log(10000.0) / dim))
        
        pos_encoding[:, 0::2] = torch.sin(position * div_term)
        pos_encoding[:, 1::2] = torch.cos(position * div_term)
        
        return pos_encoding.unsqueeze(0)
        
    def forward(self, x):
        # x shape: [batch_size, seq_len, input_dim]
        seq_len = x.size(1)
        
        # Apply bilinear normalization
        x = self.bilinear_norm(x)
        
        # Project to hidden dimension
        x = self.projection(x)
        
        # Add positional encoding
        x = x + self.pos_encoding[:, :seq_len, :]
        
        # Apply TLOB blocks
        for block in self.tlob_blocks:
            x = block(x)
        
        # Get the final time step
        x = x[:, -1, :]
        
        # Final prediction
        x = F.gelu(self.fc1(x))
        x = F.gelu(self.fc2(x))
        x = self.fc3(x)
        
        return x


class MLPLOB(nn.Module):
    """
    MLPLOB model as described in the paper.
    """
    def __init__(self, input_dim, hidden_dim, num_blocks, num_classes):
        super(MLPLOB, self).__init__()
        
        # Bilinear normalization layer
        self.bilinear_norm = BilinearNormalization(input_dim)
        
        # Initial projection
        self.projection = nn.Linear(input_dim, hidden_dim)
        
        # MLPLOB blocks
        self.blocks = nn.ModuleList()
        for _ in range(num_blocks):
            self.blocks.append(self._create_mlplob_block(hidden_dim))
        
        # Final prediction layers
        self.fc1 = nn.Linear(hidden_dim, hidden_dim // 2)
        self.fc2 = nn.Linear(hidden_dim // 2, hidden_dim // 4)
        self.fc3 = nn.Linear(hidden_dim // 4, num_classes)
        
    def _create_mlplob_block(self, dim):
        return nn.Sequential(
            # Feature-mixing MLP
            nn.LayerNorm(dim),
            nn.Linear(dim, 4 * dim),
            nn.GELU(),
            nn.Linear(4 * dim, dim),
            nn.LayerNorm(dim),
            
            # Temporal-mixing MLP
            nn.Linear(dim, 4 * dim),
            nn.GELU(),
            nn.Linear(4 * dim, dim)
        )
        
    def forward(self, x):
        # x shape: [batch_size, seq_len, input_dim]
        batch_size, seq_len, _ = x.shape
        
        # Apply bilinear normalization
        x = self.bilinear_norm(x)
        
        # Project to hidden dimension
        x = self.projection(x)
        
        # Apply MLPLOB blocks
        for block in self.blocks:
            # Feature-mixing (apply across each time step)
            # Reshape to handle each time step separately
            x_reshaped = x.reshape(batch_size * seq_len, -1)
            x_reshaped = block[:5](x_reshaped)
            x = x_reshaped.reshape(batch_size, seq_len, -1)
            
            # Temporal-mixing (apply across each feature dimension)
            # Transpose to handle each feature dimension separately
            x = x.transpose(1, 2)  # [batch_size, dim, seq_len]
            x_reshaped = x.reshape(batch_size * x.size(1), -1)
            x_reshaped = block[5:](x_reshaped)
            x = x_reshaped.reshape(batch_size, x.size(1), -1)
            x = x.transpose(1, 2)  # [batch_size, seq_len, dim]
        
        # Get the final time step
        x = x[:, -1, :]
        
        # Final prediction
        x = F.gelu(self.fc1(x))
        x = F.gelu(self.fc2(x))
        x = self.fc3(x)
        
        return x

def generate_simulated_lob_data(num_samples, num_levels=10, window_size=128):
    """
    Generate simulated LOB data based on the description in the paper.
    Each LOB record consists of ask prices, ask volumes, bid prices, and bid volumes
    for a specified number of levels.
    """
    # Initialize LOB data
    lob_data = np.zeros((num_samples, 4 * num_levels))
    
    # Generate initial mid-price
    mid_price = 100.0
    
    # Set the average spread as a percentage of mid-price
    avg_spread_pct = 0.01
    
    # Generate data for each sample
    for i in range(num_samples):
        # Randomly perturb mid-price (simulate price movement)
        mid_price += np.random.normal(0, 0.05)
        
        # Calculate spread based on mid-price
        spread = mid_price * avg_spread_pct
        
        # Calculate best bid and ask
        best_bid = mid_price - spread / 2
        best_ask = mid_price + spread / 2
        
        # Generate prices and volumes for each level
        for level in range(num_levels):
            # Ask prices increase from best ask
            lob_data[i, level] = best_ask + level * 0.01 * mid_price
            
            # Ask volumes (random with some structure)
            lob_data[i, num_levels + level] = max(10, np.random.exponential(100) * (1 - 0.05 * level))
            
            # Bid prices decrease from best bid
            lob_data[i, 2 * num_levels + level] = best_bid - level * 0.01 * mid_price
            
            # Bid volumes (random with some structure)
            lob_data[i, 3 * num_levels + level] = max(10, np.random.exponential(100) * (1 - 0.05 * level))
        
        # Add some noise
        lob_data[i] += np.random.normal(0, 0.01, 4 * num_levels)
    
    # Generate labels based on future price movements
    labels = generate_labels(lob_data, window_size=window_size, horizon=20, threshold=0.002)
    
    return lob_data, labels


def generate_labels(lob_data, window_size=128, horizon=20, threshold=0.002):
    """
    Generate labels for price movement prediction using the method described in the paper.
    
    Args:
        lob_data: numpy array of LOB data
        window_size: number of past LOB snapshots to consider
        horizon: prediction horizon
        threshold: threshold for classifying price movements
        
    Returns:
        numpy array of labels (0: down, 1: stable, 2: up)
    """
    num_samples = len(lob_data)
    labels = np.ones(num_samples) * 1  # Default: stable
    
    # Calculate mid-prices for each sample
    mid_prices = np.zeros(num_samples)
    for i in range(num_samples):
        best_ask = lob_data[i, 0]
        best_bid = lob_data[i, 2 * 10]
        mid_prices[i] = (best_ask + best_bid) / 2
    
    # Generate labels using the method from the paper
    for t in range(window_size, num_samples - horizon):
        # Past average (w-)
        w_minus = np.mean(mid_prices[t-window_size:t])
        
        # Future average (w+)
        w_plus = np.mean(mid_prices[t:t+horizon])
        
        # Calculate percentage change
        pct_change = (w_plus - w_minus) / w_minus
        
        # Classify based on threshold
        if pct_change > threshold:
            labels[t] = 2  # Up
        elif pct_change < -threshold:
            labels[t] = 0  # Down
        else:
            labels[t] = 1  # Stable
    
    return labels


def evaluate_model(model, dataloader, device):
    """
    Evaluate model performance on a dataset.
    """
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for x, y in dataloader:
            x, y = x.to(device), y.to(device)
            outputs = model(x)
            _, predicted = torch.max(outputs, 1)
            
            all_preds.extend(predicted.cpu().numpy())
            all_labels.extend(y.cpu().numpy())
    
    # Calculate F1 score
    f1 = f1_score(all_labels, all_preds, average='macro')
    
    return f1, all_preds, all_labels
