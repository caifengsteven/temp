import numpy as np
import pandas as pd

class AnalystReport:
    """Simulates an analyst report with numerical and textual data"""
    
    def __init__(self, ticker, date, recommendation=None, eps_forecast=None, target_price=None, 
                 text_embedding=None, report_type='revision', topic_weights=None):
        self.ticker = ticker
        self.date = date
        self.recommendation = recommendation  # 1-5 scale (1=Strong Buy, 5=Strong Sell)
        self.eps_forecast = eps_forecast
        self.target_price = target_price
        self.text_embedding = text_embedding if text_embedding is not None else np.random.randn(20)
        self.report_type = report_type  # 'revision' or 'reiteration'
        # Default topic weights - Income Statement Analysis has highest weight
        self.topic_weights = topic_weights if topic_weights is not None else {
            'Income Statement Analysis': 0.67,
            'Financial Ratios': 0.12,
            'Investment Thesis': 0.07,
            'Valuation': 0.05,
            'Risk Factors': 0.04,
            'Other': 0.05
        }
        
    def is_post_earnings(self, earnings_dates):
        """Check if report is released within 7 days after earnings"""
        for e_date in earnings_dates:
            if 0 <= (self.date - e_date).days <= 7:
                return True
        return False
        
    def is_bold(self, consensus_eps=None, previous_eps=None):
        """Check if analyst's forecast is bold (above/below both consensus and previous)"""
        if consensus_eps is None or previous_eps is None or self.eps_forecast is None:
            return False
        return ((self.eps_forecast > consensus_eps and self.eps_forecast > previous_eps) or 
                (self.eps_forecast < consensus_eps and self.eps_forecast < previous_eps))


class StockData:
    """Simulates stock price data and market reactions to analyst reports"""
    
    def __init__(self, ticker, start_date, end_date, initial_price=100):
        self.ticker = ticker
        self.start_date = start_date
        self.end_date = end_date
        self.initial_price = initial_price
        self.price_data = self._generate_price_data()
        self.earnings_dates = self._generate_earnings_dates()
        
    def _generate_price_data(self):
        """Generate daily price data with random walk"""
        days = (self.end_date - self.start_date).days + 1
        dates = [self.start_date + timedelta(days=i) for i in range(days)]
        
        # Generate random daily returns with slight positive drift
        daily_returns = np.random.normal(0.0003, 0.012, days)
        
        # Calculate cumulative returns and prices
        cum_returns = np.cumprod(1 + daily_returns)
        prices = self.initial_price * cum_returns
        
        # Create DataFrame
        df = pd.DataFrame({
            'date': dates,
            'price': prices,
            'daily_return': daily_returns
        })
        df['date'] = pd.to_datetime(df['date'])
        return df.set_index('date')
    
    def _generate_earnings_dates(self):
        """Generate quarterly earnings announcement dates"""
        earnings_dates = []
        current_date = self.start_date
        
        while current_date <= self.end_date:
            # Add some randomness to earnings date (around end of quarter)
            month = current_date.month
            if month in [3, 6, 9, 12]:
                # Random day in last week of month
                last_day = 31 if month in [3, 12] else 30
                earnings_day = np.random.randint(last_day-6, last_day)
                earnings_date = datetime(current_date.year, month, earnings_day)
                if self.start_date <= earnings_date <= self.end_date:
                    earnings_dates.append(earnings_date)
            
            # Move to next month
            if current_date.month == 12:
                current_date = datetime(current_date.year + 1, 1, 1)
            else:
                current_date = datetime(current_date.year, current_date.month + 1, 1)
                
        return earnings_dates
    
    def calculate_car(self, report_date, window=3):
        """Calculate cumulative abnormal return around report date"""
        # For simplicity, abnormal return = actual return (no market adjustment)
        start_idx = max(0, self.price_data.index.get_indexer([report_date])[0] - 1)
        end_idx = min(len(self.price_data) - 1, start_idx + window - 1)
        
        if start_idx >= len(self.price_data) or end_idx < 0:
            return 0.0
        
        # Calculate price at beginning and end of window
        start_price = self.price_data.iloc[start_idx]['price']
        end_price = self.price_data.iloc[end_idx]['price']
        
        # Calculate CAR
        car = (end_price / start_price) - 1.0
        return car


class AnalystReportSimulator:
    """Simulates generation of analyst reports and their impact on stock returns"""
    
    def __init__(self, tickers, start_date, end_date):
        self.tickers = tickers
        self.start_date = start_date
        self.end_date = end_date
        self.stocks = {ticker: StockData(ticker, start_date, end_date) for ticker in tickers}
        self.reports = []
        self.report_impacts = {}
        
    def generate_reports(self, num_reports_per_ticker=50):
        """Generate analyst reports for each ticker"""
        for ticker in self.tickers:
            dates = self._generate_report_dates(ticker, num_reports_per_ticker)
            prev_rec = 3  # Neutral initial recommendation
            prev_eps = 1.0  # Initial EPS forecast
            prev_target = self.stocks[ticker].initial_price * 1.1  # Initial target price
            
            for i, date in enumerate(dates):
                # Decide if this is a revision or reiteration report
                is_revision = np.random.choice([True, False], p=[0.4, 0.6])
                
                # Generate recommendations and forecasts
                if is_revision:
                    # Create revision with some randomness
                    rec_change = np.random.choice([-1, 0, 1], p=[0.3, 0.4, 0.3])
                    recommendation = max(1, min(5, prev_rec + rec_change))
                    
                    eps_change = np.random.normal(0, 0.1)
                    eps_forecast = max(0, prev_eps + eps_change)
                    
                    target_change = np.random.normal(0, 0.05)
                    target_price = max(0, prev_target * (1 + target_change))
                    
                    prev_rec = recommendation
                    prev_eps = eps_forecast
                    prev_target = target_price
                    report_type = 'revision'
                else:
                    # Reiteration - use previous values
                    recommendation = prev_rec
                    eps_forecast = prev_eps
                    target_price = prev_target
                    report_type = 'reiteration'
                
                # Create text embedding influenced by report type
                # For revisions, make embedding more impactful
                if is_revision:
                    embedding = np.random.randn(20) * 1.5
                else:
                    embedding = np.random.randn(20) * 0.8
                
                # Post-earnings reports have stronger impact - boost embedding magnitude
                if any(0 <= (date - e_date).days <= 7 for e_date in self.stocks[ticker].earnings_dates):
                    embedding = embedding * 1.8
                
                # Generate the report
                report = AnalystReport(
                    ticker=ticker,
                    date=date,
                    recommendation=recommendation,
                    eps_forecast=eps_forecast,
                    target_price=target_price,
                    text_embedding=embedding,
                    report_type=report_type
                )
                
                self.reports.append(report)
        
        return self.reports
    
    def _generate_report_dates(self, ticker, num_reports):
        """Generate plausible dates for analyst reports"""
        # Ensure some reports are issued shortly after earnings
        earnings_dates = self.stocks[ticker].earnings_dates
        report_dates = []
        
        # Add reports after earnings dates (1-5 days after)
        for e_date in earnings_dates:
            days_after = np.random.randint(1, 6)
            report_date = e_date + timedelta(days=days_after)
            if self.start_date <= report_date <= self.end_date:
                report_dates.append(report_date)
        
        # Add additional random reports
        remaining_reports = max(0, num_reports - len(report_dates))
        available_days = (self.end_date - self.start_date).days
        for _ in range(remaining_reports):
            random_day = np.random.randint(0, available_days)
            random_date = self.start_date + timedelta(days=random_day)
            
            # Avoid weekends
            while random_date.weekday() >= 5:  # 5 = Saturday, 6 = Sunday
                random_day = np.random.randint(0, available_days)
                random_date = self.start_date + timedelta(days=random_day)
                
            report_dates.append(random_date)
            
        return sorted(report_dates)
    
    def simulate_market_reactions(self):
        """Simulate market reactions to analyst reports"""
        results = []
        
        for report in self.reports:
            # Calculate actual CAR (3-day window) from stock data
            car = self.stocks[report.ticker].calculate_car(report.date, window=3)
            
            # Impact factors based on paper findings
            # Revision reports have higher impact
            revision_factor = 2.5 if report.report_type == 'revision' else 1.0
            
            # Post-earnings reports have higher impact
            post_earnings_factor = 2.0 if report.is_post_earnings(
                self.stocks[report.ticker].earnings_dates) else 1.0
            
            # Text embedding contribution (paper says this explains ~10% of returns)
            # For simplicity, we'll use the norm of the embedding
            text_impact = np.linalg.norm(report.text_embedding) * 0.01
            
            # Numerical forecast contribution
            num_impact = 0
            if report.recommendation is not None:
                # Recommendation impact (1-5 scale, with 1 being most positive)
                num_impact -= (3 - report.recommendation) * 0.01
            
            # Record results
            results.append({
                'ticker': report.ticker,
                'date': report.date,
                'report_type': report.report_type,
                'post_earnings': report.is_post_earnings(self.stocks[report.ticker].earnings_dates),
                'recommendation': report.recommendation,
                'eps_forecast': report.eps_forecast,
                'target_price': report.target_price,
                'text_impact': text_impact,
                'numerical_impact': num_impact,
                'total_simulated_impact': (text_impact + num_impact) * revision_factor * post_earnings_factor,
                'actual_car': car,
                'text_embedding': report.text_embedding
            })
        
        self.report_impacts = pd.DataFrame(results)
        return self.report_impacts

    def evaluate_information_value(self):
        """Evaluate the information value of analyst reports using Ridge regression"""
        if len(self.report_impacts) == 0:
            raise ValueError("No report impacts simulated. Run simulate_market_reactions first.")
            
        # Prepare data
        X_numerical = self.report_impacts[['recommendation', 'eps_forecast', 'target_price']].fillna(0)
        
        # Get text embeddings
        X_text = np.vstack(self.report_impacts['text_embedding'].values)
        
        # Target variable
        y = self.report_impacts['actual_car'].values
        
        # Train-test split
        X_num_train, X_num_test, X_text_train, X_text_test, y_train, y_test = train_test_split(
            X_numerical, X_text, y, test_size=0.3, random_state=42)
        
        # Fit Ridge regression models
        num_model = Ridge(alpha=1.0)
        num_model.fit(X_num_train, y_train)
        y_num_pred = num_model.predict(X_num_test)
        
        text_model = Ridge(alpha=1.0)
        text_model.fit(X_text_train, y_train)
        y_text_pred = text_model.predict(X_text_test)
        
        # Combined model
        X_combined_train = np.hstack([X_num_train, X_text_train])
        X_combined_test = np.hstack([X_num_test, X_text_test])

        combined_model.fit(X_combined_train, y_train)
        y_combined_pred = combined_model.predict(X_combined_test)
        
        # Calculate R-squared
        r2_num = r2_score(y_test, y_num_pred)
        r2_text = r2_score(y_test, y_text_pred)
        r2_combined = r2_score(y_test, y_combined_pred)
        
        # Return results
        results = {
            'R2_numerical': r2_num,
            'R2_text': r2_text,
            'R2_combined': r2_combined,
            'Sample_size': len(y),
