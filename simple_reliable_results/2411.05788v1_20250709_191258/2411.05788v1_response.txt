def generate_stock_data(days=1000, volatility=0.01, start_price=100, seed=42):
    """Generate synthetic stock price data"""
    np.random.seed(seed)
    
    # Generate dates
    end_date = datetime.datetime.now().date()
    start_date = end_date - datetime.timedelta(days=days)
    date_range = pd.date_range(start=start_date, end=end_date, freq='B')
    
    # Generate price movements
    returns = np.random.normal(0, volatility, len(date_range))
    price_factor = np.cumprod(1 + returns)
    prices = start_price * price_factor
    
    # Add seasonal patterns and trends
    t = np.arange(len(prices))
    season = 10 * np.sin(t / 252 * 2 * np.pi) # Annual seasonality
    trend = t / 100
    prices = prices + season + trend
    
    # Create volume data
    volume = np.random.randint(100000, 10000000, size=len(date_range))
    
    # Create DataFrame
    stock_data = pd.DataFrame({
        'Date': date_range,
        'Open': prices * (1 - volatility/2),
        'High': prices * (1 + volatility),
        'Low': prices * (1 - volatility),
        'Close': prices,
        'Volume': volume
    })
    
    stock_data.set_index('Date', inplace=True)
    return stock_data

def generate_news_data(stock_data, influence_strength=0.3, seed=42):
    """Generate synthetic news data with sentiment that influences stock prices"""
    np.random.seed(seed)
    
    dates = stock_data.index
    sentiments = []
    news_titles = []
    
    # News templates
    positive_templates = [
        "Company reports strong Q{} earnings, exceeding expectations",
        "Analysts upgrade stock rating to 'buy'",
        "New product launch receives positive market response",
        "Company announces strategic partnership with {}",
        "Dividend increase announced by board of directors"
    ]
    
    negative_templates = [
        "Company misses Q{} earnings expectations",
        "Regulatory challenges impact future outlook",
        "Competitors gain market share in core segment",
        "Analysts downgrade stock rating to 'sell'",
        "Cost pressures impact profit margins"
    ]
    
    neutral_templates = [
        "Company maintains previous guidance for fiscal year",
        "Industry conference highlights ongoing trends",
        "Minor changes in executive leadership announced",
        "Company participates in sector roundtable",
        "Regular quarterly dividend declared"
    ]
    
    companies = ["Microsoft", "Amazon", "Google", "Salesforce", "Adobe"]
    quarters = [1, 2, 3, 4]
    
    # For each day, generate news with sentiment
    for i, date in enumerate(dates):
        # Create price-influenced sentiment
        if i > 0:
            price_change = (stock_data['Close'].iloc[i] - stock_data['Close'].iloc[i-1]) / stock_data['Close'].iloc[i-1]
            # Influence of news sentiment on stock prices
            sentiment_base = np.clip(price_change * 5, -1, 1)  # Scale to -1 to 1
            
            # Add random noise
            sentiment = sentiment_base + np.random.normal(0, 0.2)
            sentiment = np.clip(sentiment, -1, 1)
            
            # Generate news title based on sentiment
            if sentiment > 0.3:
                template = np.random.choice(positive_templates)
            elif sentiment < -0.3:
                template = np.random.choice(negative_templates)
            else:
                template = np.random.choice(neutral_templates)
                
            # Fill in template
            if "{}" in template:
                if "Q{}" in template:
                    quarter = np.random.choice(quarters)
                    template = template.replace("Q{}", f"Q{quarter}")
                if "with {}" in template:
                    company = np.random.choice(companies)
                    template = template.replace("{}", company)
            
            news_titles.append(template)
            sentiments.append(sentiment)
        else:
            news_titles.append("Market opens with neutral outlook")
            sentiments.append(0)
    
    news_data = pd.DataFrame({
        'Date': dates,
        'Title': news_titles,
        'Sentiment': sentiments
    })
    
    news_data.set_index('Date', inplace=True)
    return news_data

class MultivariateLSTM:
    def __init__(self, look_back=30, forecast_horizon=5):
        self.look_back = look_back
        self.forecast_horizon = forecast_horizon
        self.model = None
        self.scaler = MinMaxScaler(feature_range=(0, 1))
        
    def preprocess_data(self, data):
        # Extract features
        features = data[['Open', 'High', 'Low', 'Close', 'Volume']].values
        
        # Scale the data
        scaled_features = self.scaler.fit_transform(features)
        
        # Create sequences
        X, y = [], []
        for i in range(len(scaled_features) - self.look_back - self.forecast_horizon):
            X.append(scaled_features[i:i + self.look_back])
            y.append(scaled_features[i + self.look_back:i + self.look_back + self.forecast_horizon, 3])  # Close price
        
        return np.array(X), np.array(y)
    
    def build_model(self, input_shape):
        model = Sequential()
        model.add(LSTM(50, return_sequences=True, input_shape=input_shape))
        model.add(Dropout(0.2))
        model.add(LSTM(50))
        model.add(Dropout(0.2))
        model.add(Dense(self.forecast_horizon))
        model.compile(optimizer='adam', loss='mse')
        return model
    
    def train(self, data, epochs=50, batch_size=32):
        X, y = self.preprocess_data(data)
        
        # Split into train and test sets
        train_size = int(len(X) * 0.8)
        X_train, X_test = X[:train_size], X[train_size:]
        y_train, y_test = y[:train_size], y[train_size:]
        
        # Build and train the model
        self.model = self.build_model((X_train.shape[1], X_train.shape[2]))
        history = self.model.fit(
            X_train, y_train,
            epochs=epochs,
            batch_size=batch_size,
            validation_data=(X_test, y_test),
            verbose=0
        )
        
        # Predict on test data
        y_pred = self.model.predict(X_test)
        
        # Reshape for inverse transform
        dummy = np.zeros((len(y_pred), 5))
        dummy[:, 3] = y_pred[:, 0]  # Only using the first prediction for evaluation
        
        # Inverse transform
        y_pred_actual = self.scaler.inverse_transform(dummy)[:, 3]
        
        dummy = np.zeros((len(y_test), 5))
        dummy[:, 3] = y_test[:, 0]  # Only using the first actual value
        y_test_actual = self.scaler.inverse_transform(dummy)[:, 3]
        
        # Calculate RMSE
        rmse = sqrt(mean_squared_error(y_test_actual, y_pred_actual))
        
        return rmse, history
    
    def forecast(self, data, steps=5):

class ProphetModel:
    def __init__(self):
        self.model = None
        
    def train(self, data):
        # Prepare data for Prophet
        df = data.reset_index()[['Date', 'Close']].rename(columns={'Date': 'ds', 'Close': 'y'})
        
        # Create and train model
        self.model = Prophet(daily_seasonality=True)
        self.model.fit(df)
        
        # Make future predictions
        future = self.model.make_future_dataframe(periods=30)
        forecast = self.model.predict(future)
        
        # Calculate RMSE on the training data
        y_true = df['y'].values
        y_pred = forecast.iloc[:len(y_true)]['yhat'].values
        rmse = sqrt(mean_squared_error(y_true, y_pred))
        
        return rmse, forecast
    
    def forecast(self, data, steps=5):
        # Make future predictions
        future = self.model.make_future_dataframe(periods=steps)
        forecast = self.model.predict(future)
        
        # Return only the future predictions
        return forecast.iloc[-steps:]['yhat'].values

class SARIMAModel:
    def __init__(self, order=(1, 1, 1), seasonal_order=(1, 1, 1, 12)):
        self.order = order
        self.seasonal_order = seasonal_order
        self.model = None
        
    def train(self, data):
        # Prepare data
        y = data['Close']
        
        # Create and train model
        self.model = SARIMAX(y, order=self.order, seasonal_order=self.seasonal_order)
        results = self.model.fit(disp=False)
        
        # Make in-sample predictions
        y_pred = results.predict()
        
        # Calculate RMSE
        y_true = y.values[max(self.order[0], self.seasonal_order[0]):]
        y_pred = y_pred.values[max(self.order[0], self.seasonal_order[0]):]
        rmse = sqrt(mean_squared_error(y_true, y_pred))
        
        return rmse, results
    
    def forecast(self, data, steps=5):
        # Make future predictions
        forecast = self.model.get_forecast(steps=steps)
        return forecast.predicted_mean.values

class NewsDrivenLSTM:
    def __init__(self, look_back=30, forecast_horizon=5):
        self.look_back = look_back
        self.forecast_horizon = forecast_horizon
        self.model = None
        self.scaler_features = MinMaxScaler(feature_range=(0, 1))
        self.scaler_target = MinMaxScaler(feature_range=(0, 1))
        
    def preprocess_data(self, stock_data, news_data):
        # Merge stock and news data
        merged_data = stock_data.copy()
        merged_data['Sentiment'] = news_data['Sentiment']
        
        # Fill missing values if any
        merged_data.fillna(method='ffill', inplace=True)
        
        # Extract features and target
        features = merged_data[['Open', 'High', 'Low', 'Close', 'Volume', 'Sentiment']].values
        target = merged_data[['Close']].values
        
        # Scale the data
        scaled_features = self.scaler_features.fit_transform(features)
        scaled_target = self.scaler_target.fit_transform(target)
        
        # Create sequences
        X, y = [], []
        for i in range(len(scaled_features) - self.look_back - self.forecast_horizon):
            X.append(scaled_features[i:i + self.look_back])
            y.append(scaled_target[i + self.look_back:i + self.look_back + self.forecast_horizon])
        
        return np.array(X), np.array(y)
    
    def build_model(self, input_shape):
        model = Sequential()
        model.add(Bidirectional(LSTM(64, return_sequences=True), input_shape=input_shape))
        model.add(Dropout(0.2))
        model.add(Bidirectional(LSTM(32)))
        model.add(Dropout(0.2))
        model.add(Dense(self.forecast_horizon))
        model.compile(optimizer='adam', loss='mse')
        return model
    
    def train(self, stock_data, news_data, epochs=50, batch_size=32):
        X, y = self.preprocess_data(stock_data, news_data)
        
        # Flatten target array
        y = y.reshape(y.shape[0], y.shape[1])
        
        # Split into train and test sets
        train_size = int(len(X) * 0.8)
        X_train, X_test = X[:train_size], X[train_size:]
        y_train, y_test = y[:train_size], y[train_size:]
        
        # Build and train the model
        self.model = self.build_model((X_train.shape[1], X_train.shape[2]))
        history = self.model.fit(
            X_train, y_train,
            epochs=epochs,
            batch_size=batch_size,
            validation_data=(X_test, y_test),
            verbose=0
        )
        
        # Predict on test data
        y_pred = self.model.predict(X_test)
        
        # Inverse transform
        y_pred_actual = self.scaler_target.inverse_transform(y_pred)
        y_test_actual = self.scaler_target.inverse_transform(y_test)
        
        # Calculate RMSE for first day prediction
        rmse = sqrt(mean_squared_error(y_test_actual[:, 0], y_pred_actual[:, 0]))
        
        return rmse, history
    
    def forecast(self, stock_data, news_data, steps=5):
        # Preprocess the latest data
        merged_data = stock_data.copy()
        merged_data['Sentiment'] = news_data['Sentiment']
        merged_data.fillna(method='ffill', inplace=True)
        
        features = merged_data[['Open', 'High', 'Low', 'Close', 'Volume', 'Sentiment']].values[-self.look_back:]
        scaled_features = self.scaler_features.transform(features)
        
        # Reshape for LSTM input
        X = np.array([scaled_features])
        
        # Generate prediction
        y_pred = self.model.predict(X)
        
        # Inverse transform
        y_pred_actual = self.scaler_target.inverse_transform(y_pred)
        
        return y_pred_actual[0]
