import numpy as np

class HierarchicalPortfolio:
    """
    Implements the Hierarchical Minimum Variance Portfolio (HMVP) algorithm
    based on the paper by Gamal Mograby
    """
    
    def __init__(self, level=2):
        """
        Initialize the hierarchical portfolio with a specified level
        
        Parameters:
        -----------
        level : int
            Hierarchical level of the Sierpinski graph structure
        """
        self.level = level
        self.n_nodes = self._calculate_nodes(level)
        self.n_junction_nodes = self._calculate_nodes(level - 1)
        self.n_interior_nodes = self.n_nodes - self.n_junction_nodes
        
    def _calculate_nodes(self, level):
        """Calculate the number of nodes at a given level in Sierpinski graph"""
        return int((3**(level+1) + 3) / 2)
    
    def generate_sierpinski_covariance(self, base_variance=10, cov_scale=1, seed=42):
        """
        Generate a covariance matrix with Sierpinski graph structure
        
        Parameters:
        -----------
        base_variance : float
            Base variance for diagonal elements
        cov_scale : float
            Scale factor for covariance elements
        seed : int
            Random seed for reproducibility
            
        Returns:
        --------
        np.ndarray
            Covariance matrix with hierarchical structure
        """
        np.random.seed(seed)
        
        # Initialize covariance matrix with zeros
        sigma = np.zeros((self.n_nodes, self.n_nodes))
        
        # Set diagonal elements (variances)
        for i in range(self.n_nodes):
            sigma[i, i] = base_variance * (0.7 + 0.6 * np.random.rand())
        
        # Process each level from bottom up
        nodes_per_cluster = 3  # Sierpinski has 3 nodes per base cluster
        
        # For level 2 (as in the paper example)
        if self.level == 2:
            # Junction nodes are uncorrelated with each other (first 6 nodes)
            
            # Process the three clusters
            for cluster in range(3):
                # Each cluster has 3 interior nodes
                cluster_start = self.n_junction_nodes + cluster * nodes_per_cluster
                
                # Set covariance between interior nodes in each cluster
                for i in range(cluster_start, cluster_start + nodes_per_cluster):
                    for j in range(cluster_start, cluster_start + nodes_per_cluster):
                        if i != j:  # Skip diagonal elements
                            sigma[i, j] = cov_scale * (-6 + 12 * np.random.rand())
                
                # Set covariance between junction and interior nodes
                # Each cluster connects to 3 junction nodes
                junction_indices = [cluster, (cluster + 1) % 3, (cluster + 3)]
                
                for interior_idx in range(cluster_start, cluster_start + nodes_per_cluster):
                    connected_junctions = random.sample(junction_indices, 2)  # Connect to 2 of 3 possible junctions
                    for junction_idx in connected_junctions:
                        cov_value = cov_scale * (-5 + 10 * np.random.rand())
                        sigma[interior_idx, junction_idx] = cov_value
                        sigma[junction_idx, interior_idx] = cov_value
        
        # Ensure symmetry
        sigma = (sigma + sigma.T) / 2
        
        # Ensure positive definiteness
        min_eig = np.min(np.linalg.eigvals(sigma))
        if min_eig < 0:
            sigma += (-min_eig + 0.01) * np.eye(self.n_nodes)
            
        return sigma
    
    def schur_complement(self, sigma):
        """
        Compute the Schur complement to reduce hierarchy level
        
        Parameters:
        -----------
        sigma : np.ndarray
            Full covariance matrix
            
        Returns:
        --------
        np.ndarray
            Reduced covariance matrix via Schur complement
        """
        # Extract block matrices
        n_jun = self.n_junction_nodes
        T = sigma[:n_jun, :n_jun]
        J = sigma[n_jun:, :n_jun]
        X = sigma[n_jun:, n_jun:]
        
        # Compute Schur complement S = T - J^T X^(-1) J
        X_inv = linalg.inv(X)
        S = T - J.T @ X_inv @ J
        
        return S
    
    def compute_g(self, sigma, gamma):
        """
        Compute g(gamma) as defined in the paper
        
        Parameters:
        -----------
        sigma : np.ndarray
            Covariance matrix
        gamma : np.ndarray
            Input vector
            
        Returns:
        --------
        np.ndarray
            Transformed vector g(gamma)
        """
        n_jun = self.n_junction_nodes
        
        # Extract block matrices
        gamma_jun = gamma[:n_jun]
        gamma_in = gamma[n_jun:]
        J = sigma[n_jun:, :n_jun]
        X = sigma[n_jun:, n_jun:]
        
        # Compute g(gamma)
        X_inv = linalg.inv(X)
        g_gamma = gamma_jun - J.T @ X_inv @ gamma_in
        
        return g_gamma
    
    def compute_h(self, sigma, w_jun, gamma):
        """
        Compute h(w_jun, gamma) as defined in the paper
        
        Parameters:
        -----------
        sigma : np.ndarray
            Covariance matrix
        w_jun : np.ndarray
            Junction node weights
        gamma : np.ndarray
            Input vector
            
        Returns:
        --------
        np.ndarray
            Interior node weights
        """
        n_jun = self.n_junction_nodes
        
        # Extract block matrices
        gamma_in = gamma[n_jun:]
        J = sigma[n_jun:, :n_jun]
        X = sigma[n_jun:, n_jun:]
        
        # Compute h(w_jun, gamma)
        X_inv = linalg.inv(X)
        h_w_gamma = X_inv @ gamma_in - X_inv @ J @ w_jun
        
        return h_w_gamma
    
    def compute_optimal_weights(self, sigma):
        """
        Compute optimal portfolio weights using hierarchical method
        
        Parameters:
        -----------
        sigma : np.ndarray
            Covariance matrix
            
        Returns:
        --------
        np.ndarray
            Optimal portfolio weights
        """
        # For level 2, we need to:
        # 1. Compute Schur complement to get sigma_1
        # 2. Compute gamma_1 = g_1(1)
        # 3. Compute gamma_0 = g_0(gamma_1)
        # 4. Find w_0 = sigma_0^(-1) gamma_0
        # 5. Find w_1 = [w_0, h_1(w_0, gamma_1)]
        # 6. Find w_2 = [w_1, h_2(w_1, gamma_2)]
        # 7. Normalize w_2
        
        gamma_2 = np.ones(self.n_nodes)
        
        # Step 1: Compute sigma_1 via Schur complement
        sigma_1 = self.schur_complement(sigma)
        
        # Step 2: Compute gamma_1
        gamma_1 = self.compute_g(sigma, gamma_2)
        
        # Step 3: Compute sigma_0 and gamma_0
        n_jun_1 = 3  # For Sierpinski, level 1 has 3 junction nodes
        T_1 = sigma_1[:n_jun_1, :n_jun_1]
        J_1 = sigma_1[n_jun_1:, :n_jun_1]
        X_1 = sigma_1[n_jun_1:, n_jun_1:]
        
        gamma_1_jun = gamma_1[:n_jun_1]
        gamma_1_in = gamma_1[n_jun_1:]
        
        X_1_inv = linalg.inv(X_1)
        sigma_0 = T_1 - J_1.T @ X_1_inv @ J_1
        gamma_0 = gamma_1_jun - J_1.T @ X_1_inv @ gamma_1_in
        
        # Step 4: Compute w_0
        w_0 = linalg.inv(sigma_0) @ gamma_0
        
        # Step 5: Compute w_1
        w_1_jun = w_0
        w_1_in = X_1_inv @ gamma_1_in - X_1_inv @ J_1 @ w_1_jun
        w_1 = np.concatenate([w_1_jun, w_1_in])
        
        # Step 6: Compute w_2
        w_2_jun = w_1
        w_2_in = self.compute_h(sigma, w_2_jun, gamma_2)
        w_2 = np.concatenate([w_2_jun, w_2_in])
        
        # Step 7: Normalize
        w_2_norm = w_2 / np.sum(w_2)
        
        return w_2_norm
    
    def compute_portfolio_variance(self, weights, sigma):
        """
        Compute portfolio variance
        
        Parameters:
        -----------
        weights : np.ndarray
            Portfolio weights
        sigma : np.ndarray
            Covariance matrix
            
        Returns:
        --------
        float
            Portfolio variance
        """
        return weights.T @ sigma @ weights
    
    def compute_direct_optimal_weights(self, sigma):
        """
        Compute optimal weights directly using standard formula
        
        Parameters:
        -----------
        sigma : np.ndarray
            Covariance matrix
            
        Returns:
        --------
        np.ndarray
            Optimal portfolio weights
        """
        ones = np.ones(self.n_nodes)
        sigma_inv = linalg.inv(sigma)
        
        w = sigma_inv @ ones
        w_norm = w / np.sum(w)
        
        return w_norm
