class FittedNaturalActorCritic:
    """Implementation of the Fitted Natural Actor-Critic algorithm"""
    
    def __init__(
        self, 
        state_dim: int, 
        action_type: str = "discrete",
        action_dim: int = 3,

            action_dim: Dimension of the action space (3 for discrete, 1 for continuous)

            gamma: Discount factor
            xgb_params: Parameters for XGBoost regressor
            ridge_params: Parameters for Ridge regressor
            nn_params: Parameters for neural network policy
        """

        self.action_type = action_type
        self.action_dim = action_dim
        self.learning_rate = learning_rate
        self.gamma = gamma
        
        # Default parameters
        if xgb_params is None:
            self.xgb_params = {
                'n_estimators': 100,
                'learning_rate': 0.1,
                'max_depth': 5,
                'min_child_weight': 1,
                'gamma': 0.1,
                'reg_lambda': 1.0

        else:
            self.xgb_params = xgb_params

            self.ridge_params = {
                'alpha': 0.1
            }
        else:
            self.ridge_params = ridge_params
            
        if nn_params is None:
            if action_type == "discrete":
                self.nn_params = {

                    'activation': 'relu',

                    'hidden_layer_sizes': (64, 32),
                    'activation': 'relu',
                    'learning_rate_init': 0.001,
                    'max_iter': 1000
                }
        else:
            self.nn_params = nn_params
        
        # Initialize models
        self._init_models()
        
        # Experience buffer
        self.buffer = []
        
    def _init_models(self):
        """Initialize critic and actor models"""
        # Value function estimator (critic layer 1)
        self.value_model = xgb.XGBRegressor(**self.xgb_params)
        
        # Advantage function estimator (critic layer 2)
        self.advantage_model = Ridge(**self.ridge_params)
        
        # Policy model (actor)
        if self.action_type == "discrete":
            # For discrete actions: output probabilities for each action
            self.policy_model = MLPRegressor(

            self.policy_model.fit(X_dummy, y_dummy)
        else:
            # For continuous actions: output mean and std of a normal distribution
            self.policy_model = MLPRegressor(
                hidden_layer_sizes=self.nn_params['hidden_layer_sizes'],
                activation=self.nn_params['activation'],
                learning_rate_init=self.nn_params['learning_rate_init'],
                max_iter=self.nn_params['max_iter']
            )
            # Initialize with random weights
            X_dummy = np.random.normal(0, 1, (10, self.state_dim))
            y_dummy = np.random.normal(0, 1, (10, 2))  # mean and std
            self.policy_model.fit(X_dummy, y_dummy)
    
    def select_action(self, state: np.ndarray) -> float:
        """Select an action based on current policy"""
        state = state.reshape(1, -1)
        
        if self.action_type == "discrete":
            # Get logits
            logits = self.policy_model.predict(state)[0]
            
            # Convert to probabilities with softmax
            probs = np.exp(logits - np.max(logits))
            probs = probs / np.sum(probs)
            
            # Sample action
            action_idx = np.random.choice(self.action_dim, p=probs)
            
            # Convert to actual action value (-1, 0, 1)
            action = -1.0 + action_idx * 1.0
            
            return action
        else:
            # Get mean and std
            params = self.policy_model.predict(state)[0]
            mean, log_std = params[0], params[1]
            std = np.exp(min(log_std, 2))  # Clip to avoid too large std
            
            # Sample from truncated normal
            action = np.random.normal(mean, std)
            action = np.clip(action, -1.0, 1.0)
            
            return action
    
    def store_transition(self, state: np.ndarray, action: float, reward: float, 
                         next_state: np.ndarray, done: bool):
        """Store a transition in the experience buffer"""
        self.buffer.append((state, action, reward, next_state, done))
    
    def train(self, batch_size: int = 64, iterations: int = 10):
        """Train the FNAC models using collected experiences"""
        if len(self.buffer) < batch_size:
            return
        
        # Extract transitions from buffer
        states = np.array([t[0] for t in self.buffer])
        actions = np.array([t[1] for t in self.buffer])
        rewards = np.array([t[2] for t in self.buffer])
        next_states = np.array([t[3] for t in self.buffer])
        dones = np.array([t[4] for t in self.buffer])
        
        # Step 1: Fit the value function (critic layer 1)
        if not hasattr(self.value_model, 'fitted_'):
            # First fit
            self.value_model.fit(states, rewards)
            self.value_model.fitted_ = True
        else:
            # Update fit
            for _ in range(iterations):
                # Calculate target values
                next_values = self.value_model.predict(next_states)
                # Set next_value to 0 for terminal states
                next_values[dones] = 0
                targets = rewards + self.gamma * next_values
                
                # Update value function
                self.value_model.fit(states, targets)
        
        # Step 2: Calculate advantages
        values = self.value_model.predict(states)
        next_values = self.value_model.predict(next_states)
        next_values[dones] = 0
        advantages = rewards + self.gamma * next_values - values
        
        # Step 3: Fit the advantage function with compatible features (critic layer 2)
        # Calculate compatible features
        if self.action_type == "discrete":
            # Create one-hot encoding of actions
            action_indices = ((actions + 1) / 1.0).astype(int)
            action_one_hot = np.zeros((len(actions), self.action_dim))
            for i, idx in enumerate(action_indices):
                if 0 <= idx < self.action_dim:
                    action_one_hot[i, idx] = 1
            
            # For discrete case, compatible features are gradients of log policy
            features = action_one_hot - np.ones_like(action_one_hot) / self.action_dim
            features = features.reshape(len(features), -1)
        else:
            # For continuous case, we'll use a simple feature based on the action
            # This is a simplification - in practice more complex features would be used
            features = np.column_stack([states, actions])
        
        # Fit advantage function
        self.advantage_model.fit(features, advantages)
        
        # Step 4: Update policy using natural gradient
        # For simplicity, we'll update directly using advantages as a proxy for the natural gradient
        if self.action_type == "discrete":
            # Create policy targets
            policy_targets = self.policy_model.predict(states)
            
            # For each action, adjust the logit based on its advantage
            for i, (s, a, adv) in enumerate(zip(states, actions, advantages)):
                action_idx = int((a + 1) / 1.0)
                if 0 <= action_idx < self.action_dim:
                    policy_targets[i, action_idx] += self.learning_rate * adv
            
            # Update policy
            self.policy_model.fit(states, policy_targets)
        else:
            # For continuous policy, update the mean based on advantage
            policy_outputs = self.policy_model.predict(states)
            
            # Adjust mean based on advantage and action taken
            for i, (a, adv) in enumerate(zip(actions, advantages)):
                # Move mean towards action if advantage is positive, away if negative
                policy_outputs[i, 0] += self.learning_rate * adv * (a - policy_outputs[i, 0])
                
                # Optionally adjust std based on advantage magnitude
                # policy_outputs[i, 1] -= self.learning_rate * abs(adv)  # Decrease std if advantage is large
            
            # Update policy
            self.policy_model.fit(states, policy_outputs)
    
    def clear_buffer(self):
        """Clear the experience buffer"""
        self.buffer = []

def train_and_evaluate(
    env_train,
    env_val,
    env_test,
    state_dim,
    action_type,
    episodes=200,
    batch_size=64,
    evaluate_every=10
):
    """Train FNAC agent and evaluate performance"""
    
    if action_type == "discrete":
        action_dim = 3
    else:
        action_dim = 1
    
    agent = FittedNaturalActorCritic(
        state_dim=state_dim,
        action_type=action_type,
        action_dim=action_dim
    )
    
    # Training metrics
    train_returns = []
    val_returns = []
    test_returns = []
    
    # Training loop
    for episode in range(1, episodes+1):
        # Training episode
        state = env_train.reset()
        done = False
        episode_return = 0
        
        while not done:
            action = agent.select_action(state)
            next_state, reward, done, _ = env_train.step(action)
            agent.store_transition(state, action, reward, next_state, done)
            state = next_state
            episode_return += reward
        
        # Train the agent
        agent.train(batch_size)
        
        # Store training return
        train_returns.append(episode_return)
        
        # Evaluate periodically
        if episode % evaluate_every == 0:
            # Validation
            val_return = evaluate_agent(agent, env_val)
            val_returns.append(val_return)
            
            print(f"Episode {episode}/{episodes}, "
                  f"Train Return: {episode_return:.2f}, "
                  f"Validation Return: {val_return:.2f}")
    
    # Final evaluation on test set
    test_return, test_actions, test_rewards = evaluate_agent(agent, env_test, return_actions=True)
    test_returns.append(test_return)
    
    print(f"Final Test Return: {test_return:.2f}")
    
    return agent, train_returns, val_returns, test_returns, test_actions, test_rewards

def evaluate_agent(agent, env, return_actions=False):
    """Evaluate agent performance on all days in the environment"""
    total_return = 0
    days = env.days
    all_actions = []
    all_rewards = []
    
    for day in days:
        state = env.reset(day)
        done = False
        day_return = 0
        day_actions = []
        day_rewards = []
        
        while not done:
            action = agent.select_action(state)
            next_state, reward, done, _ = env.step(action)
            state = next_state
            day_return += reward
            
            if return_actions:
                day_actions.append(action)
                day_rewards.append(reward)
        
        total_return += day_return
        
        if return_actions:
            all_actions.append(day_actions)
            all_rewards.append(day_rewards)
    
    avg_return = total_return / len(days)
    
    if return_actions:
        return avg_return, all_actions, all_rewards
    else:
        return avg_return
