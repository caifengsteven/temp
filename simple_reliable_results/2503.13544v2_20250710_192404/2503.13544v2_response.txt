class TargetPortfolioGenerator:
    """Generate target portfolios for training"""
    
    @staticmethod
    def max_sortino_portfolio(future_returns):
        """Generate Max-Sortino target portfolio"""
        # Calculate mean returns
        mean_returns = future_returns.mean()
        
        # Calculate semi-variance matrix (only negative deviations)
        neg_returns = future_returns - mean_returns
        neg_returns[neg_returns > 0] = 0
        semi_var = neg_returns.T @ neg_returns / len(future_returns)
        
        # Simple optimization for Max-Sortino
        n_assets = len(mean_returns)
        
        # Use a simplified optimization for this example
        # This is a naive approach - in practice, you'd use the Dinkelbach method
        # or another convex optimization technique as mentioned in the paper
        
        # Ensure semi_var is invertible
        semi_var = semi_var + np.eye(n_assets) * 1e-6
        
        # Simplified optimization
        weights = np.dot(np.linalg.inv(semi_var), mean_returns)
        
        # Ensure non-negative weights (long-only)
        weights = np.maximum(weights, 0)
        
        # Normalize to sum to 1
        if np.sum(weights) > 0:
            weights = weights / np.sum(weights)
        else:
            # If all weights are 0, use equal weights
            weights = np.ones(n_assets) / n_assets
            
        return weights
    
    @staticmethod
    def one_hot_portfolio(future_returns):
        """Generate One-Hot target portfolio (invest in best performing asset)"""
        mean_returns = future_returns.mean()
        weights = np.zeros_like(mean_returns)
        weights[np.argmax(mean_returns)] = 1.0
        return weights


class LSTMModel(nn.Module):
    """LSTM model for portfolio optimization"""
    
    def __init__(self, input_dim=5, hidden_dim=64, n_assets=24):
        super(LSTMModel, self).__init__()
        self.lstm = nn.LSTM(input_dim, hidden_dim, batch_first=True)
        self.fc = nn.Linear(hidden_dim, n_assets)
        self.softmax = nn.Softmax(dim=1)
        
    def forward(self, x):
        # x shape: [batch_size, seq_len, input_dim]
        _, (hidden, _) = self.lstm(x)
        out = self.fc(hidden.squeeze(0))
        return self.softmax(out)


class MambaModel(nn.Module):
    """Simplified Mamba-inspired model (since true Mamba implementation is complex)"""
    
    def __init__(self, input_dim=5, hidden_dim=64, n_assets=24):
        super(MambaModel, self).__init__()
        # Using GRU as a simplified substitute for Mamba
        self.gru = nn.GRU(input_dim, hidden_dim, batch_first=True)
        self.fc1 = nn.Linear(hidden_dim, hidden_dim)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_dim, n_assets)
        self.softmax = nn.Softmax(dim=1)
        
    def forward(self, x):
        # x shape: [batch_size, seq_len, input_dim]
        _, hidden = self.gru(x)
        out = self.fc1(hidden.squeeze(0))
        out = self.relu(out)
        out = self.fc2(out)
        return self.softmax(out)


class TransformerModel(nn.Module):
    """Simplified Transformer model for portfolio optimization"""
    
    def __init__(self, input_dim=5, hidden_dim=64, n_heads=4, n_assets=24):
        super(TransformerModel, self).__init__()
        self.embedding = nn.Linear(input_dim, hidden_dim)
        self.pos_encoder = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU()
        )
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim, nhead=n_heads, batch_first=True
        )
        self.transformer_encoder = nn.TransformerEncoder(encoder_layer, num_layers=2)
        self.fc = nn.Linear(hidden_dim, n_assets)
        self.softmax = nn.Softmax(dim=1)
        
    def forward(self, x):
        # x shape: [batch_size, seq_len, input_dim]
        x = self.embedding(x)
        x = self.pos_encoder(x)
        x = self.transformer_encoder(x)
        # Use the last sequence element for prediction
        out = self.fc(x[:, -1, :])
        return self.softmax(out)


class DeepEnsemble:
    """Implement Deep Ensemble for portfolio optimization"""
    
    def __init__(self, model_class, model_args, n_models=10):
        self.n_models = n_models
        self.models = [model_class(**model_args) for _ in range(n_models)]
        
    def train(self, train_loader, val_loader, epochs=100, lr=0.001):
        """Train all models in the ensemble"""
        for i, model in enumerate(self.models):
            optimizer = optim.Adam(model.parameters(), lr=lr)
            criterion = nn.CrossEntropyLoss()
            
            best_val_loss = float('inf')
            best_model_state = None
            
            print(f"Training model {i+1}/{self.n_models}")
            for epoch in range(epochs):
                # Training
                model.train()
                train_loss = 0
                for X_batch, y_batch in train_loader:
                    optimizer.zero_grad()
                    output = model(X_batch)
                    loss = criterion(output, y_batch)
                    loss.backward()
                    optimizer.step()
                    train_loss += loss.item()
                
                # Validation
                model.eval()
                val_loss = 0
                with torch.no_grad():
                    for X_batch, y_batch in val_loader:
                        output = model(X_batch)
                        loss = criterion(output, y_batch)
                        val_loss += loss.item()
                
                if epoch % 10 == 0:
                    print(f"Epoch {epoch}: Train Loss = {train_loss/len(train_loader):.6f}, Val Loss = {val_loss/len(val_loader):.6f}")
                
                # Save best model
                if val_loss < best_val_loss:
                    best_val_loss = val_loss
                    best_model_state = model.state_dict().copy()
            
            # Load best model
            model.load_state_dict(best_model_state)
    
    def predict(self, X):
        """Get ensemble predictions"""
        predictions = []
        
        for model in self.models:
            model.eval()
            with torch.no_grad():
                pred = model(X)
                predictions.append(pred)
        
        # Average predictions
        ensemble_pred = torch.mean(torch.stack(predictions), dim=0)
        return ensemble_pred


class PortfolioBacktester:
    """Backtest portfolio performance"""
    
    def __init__(self, returns_data, trading_cost=0.0040, rebalance_lag=2):
        """
        Initialize backtester
        
        Parameters:
        - returns_data: DataFrame with daily returns for each asset
        - trading_cost: Trading cost in decimal (40 bp = 0.0040)
        - rebalance_lag: Lag in days for rebalancing implementation
        """
        self.returns_data = returns_data
        self.trading_cost = trading_cost
        self.rebalance_lag = rebalance_lag
        
    def backtest(self, portfolio_weights, strategy_name="Strategy"):
        """
        Run backtest for a given set of portfolio weights
        
        Parameters:
        - portfolio_weights: DataFrame with portfolio weights indexed by date
        - strategy_name: Name of the strategy for plotting
        
        Returns:
        - DataFrame with backtest results
        """
        # Align dates
        common_dates = sorted(set(portfolio_weights.index) & set(self.returns_data.index))
        portfolio_weights = portfolio_weights.loc[common_dates]
        returns_subset = self.returns_data.loc[common_dates]
        
        # Apply rebalancing lag
        if self.rebalance_lag > 0:
            portfolio_weights = portfolio_weights.shift(self.rebalance_lag)
            portfolio_weights = portfolio_weights.dropna()
            common_dates = portfolio_weights.index
            returns_subset = returns_subset.loc[common_dates]
        
        # Initialize result variables
        portfolio_values = [1.0]  # Start with $1
        current_weights = portfolio_weights.iloc[0]
        
        # Run backtest
        for i in range(1, len(common_dates)):
            date = common_dates[i]
            prev_date = common_dates[i-1]
            
            # Calculate daily return based on current weights
            daily_return = np.sum(current_weights * returns_subset.loc[date])
            
            # Update portfolio value
            portfolio_values.append(portfolio_values[-1] * (1 + daily_return))
            
            # Rebalance if needed
            if i < len(common_dates) - 1:  # Skip last day
                new_weights = portfolio_weights.loc[date]
                
                # Calculate trading cost
                if not np.array_equal(current_weights, new_weights):
                    turnover = np.sum(np.abs(new_weights - current_weights))
                    cost = turnover * self.trading_cost
                    portfolio_values[-1] *= (1 - cost)
                
                current_weights = new_weights
        
        # Create results DataFrame
        results = pd.DataFrame({
            'value': portfolio_values,
            'strategy': strategy_name
        }, index=common_dates)
        
        # Calculate metrics
        daily_returns = results['value'].pct_change().dropna()
        
        # Cumulative return
        cum_return = portfolio_values[-1] / portfolio_values[0]
        
        # Sharpe ratio (annualized)
        sharpe_ratio = np.sqrt(252) * daily_returns.mean() / daily_returns.std()
        
        # Sortino ratio (annualized)
        downside_returns = daily_returns[daily_returns < 0]
        sortino_ratio = np.sqrt(252) * daily_returns.mean() / downside_returns.std()
        
        print(f"{strategy_name} Metrics:")
        print(f"Cumulative Return: {cum_return:.3f}")
        print(f"Sharpe Ratio: {sharpe_ratio:.3f}")
        print(f"Sortino Ratio: {sortino_ratio:.3f}")
        print("")
        
        return results, {
            'cum_return': cum_return,
            'sharpe_ratio': sharpe_ratio,
            'sortino_ratio': sortino_ratio
        }
