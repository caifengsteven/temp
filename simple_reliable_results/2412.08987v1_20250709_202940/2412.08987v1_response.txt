import numpy as np

class NURBSBasis:
    """Implementation of NURBS basis functions for IGA"""
    
    def __init__(self, knot_vector, p=3, weights=None):
        """
        Initialize NURBS basis
        
        Parameters:
        -----------
        knot_vector : array-like
            Knot vector
        p : int
            Polynomial degree (default: 3 for cubic NURBS)
        weights : array-like or None
            Weights for NURBS basis functions (if None, B-splines are used)
        """
        self.knot_vector = np.array(knot_vector)
        self.p = p
        self.m = len(knot_vector)
        self.n = self.m - self.p - 1
        
        # Initialize weights (if None, use constant weights = 1)
        if weights is None:
            self.weights = np.ones(self.n)
        else:
            self.weights = np.array(weights)
    
    def bspline_basis(self, i, p, xi):
        """Compute B-spline basis function N_i,p at parameter xi"""
        if p == 0:
            if self.knot_vector[i] <= xi < self.knot_vector[i+1] or \
               (xi == self.knot_vector[-1] and i == len(self.knot_vector)-2):
                return 1.0
            return 0.0
        
        # Compute first term
        if self.knot_vector[i+p] == self.knot_vector[i]:
            c1 = 0.0
        else:
            c1 = (xi - self.knot_vector[i]) / (self.knot_vector[i+p] - self.knot_vector[i])
            
        # Compute second term
        if self.knot_vector[i+p+1] == self.knot_vector[i+1]:
            c2 = 0.0
        else:
            c2 = (self.knot_vector[i+p+1] - xi) / (self.knot_vector[i+p+1] - self.knot_vector[i+1])
            
        return c1 * self.bspline_basis(i, p-1, xi) + c2 * self.bspline_basis(i+1, p-1, xi)
    
    def nurbs_basis(self, xi):
        """Compute all NURBS basis functions at parameter xi"""
        # Compute B-spline basis functions
        N = np.zeros(self.n)
        for i in range(self.n):
            N[i] = self.bspline_basis(i, self.p, xi)
        
        # Compute NURBS basis functions
        R = np.zeros(self.n)
        weighted_sum = np.sum(self.weights * N)
        if weighted_sum > 0:
            R = self.weights * N / weighted_sum
        
        return R
    
    def nurbs_basis_der(self, xi):
        """Compute first derivatives of NURBS basis functions at parameter xi"""
        # This is a simplified implementation - a full implementation would 
        # require computing derivatives of B-spline basis functions
        h = 1e-6
        R1 = self.nurbs_basis(xi)
        R2 = self.nurbs_basis(xi + h)
        return (R2 - R1) / h


class LelandModel:
    """Implementation of Leland model for European call with transaction costs"""
    
    def __init__(self, S_max=300, K=100, r=0.1, sigma=0.2, T=1.0, Le=0.8):
        """
        Initialize Leland model
        
        Parameters:
        -----------
        S_max : float
            Maximum asset price
        K : float
            Strike price
        r : float
            Risk-free interest rate
        sigma : float
            Volatility
        T : float
            Time to maturity
        Le : float
            Leland number
        """
        self.S_max = S_max
        self.K = K
        self.r = r
        self.sigma = sigma
        self.T = T
        self.Le = Le
        
        # Transformation parameters
        self.kappa = 2 * r / (sigma**2)
        self.tau_max = 0.5 * sigma**2 * T
        
        # Domain in transformed space
        self.x_min = np.log(1e-2)  # Small positive value instead of 0
        self.x_max = np.log(S_max) + self.kappa * self.tau_max
    
    def payoff(self, S):
        """Return payoff function max(S-K, 0)"""
        return np.maximum(S - self.K, 0.0)
    
    def transform_S_to_x(self, S, t):
        """Transform from (S,t) to (x,tau) space"""
        tau = 0.5 * self.sigma**2 * (self.T - t)
        x = np.log(S) + self.kappa * tau
        return x, tau
    
    def transform_x_to_S(self, x, tau):
        """Transform from (x,tau) to (S,t) space"""
        t = self.T - 2 * tau / (self.sigma**2)
        S = np.exp(x - self.kappa * tau)
        return S, t
    
    def initial_condition(self, x):
        """Initial condition in transformed space"""
        return np.maximum(np.exp(x) - self.K, 0.0)


class IGASolver:
    """Solver for Leland model using IGA with NURBS basis functions"""
    
    def __init__(self, model, num_elements=64, num_time_steps=1000, nurbs_degree=3, weights=None, knot_type='uniform'):
        """
        Initialize IGA solver
        
        Parameters:
        -----------
        model : LelandModel
            Leland model
        num_elements : int
            Number of elements in spatial discretization
        num_time_steps : int
            Number of time steps
        nurbs_degree : int
            Polynomial degree for NURBS basis (default: 3 for cubic NURBS)
        weights : array-like or None
            Weights for NURBS basis functions
        knot_type : str
            Type of knot vector ('uniform' or 'non-uniform')
        """
        self.model = model
        self.num_elements = num_elements
        self.num_time_steps = num_time_steps
        self.nurbs_degree = nurbs_degree

        self.dt = model.tau_max / num_time_steps

        if knot_type == 'uniform':
            self.create_uniform_knot_vector()
        else:
            self.create_non_uniform_knot_vector()

        self.num_basis = self.nurbs.n
        
        # Number of interior basis functions (excluding boundary)
        self.num_interior = self.num_basis - 2
        
        # Initialize solution vector
        self.v = np.zeros(self.num_basis)
        
        # Initialize matrices
        self.initialize_matrices()
    
    def create_uniform_knot_vector(self):
        """Create uniform knot vector"""
        # Open knot vector with p+1 repeated knots at endpoints
        p = self.nurbs_degree
        n = self.num_elements + p
        
        # Interior knots
        interior_knots = np.linspace(0, 1, self.num_elements + 1)
        
        # Repeat first and last knot p+1 times
        self.knot_vector = np.concatenate([
            np.zeros(p),
            interior_knots,
            np.ones(p)
        ])
    
    def create_non_uniform_knot_vector(self):
        """Create non-uniform knot vector with concentration around K"""
        p = self.nurbs_degree
        n = self.num_elements + p
        
        # Map strike price K to parameter space
        xi_K = (np.log(self.model.K) - self.model.x_min) / (self.model.x_max - self.model.x_min)
        
        # Create knots with higher density around K
        left_knots = np.linspace(0, xi_K, self.num_elements//2 + 1)
        right_knots = np.linspace(xi_K, 1, self.num_elements//2 + 1)
        interior_knots = np.unique(np.concatenate([left_knots, right_knots]))
        
        # Repeat knot at K
        interior_knots = np.sort(np.concatenate([interior_knots, [xi_K, xi_K]]))
        
        # Repeat first and last knot p+1 times
        self.knot_vector = np.concatenate([
            np.zeros(p),
            interior_knots,
            np.ones(p)
        ])
    
    def initialize_matrices(self):
        """Initialize mass and stiffness matrices using numerical integration"""
        n = self.num_basis
        
        # Initialize matrices
        self.M = np.zeros((n, n))  # Mass matrix
        self.K = np.zeros((n, n))  # Stiffness matrix
        self.N = np.zeros((n, n))  # Advection matrix
        
        # Gauss-Legendre quadrature points and weights
        num_quad_points = self.nurbs_degree + 1
        quad_points, quad_weights = np.polynomial.legendre.leggauss(num_quad_points)
        
        # Loop through elements (between unique knots)
        unique_knots = np.unique(self.knot_vector)
        for e in range(len(unique_knots) - 1):
            # Element boundaries
            xi_e = unique_knots[e]
            xi_e1 = unique_knots[e+1]
            
            # Skip zero-length elements
            if xi_e == xi_e1:
                continue
            
            # Map quadrature points to element
            xi_quad = 0.5 * (xi_e1 - xi_e) * quad_points + 0.5 * (xi_e1 + xi_e)
            w_quad = 0.5 * (xi_e1 - xi_e) * quad_weights
            
            # Loop through quadrature points
            for q in range(num_quad_points):
                xi = xi_quad[q]
                w = w_quad[q]
                
                # Evaluate basis functions and derivatives
                R = self.nurbs.nurbs_basis(xi)
                dR = self.nurbs.nurbs_basis_der(xi)
                
                # Map from parameter space to physical space
                J = (self.model.x_max - self.model.x_min)  # Jacobian
                dx_dxi = J
                dxi_dx = 1/J
                
                # Update matrices
                for i in range(n):
                    for j in range(n):
                        # Mass matrix
                        self.M[i, j] += R[i] * R[j] * w * dx_dxi
                        
                        # Stiffness matrix
                        self.K[i, j] += dR[i] * dR[j] * w * dxi_dx
                        
                        # Advection matrix
                        self.N[i, j] += R[i] * dR[j] * w
    
    def apply_boundary_conditions(self, A, b):
        """Apply boundary conditions to system matrix A and right-hand side b"""
        # Left boundary (S=0): v = 0
        A[0, :] = 0
        A[0, 0] = 1
        b[0] = 0
        
        # Right boundary (S=S_max): v ≈ S - K for large S
        A[-1, :] = 0
        A[-1, -1] = 1
        S_max, _ = self.model.transform_x_to_S(self.model.x_max, 0)
        b[-1] = S_max - self.model.K
        
        return A, b
    
    def solve(self):
        """Solve Leland model using IGA with theta-scheme"""
        # Initialize solution at t=T (tau=0)
        x_values = np.linspace(self.model.x_min, self.model.x_max, 100)
        init_values = np.array([self.model.initial_condition(x) for x in x_values])
        
        # Project initial condition onto NURBS basis
        # This is a simplification - a proper L2 projection would be better
        xi_values = np.linspace(0, 1, 100)
        for i in range(self.num_basis):
            R_i = np.array([self.nurbs.bspline_basis(i, self.nurbs_degree, xi) for xi in xi_values])
            self.v[i] = np.sum(R_i * init_values) / np.sum(R_i)
        
        # Adjust for non-zero boundary conditions
        self.v[0] = 0
        self.v[-1] = self.model.S_max - self.model.K
        
        # Time integration using Crank-Nicolson (theta=0.5)
        theta = 0.5
        
        # Store solution at each time step
        solutions = [self.v.copy()]
        
        # Use first few steps as Rannacher steps (fully implicit, theta=1)
        rannacher_steps = 4
        
        for m in range(self.num_time_steps):
            # Use fully implicit for first few steps
            if m < rannacher_steps:
                theta_step = 1.0
            else:
                theta_step = theta
            
            # Nonlinear term (simplified for test implementation)
            v_tilde = -np.dot(self.K + self.N, self.v)
            
            # Assemble system matrix
            A0 = self.M - (1 - theta_step) * self.dt * (self.K + self.N)
            A1 = self.M + theta_step * self.dt * (self.K + self.N)
            
            # Right-hand side
            b = np.dot(A0, self.v)
            b += self.dt * self.model.Le * np.dot(self.M, np.abs(v_tilde))
            
            # Apply boundary conditions
            A1, b = self.apply_boundary_conditions(A1, b)
            
            # Solve system
            self.v = np.linalg.solve(A1, b)
            
            # Store solution
            solutions.append(self.v.copy())
        
        return np.array(solutions)
    
    def transform_solution_to_S_space(self, solution, tau):
        """Transform solution from x-space to S-space"""
        # Generate S values
        S_values = np.logspace(-2, np.log10(self.model.S_max), 200)
        
        # Transform to x values
        x_values = [self.model.transform_S_to_x(S, self.model.T - 2*tau/(self.model.sigma**2))[0] for S in S_values]
        
        # Map to parameter space
        xi_values = [(x - self.model.x_min)/(self.model.x_max - self.model.x_min) for x in x_values]
        
        # Evaluate solution at xi values
        v_values = np.zeros_like(S_values)
        for i, xi in enumerate(xi_values):
            if 0 <= xi <= 1:
                R = self.nurbs.nurbs_basis(xi)
                v_values[i] = np.dot(R, solution)
            else:
                # Handle out-of-bounds
                if xi < 0:
                    v_values[i] = 0
                else:  # xi > 1
                    v_values[i] = S_values[i] - self.model.K
        
        return S_values, v_values
