class ProjectionNetwork(nn.Module):
    """Neural network to project embeddings into similarity space."""
    
    def __init__(self, input_dim=4096, hidden_dim=256, output_dim=128):
        super(ProjectionNetwork, self).__init__()
        self.projection = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, output_dim)
        )
        
    def forward(self, x):
        x = self.projection(x)
        # Normalize the output
        norm = torch.norm(x, p=2, dim=1, keepdim=True)
        x = x.div(norm.expand_as(x))
        return x

def weighted_similarity_contrastive_loss(anchor, augmented, similarity_score, margin=1.0):
    """Weighted Similarity Contrastive Loss as defined in equation (3) of the paper."""
    batch_size = anchor.size(0)
    
    # Calculate squared Euclidean distance
    distance = torch.norm(anchor - augmented, dim=1) ** 2
    
    # Calculate pull loss term
    pull_loss = similarity_score * distance
    
    # Calculate push loss term
    push_loss = (1 - similarity_score) * torch.clamp(margin - distance, min=0) ** 2
    
    # Combine losses
    loss = (pull_loss + push_loss).mean()
    
    return loss

def continuously_weighted_contrastive_loss(anchor, augmented, similarity_score, temperature=0.1):
    """Continuously Weighted Contrastive Loss as defined in equation (4) of the paper."""
    batch_size = anchor.size(0)
    
    # Calculate negative cosine similarity (distance)
    distance = -torch.sum(anchor * augmented, dim=1)
    
    # Calculate normalized weights
    exp_distance = torch.exp(-distance / temperature)
    denominator = exp_distance.sum()
    
    # Calculate loss
    loss = -torch.mean(similarity_score * torch.log(exp_distance / denominator))
    
    return loss

def evaluate_embedding_space(embeddings, labels, k=5):
    """Evaluate the quality of the embedding space using metrics from the paper."""
    
    # Convert embeddings and labels to numpy arrays
    embeddings = embeddings.cpu().detach().numpy()
    labels = np.array(labels)

    np.fill_diagonal(similarity_matrix, -np.inf)
    
    # Find k-nearest neighbors

            label_counts[label] += 1
        
        # Calculate entropy

            entropy -= prob * np.log2(prob) if prob > 0 else 0
        
        # Normalize entropy
        max_entropy = np.log2(min(k, len(np.unique(labels))))
        normalized_entropy = entropy / max_entropy if max_entropy > 0 else 0
        
        g_knn_scores.append(1 - normalized_entropy)
    
    g_knn = np.mean(g_knn_scores)
    
    # Calculate nearest neighbor accuracy
    knn_accuracy = 0
    for i, indices in enumerate(nearest_indices[:, 0:1]):  # Just the closest neighbor
        if labels[i] == labels[indices[0]]:
            knn_accuracy += 1
    knn_accuracy /= len(embeddings)
    
    # Calculate KL divergence and Jensen-Shannon divergence
    # For simplicity, we'll use a basic implementation
    global_label_dist = {}
    for label in np.unique(labels):
        global_label_dist[label] = np.sum(labels == label) / len(labels)
    
    kl_divergence_scores = []
    js_divergence_scores = []
    
    for i, indices in enumerate(nearest_indices):
        local_label_dist = {}
        for label in np.unique(labels):
            local_label_dist[label] = np.sum(labels[indices] == label) / k
        
        # KL divergence
        kl_div = 0
        for label in local_label_dist:
            if local_label_dist[label] > 0 and global_label_dist.get(label, 0) > 0:
                kl_div += local_label_dist[label] * np.log(local_label_dist[label] / global_label_dist[label])
        kl_divergence_scores.append(kl_div)
        
        # JS divergence
        m_dist = {label: (local_label_dist.get(label, 0) + global_label_dist.get(label, 0)) / 2 
                 for label in set(local_label_dist) | set(global_label_dist)}
        
        js_div = 0
        for label in m_dist:
            if local_label_dist.get(label, 0) > 0 and m_dist[label] > 0:
                js_div += 0.5 * local_label_dist[label] * np.log(local_label_dist[label] / m_dist[label])

                js_div += 0.5 * global_label_dist[label] * np.log(global_label_dist[label] / m_dist[label])
        js_divergence_scores.append(js_div)
    
    kl_divergence = np.mean(kl_divergence_scores)
    js_divergence = np.mean(js_divergence_scores)
    
    return {
        'g_knn': g_knn,
        'knn_accuracy': knn_accuracy,
        'kl_divergence': kl_divergence,
        'js_divergence': js_divergence
    }

def train_contrasim(train_loader, model, optimizer, loss_fn, epochs=50):
    """Train the ContraSim projection network."""
    model.train()
    
    for epoch in range(epochs):
        total_loss = 0
        for batch in train_loader:
            optimizer.zero_grad()
            
            # Get embeddings
            original_emb = batch['original_embedding'].float()
            augmented_emb = batch['augmented_embedding'].float()
            similarity = batch['similarity_score'].float()
            
            # Project embeddings
            original_proj = model(original_emb)
            augmented_proj = model(augmented_emb)
            
            # Calculate loss
            loss = loss_fn(original_proj, augmented_proj, similarity)
            
            # Backpropagate
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()
        
        # Print progress
        if (epoch + 1) % 10 == 0:
            print(f"Epoch {epoch+1}/{epochs}, Loss: {total_loss/len(train_loader):.4f}")
    
    return model
