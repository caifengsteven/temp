        # Instantiate decoder

        # Full autoencoder

class TD3Agent:
    """Twin Delayed DDPG agent for portfolio allocation"""
    
    def __init__(self, state_dim, action_dim, latent_dim=500):
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.latent_dim = latent_dim
        self.memory = deque(maxlen=10000)
        self.batch_size = 64
        self.gamma = 0.99
        self.tau = 0.005
        self.policy_update_freq = 2
        self.update_counter = 0

        self.actor = self._build_actor()
        self.actor_target = self._build_actor()
        self.actor_target.set_weights(self.actor.get_weights())
        
        # Critic models (Q-networks)
        self.critic_1 = self._build_critic()
        self.critic_2 = self._build_critic()
        self.critic_target_1 = self._build_critic()
        self.critic_target_2 = self._build_critic()
        self.critic_target_1.set_weights(self.critic_1.get_weights())
        self.critic_target_2.set_weights(self.critic_2.get_weights())
        
        # Optimizers
        self.actor_optimizer = keras.optimizers.Adam(learning_rate=0.0001)
        self.critic_optimizer = keras.optimizers.Adam(learning_rate=0.0002)
        
    def _build_actor(self):
        """Build the actor (policy) network"""
        inputs = keras.Input(shape=(self.latent_dim,))
        x = keras.layers.Dense(256, activation='relu')(inputs)
        x = keras.layers.Dense(256, activation='relu')(x)
        x = keras.layers.Dense(256, activation='relu')(x)
        outputs = keras.layers.Dense(self.action_dim, activation='softmax')(x)
        
        model = keras.Model(inputs, outputs)
        return model
    
    def _build_critic(self):
        """Build the critic (Q-value) network"""
        state_input = keras.Input(shape=(self.latent_dim,))
        action_input = keras.Input(shape=(self.action_dim,))
        concat = keras.layers.Concatenate()([state_input, action_input])
        
        x = keras.layers.Dense(256, activation='relu')(concat)
        x = keras.layers.Dense(256, activation='relu')(x)
        x = keras.layers.Dense(256, activation='relu')(x)
        outputs = keras.layers.Dense(1)(x)
        
        model = keras.Model([state_input, action_input], outputs)
        return model
    
    def get_action(self, state, add_noise=True):
        """Get action from actor network with optional noise"""
        state = np.expand_dims(state, axis=0)
        action = self.actor.predict(state)[0]
        
        if add_noise:
            noise = np.random.normal(0, 0.1, size=self.action_dim)
            action = action + noise
            action = np.clip(action, 0, 1)
            action = action / np.sum(action)  # Normalize to sum to 1
            
        return action
    
    def remember(self, state, action, reward, next_state, done):
        """Store experience in replay memory"""
        self.memory.append((state, action, reward, next_state, done))
    
    def train(self):
        """Train the agent using experiences from memory"""
        if len(self.memory) < self.batch_size:
            return
        
        # Sample mini-batch from memory
        mini_batch = random.sample(self.memory, self.batch_size)
        
        states = np.array([experience[0] for experience in mini_batch])
        actions = np.array([experience[1] for experience in mini_batch])
        rewards = np.array([experience[2] for experience in mini_batch])
        next_states = np.array([experience[3] for experience in mini_batch])
        dones = np.array([experience[4] for experience in mini_batch])
        
        # Update critics
        with tf.GradientTape(persistent=True) as tape:
            # Actions for next states from target actor
            target_actions = self.actor_target(next_states)
            
            # Add noise to target actions for smoothing
            noise = np.clip(np.random.normal(0, 0.2, size=(self.batch_size, self.action_dim)), -0.5, 0.5)

            q2_next = self.critic_target_2([next_states, target_actions])
            
            # Use minimum Q-value for next state
            q_next = tf.minimum(q1_next, q2_next)
            
            # Compute target Q-value
            q_target = rewards + (1 - dones) * self.gamma * q_next
            
            # Current Q-values from critics
            q1 = self.critic_1([states, actions])
            q2 = self.critic_2([states, actions])
            
            # Compute critic losses
            critic1_loss = tf.reduce_mean(tf.square(q_target - q1))
            critic2_loss = tf.reduce_mean(tf.square(q_target - q2))

        critic1_gradients = tape.gradient(critic1_loss, self.critic_1.trainable_variables)
        self.critic_optimizer.apply_gradients(zip(critic1_gradients, self.critic_1.trainable_variables))
        
        # Update critic 2
        critic2_gradients = tape.gradient(critic2_loss, self.critic_2.trainable_variables)
        self.critic_optimizer.apply_gradients(zip(critic2_gradients, self.critic_2.trainable_variables))

    def test(self, episodes=1):
        """Test the trained agent"""
        total_portfolio_values = []
        daily_returns = []
        
        for episode in range(episodes):
            state = self.env.reset('test')
            portfolio_values = [self.env.portfolio_value]
            done = False
            
            while not done:
                # Encode state
                encoded_state = self.encoder.encode(state)
                
                # Get action from agent (no exploration noise)
                action = self.agent.get_action(encoded_state, add_noise=False)
                
                # Take step in environment
                next_state, _, done, info = self.env.step(action)
                
                # Update state
                state = next_state
                
                # Record portfolio value and return
                portfolio_values.append(info['portfolio_value'])
                daily_returns.append(info['return'])
            
            total_portfolio_values.append(portfolio_values)
            
            # Print episode stats
            print(f"Test Episode {episode+1}/{episodes}, Final Portfolio Value: {portfolio_values[-1]:.2f}")
            
            # Calculate Sharpe ratio
            returns_array = np.array(daily_returns)
            sharpe_ratio = np.mean(returns_array) / (np.std(returns_array) + 1e-6) * np.sqrt(252)
            print(f"Annualized Sharpe Ratio: {sharpe_ratio:.4f}")
            
            # Calculate max drawdown
            portfolio_array = np.array(portfolio_values)
            peak = np.maximum.accumulate(portfolio_array)
            drawdown = (portfolio_array - peak) / peak
            max_drawdown = np.min(drawdown)
            print(f"Maximum Drawdown: {max_drawdown:.4f}")
        
        return total_portfolio_values, daily_returns

def test_value_weighted_portfolio(env, episodes=1):
    """Test a value-weighted portfolio strategy"""
    total_portfolio_values = []
    daily_returns = []
    
    for episode in range(episodes):
        state = env.reset('test')
        portfolio_values = [env.portfolio_value]
        done = False
        
        # Initialize with equal weights in all stocks
        weights = np.ones(env.n_stocks) / env.n_stocks
        
        while not done:
            # Create action (include cash weight of 0)
            action = np.zeros(env.n_stocks + 1)
            action[:-1] = weights
            
            # Take step in environment
            next_state, _, done, info = env.step(action)
            
            # Update state
            state = next_state
            
            # Record portfolio value and return
            portfolio_values.append(info['portfolio_value'])
            daily_returns.append(info['return'])
            
            # Update weights based on market capitalization (simulated by current prices)
            if not done:
                market_caps = env.data['prices'][env.current_step]
                weights = market_caps / np.sum(market_caps)
        
        total_portfolio_values.append(portfolio_values)
        
        # Print episode stats
        print(f"Value-Weighted Portfolio - Final Value: {portfolio_values[-1]:.2f}")
        
        # Calculate Sharpe ratio
        returns_array = np.array(daily_returns)
        sharpe_ratio = np.mean(returns_array) / (np.std(returns_array) + 1e-6) * np.sqrt(252)
        print(f"Annualized Sharpe Ratio: {sharpe_ratio:.4f}")
    
    return total_portfolio_values, daily_returns

This code implements a simplified version of the DERL framework described in the paper. It includes:
