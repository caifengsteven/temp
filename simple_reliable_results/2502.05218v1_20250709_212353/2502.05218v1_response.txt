class FeatureExtractor(nn.Module):
    def __init__(self, input_dim, hidden_dim):
        super(FeatureExtractor, self).__init__()
        self.gru = nn.GRU(input_dim, hidden_dim, batch_first=True)
        self.bn = nn.BatchNorm1d(hidden_dim)
        
    def forward(self, x):
        # x: (batch_size, seq_len, input_dim)
        _, h_n = self.gru(x)  # h_n: (1, batch_size, hidden_dim)
        h_n = h_n.squeeze(0)  # (batch_size, hidden_dim)
        return self.bn(h_n)

class HyperGCN(nn.Module):
    def __init__(self, node_dim, out_dim):
        super(HyperGCN, self).__init__()
        self.w = nn.Linear(node_dim, out_dim, bias=False)
        
    def forward(self, node_features, incidence_matrix):
        # node_features: (batch_size, N, node_dim)
        # incidence_matrix: (batch_size, N, E)
        
        batch_size, N, _ = node_features.shape
        E = incidence_matrix.shape[2]
        
        # Calculate degree matrices
        D_n = torch.sum(incidence_matrix, dim=2)  # (batch_size, N)
        D_n = torch.diag_embed(torch.pow(D_n + 1e-8, -0.5))  # (batch_size, N, N)
        
        D_e = torch.sum(incidence_matrix, dim=1)  # (batch_size, E)
        D_e = torch.diag_embed(torch.pow(D_e + 1e-8, -1))  # (batch_size, E, E)
        
        # Apply HyperGCN formula
        x = torch.matmul(node_features, self.w.weight.T)  # (batch_size, N, out_dim)
        
        # D_n^(-1/2) H
        x = torch.bmm(D_n, torch.bmm(incidence_matrix, D_e))  # (batch_size, N, E)
        
        # D_n^(-1/2) H D_e^(-1) H^T D_n^(-1/2) X
        x = torch.bmm(x, torch.transpose(incidence_matrix, 1, 2))  # (batch_size, N, N)
        x = torch.bmm(x, D_n)  # (batch_size, N, N)
        x = torch.bmm(x, node_features)  # (batch_size, N, node_dim)
        
        return F.leaky_relu(x)

class FactorGCL(nn.Module):
    def __init__(self, input_dim, hidden_dim, num_stocks, num_prior_factors, num_hidden_factors, pred_periods):
        super(FactorGCL, self).__init__()
        self.hidden_dim = hidden_dim
        self.num_hidden_factors = num_hidden_factors
        self.pred_periods = pred_periods
        
        # Feature extractor
        self.feature_extractor = FeatureExtractor(input_dim, hidden_dim)
        self.feature_extractor_future = FeatureExtractor(input_dim, hidden_dim)
        
        # Prior beta module
        self.prior_beta = HyperGCN(hidden_dim, hidden_dim)
        
        # Hidden beta module
        self.hidden_factor_prototypes = nn.Parameter(torch.randn(num_hidden_factors, hidden_dim))
        self.hidden_beta = HyperGCN(hidden_dim, hidden_dim)
        
        # Individual alpha module
        self.alpha = nn.Linear(hidden_dim, hidden_dim)
        
        # Output layers for multiple prediction periods
        self.output_layers = nn.ModuleList([
            nn.ModuleList([
                nn.Linear(hidden_dim, 1),  # Prior beta output
                nn.Linear(hidden_dim, 1),  # Hidden beta output
                nn.Linear(hidden_dim, 1)   # Alpha output
            ]) for _ in range(len(pred_periods))
        ])
        
        # Contrastive learning projector
        self.projector = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.LeakyReLU(),
            nn.Linear(hidden_dim, hidden_dim)
        )
        
    def forward(self, x, x_future=None, prior_factors=None, training=True):
        batch_size, N, T, D = x.shape
        
        # Reshape for feature extractor
        x_reshaped = x.view(batch_size * N, T, D)
        es = self.feature_extractor(x_reshaped)  # (batch_size * N, hidden_dim)
        es = es.view(batch_size, N, -1)  # (batch_size, N, hidden_dim)
        
        # Prior beta module
        ep = self.prior_beta(es, prior_factors)
        
        # Residual after prior beta
        er = es - ep
        
        # Hidden beta module - generate hidden factor exposures
        hidden_factors = torch.matmul(er, self.hidden_factor_prototypes.t())  # (batch_size, N, num_hidden_factors)
        hidden_factors = torch.sigmoid(hidden_factors)
        
        # Apply hidden beta module
        eh = self.hidden_beta(er, hidden_factors)
        
        # Individual alpha module
        eα = F.leaky_relu(self.alpha(es - ep - eh))
        
        # Prediction for multiple periods
        predictions = []
        for i, layer_set in enumerate(self.output_layers):
            pred_p = layer_set[0](ep)  # Prior beta contribution
            pred_h = layer_set[1](eh)  # Hidden beta contribution
            pred_a = layer_set[2](eα)  # Alpha contribution
            pred = pred_p + pred_h + pred_a
            predictions.append(pred.squeeze(-1))
        
        # If training, compute future embeddings for contrastive learning
        if training and x_future is not None:
            # Process future data
            x_future_reshaped = x_future.view(batch_size * N, -1, D)
            es_future = self.feature_extractor_future(x_future_reshaped)
            es_future = es_future.view(batch_size, N, -1)
            
            # Apply prior and hidden beta modules with same factors
            ep_future = self.prior_beta(es_future, prior_factors)
            er_future = es_future - ep_future
            eh_future = self.hidden_beta(er_future, hidden_factors)
            
            # Calculate future alpha embeddings
            eα_future = F.leaky_relu(self.alpha(es_future - ep_future - eh_future))
            
            # Project alpha embeddings for contrastive learning
            p_eα = self.projector(eα.view(batch_size * N, -1))
            p_eα_future = self.projector(eα_future.view(batch_size * N, -1))
            
            return predictions, p_eα, p_eα_future
        
        return predictions
    
    def contrastive_loss(self, p_eα, p_eα_future, temperature=0.1):
        # p_eα, p_eα_future: (batch_size * N, hidden_dim)
        batch_size_n = p_eα.shape[0]
        
        # Normalize the embeddings
        p_eα = F.normalize(p_eα, dim=1)
        p_eα_future = F.normalize(p_eα_future, dim=1)
        
        # Calculate similarity matrix
        sim_matrix = torch.matmul(p_eα, p_eα_future.t()) / temperature  # (batch_size * N, batch_size * N)
        
        # Positive pairs are on diagonal
        positives = torch.diag(sim_matrix)  # (batch_size * N)
        
        # Calculate InfoNCE loss
        loss = -torch.mean(
            positives - torch.logsumexp(sim_matrix, dim=1)
        )
        
        return loss

def calculate_ic(predictions, targets):
    """Calculate Information Coefficient (Spearman rank correlation)"""
    batch_size, num_stocks = predictions.shape
    ics = []
    
    for i in range(batch_size):
        pred = predictions[i].detach().cpu().numpy()
        tgt = targets[i].detach().cpu().numpy()
        
        # Calculate Spearman rank correlation
        pred_rank = pd.Series(pred).rank()
        tgt_rank = pd.Series(tgt).rank()
        corr = pred_rank.corr(tgt_rank, method='spearman')
        
        if not np.isnan(corr):
            ics.append(corr)
    
    return np.mean(ics) if ics else 0

def calculate_icir(predictions, targets):
    """Calculate ICIR (Information Coefficient Information Ratio)"""
    batch_size, num_stocks = predictions.shape
    ics = []
    
    for i in range(batch_size):
        pred = predictions[i].detach().cpu().numpy()
        tgt = targets[i].detach().cpu().numpy()
        
        # Calculate Spearman rank correlation
        pred_rank = pd.Series(pred).rank()
        tgt_rank = pd.Series(tgt).rank()
        corr = pred_rank.corr(tgt_rank, method='spearman')
        
        if not np.isnan(corr):
            ics.append(corr)
    
    return np.mean(ics) / (np.std(ics) + 1e-8) if ics else 0

def generate_simulated_data(num_stocks=300, time_steps=500, num_features=6, num_industry_factors=10):
    """Generate simulated stock data and industry factors"""
    # Generate stock prices with random walk
    data = np.zeros((num_stocks, time_steps, num_features))
    
    # Initial prices
    initial_prices = np.random.uniform(50, 200, num_stocks)
    
    # Generate industry factors (one-hot encoding)
    prior_factors = np.zeros((num_stocks, num_industry_factors))
    for i in range(num_stocks):
        industry = np.random.randint(0, num_industry_factors)
        prior_factors[i, industry] = 1
    
    # Generate stock data based on industry factors and some hidden factors
    hidden_factors = np.random.normal(0, 1, (num_stocks, 5))  # 5 hidden factors
    
    # Generate price paths
    for stock_idx in range(num_stocks):
        price = initial_prices[stock_idx]
        
        # Industry effect
        industry_idx = np.argmax(prior_factors[stock_idx])
        industry_volatility = 0.01 + 0.005 * industry_idx  # Different industries have different volatilities
        
        # Hidden factor effect
        hidden_volatility = 0.005 * np.sum(hidden_factors[stock_idx])
        
        for t in range(time_steps):
            # Price (VWAP)
            price = price * (1 + np.random.normal(0.0002, industry_volatility + hidden_volatility))
            data[stock_idx, t, 0] = price
            
            # High price
            data[stock_idx, t, 1] = price * (1 + abs(np.random.normal(0, 0.005)))
            
            # Open price
            data[stock_idx, t, 2] = price * (1 + np.random.normal(0, 0.003))
            
            # Low price
            data[stock_idx, t, 3] = price * (1 - abs(np.random.normal(0, 0.005)))
            
            # Close price
            data[stock_idx, t, 4] = price * (1 + np.random.normal(0, 0.003))
            
            # Volume
            base_volume = np.random.normal(100000, 30000)
            if t > 0 and abs(data[stock_idx, t, 0] - data[stock_idx, t-1, 0]) > 0.01 * data[stock_idx, t-1, 0]:
                # Higher volume on bigger price moves
                base_volume *= 1.5
            data[stock_idx, t, 5] = max(base_volume, 10000)  # Ensure positive volume
    
    return data, prior_factors

def train_model(model, train_loader, optimizer, device, num_epochs=10):
    model.train()
    history = {'loss': [], 'ic': [], 'icir': []}
    
    for epoch in range(num_epochs):
        epoch_loss = 0
        epoch_ic = 0
        epoch_icir = 0
        num_batches = 0
        
        for x, x_future, prior_factors, y in tqdm(train_loader, desc=f"Epoch {epoch+1}/{num_epochs}"):
            x = x.to(device).float()
            x_future = x_future.to(device).float()
            prior_factors = prior_factors.to(device).float()
            y = y.to(device).float()
            
            # Forward pass
            predictions, p_eα, p_eα_future = model(x, x_future, prior_factors, training=True)
            
            # MSE Loss
            mse_loss = 0
            for i, pred in enumerate(predictions):
                mse_loss += F.mse_loss(pred, y[:, :, i])
            mse_loss /= len(predictions)
            
            # Contrastive Learning Loss
            cl_loss = model.contrastive_loss(p_eα, p_eα_future)
            
            # Total loss
            loss = mse_loss + 0.1 * cl_loss  # gamma=0.1
            
            # Backward and optimize
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            # Calculate metrics
            ic_values = [calculate_ic(pred, y[:, :, i]) for i, pred in enumerate(predictions)]
            icir_values = [calculate_icir(pred, y[:, :, i]) for i, pred in enumerate(predictions)]
            
            epoch_loss += loss.item()
            epoch_ic += np.mean(ic_values)
            epoch_icir += np.mean(icir_values)
            num_batches += 1
        
        # Log epoch metrics
        avg_loss = epoch_loss / num_batches
        avg_ic = epoch_ic / num_batches
        avg_icir = epoch_icir / num_batches
        
        history['loss'].append(avg_loss)
        history['ic'].append(avg_ic)
        history['icir'].append(avg_icir)
        
        print(f"Epoch {epoch+1}/{num_epochs}, Loss: {avg_loss:.4f}, IC: {avg_ic:.4f}, ICIR: {avg_icir:.4f}")
    
    return history

def evaluate_model(model, test_loader, device):
    model.eval()
    all_ics = [[] for _ in range(len(model.pred_periods))]
    all_icirs = [[] for _ in range(len(model.pred_periods))]
    
    with torch.no_grad():
        for x, x_future, prior_factors, y in tqdm(test_loader, desc="Evaluating"):
            x = x.to(device).float()
            prior_factors = prior_factors.to(device).float()
            y = y.to(device).float()
            
            # Forward pass
            predictions = model(x, prior_factors=prior_factors, training=False)
            
            # Calculate metrics for each prediction period
            for i, pred in enumerate(predictions):
                ic = calculate_ic(pred, y[:, :, i])
                icir = calculate_icir(pred, y[:, :, i])
                
                all_ics[i].append(ic)
                all_icirs[i].append(icir)
    
    # Calculate average metrics
    avg_ics = [np.mean(ics) for ics in all_ics]
    avg_icirs = [np.mean(icirs) for icirs in all_icirs]
    
    return avg_ics, avg_icirs

def backtest_strategy(model, test_data, test_prior_factors, device, pred_period_idx=2, topk=30, transaction_cost=0.003):
    """Simulate investment returns using the model predictions"""
    model.eval()
    
    N, T, D = test_data.shape
    seq_length = 60
    pred_period = model.pred_periods[pred_period_idx]
    
    # Create a time series of returns
    returns = np.zeros(T - seq_length - pred_period)
    benchmark_returns = np.zeros(T - seq_length - pred_period)
    
    for t in tqdm(range(T - seq_length - pred_period), desc="Backtesting"):
        # Extract features for current time step
        x = test_data[:, t:t+seq_length, :]
        x = torch.tensor(x).unsqueeze(0).to(device).float()
        
        # Get factor exposure
        prior_factors = torch.tensor(test_prior_factors).unsqueeze(0).to(device).float()
        
        # Get model predictions
        with torch.no_grad():
            predictions = model(x, prior_factors=prior_factors, training=False)
            pred = predictions[pred_period_idx].squeeze(0).cpu().numpy()
        
        # Select top K stocks
        selected_stocks = np.argsort(pred)[-topk:]
        
        # Calculate returns (equal weighted portfolio)
        portfolio_return = 0
        for stock_idx in selected_stocks:
            current_price = test_data[stock_idx, t+seq_length, 0]
            future_price = test_data[stock_idx, t+seq_length+pred_period, 0]
            stock_return = (future_price - current_price) / current_price - transaction_cost
            portfolio_return += stock_return / topk
        
        # Calculate benchmark return (equally weighted all stocks)
        benchmark_return = 0
        for stock_idx in range(N):
            current_price = test_data[stock_idx, t+seq_length, 0]
            future_price = test_data[stock_idx, t+seq_length+pred_period, 0]
            stock_return = (future_price - current_price) / current_price
            benchmark_return += stock_return / N
        
        returns[t] = portfolio_return
        benchmark_returns[t] = benchmark_return
    
    # Calculate cumulative returns
    cum_returns = (1 + returns).cumprod() - 1
    cum_benchmark_returns = (1 + benchmark_returns).cumprod() - 1
    excess_returns = returns - benchmark_returns
    cum_excess_returns = (1 + excess_returns).cumprod() - 1
    
    # Calculate performance metrics
    ann_return = (1 + np.mean(returns)) ** (252 / pred_period) - 1
    ann_benchmark_return = (1 + np.mean(benchmark_returns)) ** (252 / pred_period) - 1
    ann_excess_return = ann_return - ann_benchmark_return
    
    ann_vol = np.std(returns) * np.sqrt(252 / pred_period)
    ann_benchmark_vol = np.std(benchmark_returns) * np.sqrt(252 / pred_period)
    ann_excess_vol = np.std(excess_returns) * np.sqrt(252 / pred_period)
    
    information_ratio = ann_excess_return / ann_excess_vol if ann_excess_vol != 0 else 0
    
    # Calculate maximum drawdown
    def calculate_max_drawdown(returns):
        cum_returns = (1 + returns).cumprod()
        running_max = np.maximum.accumulate(cum_returns)
        drawdown = (cum_returns - running_max) / running_max
        return np.min(drawdown)
    
    max_drawdown = calculate_max_drawdown(returns)
    romad = ann_return / abs(max_drawdown) if max_drawdown != 0 else 0
    
    # Return results
    results = {
        'cumulative_returns': cum_returns,
        'cumulative_benchmark_returns': cum_benchmark_returns,
        'cumulative_excess_returns': cum_excess_returns,
        'annualized_return': ann_return,
        'annualized_benchmark_return': ann_benchmark_return,
        'annualized_excess_return': ann_excess_return,
        'information_ratio': information_ratio,
        'romad': romad,
        'max_drawdown': max_drawdown
    }
    
    return results

def run_ablation_study(model_class, train_loader, test_loader, device, hidden_dim=64, num_stocks=300,
