class AbstractShock:
    """Base class for economic shocks"""
    def __call__(self, model: Model) -> None:
        pass


class NoShock(AbstractShock):
    """Default no-op shock"""
    def __call__(self, model: Model) -> None:
        pass


class ConsumptionShock(AbstractShock):
    """Shock that affects consumption behavior"""
    def __init__(self, multiplier: float, final_time: int):
        self.multiplier = multiplier
        self.final_time = final_time
        self.original_psi = None
    
    def __call__(self, model: Model) -> None:
        if model.agg.t == 1:
            self.original_psi = model.prop.psi
            model.prop.psi *= self.multiplier
        elif model.agg.t == self.final_time:
            model.prop.psi = self.original_psi

def update_consumption(model: Model) -> None:
    """Update consumption for all households based on income and propensity to consume"""
    # Update active workers' consumption and savings
    model.w_act.C_h = model.prop.psi * model.w_act.Y_h
    model.w_act.S_h = model.w_act.Y_h - model.w_act.C_h
    
    # Update inactive workers' consumption and savings
    model.w_inact.C_h = model.prop.psi * model.w_inact.Y_h
    model.w_inact.S_h = model.w_inact.Y_h - model.w_inact.C_h


def firms_profits(firms: Firms, model: Model) -> np.ndarray:
    """Calculate firm profits"""
    # Simple profit calculation (revenue - costs)
    price = 1.0
    wage = 0.7
    
    revenue = price * firms.Y_i
    labor_cost = wage * firms.L_i
    capital_cost = 0.05 * firms.K_i  # 5% cost of capital
    
    return revenue - labor_cost - capital_cost


def update_central_bank_rate(cb: CentralBank, model: Model) -> None:
    """Update the central bank policy rate using Taylor rule"""
    inflation = model.agg.inflation
    
    # Simple output gap calculation
    potential_output = model.firms.total_production * 1.05  # Assume potential is 5% higher
    output_gap = (model.agg.GDP / potential_output) - 1
    
    # Apply Taylor rule
    cb.rate = cb.taylor_rule(inflation, output_gap)


def step(model: Model) -> None:
    """Perform one simulation step (one quarter)"""
    model.agg.t += 1
    
    # Update firm production and profits
    for i in range(model.firms.count):
        # Simple production function update
        model.firms.Y_i[i] = (model.firms.K_i[i] ** 0.3) * (model.firms.L_i[i] ** 0.7)
    
    # Update firm profits
    model.firms.Pi_i = firms_profits(model.firms, model)
    
    # Update bank profits and equity
    model.bank.Pi_k = model.bank.bank_profits(model)
    model.bank.E_k += model.bank.Pi_k
    
    # Update household consumption
    update_consumption(model)
    
    # Update government
    model.gov.T = model.gov.collect_taxes(model)
    model.gov.B += model.gov.G - model.gov.T  # Update debt
    
    # Update trade
    domestic_demand = model.w_act.total_consumption + model.w_inact.total_consumption + model.gov.G
    model.rotw.update_trade(domestic_demand)
    
    # Update aggregate variables
    model.agg.update_gdp(model)
    
    # Calculate inflation (simple AR(1) process with random shock)
    model.agg.inflation = 0.7 * model.agg.inflation + 0.3 * 0.02 + np.random.normal(0, 0.005)
    
    # Calculate unemployment rate (simple model)
    total_labor_force = model.w_act.count + model.w_inact.count
    employed = model.firms.total_labor
    model.agg.unemployment = 1 - (employed / total_labor_force)
    
    # Update central bank policy rate
    update_central_bank_rate(model.cb, model)


def update_data(data: SimulationData, model: Model) -> None:
    """Update the data object with the current state of the model"""
    data.nominal_gdp.append(model.agg.GDP)
    
    # Simple calculation of real GDP (adjusting for inflation)
    if model.agg.t == 1:
        data.real_gdp.append(model.agg.GDP)
    else:
        data.real_gdp.append(model.agg.GDP / (1 + model.agg.inflation))
    
    data.real_household_consumption.append(model.w_act.total_consumption + model.w_inact.total_consumption)
    data.inflation.append(model.agg.inflation)
    data.unemployment.append(model.agg.unemployment)
    data.policy_rate.append(model.cb.rate)


def run(model: Model, shock: AbstractShock = NoShock()) -> SimulationData:
    """Run a full simulation and collect data"""
    data = SimulationData()
    
    for _ in range(model.prop.T):
        # Apply any shocks
        shock(model)
        
        # Perform one simulation step
        step(model)
        
        # Update data
        update_data(data, model)
    
    return data

def test_taylor_rule() -> bool:
    """Test that the Taylor rule produces expected values"""
    cb = CentralBank(rate=0.02)
    
    # Test cases
    test_cases = [
        {"inflation": 0.02, "output_gap": 0.0, "expected": 0.02},
        {"inflation": 0.03, "output_gap": 0.0, "expected": 0.035},
        {"inflation": 0.02, "output_gap": 0.01, "expected": 0.025},
        {"inflation": 0.03, "output_gap": -0.01, "expected": 0.03}
    ]
    
    all_passed = True
    for case in test_cases:
        result = cb.taylor_rule(case["inflation"], case["output_gap"])
        passed = abs(result - case["expected"]) < 1e-10
        
        if not passed:
            print(f"Taylor rule test failed: expected {case['expected']}, got {result}")
            all_passed = False
    
    return all_passed
