class TeMoP:
    """
    Trend-encoded Probabilistic Multi-Order Model as described in the paper.
    This is a simplified implementation focusing on the core concepts.
    """
    
    def __init__(self):
        self.max_lag_order = None
        self.trend_subsets = {}
        self.subset_stats = {}
        self.m = 50  # Threshold for determining large sample
        
    def trend_encode(self, values):
        """Convert a series to trend encoding (-1 for down, 1 for up)"""
        result = []
        for i in range(1, len(values)):
            if values[i] >= values[i-1]:
                result.append(1)
            else:
                result.append(-1)
        return np.array(result)
    
    def create_lagged_samples(self, data, lag_order):
        """Create samples with specified lag order"""
        X = []
        y = []
        
        for i in range(len(data) - lag_order):
            sample = data[i:i+lag_order]
            label = 1 if data[i+lag_order] >= data[i+lag_order-1] else -1
            X.append(sample)
            y.append(label)
            
        return np.array(X), np.array(y)
    
    def fit(self, Z):
        """Train the model on time series data Z"""
        i = 1
        
        while True:
            # Create samples with lag order i
            X, y = self.create_lagged_samples(Z, i)
            
            # Create trend encoded samples
            trend_encoded = []
            for sample in X:
                trend_encoded.append(self.trend_encode(sample))
            trend_encoded = np.array(trend_encoded)
            
            # Group samples by their trend encoding
            unique_trends = {}
            for j, trend in enumerate(trend_encoded):
                trend_key = tuple(trend)
                if trend_key not in unique_trends:
                    unique_trends[trend_key] = []
                unique_trends[trend_key].append(j)
            
            # Check if any subset has fewer than m samples
            if any(len(indices) < self.m for indices in unique_trends.values()):
                i += 1
                if i > 30:  # Set a maximum lag to avoid infinite loop
                    i -= 1

            # Split each subset by label
            subset_stats = {}
            for trend_key, indices in unique_trends.items():
                pos_indices = [idx for idx in indices if y[idx] == 1]
                neg_indices = [idx for idx in indices if y[idx] == -1]

                pos_samples = X[pos_indices] if pos_indices else np.array([])
                neg_samples = X[neg_indices] if neg_indices else np.array([])
                
                subset_stats[trend_key] = {
                    'pos_count': len(pos_indices),
                    'neg_count': len(neg_indices),
                    'pos_mean': np.mean(pos_samples, axis=0) if pos_samples.size > 0 else None,
                    'neg_mean': np.mean(neg_samples, axis=0) if neg_samples.size > 0 else None,
                    'pos_cov': np.cov(pos_samples.T) if pos_samples.size > 0 else None,
                    'neg_cov': np.cov(neg_samples.T) if neg_samples.size > 0 else None,
                }
            
            self.trend_subsets[i] = unique_trends
            self.subset_stats[i] = subset_stats
            i += 1

                break
        
        self.max_lag_order = i - 1
        return self
    
    def overlap(self, x, y):
        """Calculate overlap between two trend vectors"""
        return np.mean(x == y)

        # Calculate scores for each lag order
        pos_scores = []
        neg_scores = []

                continue
                
            # Extract current sample
            x_i = Z_test[-i:]
            trend_x_i = self.trend_encode(x_i)
            
            # Calculate scores for each subset
            s_pos = 0
            s_neg = 0
            
            for trend_key, stats in self.subset_stats[i].items():
                trend_subset = np.array(trend_key)
                
                # Calculate similarity (membership degree)
                if len(trend_subset) == len(trend_x_i):
                    overlap_val = self.overlap(trend_subset, trend_x_i)
                    
                    # Calculate score based on trend features
                    s_pos_trend = stats['pos_count'] / (stats['pos_count'] + stats['neg_count'] + 2)
                    s_neg_trend = stats['neg_count'] / (stats['pos_count'] + stats['neg_count'] + 2)
                    
                    # Simplified Mahalanobis distance score
                    s_pos_orig = 0.5  # Simplified
                    s_neg_orig = 0.5  # Simplified
                    
                    # Combine scores
                    s_pos += overlap_val * (s_pos_trend + s_pos_orig)
                    s_neg += overlap_val * (s_neg_trend + s_neg_orig)
            
            pos_scores.append(s_pos)
            neg_scores.append(s_neg)
        
        # Combine scores from all lag orders
        total_pos_score = np.sum(pos_scores)
        total_neg_score = np.sum(neg_scores)
        
        # Calculate probability
        prob = np.exp(total_pos_score) / (np.exp(total_pos_score) + np.exp(total_neg_score))

def generate_synthetic_stock_data(n_points, volatility=0.01, trend=0.0001):
    """Generate synthetic stock price data"""
    # Start with initial price
    price = 100
    prices = [price]
    
    # Generate subsequent prices with random walk + trend
    for _ in range(n_points - 1):
        change = np.random.normal(trend, volatility) * price
        price += change
        prices.append(price)
    
    return np.array(prices)


def simulate_trading(predictions, price_data, initial_capital=10000):
    """Simulate trading based on predictions"""
    capital = initial_capital
    position = 0  # 0: no position, 1: long
    
    returns = []
    
    for i in range(1, len(predictions)):
        # Previous prediction determines today's position
        if predictions[i-1] == 1 and position == 0:
            # Buy at open
            position = 1
            entry_price = price_data[i]
        elif predictions[i-1] == -1 and position == 1:
            # Sell at open
            position = 0
            exit_price = price_data[i]
            returns.append((exit_price / entry_price) - 1)
    
    # Close any open position at the end
    if position == 1:
        exit_price = price_data[-1]
        returns.append((exit_price / entry_price) - 1)
    
    # Calculate Sharpe Ratio (simplified, assuming risk-free rate = 0)
    if returns:
        sharpe_ratio = np.mean(returns) / np.std(returns) if np.std(returns) > 0 else 0
    else:
        sharpe_ratio = 0
        
    return sharpe_ratio
