@dataclass
class HestonParameters:
    kappa: float  # mean reversion rate
    theta: float  # long-term mean of variance (μ_ν in the paper)
    sigma: float  # volatility of variance (σ_ν in the paper)
    rho: float    # correlation between asset and variance
    v0: float     # initial variance
    
    def __post_init__(self):
        # Check Feller condition
        if 2 * self.kappa * self.theta < self.sigma ** 2:
            print(f"Warning: Feller condition not satisfied: 2κθ = {2 * self.kappa * self.theta} < σ² = {self.sigma ** 2}")
        
    def as_tuple(self):
        return (self.kappa, self.theta, self.sigma, self.rho)
    
    def __str__(self):
        return f"κ={self.kappa:.4f}, θ={self.theta:.4f}, σ={self.sigma:.4f}, ρ={self.rho:.4f}, v0={self.v0:.4f}"


class HestonSDE:
    """Fine model: Heston SDE for Asian option pricing"""
    
    def __init__(self, S0, K, r, q, T, params: HestonParameters, n_paths=10000, n_steps=100):
        self.S0 = S0
        self.K = K
        self.r = r
        self.q = q
        self.T = T
        self.params = params
        self.n_paths = n_paths
        self.n_steps = n_steps
        self.dt = T / n_steps
        
    def simulate_paths(self):
        """Simulate paths using Euler-Maruyama scheme with antithetic variables"""
        dt = self.dt
        sqrt_dt = np.sqrt(dt)
        
        # Initialize arrays for stock price, variance, and their antithetic counterparts
        S = np.zeros((self.n_paths, self.n_steps + 1))
        v = np.zeros((self.n_paths, self.n_steps + 1))
        S_anti = np.zeros((self.n_paths, self.n_steps + 1))
        v_anti = np.zeros((self.n_paths, self.n_steps + 1))
        
        # Initial values
        S[:, 0] = self.S0
        v[:, 0] = self.params.v0
        S_anti[:, 0] = self.S0
        v_anti[:, 0] = self.params.v0
        
        # Extract parameters
        kappa = self.params.kappa
        theta = self.params.theta
        sigma = self.params.sigma
        rho = self.params.rho
        
        # Simulate paths
        for i in range(self.n_steps):
            # Generate random numbers
            z1 = np.random.normal(0, 1, self.n_paths)
            z2 = np.random.normal(0, 1, self.n_paths)
            
            # Correlation
            z2_corr = rho * z1 + np.sqrt(1 - rho**2) * z2
            
            # Ensure variance is positive
            v[:, i] = np.maximum(v[:, i], 0)
            v_anti[:, i] = np.maximum(v_anti[:, i], 0)
            
            # Update variance
            v[:, i+1] = v[:, i] + kappa * (theta - v[:, i]) * dt + sigma * np.sqrt(v[:, i]) * sqrt_dt * z2_corr
            v_anti[:, i+1] = v_anti[:, i] + kappa * (theta - v_anti[:, i]) * dt - sigma * np.sqrt(v_anti[:, i]) * sqrt_dt * z2_corr
            
            # Update stock price (log-space)
            log_S_increment = (self.r - self.q - 0.5 * v[:, i]) * dt + np.sqrt(v[:, i]) * sqrt_dt * z1
            log_S_anti_increment = (self.r - self.q - 0.5 * v_anti[:, i]) * dt - np.sqrt(v_anti[:, i]) * sqrt_dt * z1
            
            S[:, i+1] = S[:, i] * np.exp(log_S_increment)
            S_anti[:, i+1] = S_anti[:, i] * np.exp(log_S_anti_increment)
            
        return S, S_anti, v, v_anti
    
    def price_asian_option(self):
        """Price Asian put option using Monte Carlo with antithetic variates"""
        S, S_anti, _, _ = self.simulate_paths()
        
        # Calculate average stock prices for each path
        avg_S = np.mean(S, axis=1)
        avg_S_anti = np.mean(S_anti, axis=1)
        
        # Calculate payoffs
        payoff = np.maximum(self.K - avg_S, 0)
        payoff_anti = np.maximum(self.K - avg_S_anti, 0)
        
        # Average payoffs and discount
        avg_payoff = 0.5 * (payoff + payoff_anti)
        price = np.exp(-self.r * self.T) * np.mean(avg_payoff)
        
        return price


class HestonPDE:

    def __init__(self, S0, K, r, q, T, params: HestonParameters, nx=100, nv=50, nt=50):

        self.K = K
        self.r = r
        self.q = q
        self.T = T
        self.params = params

        self.nx = nx
        self.nv = nv
        self.nt = nt
        
        # Domain boundaries
        self.x_min = np.log(S0 / 5)
        self.x_max = np.log(S0 * 3)
        self.v_max = 1.0
        
        # Grid spacing
        self.dx = (self.x_max - self.x_min) / (nx - 1)
        self.dv = self.v_max / (nv - 1)
        self.dt = T / nt
        
        # Create grid
        self.x_grid = np.linspace(self.x_min, self.x_max, nx)
        self.v_grid = np.linspace(0, self.v_max, nv)
        
        # Find index for S0
        self.i_S0 = np.argmin(np.abs(np.exp(self.x_grid) - S0))
        
    def price_european_option(self):
        """
        Simplified finite difference method for the Heston PDE
        This is a basic implementation - in practice a more sophisticated scheme 
        like Hundsdorfer-Verwer would be used as mentioned in the paper
        """
        # Extract parameters
        kappa = self.params.kappa
        theta = self.params.theta
        sigma = self.params.sigma
        rho = self.params.rho
        v0 = self.params.v0
        
        # Initialize grid
        V = np.zeros((self.nx, self.nv))
        
        # Terminal condition (put option payoff)
        for i in range(self.nx):
            S = np.exp(self.x_grid[i])
            V[i, :] = np.maximum(self.K - S, 0)
        
        # Time stepping (backward in time)
        for n in range(self.nt):
            V_new = np.copy(V)
            
            # Interior points
            for i in range(1, self.nx-1):
                for j in range(1, self.nv-1):
                    x = self.x_grid[i]
                    v = self.v_grid[j]
                    
                    # Coefficients
                    a = 0.5 * v
                    b = self.r - self.q - 0.5 * v
                    c = kappa * (theta - v)
                    d = 0.5 * sigma**2 * v
                    e = rho * sigma * v
                    
                    # Finite difference approximations
                    d_x = (V[i+1, j] - V[i-1, j]) / (2 * self.dx)
                    d_xx = (V[i+1, j] - 2*V[i, j] + V[i-1, j]) / (self.dx**2)
                    d_v = (V[i, j+1] - V[i, j-1]) / (2 * self.dv)
                    d_vv = (V[i, j+1] - 2*V[i, j] + V[i, j-1]) / (self.dv**2)
                    d_xv = (V[i+1, j+1] - V[i+1, j-1] - V[i-1, j+1] + V[i-1, j-1]) / (4 * self.dx * self.dv)
                    
                    # PDE discretization
                    V_new[i, j] = V[i, j] + self.dt * (a * d_xx + b * d_x + c * d_v + d * d_vv + e * d_xv - self.r * V[i, j])
            
            # Boundary conditions
            # x_min (large put value)
            V_new[0, :] = self.K * np.exp(-self.r * (self.T - n*self.dt))
            # x_max (small put value)
            V_new[-1, :] = 0
            # v = 0 (degenerate case)
            V_new[:, 0] = V_new[:, 1]  # Simple approximation
            # v = v_max
            V_new[:, -1] = 2*V_new[:, -2] - V_new[:, -3]  # Extrapolation
            
            V = V_new
        
        # Find option price at S0, v0
        j_v0 = np.argmin(np.abs(self.v_grid - v0))
        price = V[self.i_S0, j_v0]
        
        return price
