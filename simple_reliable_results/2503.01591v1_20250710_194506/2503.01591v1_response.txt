import numpy as np

class FinancialDataGenerator:
    """Generate synthetic financial data for testing."""
    
    def __init__(self, n_assets=5, n_days=1000, volatility=0.015, seed=42):
        self.n_assets = n_assets
        self.n_days = n_days
        self.volatility = volatility
        np.random.seed(seed)
        
    def generate_prices(self):
        """Generate synthetic price data for multiple assets."""
        # Start with prices around 100
        prices = np.zeros((self.n_days, self.n_assets))
        prices[0] = 100 * np.random.uniform(0.8, 1.2, size=self.n_assets)
        
        # Generate random walk with drift
        for i in range(1, self.n_days):
            # Random returns with some correlation between assets
            correlation_matrix = np.random.uniform(0.3, 0.7, size=(self.n_assets, self.n_assets))
            np.fill_diagonal(correlation_matrix, 1)
            correlation_matrix = (correlation_matrix + correlation_matrix.T) / 2  # Make symmetric
            
            drift = np.random.uniform(0.0001, 0.0005, size=self.n_assets)
            daily_returns = np.random.multivariate_normal(
                drift, 
                self.volatility**2 * correlation_matrix
            )
            
            prices[i] = prices[i-1] * (1 + daily_returns)
        
        # Convert to DataFrame
        dates = [datetime.datetime.now() - datetime.timedelta(days=self.n_days-i) for i in range(self.n_days)]
        df = pd.DataFrame(prices, index=dates, columns=[f'Asset_{i+1}' for i in range(self.n_assets)])
        
        # Calculate returns
        returns = df.pct_change().dropna()
        
        return df, returns
    
    def add_technical_indicators(self, df):
        """Add technical indicators to the dataframe."""
        for col in df.columns:
            # Simple Moving Averages
            df[f'{col}_SMA_5'] = df[col].rolling(window=5).mean()
            df[f'{col}_SMA_20'] = df[col].rolling(window=20).mean()
            
            # Relative Strength Index (simplified)
            delta = df[col].diff()
            gain = (delta.where(delta > 0, 0)).rolling(window=14).mean()
            loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean()
            rs = gain / loss
            df[f'{col}_RSI'] = 100 - (100 / (1 + rs))
            
            # Bollinger Bands
            sma_20 = df[col].rolling(window=20).mean()
            std_20 = df[col].rolling(window=20).std()
            df[f'{col}_BB_upper'] = sma_20 + (std_20 * 2)
            df[f'{col}_BB_lower'] = sma_20 - (std_20 * 2)
            
        return df.dropna()


class DeepLearningModels:
    """Implement deep learning models for financial forecasting."""
    
    def __init__(self):
        self.scaler = MinMaxScaler(feature_range=(0, 1))
        self.model = None
        
    def prepare_data(self, data, look_back=60):
        """Prepare data for time series prediction."""
        scaled_data = self.scaler.fit_transform(data)
        X, y = [], []
        
        for i in range(look_back, len(scaled_data)):
            X.append(scaled_data[i-look_back:i])
            y.append(scaled_data[i])
            
        return np.array(X), np.array(y)
    
    def build_lstm_model(self, input_shape, output_shape):
        """Build an LSTM model for time series prediction."""
        model = Sequential()
        model.add(LSTM(units=50, return_sequences=True, input_shape=input_shape))
        model.add(Dropout(0.2))
        model.add(LSTM(units=50, return_sequences=False))
        model.add(Dropout(0.2))
        model.add(Dense(units=output_shape))
        
        model.compile(optimizer=Adam(learning_rate=0.001), loss='mean_squared_error')
        return model
    
    def train_model(self, X_train, y_train, epochs=50, batch_size=32, validation_split=0.2):
        """Train the LSTM model."""
        self.model = self.build_lstm_model((X_train.shape[1], X_train.shape[2]), y_train.shape[1])
        history = self.model.fit(
            X_train, y_train,
            epochs=epochs,
            batch_size=batch_size,
            validation_split=validation_split,
            verbose=0
        )
        return history
    
    def predict(self, X_test):
        """Make predictions using the trained model."""
        predictions = self.model.predict(X_test)
        return self.scaler.inverse_transform(predictions)


class PortfolioOptimization:
    """Portfolio optimization using deep learning predictions."""
    
    def __init__(self, prices, returns):
        self.prices = prices
        self.returns = returns
        
    def calculate_expected_returns(self, predictions, actual):
        """Calculate expected returns based on predictions and compare with actual."""
        predicted_returns = (predictions - self.prices.iloc[-len(predictions):].values) / self.prices.iloc[-len(predictions):].values
        
        print(f"Mean Squared Error: {mean_squared_error(actual, predictions):.6f}")
        print(f"Mean Absolute Error: {mean_absolute_error(actual, predictions):.6f}")
        print(f"R² Score: {r2_score(actual, predictions):.6f}")
        
        return predicted_returns
    
    def equal_weight_portfolio(self):
        """Create an equal-weight portfolio."""
        weights = np.ones(self.returns.shape[1]) / self.returns.shape[1]
        return weights
    
    def optimal_weight_portfolio(self, expected_returns, risk_aversion=2.0):
        """Create a simplified optimal weight portfolio based on expected returns and covariance."""
        cov_matrix = self.returns.cov() * 252  # Annualized covariance
        
        # Mean expected return
        mean_returns = np.mean(expected_returns, axis=0)
        
        # Simple optimization based on Markowitz model
        inv_cov = np.linalg.inv(cov_matrix.values)
        weights = (1/risk_aversion) * np.dot(inv_cov, mean_returns)
        
        # Normalize weights to sum to 1
        weights = weights / np.sum(np.abs(weights))
        
        return weights
    
    def calculate_portfolio_performance(self, weights, test_returns):
        """Calculate portfolio performance metrics."""
        # Portfolio returns
        portfolio_returns = np.dot(test_returns, weights)
        
        # Cumulative returns
        cumulative_returns = np.cumprod(1 + portfolio_returns) - 1
        
        # Annualized return
        annualized_return = (1 + portfolio_returns.mean()) ** 252 - 1
        
        # Annualized volatility
        annualized_vol = portfolio_returns.std() * np.sqrt(252)
        
        # Sharpe Ratio (assuming risk-free rate of 0 for simplicity)
        sharpe_ratio = annualized_return / annualized_vol
        
        # Maximum Drawdown
        cumulative = np.cumprod(1 + portfolio_returns)
        peak = cumulative.cummax()
        drawdown = (cumulative - peak) / peak
        max_drawdown = drawdown.min()
        
        performance = {
            "Annualized Return": annualized_return,
            "Annualized Volatility": annualized_vol,
            "Sharpe Ratio": sharpe_ratio,
            "Maximum Drawdown": max_drawdown,
            "Cumulative Returns": cumulative_returns
        }
        
        return performance
