import numpy as np
import matplotlib.pyplot as plt

class GibsonSchwartzModel:
    """
    Implementation of the Gibson-Schwartz model for commodity futures
    with convenience yield as described in the paper.
    """
    def __init__(
        self, 
        r: float = 0.03,                # risk-free rate
        s: float = 0.01,                # storage cost 
        sigma: float = 0.2,             # spot price volatility
        c0: float = 0.05,               # initial convenience yield
        kappa: float = 2.0,             # mean reversion speed
        theta: float = 0.05,            # long-term mean of convenience yield
        gamma: float = 0.3,             # convenience yield volatility
        rho: float = 0.3,               # correlation between spot and convenience yield
        x0: float = 4.6,                # initial log spot price
        delta: float = 1.0,             # time-scale parameter
    ):
        self.r = r
        self.s = s
        self.sigma = sigma
        self.c0 = c0
        self.kappa = kappa
        self.theta = theta
        self.gamma = gamma
        self.rho = rho
        self.x0 = x0
        self.delta = delta
    
    def simulate_paths(self, n_steps: int = 252, T: float = 1.0, 
                     seed: int = None) -> Tuple[np.ndarray, np.ndarray]:
        """
        Simulate paths for spot price and convenience yield.
        
        Args:
            n_steps: Number of time steps
            T: Time horizon
            seed: Random seed
            
        Returns:
            X: Log spot price path
            C: Convenience yield path
        """
        if seed is not None:
            np.random.seed(seed)
            
        dt = T / n_steps
        sqrt_dt = np.sqrt(dt)
        
        # Initialize paths
        X = np.zeros(n_steps + 1)
        C = np.zeros(n_steps + 1)
        
        # Set initial values
        X[0] = self.x0
        C[0] = self.c0
        
        # Generate correlated random numbers
        z1 = np.random.normal(0, 1, n_steps)
        z2 = self.rho * z1 + np.sqrt(1 - self.rho**2) * np.random.normal(0, 1, n_steps)
        
        # Simulate paths
        for i in range(n_steps):
            # Convenience yield (Ornstein-Uhlenbeck process)
            C[i+1] = C[i] + self.delta * self.kappa * (self.theta - C[i]) * dt + \
                    np.sqrt(self.delta) * self.gamma * sqrt_dt * z2[i]
            
            # Log spot price 
            X[i+1] = X[i] + (self.r - 0.5 * self.sigma**2 - C[i]) * dt + \
                    self.sigma * sqrt_dt * z1[i]
                    
        return X, C
    
    def compute_futures_curve(self, X: np.ndarray, C: np.ndarray, 
                            maturities: List[float]) -> np.ndarray:
        """
        Compute futures term structure for given spot and convenience yield paths.
        
        Args:
            X: Log spot price path
            C: Convenience yield path
            maturities: List of times to maturity
            
        Returns:
            F: Futures log price paths for each maturity
        """
        n_steps = len(X) - 1
        n_maturities = len(maturities)
        F = np.zeros((n_steps + 1, n_maturities))
        
        for t in range(n_steps + 1):
            for i, T_k in enumerate(maturities):
                # B(T_k) term from the Gibson-Schwartz model
                B_T = (1 - np.exp(-self.kappa * T_k)) / self.kappa
                
                # Futures log price
                F[t, i] = X[t] - B_T * C[t] + (self.r + self.s) * T_k
                
        return F


class SignatureCalculator:
    """
    Calculate truncated signatures for paths
    """
    def __init__(self, max_level: int = 3):
        """
        Initialize signature calculator with maximum level of iterated integrals.
        
        Args:
            max_level: Maximum level of signatures to compute
        """
        self.max_level = max_level
    
    def compute_signature_terms(self, path: np.ndarray) -> Dict[str, float]:
        """
        Compute truncated signature terms for a given path.
        This is a simplified implementation that focuses on key terms
        from the paper's expansion.
        
        Args:
            path: Path of shape (n_steps, n_dimensions)
            
        Returns:
            Dictionary of signature terms and their values
        """
        n_steps, n_dim = path.shape
        # Calculate increments
        increments = np.diff(path, axis=0)
        
        # Initialize signature dictionary
        sig_terms = {"1": 1.0}  # Level 0 term is always 1
        
        # Level 1 terms
        for i in range(n_dim):
            sig_terms[f"{i+1}"] = np.sum(increments[:, i])
        
        # Level 2 terms
        for i in range(n_dim):
            for j in range(n_dim):
                term = 0.0
                for t in range(n_steps-1):
                    term += increments[t, i] * np.sum(increments[t+1:, j])
                sig_terms[f"{i+1},{j+1}"] = term
        
        # For simplicity, we only implement up to level 2 signatures
        # Higher levels would follow similar pattern
        
        return sig_terms


def simulate_commodity(commodity_type: str, delta: float = 1.0, 
                     n_steps: int = 252, seed: int = None) -> Dict:
    """
    Simulate a commodity with parameters based on its type.
    
    Args:
        commodity_type: Type of commodity (e.g., 'metal', 'energy', 'soft')
        delta: Time-scale parameter
        n_steps: Number of time steps
        seed: Random seed
        
    Returns:
        Dictionary with simulation results
    """
    # Set parameters based on commodity type
    if commodity_type == 'metal':  # Highly storable
        params = {
            'gamma': 0.1,  # Low convenience yield volatility
            'kappa': 1.5
        }
    elif commodity_type == 'grain':  # Moderately storable
        params = {
            'gamma': 0.3,  # Medium convenience yield volatility
            'kappa': 2.0
        }
    elif commodity_type == 'energy':  # Less storable
        params = {
            'gamma': 0.8,  # High convenience yield volatility
            'kappa': 3.0
        }
    else:  # Default
        params = {
            'gamma': 0.3,
            'kappa': 2.0
        }
    
    # Create model
    model = GibsonSchwartzModel(
        gamma=params['gamma'],
        kappa=params['kappa'],
        delta=delta,
        seed=seed
    )
    
    # Simulate paths
    X, C = model.simulate_paths(n_steps=n_steps, seed=seed)
    
    # Define maturities (e.g., 1 month, 3 months, 6 months, 12 months)
    maturities = [1/12, 3/12, 6/12, 1.0]
    
    # Compute futures curve
    F = model.compute_futures_curve(X, C, maturities)
    
    # Compute signatures
    sig_calc = SignatureCalculator(max_level=2)
    
    # Prepare path data for signature calculation
    # Using log returns rather than levels
    F_returns = np.diff(F, axis=0)
    
    # Calculate signatures
    signatures = sig_calc.compute_signature_terms(F_returns)
    
    return {
        'commodity_type': commodity_type,
        'X': X,
        'C': C,
        'F': F,
        'returns': F_returns,
        'signatures': signatures,
        'params': {**model.__dict__, **params}
    }


def test_theoretical_result(n_simulations: int = 100):
    """
    Test the theoretical result from the paper by comparing 
    signature variations across commodities with different 
    convenience yield volatilities.
    """
    commodities = ['metal', 'grain', 'energy']
    results = []
    
    for commodity in commodities:
        # Run multiple simulations
        sig_variations = []
        
        for i in range(n_simulations):
            sim = simulate_commodity(commodity, seed=i)
            # Extract level 1 signature terms
            level1_terms = [v for k, v in sim['signatures'].items() if len(k) == 1 and k != '1']
            # Calculate variation
            sig_variations.append(np.std(level1_terms))
        
        # Store results
        results.append({
            'commodity': commodity,
            'mean_sig_variation': np.mean(sig_variations),
            'std_sig_variation': np.std(sig_variations),
            'gamma': simulate_commodity(commodity)['params']['gamma']
        })
    
    # Print results
    print("\n=== Signature Variation Analysis ===")
    results_df = pd.DataFrame(results)
    print(results_df)
    
    # Plot results
    plt.figure(figsize=(10, 6))
    plt.bar(
        results_df['commodity'], 
        results_df['mean_sig_variation'],
        yerr=results_df['std_sig_variation'],
        capsize=5
    )
    plt.xlabel('Commodity Type')
    plt.ylabel('Signature Variation (Std Dev of Level 1 Terms)')
    plt.title('Signature Variation vs Commodity Type')
    
    # Add convenience yield volatility as text
    for i, row in results_df.iterrows():
        plt.text(
            i, 
            row['mean_sig_variation'] + row['std_sig_variation'] + 0.01,
            f"γ={row['gamma']:.1f}",
            ha='center'
        )
    
    plt.tight_layout()
    plt.savefig('signature_variation.png')
    plt.close()
    
    return results_df
