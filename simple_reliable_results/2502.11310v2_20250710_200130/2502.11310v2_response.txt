class PCAlayer(nn.Module):
    """Implementation of differentiable PCA layer as described in the paper"""
    def __init__(self, input_dim, output_dim):
        super(PCAlayer, self).__init__()
        self.input_dim = input_dim
        self.output_dim = output_dim
        self.projection = nn.Parameter(torch.randn(input_dim, output_dim) / np.sqrt(input_dim))
        self.initialized = False
        
    def forward(self, x, initializing=False):
        if initializing or not self.initialized:
            # Compute covariance matrix
            x_centered = x - x.mean(dim=0, keepdim=True)
            cov_matrix = torch.mm(x_centered.t(), x_centered) / (x.shape[0] - 1)
            
            # Eigendecomposition with stability check
            try:
                eigenvalues, eigenvectors = torch.linalg.eigh(cov_matrix)
                # Sort in descending order
                idx = torch.argsort(eigenvalues, descending=True)
                eigenvalues = eigenvalues[idx]
                eigenvectors = eigenvectors[:, idx]
                
                # Check eigenvalue separation for stability
                min_diff = float('inf')
                for i in range(len(eigenvalues)-1):
                    diff = abs(eigenvalues[i] - eigenvalues[i+1])
                    if diff < min_diff:
                        min_diff = diff

                if min_diff <= 1e-10:

                    cov_matrix += noise
                    eigenvalues, eigenvectors = torch.linalg.eigh(cov_matrix)
                    idx = torch.argsort(eigenvalues, descending=True)
                    eigenvalues = eigenvalues[idx]
                    eigenvectors = eigenvectors[:, idx]
                
                # Take top k eigenvectors
                self.projection.data = eigenvectors[:, :self.output_dim] / np.sqrt(self.input_dim)
                self.initialized = True

                print(f"Error in PCA decomposition: {e}")
                # Fall back to random projection if decomposition fails
                self.projection.data = torch.randn(self.input_dim, self.output_dim) / np.sqrt(self.input_dim)
        
        # Project the data
        return torch.mm(x, self.projection)


class SoftPCAlayer(nn.Module):
    """Implementation of Soft PCA layer as described in the paper"""
    def __init__(self, input_dim, output_dim):
        super(SoftPCAlayer, self).__init__()
        self.input_dim = input_dim

        # Project the data
        output = torch.mm(x, self.projection)
        
        # Calculate losses for variance preservation and orthogonality
        # These would normally be added to the total loss, but for simplicity
        # we just compute them here and don't use them in the loss function
        
        # Variance preservation
        x_var = torch.var(x, dim=0, unbiased=True)
        output_var = torch.var(output, dim=0, unbiased=True)
        variance_loss = torch.norm(x_var.sum() - output_var.sum())**2
        
        # Orthogonality
        output_centered = output - output.mean(dim=0, keepdim=True)
        cov_matrix = torch.mm(output_centered.t(), output_centered) / (output.shape[0] - 1)
        diag_cov = torch.diag(torch.diag(cov_matrix))
        orthogonality_loss = torch.norm(cov_matrix - diag_cov, 'fro')**2
        
        return output


class AdditiveLayer(nn.Module):
    """Implementation of additive layer as described in the paper"""
    def __init__(self, input_dims, output_dims):
        super(AdditiveLayer, self).__init__()
        self.input_dims = input_dims
        self.output_dims = output_dims
        self.n_subnets = len(input_dims)
        
        # Create subnetworks
        self.subnets = nn.ModuleList([
            nn.Linear(input_dims[i], output_dims[i]) 
            for i in range(self.n_subnets)
        ])
        
    def forward(self, x):
        # Split input into segments
        start_idx = 0
        outputs = []
        
        for i in range(self.n_subnets):
            end_idx = start_idx + self.input_dims[i]
            subnet_input = x[:, start_idx:end_idx]
            outputs.append(self.subnets[i](subnet_input))
            start_idx = end_idx
            
        # Concatenate the outputs
        return torch.cat(outputs, dim=1)


class GFADNN(nn.Module):
    """Generalized Factor Additive Neural Network Model"""
    def __init__(self, input_dim, latent_dim1, latent_dim2, hidden_dim, output_dim, use_soft_pca=False):
        super(GFADNN, self).__init__()

        self.latent_dim1 = latent_dim1

            self.pca1 = SoftPCAlayer(input_dim, latent_dim1)
        else:
            self.pca1 = PCAlayer(input_dim, latent_dim1)

            nn.LeakyReLU(0.1)
        )

            self.pca2 = SoftPCAlayer(hidden_dim, latent_dim2)

        # Additive layers
        # Split latent_dim2 into 3 equal parts for simplicity
        part_size = latent_dim2 // 3
        remaining = latent_dim2 - part_size * 3
        self.input_split = [part_size, part_size, part_size + remaining]
        self.output_split = [1, 1, 1]  # Each subnet produces 1 output
        
        self.additive = AdditiveLayer(self.input_split, self.output_split)
        
    def forward(self, x, initializing=False):
        # First PCA layer
        if isinstance(self.pca1, PCAlayer):
            x = self.pca1(x, initializing)

            x = self.pca1(x)

        x = self.additive(x)

        return torch.sum(x, dim=1, keepdim=True)


def generate_latent_factors(n_samples, n_factors, distribution="uniform"):
    """Generate latent factors according to the specified distribution"""
    if distribution == "uniform":
        return np.random.uniform(-1, 1, (n_samples, n_factors))
    elif distribution == "normal":
        return np.random.normal(0, 1, (n_samples, n_factors))
    else:
        raise ValueError(f"Unknown distribution: {distribution}")


def generate_factor_loadings(n_factors, n_features, distribution="uniform"):
    """Generate factor loadings according to the specified distribution"""
    if distribution == "uniform":
        return np.random.uniform(-np.sqrt(3), np.sqrt(3), (n_factors, n_features))
    elif distribution == "normal":
        return np.random.normal(0, 1, (n_factors, n_features))
    else:
        raise ValueError(f"Unknown distribution: {distribution}")

        2: Non-linear factor model (B * g(Z))
        
    Returns:
    --------
    X : ndarray
        Feature matrix of shape (n_samples, n_features)
    y : ndarray
        Target vector of shape (n_samples,)
    Z : ndarray
        Latent factors of shape (n_samples, n_factors)
    """

    Z = generate_latent_factors(n_samples, n_factors)

    # Generate X based on case
    if case == 1:
        # Linear factor model: X = Z*B + u
        X = Z @ B + u
    elif case == 2:
        # Non-linear factor model: X = g(Z)*B + u

        g_Z = np.exp(Z)

def train_model(model, X_train, y_train, X_val, y_val, batch_size=32, epochs=100, lr=0.001, operating_schedule=None):
    """
    Train a PyTorch model with the specified parameters
    
    Parameters:
    -----------
    model : torch.nn.Module
        The model to train
    X_train, y_train : ndarray
        Training data
    X_val, y_val : ndarray
        Validation data
    batch_size : int
        Batch size for training

        Number of epochs to train
    lr : float
        Learning rate
    operating_schedule : list or None

    Returns:
    --------
    model : torch.nn.Module
        Trained model
    history : dict
        Training history
    """
    # Convert data to PyTorch tensors
    X_train_tensor = torch.FloatTensor(X_train)
    y_train_tensor = torch.FloatTensor(y_train).view(-1, 1)
    X_val_tensor = torch.FloatTensor(X_val)
    y_val_tensor = torch.FloatTensor(y_val).view(-1, 1)
    
    # Create data loaders
    train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
    
    # Define loss function and optimizer
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=lr)
    
    # Training loop
    history = {
        'train_loss': [],
        'val_loss': [],
        'explained_variance': []
    }
    
    best_val_loss = float('inf')
    patience = 10
    patience_counter = 0
    
    for epoch in range(1, epochs + 1):
        model.train()
        epoch_loss = 0
        
        # Determine if this epoch is in the operating schedule
        initializing = operating_schedule is not None and epoch in operating_schedule
        
        for batch_X, batch_y in train_loader:
            optimizer.zero_grad()
            
            # Forward pass with initialization flag
            output = model(batch_X, initializing)
            
            # Calculate loss
            loss = criterion(output, batch_y)
            
            # Backward pass and optimization
            loss.backward()
            optimizer.step()
            
            epoch_loss += loss.item() * batch_X.size(0)
        
        # Calculate training loss
        epoch_loss /= len(train_loader.dataset)
        history['train_loss'].append(epoch_loss)
        
        # Evaluate on validation set
        model.eval()
        with torch.no_grad():
            val_output = model(X_val_tensor)
            val_loss = criterion(val_output, y_val_tensor).item()
            history['val_loss'].append(val_loss)
        
        # Early stopping
        if val_loss < best_val_loss:
            best_val_loss = val_loss
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                print(f"Early stopping at epoch {epoch}")
                break
        
        # Print progress
        if epoch % 10 == 0:
            print(f"Epoch {epoch}/{epochs}, Train Loss: {epoch_loss:.6f}, Val Loss: {val_loss:.6f}")

def test_model_performance(model_name, model, X_test, y_test):
    """Test a model and return performance metrics"""
    model.eval()
    X_test_tensor = torch.FloatTensor(X_test)
    
    with torch.no_grad():
        y_pred = model(X_test_tensor).cpu().numpy().flatten()
    
    mse = mean_squared_error(y_test, y_pred)
    print(f"{model_name} Test MSE: {mse:.6f}")
    
    return {
        'model_name': model_name,
        'mse': mse,
        'y_pred': y_pred
