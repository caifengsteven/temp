class RealizedVolatilitySimulator:

class AutoencoderRealizedMeasure:
    """Creates a synthetic realized measure using an autoencoder"""
    
    def __init__(self, sparsity_reg=0.001, weight_reg=0.001, sparsity_param=0.05):
        self.sparsity_reg = sparsity_reg
        self.weight_reg = weight_reg
        self.sparsity_param = sparsity_param
        self.model = None
        self.encoder = None
        
    def build_model(self, input_dim):
        # Define custom sparsity regularization
        def kl_divergence(rho, rho_hat):
            return rho * tf.math.log(rho / rho_hat) + (1 - rho) * tf.math.log((1 - rho) / (1 - rho_hat))
        
        def sparse_regularization(activation):
            rho_hat = K.mean(activation)
            return self.sparsity_reg * kl_divergence(self.sparsity_param, rho_hat)
        
        # Input layer
        input_layer = Input(shape=(input_dim,))
        
        # Encoder (with l2 regularization and sparse activity)
        encoded = Dense(1, activation='sigmoid', 
                        activity_regularizer=sparse_regularization,
                        kernel_regularizer=l2(self.weight_reg))(input_layer)
        
        # Decoder
        decoded = Dense(input_dim, activation='sigmoid', 
                        kernel_regularizer=l2(self.weight_reg))(encoded)
        
        # Full model (autoencoder)
        autoencoder = Model(input_layer, decoded)
        
        # Encoder model (for extracting the synthetic measure)
        encoder = Model(input_layer, encoded)
        
        # Compile the model
        autoencoder.compile(optimizer=Adam(learning_rate=0.001), loss='mse')
        
        self.model = autoencoder
        self.encoder = encoder
        
        return autoencoder, encoder
    
    def fit(self, X, epochs=100, batch_size=32, verbose=0):
        self.model.fit(X, X, epochs=epochs, batch_size=batch_size, verbose=verbose)
    
    def get_synthetic_measure(self, X):
        encoded_values = self.encoder.predict(X).flatten()
        
        # Scale to range of original data (as done in the paper)
        min_val = np.min(X)
        max_val = np.max(X)
        encoded_scaled = (encoded_values - np.min(encoded_values)) / (np.max(encoded_values) - np.min(encoded_values))
        encoded_scaled = encoded_scaled * (max_val - min_val) + min_val
        
        return encoded_scaled

class RealGARCH:
    """Implementation of RealGARCH model"""
    
    def __init__(self):
        self.params = None
        
    def log_likelihood(self, params, returns, x_measure):
        """Log-likelihood function for RealGARCH"""
        omega, beta, gamma, xi, phi, tau1, tau2, sigma_e = params
        
        n = len(returns)
        log_sigma2 = np.zeros(n)
        z = np.zeros(n)
        log_x = np.log(x_measure)
        
        # Initialize
        log_sigma2[0] = np.log(np.var(returns))
        
        # GARCH equation
        for t in range(1, n):
            log_sigma2[t] = omega + beta * log_sigma2[t-1] + gamma * log_x[t-1]
        
        sigma = np.sqrt(np.exp(log_sigma2))
        z = returns / sigma
        
        # Measurement equation
        mu_x = xi + phi * log_sigma2 + tau1 * z + tau2 * (z**2 - 1)
        eps = log_x - mu_x
        
        # Log-likelihood for returns
        ll_r = -0.5 * (np.log(2*np.pi) + log_sigma2 + (returns**2) / np.exp(log_sigma2))
        
        # Log-likelihood for measurement
        ll_x = -0.5 * (np.log(2*np.pi) + np.log(sigma_e**2) + eps**2 / (sigma_e**2))
        
        # Total log-likelihood
        ll = np.sum(ll_r) + np.sum(ll_x)
        
        # Check for stationary condition
        if abs(beta + gamma * phi) >= 1:
            return -np.inf
        
        return ll
    
    def negative_log_likelihood(self, params, returns, x_measure):
        """Negative log-likelihood for minimization"""
        return -self.log_likelihood(params, returns, x_measure)
    
    def fit(self, returns, x_measure):
        """Fit RealGARCH model"""
        # Initial parameter guesses based on paper
        initial_params = [0.1, 0.6, 0.3, -0.4, 1.0, -0.1, 0.1, 0.5]
        
        # Parameter bounds
        bounds = [
            (0.001, 1.0),    # omega: positive
            (0.001, 0.999),  # beta: between 0 and 1
            (0.001, 0.999),  # gamma: between 0 and 1
            (-2.0, 2.0),     # xi
            (0.001, 2.0),    # phi: positive
            (-1.0, 1.0),     # tau1
            (-1.0, 1.0),     # tau2
            (0.001, 2.0)     # sigma_e: positive
        ]
        
        # Constraint to ensure stationarity: |beta + gamma*phi| < 1
        def constraint(params):
            beta, gamma, phi = params[1], params[2], params[4]
            return 0.999 - abs(beta + gamma * phi)
        
        constraints = {'type': 'ineq', 'fun': constraint}
        
        # Optimize
        try:
            result = minimize(self.negative_log_likelihood, initial_params, 
                             args=(returns, x_measure),
                             bounds=bounds,
                             constraints=constraints,
                             method='SLSQP')
            
            self.params = result.x
            return result
        except Exception as e:
            print(f"Optimization failed: {e}")
            return None
    
    def forecast_volatility(self, returns, x_measure, h=1):
        """Forecast volatility h steps ahead"""
        if self.params is None:
            raise ValueError("Model must be fit before forecasting")
            
        omega, beta, gamma, xi, phi, tau1, tau2, sigma_e = self.params
        
        n = len(returns)
        log_sigma2 = np.zeros(n+h)
        
        # Initialize
        log_x = np.log(x_measure)
        
        # GARCH equation for observed data
        for t in range(1, n):
            log_sigma2[t] = omega + beta * log_sigma2[t-1] + gamma * log_x[t-1]
        
        # Forecast future volatility
        for t in range(n, n+h):
            log_sigma2[t] = omega + beta * log_sigma2[t-1] + gamma * log_x[n-1]
        
        # Convert to volatility
        volatility = np.sqrt(np.exp(log_sigma2))
        
        return volatility[-h:]


class VolatilityForecaster:
    """Implements forecasting with different RealGARCH models"""
    
    def __init__(self):

    def prepare_data(self, data, in_sample_size):
        """Prepare training and testing data"""
        returns = data['returns']
        rv_measures = data['rv_measures'].values
        
        train_returns = returns[:in_sample_size]
        test_returns = returns[in_sample_size:]
        
        train_measures = rv_measures[:in_sample_size]
        test_measures = rv_measures[in_sample_size:]
        
        return {
            'train_returns': train_returns,
            'test_returns': test_returns,
            'train_measures': train_measures,
            'test_measures': test_measures
        }
    
    def forecast_one_step_ahead(self, model_type, train_data, test_data, window_size=None):
        """Rolling one-step-ahead forecasting with different models"""
        returns = np.concatenate([train_data['train_returns'], test_data['test_returns']])
        measures = np.concatenate([train_data['train_measures'], test_data['test_measures']], axis=0)
        
        n_train = len(train_data['train_returns'])
        n_test = len(test_data['test_returns'])
        
        # If window_size is None, use the entire training set
        if window_size is None:
            window_size = n_train
        
        forecasts = np.zeros(n_test)
        neg_log_likelihood = 0
        
        generator = SyntheticMeasureGenerator()
        
        # For each test point, train model and make one-step forecast
        for i in range(n_test):
            # Define rolling window
            start_idx = n_train + i - window_size
            end_idx = n_train + i
            
            window_returns = returns[start_idx:end_idx]
            window_measures = measures[start_idx:end_idx]
            
            # Generate synthetic measure based on model type
            if model_type == 'rv':
                # Use the first realized measure (RV5)
                x_measure = window_measures[:, 0]
            elif model_type == 'pca':
                x_measure = generator.generate_pca_measure(window_measures)
            elif model_type == 'ica':
                x_measure = generator.generate_ica_measure(window_measures)
            elif model_type == 'avg':
                x_measure = generator.generate_avg_measure(window_measures)
            elif model_type == 'ae':
                x_measure = generator.generate_ae_measure(window_measures, epochs=50)
            else:
                raise ValueError(f"Unknown model type: {model_type}")
            
            # Fit RealGARCH model
            model = RealGARCH()
            result = model.fit(window_returns, x_measure)
            
            if result is None or not result.success:
                print(f"Warning: Optimization failed at step {i}. Using previous forecast.")
                if i > 0:
                    forecasts[i] = forecasts[i-1]
                else:
                    forecasts[i] = np.sqrt(np.var(window_returns))
                continue
            
            # Forecast volatility for the next point
            forecast_vol = model.forecast_volatility(window_returns, x_measure)[0]
            forecasts[i] = forecast_vol
            
            # Calculate negative log-likelihood for evaluation
            actual_return = test_data['test_returns'][i]
            neg_log_likelihood += 0.5 * (np.log(forecast_vol**2) + actual_return**2 / forecast_vol**2)
            
            # Print progress
            if (i+1) % 10 == 0:
                print(f"Completed {i+1}/{n_test} forecasts for {model_type.upper()}")
        
        return {'forecasts': forecasts, 'neg_log_likelihood': neg_log_likelihood}
