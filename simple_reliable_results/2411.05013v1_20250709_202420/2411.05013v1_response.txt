import numpy as np

class SimulatedDataGenerator:
    """Generate synthetic financial data for testing algorithmic trading strategies."""
    
    def __init__(self, n_samples=1000, time_steps=60, n_features=5, freq='D'):
        """
        Initialize the data generator.
        
        Args:
            n_samples: Number of trading days to generate
            time_steps: Number of past days to use for prediction
            n_features: Number of features (OHLCV + technical indicators)
            freq: Frequency of data ('D' for daily, 'min' for minute)
        """
        self.n_samples = n_samples
        self.time_steps = time_steps
        self.n_features = n_features
        self.freq = freq
        
    def generate_price_series(self, volatility=0.01, drift=0.0001, starting_price=100.0):
        """Generate a price series with random walk and some seasonality."""
        np.random.seed(42)
        
        # Generate dates
        if self.freq == 'D':
            start_date = dt.datetime(2020, 1, 1)
            dates = [start_date + dt.timedelta(days=i) for i in range(self.n_samples)]
        else:
            start_date = dt.datetime(2020, 1, 1, 9, 30)
            dates = [start_date + dt.timedelta(minutes=i) for i in range(self.n_samples)]
        
        # Generate price with random walk
        returns = np.random.normal(drift, volatility, self.n_samples)
        
        # Add some seasonality/cycles to make it more realistic
        t = np.arange(self.n_samples)
        seasonality = 0.001 * np.sin(t/50) + 0.0005 * np.sin(t/20)
        returns = returns + seasonality
        
        # Calculate price series
        price_series = starting_price * (1 + returns).cumprod()
        
        # Create OHLCV data
        data = {
            'date': dates,
            'open': price_series * (1 + np.random.normal(0, 0.002, self.n_samples)),
            'high': price_series * (1 + abs(np.random.normal(0.002, 0.003, self.n_samples))),
            'low': price_series * (1 - abs(np.random.normal(0.002, 0.003, self.n_samples))),
            'close': price_series,
            'volume': np.random.normal(1000000, 200000, self.n_samples) * (1 + 0.1 * abs(returns))
        }
        
        # Ensure high is the highest and low is the lowest
        for i in range(self.n_samples):
            data['high'][i] = max(data['high'][i], data['open'][i], data['close'][i])
            data['low'][i] = min(data['low'][i], data['open'][i], data['close'][i])
        
        df = pd.DataFrame(data)
        df.set_index('date', inplace=True)
        
        # Add technical indicators
        df['sma_5'] = df['close'].rolling(window=5).mean()
        df['sma_20'] = df['close'].rolling(window=20).mean()
        df['rsi'] = self._compute_rsi(df['close'], 14)
        df['macd'] = self._compute_macd(df['close'])
        df['atr'] = self._compute_atr(df, 14)
        
        # Fill NaN values
        df.fillna(method='bfill', inplace=True)
        
        return df
    
    def _compute_rsi(self, prices, period=14):
        """Compute the RSI technical indicator."""
        delta = prices.diff()
        gain = delta.where(delta > 0, 0).rolling(window=period).mean()
        loss = -delta.where(delta < 0, 0).rolling(window=period).mean()
        
        rs = gain / loss
        rsi = 100 - (100 / (1 + rs))
        return rsi
    
    def _compute_macd(self, prices, fast=12, slow=26, signal=9):
        """Compute the MACD technical indicator."""
        ema_fast = prices.ewm(span=fast, min_periods=fast).mean()
        ema_slow = prices.ewm(span=slow, min_periods=slow).mean()
        macd_line = ema_fast - ema_slow
        return macd_line
    
    def _compute_atr(self, df, period=14):
        """Compute the Average True Range technical indicator."""
        high = df['high']
        low = df['low']
        close = df['close']
        
        tr1 = high - low
        tr2 = abs(high - close.shift())
        tr3 = abs(low - close.shift())
        
        tr = pd.DataFrame({'tr1': tr1, 'tr2': tr2, 'tr3': tr3}).max(axis=1)
        atr = tr.rolling(window=period).mean()
        return atr
    
    def prepare_lstm_data(self, df, target_col='close', feature_cols=None):
        """Prepare data for LSTM model."""
        if feature_cols is None:
            feature_cols = ['open', 'high', 'low', 'close', 'volume', 
                           'sma_5', 'sma_20', 'rsi', 'macd', 'atr']
        
        # Scale features
        scaler = MinMaxScaler()
        df_scaled = pd.DataFrame(scaler.fit_transform(df[feature_cols]), 
                                 columns=feature_cols, 
                                 index=df.index)
        
        # Create sequences for LSTM
        X, y = [], []
        for i in range(self.time_steps, len(df_scaled)):
            X.append(df_scaled.iloc[i-self.time_steps:i].values)
            y.append(df_scaled[target_col].iloc[i])
        
        X, y = np.array(X), np.array(y)
        
        # Return also the price scaler for inverse transformation
        price_scaler = MinMaxScaler()
        price_scaler.fit(df[[target_col]])
        
        return X, y, price_scaler


class LSTMModel:
    """LSTM model for time series prediction."""
    
    def __init__(self, time_steps, n_features, learning_rate=0.001):
        """
        Initialize the LSTM model.
        
        Args:
            time_steps: Number of past time steps to use for prediction
            n_features: Number of features in the input data
            learning_rate: Learning rate for the optimizer
        """
        self.time_steps = time_steps
        self.n_features = n_features
        self.learning_rate = learning_rate
        self.model = self._build_model()
    
    def _build_model(self):
        """Build the LSTM model architecture."""
        model = Sequential()
        model.add(LSTM(units=50, return_sequences=True, 
                       input_shape=(self.time_steps, self.n_features)))
        model.add(Dropout(0.2))
        model.add(LSTM(units=50, return_sequences=False))
        model.add(Dropout(0.2))
        model.add(Dense(units=25))
        model.add(Dense(units=1))
        
        model.compile(optimizer=Adam(learning_rate=self.learning_rate), loss='mean_squared_error')
        return model
    
    def train(self, X_train, y_train, X_val, y_val, epochs=50, batch_size=32, verbose=1):
        """Train the LSTM model."""
        history = self.model.fit(
            X_train, y_train,
            epochs=epochs,
            batch_size=batch_size,
            validation_data=(X_val, y_val),
            verbose=verbose
        )
        return history
    
    def predict(self, X):
        """Make predictions with the LSTM model."""
        return self.model.predict(X)


class TradingStrategy:
    """Trading strategy using LSTM predictions."""
    
    def __init__(self, model, price_scaler):
        """
        Initialize the trading strategy.
        
        Args:
            model: Trained LSTM model
            price_scaler: Scaler used for price normalization
        """
        self.model = model
        self.price_scaler = price_scaler
        
    def generate_signals(self, X_test, y_test, threshold=0.001):
        """
        Generate buy/sell signals based on predicted price changes.
        
        Args:
            X_test: Test features
            y_test: True target values
            threshold: Threshold for buy/sell decisions
        
        Returns:
            DataFrame with signals and performance metrics
        """
        # Make predictions
        y_pred = self.model.predict(X_test)
        
        # Inverse transform predictions and actual values
        y_pred_inv = self.price_scaler.inverse_transform(y_pred.reshape(-1, 1)).flatten()
        y_test_inv = self.price_scaler.inverse_transform(y_test.reshape(-1, 1)).flatten()
        
        # Calculate price changes
        actual_changes = np.diff(y_test_inv)
        predicted_changes = y_pred_inv[1:] - y_test_inv[:-1]
        
        # Generate signals
        signals = np.zeros(len(predicted_changes))
        signals[predicted_changes > threshold] = 1  # Buy
        signals[predicted_changes < -threshold] = -1  # Sell
        
        # Calculate returns
        strategy_returns = signals * actual_changes
        
        # Buy and hold returns
        buy_hold_returns = actual_changes
        
        # Create results DataFrame
        results = pd.DataFrame({
            'Actual_Price': y_test_inv[1:],
            'Predicted_Price': y_pred_inv[1:],
            'Actual_Change': actual_changes,
            'Predicted_Change': predicted_changes,
            'Signal': signals,
            'Strategy_Return': strategy_returns,
            'Buy_Hold_Return': buy_hold_returns
        })
        
        return results
    
    def evaluate_performance(self, results):
        """
        Evaluate the performance of the trading strategy.
        
        Args:
            results: DataFrame with trading signals and returns
            
        Returns:
            Dictionary with performance metrics
        """
        # Calculate cumulative returns
        cum_strategy_returns = results['Strategy_Return'].cumsum()
        cum_buy_hold_returns = results['Buy_Hold_Return'].cumsum()
        
        # Calculate Sharpe ratio (annualized)
        strategy_sharpe = np.sqrt(252) * cum_strategy_returns.iloc[-1] / (results['Strategy_Return'].std() * np.sqrt(len(results)))
        buy_hold_sharpe = np.sqrt(252) * cum_buy_hold_returns.iloc[-1] / (results['Buy_Hold_Return'].std() * np.sqrt(len(results)))
        
        # Calculate maximum drawdown
        strategy_cum_returns = cum_strategy_returns + 1
        buy_hold_cum_returns = cum_buy_hold_returns + 1
        
        strategy_max_drawdown = (strategy_cum_returns / strategy_cum_returns.cummax() - 1).min()
        buy_hold_max_drawdown = (buy_hold_cum_returns / buy_hold_cum_returns.cummax() - 1).min()
        
        # Calculate win rate
        win_trades = (results['Strategy_Return'] > 0).sum()
        total_trades = (results['Signal'] != 0).sum()
        win_rate = win_trades / total_trades if total_trades > 0 else 0
        
        # Calculate metrics
        metrics = {
            'Total_Return_Strategy': cum_strategy_returns.iloc[-1],
            'Total_Return_Buy_Hold': cum_buy_hold_returns.iloc[-1],
            'Sharpe_Ratio_Strategy': strategy_sharpe,
            'Sharpe_Ratio_Buy_Hold': buy_hold_sharpe,
            'Max_Drawdown_Strategy': strategy_max_drawdown,
            'Max_Drawdown_Buy_Hold': buy_hold_max_drawdown,
            'Win_Rate': win_rate,
            'Total_Trades': total_trades
        }
        
        return metrics
    
    def plot_results(self, results):
        """Plot the trading strategy results."""
        fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(12, 10))

def hyperparameter_optimization(X_train, y_train, X_val, y_val, time_steps, n_features):
    """
    Simple hyperparameter optimization for the LSTM model.
    
    Args:
        X_train, y_train: Training data
        X_val, y_val: Validation data
        time_steps: Number of time steps
        n_features: Number of features
        
    Returns:
        Best model and its validation loss
    """
    learning_rates = [0.01, 0.001, 0.0001]
    batch_sizes = [16, 32, 64]
    
    best_val_loss = float('inf')
    best_model = None
    best_params = {}
    
    for lr in learning_rates:
        for bs in batch_sizes:
            print(f"Training with learning_rate={lr}, batch_size={bs}")
            
            model = LSTMModel(time_steps, n_features, learning_rate=lr)
            history = model.train(X_train, y_train, X_val, y_val, 
                                 epochs=20, batch_size=bs, verbose=0)
            
            val_loss = history.history['val_loss'][-1]
            
            if val_loss < best_val_loss:
                best_val_loss = val_loss
                best_model = model
                best_params = {'learning_rate': lr, 'batch_size': bs}
                
    print(f"Best parameters: {best_params}, Validation loss: {best_val_loss}")
    return best_model, best_val_loss
