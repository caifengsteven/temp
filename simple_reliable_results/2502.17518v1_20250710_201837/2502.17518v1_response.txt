import numpy as np

class Stock:
    """Simulates a stock with price movements"""
    def __init__(self, symbol, initial_price, volatility=0.01):
        self.symbol = symbol
        self.price = initial_price
        self.volatility = volatility
        self.history = [initial_price]
    
    def update_price(self):
        """Update stock price with random movement"""
        change = np.random.normal(0, self.volatility)
        self.price *= (1 + change)
        self.history.append(self.price)
        return self.price

class Portfolio:
    """Manages stock holdings and cash balance"""
    def __init__(self, initial_balance=100000):
        self.holdings = {}  # Stock symbol -> quantity
        self.balance = initial_balance
        self.history = [initial_balance]
        self.value_history = [initial_balance]
    
    def get_state(self, prices):
        """Return current state as vector [prices, holdings, balance]"""
        p = np.array([prices[s] for s in prices])
        h = np.array([self.holdings.get(s, 0) for s in prices])
        return np.concatenate([p, h, [self.balance]])
    
    def execute_action(self, action, prices, transaction_cost_pct=0.001):
        """Execute buy/sell/hold actions for each stock"""
        old_value = self.get_total_value(prices)
        
        for stock_symbol, quantity in action.items():
            if quantity > 0:  # Buy
                cost = prices[stock_symbol] * quantity
                transaction_cost = cost * transaction_cost_pct
                total_cost = cost + transaction_cost
                
                if total_cost <= self.balance:
                    self.holdings[stock_symbol] = self.holdings.get(stock_symbol, 0) + quantity
                    self.balance -= total_cost
            
            elif quantity < 0:  # Sell
                current_holdings = self.holdings.get(stock_symbol, 0)
                sell_quantity = min(abs(quantity), current_holdings)
                
                if sell_quantity > 0:
                    revenue = prices[stock_symbol] * sell_quantity
                    transaction_cost = revenue * transaction_cost_pct
                    net_revenue = revenue - transaction_cost
                    
                    self.holdings[stock_symbol] = current_holdings - sell_quantity
                    if self.holdings[stock_symbol] == 0:
                        del self.holdings[stock_symbol]
                    
                    self.balance += net_revenue
        
        new_value = self.get_total_value(prices)
        reward = new_value - old_value
        
        self.history.append(self.balance)
        self.value_history.append(new_value)
        
        return reward
    
    def get_total_value(self, prices):
        """Calculate total portfolio value"""
        holdings_value = sum(self.holdings.get(s, 0) * prices[s] for s in prices)
        return self.balance + holdings_value


class RLAgent:
    """Base class for RL agents"""
    def __init__(self, name, stock_symbols):

        self.stock_symbols = stock_symbols
    
    def select_action(self, state, exploration_rate=0.1):
        """Select an action based on current state"""
        # For simulation, we'll use a simple random strategy with some bias
        action = {}
        
        for symbol in self.stock_symbols:
            # Extract price and current holdings for this stock
            idx = self.stock_symbols.index(symbol)
            price = state[idx]
            holdings = state[len(self.stock_symbols) + idx]
            
            # Simple random action with bias
            if random.random() < exploration_rate:
                # Explore: completely random action
                action[symbol] = random.randint(-int(holdings), 5)
            else:
                # Exploit: slightly biased action based on agent type
                if self.name == "A2C":
                    bias = 0.6  # Slightly bullish
                elif self.name == "PPO":
                    bias = 0.4  # Neutral
                else:  # SAC
                    bias = 0.2  # Slightly bearish
                
                if random.random() < bias:
                    action[symbol] = random.randint(0, 5)  # Buy
                elif random.random() < 0.5:
                    action[symbol] = 0  # Hold
                else:
                    action[symbol] = -random.randint(0, int(holdings))  # Sell
        
        return action


class ClassifierModel:
    """Classifier model for predicting agent actions"""
    def __init__(self, classifier_type, params=None):
        self.type = classifier_type
        self.params = params or {}
        
        if classifier_type == "SVM":
            self.model = SVC(probability=True, **self.params)
        elif classifier_type == "DecisionTree":
            self.model = DecisionTreeClassifier(**self.params)
        elif classifier_type == "LogisticRegression":
            self.model = LogisticRegression(**self.params)
        else:
            raise ValueError(f"Unsupported classifier type: {classifier_type}")
        
        self.scaler = StandardScaler()
    
    def fit(self, X, y):
        """Train the classifier"""
        X_scaled = self.scaler.fit_transform(X)
        self.model.fit(X_scaled, y)
    
    def predict_proba(self, X):
        """Predict probabilities for each class"""
        X_scaled = self.scaler.transform(X)
        return self.model.predict_proba(X_scaled)


class EnsembleRL:
    """Implements the ensemble RL method from the paper"""
    def __init__(self, agents, classifier_groups, variance_threshold=0.3):
        self.agents = agents
        self.classifier_groups = classifier_groups
        self.variance_threshold = variance_threshold
        self.features_history = []
        self.agent_actions_history = []
    
    def generate_features(self, state, prices_history, window=5):
        """Generate features for classifiers based on state and price history"""
        # Extract prices and holdings
        n_stocks = len(self.agents[0].stock_symbols)
        prices = state[:n_stocks]
        holdings = state[n_stocks:2*n_stocks]
        balance = state[-1]
        
        features = []
        
        # Price-based features
        if len(prices_history) >= window:
            # Returns over different windows
            for w in [1, 3, 5]:
                if len(prices_history) >= w:
                    returns = [(prices[i] / prices_history[-w][i]) - 1 for i in range(n_stocks)]
                    features.extend(returns)
            
            # Volatility over window
            for i in range(n_stocks):
                price_history = [ph[i] for ph in prices_history[-window:]]
                volatility = np.std(price_history) / np.mean(price_history)
                features.append(volatility)
        
        # Portfolio-based features
        total_value = balance + sum(prices[i] * holdings[i] for i in range(n_stocks))
        for i in range(n_stocks):
            # Position size as % of portfolio
            features.append((prices[i] * holdings[i]) / total_value if total_value > 0 else 0)
        
        # Diversification metric
        non_zero_positions = sum(1 for h in holdings if h > 0)
        features.append(non_zero_positions / n_stocks)
        
        return np.array(features)
    
    def train_classifiers(self, X, y):
        """Train all classifier groups"""
        for group in self.classifier_groups:
            for classifier in group:
                classifier.fit(X, y)
    
    def compute_candidate_matrix(self, features, agent_idx):
        """Compute candidate matrix Q as described in the paper"""
        Q = np.zeros((len(self.classifier_groups), len(self.agents)))
        
        for i, group in enumerate(self.classifier_groups):
            for j, agent in enumerate(self.agents):
                # Average probability across classifiers in this group
                probs = np.mean([clf.predict_proba([features])[0][agent_idx] 
                                for clf in group])
                Q[i, j] = probs
        
        return Q
    
    def decide_action(self, state, prices_history):
        """Decide which agent's action to take based on ensemble method"""
        features = self.generate_features(state, prices_history)
        self.features_history.append(features)
        
        # Get actions from each agent
        agent_actions = [agent.select_action(state) for agent in self.agents]
        self.agent_actions_history.append(agent_actions)
        
        # For training data collection only - use random agent
        if len(self.features_history) < 30:  # Initial collection phase
            return random.choice(agent_actions)
        
        # Train classifiers periodically
        if len(self.features_history) % 10 == 0 and len(self.features_history) > 30:
            # Generate synthetic training data based on past performance
            X_train = np.array(self.features_history[:-10])
            
            # Simple labeling - just for simulation
            # In reality, you'd use actual performance metrics to label which agent was best
            y_train = np.random.randint(0, len(self.agents), size=len(X_train))
            
            self.train_classifiers(X_train, y_train)
        
        # Compute stock holdings for each agent
        holdings = np.zeros((len(self.agents), len(self.agents[0].stock_symbols)))
        
        # Simulate what holdings would be after each agent's actions
        for i, actions in enumerate(agent_actions):
            for j, symbol in enumerate(self.agents[0].stock_symbols):
                current_holdings = state[len(self.agents[0].stock_symbols) + j]
                holdings[i, j] = current_holdings + actions.get(symbol, 0)
        
        # Compute standard deviation across agents for each stock
        std_devs = np.std(holdings, axis=0)
        
        # Normalize standard deviations
        if np.max(std_devs) > np.min(std_devs):
            normalized_std_devs = (std_devs - np.min(std_devs)) / (np.max(std_devs) - np.min(std_devs) + 1e-10)
        else:
            normalized_std_devs = np.zeros_like(std_devs)
        
        # Compute average normalized standard deviation
        avg_std_dev = np.mean(normalized_std_devs)
        
        # If classifiers aren't trained yet, use random agent
        if len(self.features_history) <= 30:
            return random.choice(agent_actions)
        
        # Compute candidate matrix for each agent
        candidate_matrices = [self.compute_candidate_matrix(features, i) 
                              for i in range(len(self.agents))]
        
        # Decision block based on variance threshold
        if avg_std_dev < self.variance_threshold:
            # Low variance - select action with highest probability
            agent_scores = np.array([np.mean(Q.max(axis=0)) for Q in candidate_matrices])
        else:
            # High variance - select action with lowest probability (more conservative)
            agent_scores = np.array([np.mean(Q.min(axis=0)) for Q in candidate_matrices])
        
        # Select agent with highest score
        selected_agent_idx = np.argmax(agent_scores)
        
        return agent_actions[selected_agent_idx]


def create_classifier_groups():
    """Create the classifier groups mentioned in the paper"""
    # Group 1: SVM with different kernels
    group1 = [

    # Group 2: Logistic Regression with different regularizations
    group2 = [
        ClassifierModel("LogisticRegression", {"penalty": "l1", "solver": "liblinear"}),
        ClassifierModel("LogisticRegression", {"penalty": "l2"}),
        ClassifierModel("LogisticRegression", {"penalty": "elasticnet", "solver": "saga", "l1_ratio": 0.5})

    group3 = [

        ClassifierModel("DecisionTree", {"criterion": "entropy"})
    ]
    
    # Group 4: Hybrid of SVM and Logistic Regression
    group4 = group1 + group2
    
    # Group 5: All models combined
    group5 = group1 + group2 + group3
    
    return [group1, group2, group3, group4, group5]


def calculate_metrics(portfolio_values):
    """Calculate financial metrics as described in the paper"""
    daily_returns = np.diff(portfolio_values) / portfolio_values[:-1]
    
    # Cumulative return
    cumulative_return = (portfolio_values[-1] / portfolio_values[0]) - 1
    
    # Max Drawdown
    peak = np.maximum.accumulate(portfolio_values)
    drawdown = (portfolio_values - peak) / peak
    max_drawdown = np.min(drawdown)
    
    # Sharpe Ratio (assuming risk-free rate of 0 for simplicity)
    sharpe_ratio = np.mean(daily_returns) / np.std(daily_returns) * np.sqrt(252)  # Annualized
    
    # Calmar Ratio
    calmar_ratio = cumulative_return / abs(max_drawdown) if max_drawdown != 0 else 0
    
    return {
        "Cumulative Returns": cumulative_return,
        "Sharpe Ratio": sharpe_ratio,
        "Max Drawdown": max_drawdown,
        "Calmar Ratio": calmar_ratio
    }


def run_simulation(agents, ensemble=None, days=252, initial_balance=100000):
    """Run a trading simulation for the specified number of days"""
    # Create stocks
    stocks = {
        "AAPL": Stock("AAPL", 150, 0.015),
        "MSFT": Stock("MSFT", 300, 0.012),
        "GOOGL": Stock("GOOGL", 2800, 0.018),
        "AMZN": Stock("AMZN", 3300, 0.02),
        "TSLA": Stock("TSLA", 900, 0.025)
    }
    stock_symbols = list(stocks.keys())
    
    # Initialize portfolio
    portfolio = Portfolio(initial_balance)
    
    # Track price history for feature generation
    prices_history = []
    
    # Simulation loop
    for day in range(days):
        # Update stock prices
        current_prices = {s: stocks[s].update_price() for s in stocks}
        prices_history.append([current_prices[s] for s in stock_symbols])
        
        # Get current state
        state = portfolio.get_state(current_prices)
        
        # Select action
        if ensemble:
            action = ensemble.decide_action(state, prices_history)
        else:
            # If no ensemble, use the first agent
            action = agents[0].select_action(state)
        
        # Execute action and get reward
        reward = portfolio.execute_action(action, current_prices)
    
    # Calculate performance metrics
    metrics = calculate_metrics(portfolio.value_history)
    
    return portfolio, metrics
