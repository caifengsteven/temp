class MomentumTransformer:
    def __init__(self, window_size=252, n_attention_heads=4, learning_rate=0.001):
        """
        Initialize the Momentum Transformer model
        
        Args:
            window_size: Number of time steps to include in each training instance
            n_attention_heads: Number of attention heads in the transformer
            learning_rate: Learning rate for the Adam optimizer
        """
        self.window_size = window_size
        self.n_attention_heads = n_attention_heads
        self.learning_rate = learning_rate
        self.model = None
        self.scaler = StandardScaler()
    
    def build_model(self, input_shape):
        """
        Build the Momentum Transformer model
        
        Args:
            input_shape: Shape of the input data (window_size, features)
        """
        # Input layer
        inputs = Input(shape=input_shape)
        
        # LSTM encoding layer (as mentioned in the paper)
        x = LSTM(64, return_sequences=True)(inputs)
        
        # Add & Norm with skip connection
        lstm_output = x
        
        # Multi-head attention block
        attention_output = MultiHeadAttention(
            num_heads=self.n_attention_heads, 
            key_dim=64 // self.n_attention_heads
        )(lstm_output, lstm_output)
        
        # Add & Norm
        x = LayerNormalization()(lstm_output + attention_output)
        
        # Feed-forward network
        ffn_output = Dense(128, activation='relu')(x)
        ffn_output = Dropout(0.1)(ffn_output)
        ffn_output = Dense(64)(ffn_output)
        
        # Add & Norm
        x = LayerNormalization()(x + ffn_output)
        
        # Final dense layer with tanh activation for position sizing (-1 to 1)
        outputs = Dense(1, activation='tanh')(x)
        
        # Create model
        model = Model(inputs=inputs, outputs=outputs)
        
        # Custom Sharpe ratio loss function (negative Sharpe to minimize)
        def sharpe_loss(y_true, y_pred):
            # Calculate returns using positions and true returns
            returns = tf.multiply(y_pred, y_true)
            mean_returns = tf.reduce_mean(returns)
            std_returns = tf.math.reduce_std(returns) + 1e-6  # Add small constant to avoid division by zero
            sharpe = mean_returns / std_returns
            return -sharpe  # Negative Sharpe since we want to maximize it
        
        # Compile model
        model.compile(optimizer=Adam(learning_rate=self.learning_rate), loss=sharpe_loss)
        
        self.model = model
        return model
    
    def create_sequences(self, data, returns):
        """
        Create sequences for training the model
        
        Args:
            data: Feature data
            returns: Target returns
            
        Returns:
            X: Sequences of features
            y: Corresponding returns
        """
        X, y = [], []
        for i in range(len(data) - self.window_size):
            X.append(data[i:i+self.window_size])
            y.append(returns[i+self.window_size])
        return np.array(X), np.array(y)
    
    def fit(self, X, y, validation_split=0.2, epochs=100, batch_size=32):
        """
        Train the model
        
        Args:
            X: Training features
            y: Training targets
            validation_split: Portion of data to use for validation
            epochs: Number of training epochs
            batch_size: Batch size for training
            
        Returns:
            Training history
        """
        if self.model is None:
            self.build_model(X.shape[1:])
        
        early_stopping = EarlyStopping(
            monitor='val_loss',
            patience=10,
            restore_best_weights=True
        )
        
        history = self.model.fit(
            X, y,
            validation_split=validation_split,
            epochs=epochs,
            batch_size=batch_size,
            callbacks=[early_stopping],
            verbose=0
        )
        
        return history
    
    def predict_positions(self, X):
        """
        Predict positions using the trained model
        
        Args:
            X: Input features
            
        Returns:
            Predicted positions
        """
        if self.model is None:
            raise ValueError("Model has not been trained yet")
        
        return self.model.predict(X, verbose=0)

class ChangePointDetection:
    """
    Simplified change point detection class that simulates the CPD module mentioned in the paper
    """
    def __init__(self, window_size=21):
        self.window_size = window_size
    
    def detect_change_points(self, returns):
        """
        Detect change points in the returns series
        
        Args:
            returns: Array of returns
            
        Returns:
            Change point features
        """
        cp_features = np.zeros(len(returns))
        
        # For simulation, we'll mark change points based on volatility changes
        for i in range(self.window_size, len(returns)):
            window_std = np.std(returns[i-self.window_size:i])
            prev_window_std = np.std(returns[i-self.window_size*2:i-self.window_size]) if i >= self.window_size*2 else window_std
            
            # If volatility changes significantly, mark as change point
            if window_std > prev_window_std * 1.5 or window_std < prev_window_std * 0.5:
                cp_score = abs(window_std - prev_window_std) / prev_window_std
                cp_features[i] = min(cp_score, 1.0)  # Normalize to 0-1
        
        return cp_features

            volatility = 0.01  # Daily volatility

                # Change regime parameters at specific points
                if i in regime_changes:
                    drift = random.uniform(-0.0005, 0.0005)

                # Add covid effect
                if '2020-03' in date_range[i].strftime('%Y-%m'):

                # Generate next price
                daily_return = np.random.normal(drift, volatility)
                price = prices[-1] * (1 + daily_return)
                prices.append(price)
            
            price_data[ticker] = prices
        
        # Create DataFrame
        df = pd.DataFrame(price_data, index=date_range)

    def calculate_features(self, prices_df):
        """
        Calculate features from price data

            DataFrame with features

        features_dfs = []
        
        for ticker in prices_df.columns:
            prices = prices_df[ticker]
            
            # Calculate returns
            returns = prices.pct_change().fillna(0)
            
            # Calculate MACD (as mentioned in the paper)
            ema12 = prices.ewm(span=12).mean()
            ema26 = prices.ewm(span=26).mean()
            macd = ema12 - ema26
            signal = macd.ewm(span=9).mean()
            macd_hist = macd - signal

            })
            
            features_dfs.append(ticker_df)
        
        # Combine all tickers
        features_df = pd.concat(features_dfs)

class BacktestEngine:
    """
    Engine for backtesting trading strategies
    """
    def __init__(self, transaction_cost=0.001):
        self.transaction_cost = transaction_cost
    
    def backtest_strategy(self, positions, returns, prices):
        """
        Backtest a trading strategy
        
        Args:
            positions: Array of positions (-1 to 1)
            returns: Array of returns
            prices: Array of prices
            
        Returns:
            Dictionary with performance metrics
        """
        # Initialize portfolio
        portfolio_value = [1.0]  # Start with $1
        position_sizes = []
        
        for i in range(len(positions)):
            # Calculate position change and transaction costs
            if i > 0:
                position_change = abs(positions[i] - positions[i-1])
                transaction_cost = position_change * self.transaction_cost
                
                # Calculate daily return with transaction costs
                daily_return = positions[i-1] * returns[i] - transaction_cost
                new_value = portfolio_value[-1] * (1 + daily_return)
                portfolio_value.append(new_value)
            
            position_sizes.append(positions[i] * portfolio_value[-1])
        
        # Calculate performance metrics
        portfolio_returns = np.diff(portfolio_value) / portfolio_value[:-1]
        
        annual_return = np.mean(portfolio_returns) * 252
        annual_volatility = np.std(portfolio_returns) * np.sqrt(252)
        sharpe_ratio = annual_return / annual_volatility if annual_volatility > 0 else 0
        
        # Calculate maximum drawdown
        cumulative_returns = np.cumprod(1 + portfolio_returns)
        peak = np.maximum.accumulate(cumulative_returns)
        drawdown = (peak - cumulative_returns) / peak
        max_drawdown = np.max(drawdown)
        
        # Calculate downside risk (standard deviation of negative returns)
        negative_returns = portfolio_returns[portfolio_returns < 0]
        downside_risk = np.std(negative_returns) * np.sqrt(252) if len(negative_returns) > 0 else 0
        
        # Calculate sortino ratio
        sortino_ratio = annual_return / downside_risk if downside_risk > 0 else 0
        
        # Calculate calmar ratio
        calmar_ratio = annual_return / max_drawdown if max_drawdown > 0 else 0
        
        # Calculate percentage of positive returns
        positive_returns_pct = np.mean(portfolio_returns > 0) * 100
        
        # Calculate profit/loss ratio
        avg_profit = np.mean(portfolio_returns[portfolio_returns > 0]) if any(portfolio_returns > 0) else 0
        avg_loss = np.mean(portfolio_returns[portfolio_returns < 0]) if any(portfolio_returns < 0) else -1e-6
        profit_loss_ratio = abs(avg_profit / avg_loss) if avg_loss != 0 else 1.0
        
        return {
            'portfolio_value': portfolio_value,
            'annual_return': annual_return * 100,  # Convert to percentage
            'annual_volatility': annual_volatility * 100,  # Convert to percentage
            'sharpe_ratio': sharpe_ratio,
            'max_drawdown': max_drawdown * 100,  # Convert to percentage
            'downside_risk': downside_risk * 100,  # Convert to percentage
            'sortino_ratio': sortino_ratio,
            'calmar_ratio': calmar_ratio,
            'positive_returns_pct': positive_returns_pct,
            'profit_loss_ratio': profit_loss_ratio
        }
