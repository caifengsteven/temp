import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from datetime import datetime, timedelta
import random

class LimitOrderBook:
    """Simplified Limit Order Book implementation"""
    
    def __init__(self, initial_mid_price: float = 200.0, levels: int = 10):
        self.levels = levels
        self.mid_price = initial_mid_price
        self.spread = 0.1
        
        # Initialize empty order book with specified levels
        self.bids = {i: {'price': self.mid_price - (i+0.5)*self.spread, 'size': 100 + i*20} 
                    for i in range(levels)}
        self.asks = {i: {'price': self.mid_price + (i+0.5)*self.spread, 'size': 100 + i*20} 
                    for i in range(levels)}
        
    def get_snapshot(self) -> Dict:
        """Return current LOB snapshot"""
        snapshot = {}
        
        for i in range(self.levels):
            snapshot[f'bid_price_{i+1}'] = self.bids[i]['price']
            snapshot[f'bid_size_{i+1}'] = self.bids[i]['size']
            snapshot[f'ask_price_{i+1}'] = self.asks[i]['price']
            snapshot[f'ask_size_{i+1}'] = self.asks[i]['size']
            
        snapshot['mid_price'] = self.mid_price
        snapshot['spread'] = self.asks[0]['price'] - self.bids[0]['price']
        
        # Calculate volume imbalance at first level
        total_vol = self.bids[0]['size'] + self.asks[0]['size']
        if total_vol > 0:
            snapshot['volume_imbalance'] = (self.bids[0]['size'] - self.asks[0]['size']) / total_vol
        else:
            snapshot['volume_imbalance'] = 0
            
        snapshot['vwap'] = self._calculate_vwap()
        
        return snapshot
    
    def _calculate_vwap(self) -> float:
        """Calculate Volume-Weighted Average Price"""
        total_volume = 0
        weighted_sum = 0
        
        for i in range(self.levels):
            weighted_sum += self.bids[i]['price'] * self.bids[i]['size']
            weighted_sum += self.asks[i]['price'] * self.asks[i]['size']
            total_volume += self.bids[i]['size'] + self.asks[i]['size']
            
        if total_volume > 0:
            return weighted_sum / total_volume
        return self.mid_price
    
    def process_order(self, order: Dict) -> None:
        """Process an order and update the LOB"""
        order_type = order['type']
        direction = order['direction']
        price = order.get('price', None)
        quantity = order['quantity']
        
        if order_type == 'market':
            self._process_market_order(direction, quantity)
        elif order_type == 'limit':
            self._process_limit_order(direction, price, quantity)
        elif order_type == 'cancel':
            self._process_cancel_order(direction, price, quantity)
    
    def _process_market_order(self, direction: int, quantity: float) -> None:
        """Process market order and update LOB"""
        remaining_qty = quantity
        
        if direction > 0:  # Buy order
            # Match with asks starting from the best price
            for level in range(self.levels):
                if remaining_qty <= 0:
                    break
                    
                if self.asks[level]['size'] >= remaining_qty:
                    self.asks[level]['size'] -= remaining_qty
                    remaining_qty = 0
                else:
                    remaining_qty -= self.asks[level]['size']
                    self.asks[level]['size'] = 0
                
                # Update mid price if level 0 price changes
                if level == 0:
                    self.mid_price = (self.bids[0]['price'] + self.asks[0]['price']) / 2
                    
        else:  # Sell order
            # Match with bids starting from the best price
            for level in range(self.levels):
                if remaining_qty <= 0:
                    break
                    
                if self.bids[level]['size'] >= remaining_qty:
                    self.bids[level]['size'] -= remaining_qty
                    remaining_qty = 0
                else:
                    remaining_qty -= self.bids[level]['size']
                    self.bids[level]['size'] = 0
                
                # Update mid price if level 0 price changes
                if level == 0:
                    self.mid_price = (self.bids[0]['price'] + self.asks[0]['price']) / 2
    
    def _process_limit_order(self, direction: int, price: float, quantity: float) -> None:
        """Process limit order and update LOB"""
        if direction > 0:  # Buy limit order
            # Find the appropriate level
            for level in range(self.levels):
                if abs(self.bids[level]['price'] - price) < 0.0001:
                    self.bids[level]['size'] += quantity
                    break
        else:  # Sell limit order
            # Find the appropriate level
            for level in range(self.levels):
                if abs(self.asks[level]['price'] - price) < 0.0001:
                    self.asks[level]['size'] += quantity
                    break
    
    def _process_cancel_order(self, direction: int, price: float, quantity: float) -> None:
        """Process cancel order and update LOB"""
        if direction > 0:  # Cancel buy order
            for level in range(self.levels):
                if abs(self.bids[level]['price'] - price) < 0.0001:
                    self.bids[level]['size'] = max(0, self.bids[level]['size'] - quantity)
                    break
        else:  # Cancel sell order
            for level in range(self.levels):
                if abs(self.asks[level]['price'] - price) < 0.0001:
                    self.asks[level]['size'] = max(0, self.asks[level]['size'] - quantity)
                    break


class TRADESSimulator:
    """Simulated version of TRADES model for testing purposes"""
    
    def __init__(self, initial_mid_price: float = 200.0, volatility: float = 0.001):
        self.lob = LimitOrderBook(initial_mid_price)
        self.volatility = volatility
        self.order_history = []
        self.lob_snapshots = []
        self.current_time = datetime.now().replace(hour=9, minute=30, second=0, microsecond=0)
        
        # Create initial LOB snapshots and order history
        for _ in range(256):
            self._generate_order()
            self.lob_snapshots.append(self.lob.get_snapshot())
            
    def _generate_order(self) -> Dict:
        """Generate a simulated order based on LOB state"""
        order_types = ['limit', 'market', 'cancel']
        weights = [0.6, 0.2, 0.2]  # More limit orders typically
        order_type = random.choices(order_types, weights=weights)[0]
        
        # Generate direction with slight bias based on volume imbalance
        imbalance = self.lob.get_snapshot().get('volume_imbalance', 0)
        buy_prob = 0.5 - imbalance * 0.2  # Slight mean reversion
        direction = 1 if random.random() < buy_prob else -1
        
        # Generate quantity with log-normal distribution
        quantity = max(1, int(np.random.lognormal(3, 1)))
        
        # Random time offset between 0.01 and 5 seconds
        time_offset = random.uniform(0.01, 5)
        self.current_time += timedelta(seconds=time_offset)
        
        order = {
            'timestamp': self.current_time,
            'type': order_type,
            'direction': direction,
            'quantity': quantity,
            'time_offset': time_offset
        }
        
        if order_type == 'limit':
            # Depth for limit orders
            depth = random.randint(0, 5)
            if direction > 0:  # Buy
                price = self.lob.bids[depth]['price']
            else:  # Sell
                price = self.lob.asks[depth]['price']
            order['price'] = price
            order['depth'] = depth
            
        elif order_type == 'cancel':
            # Cancel orders reference existing limit orders
            depth = random.randint(0, 5)
            if direction > 0:  # Cancel buy
                price = self.lob.bids[depth]['price']
                quantity = min(quantity, self.lob.bids[depth]['size'])
            else:  # Cancel sell
                price = self.lob.asks[depth]['price']
                quantity = min(quantity, self.lob.asks[depth]['size'])
            order['price'] = price
            order['depth'] = depth
            order['quantity'] = quantity
        
        # Process the order in the LOB
        self.lob.process_order(order)
        self.order_history.append(order)
        
        # Apply a small random walk to the mid price to simulate market movement
        price_change = np.random.normal(0, self.volatility * self.lob.mid_price)
        self.lob.mid_price += price_change
        
        return order
    
    def generate_next_order(self) -> Dict:
        """Generate the next order based on current LOB state"""
        order = self._generate_order()
        self.lob_snapshots.append(self.lob.get_snapshot())
        return order
    
    def run_simulation(self, duration_minutes: int = 120) -> Tuple[List[Dict], List[Dict]]:
        """Run a simulation for the specified duration"""
        end_time = self.current_time + timedelta(minutes=duration_minutes)
        
        while self.current_time < end_time:
            self.generate_next_order()
            
        return self.order_history, self.lob_snapshots


class PredictiveModel:
    """Simple predictive model for mid-price forecasting"""
    
    def __init__(self, lookback: int = 100, horizon: int = 10):
        self.lookback = lookback
        self.horizon = horizon
        self.weights = np.random.normal(0, 0.1, 5)  # Random initial weights for 5 features
    
    def train(self, data: pd.DataFrame, epochs: int = 10):
        """Train the model on historical data"""
        print(f"Training model on {len(data)} data points for {epochs} epochs...")
        
        X, y = self._prepare_data(data)
        
        for epoch in range(epochs):
            predictions = self._predict_internal(X)
            errors = y - predictions
            
            # Simple gradient descent
            gradient = -2 * np.mean(np.array([errors]).T * X, axis=0)
            self.weights -= 0.01 * gradient
            
            mae = np.mean(np.abs(errors))
            if epoch % 2 == 0:
                print(f"Epoch {epoch}, MAE: {mae:.4f}")
    
    def _prepare_data(self, data: pd.DataFrame) -> Tuple[np.ndarray, np.ndarray]:
        """Prepare features and target from data"""
        X = []
        y = []
        
        for i in range(self.lookback, len(data) - self.horizon):
            # Extract features
            features = [
                data.iloc[i]['mid_price'],
                data.iloc[i]['volume_imbalance'],
                data.iloc[i]['spread'],
                data.iloc[i]['bid_size_1'] / data.iloc[i]['ask_size_1'],
                (data.iloc[i]['mid_price'] - data.iloc[i-5]['mid_price']) / data.iloc[i-5]['mid_price']
            ]
            X.append(features)
            
            # Target is the mid price after horizon steps
            y.append(data.iloc[i + self.horizon]['mid_price'])
        
        return np.array(X), np.array(y)
    
    def _predict_internal(self, X: np.ndarray) -> np.ndarray:
        """Make predictions using the model weights"""
        return np.dot(X, self.weights)
    
    def predict(self, data: pd.DataFrame) -> float:
        """Make a prediction for the next price"""
        if len(data) < self.lookback:
            raise ValueError(f"Not enough data points. Need at least {self.lookback}.")
        
        # Extract the last lookback points
        recent_data = data.iloc[-self.lookback:]
        
        # Extract features
        features = np.array([
            recent_data.iloc[-1]['mid_price'],
            recent_data.iloc[-1]['volume_imbalance'],
            recent_data.iloc[-1]['spread'],
            recent_data.iloc[-1]['bid_size_1'] / recent_data.iloc[-1]['ask_size_1'],
            (recent_data.iloc[-1]['mid_price'] - recent_data.iloc[-5]['mid_price']) / recent_data.iloc[-5]['mid_price']
        ])
        
        return np.dot(features, self.weights)
    
    def evaluate(self, data: pd.DataFrame) -> float:
        """Evaluate the model on test data and return MAE"""
        X, y = self._prepare_data(data)
        predictions = self._predict_internal(X)
        mae = np.mean(np.abs(y - predictions))
        return mae
