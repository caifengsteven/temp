class ReturnSimulator:
    """Generate returns based on different models described in the paper."""
    
    @staticmethod
    def independent_returns(n_days: int, mu: float, sigma: float) -> np.ndarray:
        """Generate independent returns with given mean and standard deviation."""
        return np.random.normal(mu, sigma, n_days)
    
    @staticmethod
    def ar1_returns(n_days: int, phi: float, sigma: float, mu: float = 0) -> np.ndarray:
        """Generate returns following AR(1) process: X_t = mu + phi*X_{t-1} + epsilon_t."""
        returns = np.zeros(n_days)
        epsilon = np.random.normal(0, sigma, n_days)
        
        # Initialize first return
        returns[0] = epsilon[0]
        
        # Generate subsequent returns based on AR(1) process
        for t in range(1, n_days):
            returns[t] = mu + phi * returns[t-1] + epsilon[t]
            
        return returns
    
    @staticmethod
    def ar1_garch_returns(n_days: int, mu: float, phi: float, 
                          omega: float, alpha: float, beta: float) -> Tuple[np.ndarray, np.ndarray]:
        """
        Generate returns following AR(1)-GARCH(1,1) process:
        X_t = mu + phi*X_{t-1} + epsilon_t
        epsilon_t = sigma_t * z_t, z_t ~ N(0, 1)
        sigma_t^2 = omega + alpha*epsilon_{t-1}^2 + beta*sigma_{t-1}^2
        """
        returns = np.zeros(n_days)
        epsilon = np.zeros(n_days)
        sigma2 = np.zeros(n_days)
        
        # Initialize
        sigma2[0] = omega / (1 - alpha - beta)  # Unconditional variance
        z = np.random.normal(0, 1, n_days)
        epsilon[0] = np.sqrt(sigma2[0]) * z[0]
        returns[0] = mu + epsilon[0]
        
        # Generate subsequent returns
        for t in range(1, n_days):
            sigma2[t] = omega + alpha * epsilon[t-1]**2 + beta * sigma2[t-1]
            epsilon[t] = np.sqrt(sigma2[t]) * z[t]
            returns[t] = mu + phi * returns[t-1] + epsilon[t]
            
        return returns, np.sqrt(sigma2)

    @staticmethod
    def regime_switching_returns(n_days: int, regimes: Dict[int, Dict], 
                                transition_matrix: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
        """
        Generate returns following a regime-switching model:
        dS_t = mu_Z_t * S_t * dt + sigma_Z_t * S_t * dW_t
        
        Args:
            n_days: Number of days to simulate
            regimes: Dictionary with regime parameters {regime_id: {'mu': float, 'sigma': float}}
            transition_matrix: Matrix of transition probabilities between regimes
        """
        returns = np.zeros(n_days)
        regime_path = np.zeros(n_days, dtype=int)
        
        # Initialize with random regime
        n_regimes = len(regimes)
        regime_path[0] = np.random.choice(range(n_regimes))
        
        # Generate regime path
        for t in range(1, n_days):
            current_regime = regime_path[t-1]
            regime_path[t] = np.random.choice(range(n_regimes), p=transition_matrix[current_regime])
        
        # Generate returns based on regime path
        dt_val = 1/252  # Assuming daily returns (annualized)
        for t in range(n_days):
            regime = regime_path[t]
            mu = regimes[regime]['mu']
            sigma = regimes[regime]['sigma']
            
            # Discretized GBM for the regime
            returns[t] = mu * dt_val + sigma * np.sqrt(dt_val) * np.random.normal(0, 1)
            
        return returns, regime_path


class LETF:
    """Simulates leveraged ETF performance."""
    
    def __init__(self, leverage: int, fee: float = 0.0, tracking_error_std: float = 0.0):
        """
        Initialize LETF with given parameters.
        
        Args:
            leverage: Leverage ratio (e.g., -3, -2, -1, 2, 3)
            fee: Daily expense ratio
            tracking_error_std: Standard deviation of daily tracking error
        """
        self.leverage = leverage
        self.fee = fee
        self.tracking_error_std = tracking_error_std
    
    def simulate_returns(self, etf_returns: np.ndarray, rebalance_frequency: int = 1) -> np.ndarray:
        """
        Simulate LETF returns given ETF returns.
        
        Args:
            etf_returns: Daily returns of the underlying ETF
            rebalance_frequency: Number of days between rebalancing (1=daily, 5=weekly, etc.)
        
        Returns:
            Array of LETF daily returns
        """
        n_days = len(etf_returns)
        letf_returns = np.zeros(n_days)
        
        # If rebalancing is daily
        if rebalance_frequency == 1:
            # Add tracking error if specified
            tracking_error = np.random.normal(0, self.tracking_error_std, n_days) if self.tracking_error_std > 0 else 0
            
            # Calculate daily LETF returns with leverage, fees, and tracking error
            letf_returns = self.leverage * etf_returns - self.fee + tracking_error
            
            return letf_returns
        
        # If rebalancing is less frequent (weekly, monthly)
        rebalance_dates = range(0, n_days, rebalance_frequency)
        current_leverage = self.leverage
        
        for i in range(n_days):
            # Add tracking error if specified
            tracking_error = np.random.normal(0, self.tracking_error_std) if self.tracking_error_std > 0 else 0
            
            # Apply current leverage and fees
            letf_returns[i] = current_leverage * etf_returns[i] - self.fee + tracking_error
            
            # Update leverage if this is not a rebalance date and not the last day
            if i not in rebalance_dates and i < n_days - 1:
                # Update effective leverage based on today's return
                current_leverage = current_leverage * (1 + etf_returns[i]) / (1 + letf_returns[i])
            elif i in rebalance_dates:
                # Reset leverage on rebalance dates
                current_leverage = self.leverage
                
        return letf_returns


def calculate_compounding_effect(etf_returns: np.ndarray, letf_returns: np.ndarray, leverage: int) -> float:
    """
    Calculate the compounding effect as defined in the paper:
    CE_n = R^{LETF,β}_n - β*R^{ETF}_n
    
    Args:
        etf_returns: Daily returns of the underlying ETF
        letf_returns: Daily returns of the LETF
        leverage: Leverage ratio
    
    Returns:
        Compounding effect
    """
    # Calculate cumulative returns
    etf_cum_return = np.prod(1 + etf_returns) - 1
    letf_cum_return = np.prod(1 + letf_returns) - 1
    
    # Calculate compounding effect
    ce = letf_cum_return - leverage * etf_cum_return
    
    return ce


def run_monte_carlo(n_simulations: int, n_days: int, 
                   return_generator, return_params: Dict,
                   leverage_ratios: List[int],
                   rebalance_freq: int = 1,
                   fee: float = 0.0) -> Dict[int, List[float]]:
    """
    Run Monte Carlo simulations to evaluate expected compounding effect.
    
    Args:
        n_simulations: Number of simulations to run
        n_days: Number of days in each simulation
        return_generator: Function to generate returns
        return_params: Parameters for the return generator
        leverage_ratios: List of leverage ratios to test
        rebalance_freq: Rebalancing frequency (1=daily, 5=weekly, 22=monthly)
        fee: Daily fee for the LETF
    
    Returns:
        Dictionary mapping leverage ratios to lists of compounding effects
    """
    results = {leverage: [] for leverage in leverage_ratios}
    
    for _ in range(n_simulations):
        # Generate ETF returns for this simulation
        etf_returns = return_generator(**return_params, n_days=n_days)
        if isinstance(etf_returns, tuple):  # Handle case when return_generator returns additional info
            etf_returns = etf_returns[0]
            
        # For each leverage ratio, calculate LETF returns and compounding effect
        for leverage in leverage_ratios:
            letf = LETF(leverage=leverage, fee=fee)
            letf_returns = letf.simulate_returns(etf_returns, rebalance_frequency=rebalance_freq)
            ce = calculate_compounding_effect(etf_returns, letf_returns, leverage)
            results[leverage].append(ce)
    
    return results


def analyze_results(results: Dict[int, List[float]]) -> pd.DataFrame:
    """
    Analyze Monte Carlo simulation results.
    
    Args:
        results: Dictionary mapping leverage ratios to lists of compounding effects
    
    Returns:
        DataFrame with summary statistics
    """
    summary = []
    
    for leverage, ce_values in results.items():
        ce_array = np.array(ce_values)
        summary.append({
            'Leverage': leverage,
            'Mean CE': ce_array.mean(),
            'Std Dev': ce_array.std(),
            '5th Percentile': np.percentile(ce_array, 5),
            'Median': np.percentile(ce_array, 50),
            '95th Percentile': np.percentile(ce_array, 95)
        })
    
    return pd.DataFrame(summary)
