class ReportAgent:
    """Simulates the AI Expert that combines analysis and incorporates risk preferences"""
    
    def __init__(self, name="GPT-4o-mini"):
        self.name = name
        
    def make_decision(self, time_series_pred, news_pred, statement_pred, risk_preference):
        """
        Combines predictions from all agents and risk preference to make a final decision
        Returns: decision (1 for buy, 0 for hold, -1 for sell) and confidence
        """
        # Extract statement confidence
        statement_result, statement_confidence = statement_pred
        
        # Weighted combination based on risk preference
        weights = {
            'conservative': {'time_series': 0.2, 'news': 0.3, 'statement': 0.5},
            'moderately_conservative': {'time_series': 0.3, 'news': 0.3, 'statement': 0.4},
            'moderately_aggressive': {'time_series': 0.4, 'news': 0.3, 'statement': 0.3},
            'aggressive': {'time_series': 0.5, 'news': 0.3, 'statement': 0.2}
        }
        
        w = weights.get(risk_preference, weights['moderately_conservative'])
        
        # Combine predictions with weights
        combined_score = (
            w['time_series'] * (time_series_pred * 2 - 1) +  # Convert 0/1 to -1/1
            w['news'] * (news_pred * 2 - 1) +                # Convert 0/1 to -1/1
            w['statement'] * (statement_result * 2 - 1) * statement_confidence
        )
        
        # Determine confidence based on agreement and statement confidence
        agent_agreement = 1 - (abs(time_series_pred - news_pred) + 
                             abs(news_pred - statement_result) + 
                             abs(statement_result - time_series_pred)) / 3
        
        confidence = (agent_agreement * 0.6 + statement_confidence * 0.4)
        
        # Decision thresholds depend on risk preference
        thresholds = {
            'conservative': {'buy': 0.3, 'sell': -0.3},
            'moderately_conservative': {'buy': 0.2, 'sell': -0.2},

            'aggressive': {'buy': 0.05, 'sell': -0.05}
        }
        
        t = thresholds.get(risk_preference, thresholds['moderately_conservative'])
        
        # Make decision
        if combined_score > t['buy']:
            return 1, confidence  # Buy
        elif combined_score < t['sell']:
            return -1, confidence  # Sell
        else:
            return 0, confidence  # Hold


class FinArena:
    """Main framework that coordinates all agents"""
    
    def __init__(self, use_rag=True, time_series_model="TimeGPT", text_model="DeepSeek-v2"):
        self.time_series_agent = TimeSeriesAgent(name=time_series_model)
        self.news_agent = NewsAgent(use_rag=use_rag, name=text_model)
        self.statement_agent = StatementAgent(name=text_model)
        self.report_agent = ReportAgent()
        
    def predict_movement(self, data):
        """Predicts stock movement (up/down) without considering risk preference"""
        time_series_pred = self.time_series_agent.predict(data['prices'])
        news_pred = self.news_agent.predict(data['news'])
        statement_pred = self.statement_agent.predict(data['financial_statements'])
        
        # Simple majority voting for movement prediction
        votes = [time_series_pred, news_pred, statement_pred[0]]
        return 1 if sum(votes) >= 2 else 0
    
    def make_investment_decision(self, data, risk_preference):
        """Makes investment decision based on all data and risk preference"""
        time_series_pred = self.time_series_agent.predict(data['prices'])
        news_pred = self.news_agent.predict(data['news'])
        statement_pred = self.statement_agent.predict(data['financial_statements'])
        
        decision, confidence = self.report_agent.make_decision(
            time_series_pred, news_pred, statement_pred, risk_preference
        )
        
        return decision, confidence

class BacktestSimulator:
    """Simulates backtesting of trading strategies"""
    
    def __init__(self, initial_capital=100000):
        self.initial_capital = initial_capital
        
    def simulate(self, prices, decisions, risk_profile):
        """
        Simulates trading based on decisions and risk profile
        Returns: final capital, returns, drawdowns
        """
        capital = self.initial_capital
        position = 0  # Shares held
        trades = []
        equity_curve = [capital]
        
        # Define risk profile parameters
        risk_params = {
            'conservative': {'buy_pct': 0.5, 'sell_pct': 1.0},
            'moderately_conservative': {'buy_pct': 0.7, 'sell_pct': 1.0},
            'moderately_aggressive': {'buy_pct': 1.0, 'sell_pct': 0.5},
            'aggressive': {'buy_pct': 1.0, 'sell_pct': 0.3}
        }
        
        params = risk_params.get(risk_profile, risk_params['moderately_conservative'])
        
        for i in range(len(decisions)):
            decision = decisions[i]
            price = prices[i]
            next_price = prices[i+1] if i < len(prices)-1 else price
            
            if decision == 1:  # Buy
                # Calculate amount to invest
                amount_to_invest = capital * params['buy_pct']
                if amount_to_invest > 0:
                    shares_to_buy = amount_to_invest / price
                    position += shares_to_buy
                    capital -= shares_to_buy * price
                    trades.append(('BUY', price, shares_to_buy))
            
            elif decision == -1:  # Sell
                # Calculate amount to sell
                shares_to_sell = position * params['sell_pct']
                if shares_to_sell > 0:
                    position -= shares_to_sell
                    capital += shares_to_sell * price
                    trades.append(('SELL', price, shares_to_sell))
            
            # Update equity at end of day
            equity = capital + position * next_price
            equity_curve.append(equity)
        
        # Calculate returns and drawdowns
        returns = [(equity_curve[i] - equity_curve[i-1]) / equity_curve[i-1] 
                   for i in range(1, len(equity_curve))]
        
        cumulative_returns = [equity / self.initial_capital - 1 for equity in equity_curve]
        
        # Calculate drawdowns
        peak = equity_curve[0]
        drawdowns = []
        
        for equity in equity_curve:
            if equity > peak:
                peak = equity
            drawdown = (peak - equity) / peak
            drawdowns.append(drawdown)
        
        final_capital = equity_curve[-1]
        annualized_return = ((final_capital / self.initial_capital) ** (252 / len(prices)) - 1) if len(prices) > 0 else 0
        max_drawdown = max(drawdowns) if drawdowns else 0
        
        # Calculate Sharpe ratio (assuming risk-free rate = 0 for simplicity)
        daily_returns = returns
        sharpe_ratio = (np.mean(daily_returns) / np.std(daily_returns) * np.sqrt(252)) if len(daily_returns) > 0 and np.std(daily_returns) > 0 else 0
        
        results = {
            'final_capital': final_capital,
            'annualized_return': annualized_return,
            'max_drawdown': max_drawdown,
            'sharpe_ratio': sharpe_ratio,
            'equity_curve': equity_curve,
            'trades': trades
        }
        
        return results
