# Simulate stochastic factor process
def simulate_factor_process(Y0, b, beta, q, sigma, T, dt, num_paths=1):
    steps = int(T/dt)
    Y = np.zeros((num_paths, steps+1))
    Y[:, 0] = Y0
    
    # Generate correlated Brownian motions
    n = sigma.shape[0]  # number of assets
    dW1 = np.random.normal(0, np.sqrt(dt), (num_paths, steps, n))
    dW2 = np.random.normal(0, np.sqrt(dt), (num_paths, steps))
    
    for t in range(steps):
        # Factor process evolution
        dY = b * dt + beta * (np.sum(q * dW1[:, t, :], axis=1) + 
                             np.sqrt(1 - np.sum(q**2)) * dW2[:, t])
        Y[:, t+1] = Y[:, t] + dY
        
    return Y

# Power utility function
def power_utility(x, alpha, h=1.0):
    if alpha == 0:
        return np.log(x) + h
    else:
        return (x**alpha * h) / alpha

# Modified utility with ratio evaluation
def modified_utility(x, xprev, gamma, alpha, h=1.0, rho=0.05, tau=1.0):
    ratio = x / (np.exp(rho * tau) * xprev**gamma)
    return power_utility(ratio, alpha, h)

# Optimize portfolio for a single period
def optimize_portfolio(market_params, eval_params, constraint_params, Y_curr, X_curr, verbose=False):
    """Optimize portfolio for a single period"""
    n = len(market_params.mu)
    
    # Define objective function (negative of expected utility)
    def objective(weights):
        # Simple approximation of expected utility 
        # In practice, would need to compute expected utility via simulation/quadrature
        expected_return = np.sum(weights * market_params.mu)
        variance = weights.T @ market_params.sigma @ weights
        
        # Approximate expected utility using mean-variance for power utility
        approx_expected_utility = (
            (1 + expected_return)**eval_params.alpha - 
            0.5 * eval_params.alpha * (eval_params.alpha - 1) * 
            (1 + expected_return)**(eval_params.alpha - 2) * variance
        )
        
        return -approx_expected_utility
    
    # Define constraints
    constraints = []
    bounds = None
    
    # Sum of weights equals 1
    constraints.append({'type': 'eq', 'fun': lambda x: np.sum(x) - 1})
    
    # Apply trading constraints
    if constraint_params.constraints_type == "shortselling":
        # No short-selling
        bounds = [(0, None) for _ in range(n)]
    elif constraint_params.constraints_type == "borrowing":
        # No borrowing (sum of positive weights <= 1)
        constraints.append({'type': 'ineq', 'fun': lambda x: constraint_params.a - np.sum(np.maximum(x, 0))})
    elif constraint_params.constraints_type == "both":
        # Both no short-selling and no borrowing
        bounds = [(0, None) for _ in range(n)]
        constraints.append({'type': 'ineq', 'fun': lambda x: constraint_params.a - np.sum(x)})
    else:
        # No constraints
        bounds = [(None, None) for _ in range(n)]
    
    # Initial guess - equal weight
    initial_weights = np.ones(n) / n
    
    # Run optimization
    result = minimize(
        objective, 
        initial_weights,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )
    
    if verbose:
        print(f"Optimization result: {result.success}")
        print(f"Optimal weights: {result.x}")
    
    return result.x

def simulate_wealth_process(S, market_params, eval_params, constraint_params, rebalance_freq=1):
    num_paths, steps, n = S.shape
    
    # Initialize wealth process and portfolio weights
    X = np.ones((num_paths, steps))

        Y[:, t] = np.random.normal(0, 1, num_paths)  # Simplified factor

            for p in range(num_paths):
                weights[p, t, :] = optimize_portfolio(
                    market_params, eval_params, constraint_params, 
                    Y[p, t], X[p, t]

            weights[:, t, :] = weights[:, t-1, :]
        
        # Calculate returns

        # Update wealth
        X[:, t+1] = X[:, t] * (1 + np.sum(weights[:, t, :] * returns, axis=1))
    
    return X, weights

# Evaluate portfolio performance with periodic evaluation
def evaluate_periodic_performance(X, eval_params):
    """Evaluate portfolio performance using ratio-type periodic evaluation"""
    num_paths, steps = X.shape
    periods = int(steps / eval_params.tau)
    
    # Get wealth at evaluation points
    eval_points = [int(i * eval_params.tau) for i in range(periods+1)]
    X_eval = X[:, eval_points]
    
    # Calculate utility at each evaluation point
    utilities = np.zeros((num_paths, periods))
    for i in range(periods):
        curr_X = X_eval[:, i+1]
        prev_X = X_eval[:, i]
        utilities[:, i] = modified_utility(
            curr_X, prev_X, eval_params.gamma, eval_params.alpha,
            h=1.0, rho=eval_params.rho, tau=eval_params.tau
        )
    
    # Apply discount factor
    discount = np.exp(-eval_params.rho * np.arange(1, periods+1) * eval_params.tau)
    discounted_utilities = utilities * discount
    
    # Sum across periods
    total_utility = np.sum(discounted_utilities, axis=1)
    
    return total_utility.mean()
