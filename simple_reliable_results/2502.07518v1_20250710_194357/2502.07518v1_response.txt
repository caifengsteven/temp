import numpy as np

class AdSModel:
    """
    Angelini-di Sciorio (AdS) model for implied volatility
    """
    def __init__(self, alpha=0.1, beta=0.5, delta=0.5, epsilon=0.001):
        self.alpha = alpha
        self.beta = beta
        self.delta = delta
        self.epsilon = epsilon
        self.kmin = None

    def fit(self, moneyness, implied_vol):
        """Calibrate model parameters using optimization"""
        # Estimate Kmin as the moneyness with minimum IV
        self.kmin = moneyness[np.argmin(implied_vol)]
        
        # Define objective function for optimization
        def objective(params):
            self.alpha, self.beta, self.delta, self.epsilon = params
            predicted = self.predict(moneyness)
            return np.mean((predicted - implied_vol) ** 2)  # MSE
        
        # Initial parameter guess
        initial_params = [self.alpha, self.beta, self.delta, self.epsilon]
        
        # Parameter bounds
        bounds = [(0.001, 1.0), (-1.0, 1.0), (0.001, 0.999), (0.0001, 0.1)]
        
        # Minimize objective function
        result = minimize(objective, initial_params, bounds=bounds, method='L-BFGS-B')
        
        # Update model parameters
        self.alpha, self.beta, self.delta, self.epsilon = result.x
        
        return self
    
    def hurst_exponent(self, moneyness):
        """Calculate Hurst exponent H(S/K) based on moneyness"""
        term1 = 0.5 * (1 + (1 - self.kmin) ** self.delta)
        term2 = 1 / (1 + abs(moneyness - self.kmin) ** self.delta)
        return term1 * term2
    
    def predict(self, moneyness):
        """Predict implied volatility using AdS model"""
        if self.kmin is None:
            raise ValueError("Model needs to be fitted first")
        
        h_values = self.hurst_exponent(moneyness)
        
        quadratic_term = (moneyness - self.kmin) ** 2
        exponential_term = np.exp(-self.beta * h_values * abs(moneyness - self.kmin))
        
        return self.alpha * quadratic_term * exponential_term + self.epsilon

def calculate_metrics(actual, predicted):
    """Calculate error metrics for model evaluation"""
    mse = np.mean((predicted - actual) ** 2)
    mae = np.mean(np.abs(predicted - actual))
    
    # Calculate curvature
    n = len(actual)
    actual_curvature = np.zeros(n-2)
    predicted_curvature = np.zeros(n-2)
    
    for i in range(1, n-1):
        actual_curvature[i-1] = actual[i+1] - 2*actual[i] + actual[i-1]
        predicted_curvature[i-1] = predicted[i+1] - 2*predicted[i] + predicted[i-1]
    
    # Calculate curvature error metrics
    rmsce = np.sqrt(np.mean((predicted_curvature - actual_curvature) ** 2))
    ace = np.mean(np.abs(predicted_curvature - actual_curvature))
    
    return {
        'MSE': mse,
        'MAE': mae,
        'RMSCE': rmsce,
        'ACE': ace
    }
