import numpy as np

class SfBM:
    """Stationary fractional Brownian Motion process for log-volatility modeling"""
    
    def __init__(self, H, lambda_sq, T, L):
        """
        Initialize SfBM process
        
        Parameters:
        - H: Hurst exponent (0 < H < 0.5)
        - lambda_sq: Intermittency parameter
        - T: Decorrelation time
        - L: Total length of the time series
        """
        self.H = H
        self.lambda_sq = lambda_sq
        self.T = T
        self.L = L
        self.sigma_sq = lambda_sq / (H * (1 - 2 * H))
        # Mean value adjustment to ensure E[exp(chi)] = 1
        self.mu = -self.sigma_sq / 4
        
    def generate_path(self):
        """Generate a path of the SfBM process"""
        # Create covariance matrix
        t = np.arange(self.L)
        cov_matrix = np.zeros((self.L, self.L))
        
        for i in range(self.L):
            for j in range(self.L):
                tau = abs(i - j)
                if tau < self.T:
                    cov_matrix[i, j] = (self.sigma_sq/2) * (1 - (tau/self.T)**(2*self.H))
                else:
                    cov_matrix[i, j] = 0
                    
        # Ensure the matrix is positive definite
        cov_matrix = cov_matrix + 1e-10 * np.eye(self.L)
        
        # Generate multivariate normal sample with this covariance
        path = np.random.multivariate_normal(
            mean=np.ones(self.L) * self.mu, 
            cov=cov_matrix
        )
        
        return path
    
    def volatility_process(self, path=None):
        """Convert SfBM path to volatility process"""
        if path is None:
            path = self.generate_path()
        return np.exp(path)


class NSfFM:
    """Nested log S-fBM Factor Model"""
    
    def __init__(self, N, H_factor, H_idio, lambda_sq, T, L):
        """
        Initialize N-SfFM model
        
        Parameters:
        - N: Number of assets

        - H_idio: Hurst exponent of idiosyncratic log-volatility
        - lambda_sq: Intermittency parameter

        - L: Length of time series
        """

        self.sigma = np.ones(N)

        self.gamma = np.ones(N) * 0.2  # Set to 0.2 as per paper
        
        # Create factor and idiosyncratic log-volatility processes
        self.factor_vol = SfBM(H_factor, lambda_sq, T, L)
        self.idio_vol = [SfBM(H_idio, lambda_sq, T, L) for _ in range(N)]
        
    def generate_returns(self, s=256):
        """
        Generate asset returns according to the N-SfFM
        
        Parameters:
        - s: Number of intraday steps
        
        Returns:
        - factor_returns: Market factor returns
        - stock_returns: Individual stock returns
        - index_returns: Equal-weighted index returns
        """
        # Generate log-volatility paths
        factor_logvol_path = self.factor_vol.generate_path()
        idio_logvol_paths = [vol.generate_path() for vol in self.idio_vol]
        
        # Initialize arrays for returns
        factor_returns = np.zeros(self.L)
        stock_returns = np.zeros((self.N, self.L))
        
        # Generate returns for each time step
        for t in range(self.L):
            # Factor returns
            factor_vol_t = np.exp(factor_logvol_path[t]/2)
            factor_returns[t] = factor_vol_t * np.random.normal(0, 1/np.sqrt(s), s).sum()
            
            # Individual stock returns
            for i in range(self.N):
                # Combined log-volatility for residuals
                residual_logvol = self.gamma[i] * factor_logvol_path[t] + idio_logvol_paths[i][t]
                residual_vol = np.exp(residual_logvol/2)
                
                # Generate stock returns as factor + idiosyncratic component
                factor_component = self.beta[i] * factor_returns[t]
                idio_component = self.sigma[i] * residual_vol * np.random.normal(0, 1/np.sqrt(s), s).sum()
                stock_returns[i, t] = factor_component + idio_component
        
        # Calculate index returns (equal-weighted)
        index_returns = np.mean(stock_returns, axis=0)
        
        return factor_returns, stock_returns, index_returns
    
    
def estimate_volatility(returns):
    """Estimate volatility using log-returns squared (simplified)"""
    return np.log(returns**2 + 1e-8)  # Add small constant to avoid log(0)


def estimate_hurst(logvol, lags=None):
    """
    Estimate Hurst exponent from log-volatility series
    using a simplified version of the approach in the paper
    """
    if lags is None:
        lags = np.array([1, 2, 4, 8, 16, 32])
    
    var_increments = []
    for lag in lags:
        if lag >= len(logvol):
            continue
        increments = logvol[lag:] - logvol[:-lag]
        var_increments.append(np.var(increments))
    
    # In log-log space, the slope is 2H
    logs = np.log(var_increments)
    log_lags = np.log(lags[:len(logs)])
    
    # Linear regression to find slope
    slope, _ = np.polyfit(log_lags, logs, 1)
    
    return slope / 2


def gaussianize(data):
    """
    Apply Gaussian transformation to data to mitigate skewness
    as described in the paper's Gaussian trick
    """
    ranks = np.argsort(np.argsort(data))
    n = len(data)
    quantiles = (ranks + 0.5) / n
    return norm.ppf(quantiles)
