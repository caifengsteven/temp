class FinRobot:
    """Main FinRobot class that orchestrates the multi-agent Chain-of-Thought system"""
    
    def __init__(self, ticker: str):
        self.ticker = ticker
        self.data_agent = DataCoTAgent(ticker)
        self.financial_data = None
        self.metrics = None
        self.concept_agent = None
        self.insights = None
        self.thesis_agent = None
        self.report = None
        
    def run_analysis(self) -> Dict:
        """Run the complete analysis pipeline"""
        print(f"Starting FinRobot analysis for {self.ticker}...")
        
        try:
            # Data-CoT: Process financial data
            self.financial_data = self.data_agent.fetch_financial_data()
            sec_filings = self.data_agent.fetch_sec_filings()
            earnings_calls = self.data_agent.fetch_earnings_calls()
            self.metrics = self.data_agent.calculate_financial_metrics()
            
            # Concept-CoT: Generate financial insights
            self.concept_agent = ConceptCoTAgent(self.financial_data)
            self.insights = self.concept_agent.generate_insights(self.metrics)
            
            # Thesis-CoT: Generate investment thesis and report
            self.thesis_agent = ThesisCoTAgent(
                self.financial_data, 
                self.insights, 
                sec_filings, 
                earnings_calls
            )
            
            self.report = self.thesis_agent.generate_report()
            
            print("FinRobot analysis completed successfully!")
            return self.report
            
        except Exception as e:
            print(f"Error in FinRobot analysis: {str(e)}")
            raise
