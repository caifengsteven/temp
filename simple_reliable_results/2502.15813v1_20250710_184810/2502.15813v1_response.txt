class LSTM(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers):
        super(LSTM, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.lstm = nn.LSTM(input_size, hidden_size, num_layers, batch_first=True)
        self.fc = nn.Linear(hidden_size, 32)
        
    def forward(self, x):
        h0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size).to(x.device)
        c0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size).to(x.device)
        
        out, _ = self.lstm(x, (h0, c0))
        out = self.fc(out[:, -1, :])
        return out

class GNN(nn.Module):
    def __init__(self, num_features, hidden_channels):
        super(GNN, self).__init__()
        self.conv1 = GCNConv(num_features, hidden_channels)
        self.conv2 = GCNConv(hidden_channels, 32)
        self.relu = nn.ReLU()
        self.dropout = nn.Dropout(0.5)
        
    def forward(self, x, edge_index, edge_weight=None):
        x = self.conv1(x, edge_index, edge_weight)
        x = self.relu(x)
        x = self.dropout(x)
        x = self.conv2(x, edge_index, edge_weight)
        return x

class HybridLSTM_GNN(nn.Module):
    def __init__(self, lstm_input_size, lstm_hidden_size, lstm_num_layers, 
                 gnn_num_features, gnn_hidden_channels):
        super(HybridLSTM_GNN, self).__init__()
        self.lstm = LSTM(lstm_input_size, lstm_hidden_size, lstm_num_layers)
        self.gnn = GNN(gnn_num_features, gnn_hidden_channels)
        self.fc1 = nn.Linear(64, 32)
        self.fc2 = nn.Linear(32, 1)
        self.relu = nn.ReLU()
        
    def forward(self, lstm_input, gnn_x, gnn_edge_index, gnn_edge_weight=None):
        lstm_output = self.lstm(lstm_input)
        gnn_output = self.gnn(gnn_x, gnn_edge_index, gnn_edge_weight)
        
        # Assuming we're predicting for the first node in the graph
        combined = torch.cat((lstm_output, gnn_output[0].unsqueeze(0)), dim=1)
        
        out = self.relu(self.fc1(combined))
        out = self.fc2(out)
        return out

def generate_simulated_stock_data(num_stocks=10, days=2000, volatility=0.01):
    """Generate simulated stock price data"""
    stocks = {}
    
    # Generate correlated stock prices
    correlation_matrix = np.random.uniform(0.5, 0.9, size=(num_stocks, num_stocks))
    np.fill_diagonal(correlation_matrix, 1)
    
    # Make the correlation matrix symmetric
    correlation_matrix = (correlation_matrix + correlation_matrix.T) / 2
    
    # Generate random walks with correlation
    cholesky = np.linalg.cholesky(correlation_matrix)
    
    # Initial prices
    initial_prices = np.random.uniform(50, 200, num_stocks)
    
    # Generate price series
    prices = np.zeros((days, num_stocks))
    prices[0] = initial_prices
    
    for i in range(1, days):
        # Generate correlated random changes
        random_changes = np.random.normal(0, volatility, num_stocks)
        correlated_changes = np.dot(cholesky, random_changes)
        
        # Calculate new prices
        prices[i] = prices[i-1] * (1 + correlated_changes)
    
    # Convert to DataFrame
    stock_names = [f'STOCK_{i}' for i in range(num_stocks)]
    df = pd.DataFrame(prices, columns=stock_names)
    
    # Add date index
    date_rng = pd.date_range(start='2015-01-01', periods=days, freq='B')
    df.index = date_rng
    
    return df

def create_graph_from_stocks(stock_data, threshold=0.7):
    """Create a graph based on stock correlations"""
    # Calculate daily returns
    returns = stock_data.pct_change().dropna()
    
    # Calculate correlation matrix
    corr_matrix = returns.corr()
    
    # Create graph
    G = nx.Graph()
    
    # Add nodes
    for stock in stock_data.columns:
        G.add_node(stock)
    
    # Add edges based on correlation threshold
    for i, stock1 in enumerate(stock_data.columns):
        for j, stock2 in enumerate(stock_data.columns):
            if i < j and abs(corr_matrix.loc[stock1, stock2]) > threshold:
                G.add_edge(stock1, stock2, weight=abs(corr_matrix.loc[stock1, stock2]))
    
    return G, corr_matrix

def prepare_data_for_lstm(data, seq_length=10):
    """Prepare sequences for LSTM"""
    X, y = [], []
    for i in range(len(data) - seq_length):
        X.append(data[i:i + seq_length])
        y.append(data[i + seq_length])
    return np.array(X), np.array(y)

def prepare_data_for_expanding_window(stock_data, target_stock, train_size, test_size, seq_length=10):
    """Prepare data for expanding window validation"""
    # Scale data
    scaler = MinMaxScaler()
    stock_data_scaled = scaler.fit_transform(stock_data)
    
    # Select data for the target stock
    target_idx = list(stock_data.columns).index(target_stock)
    target_data = stock_data_scaled[:, target_idx]
    
    # Split into train and test
    train_data = target_data[:train_size]
    test_data = target_data[train_size:train_size+test_size]
    
    # Prepare sequences
    X_train, y_train = prepare_data_for_lstm(train_data, seq_length)
    
    # For test data, we'll prepare one at a time during expanding window
    return X_train, y_train, test_data, scaler

def prepare_gnn_data(graph, stock_data, day_idx):
    """Prepare data for GNN using the graph and stock data for a specific day"""
    # Extract features for each node (stock)
    features = []
    for stock in graph.nodes():
        stock_idx = list(stock_data.columns).index(stock)
        # Use the previous 5 days as features
        if day_idx >= 5:
            feature = stock_data.iloc[day_idx-5:day_idx, stock_idx].values
            features.append(feature)
        else:
            # Fallback for early days
            features.append(np.zeros(5))
    
    # Convert to PyTorch tensors
    x = torch.tensor(features, dtype=torch.float)
    
    # Create edge index
    edge_index = []
    edge_weight = []
    for u, v, data in graph.edges(data=True):
        u_idx = list(graph.nodes()).index(u)
        v_idx = list(graph.nodes()).index(v)
        edge_index.append([u_idx, v_idx])
        edge_index.append([v_idx, u_idx])  # Undirected graph
        edge_weight.append(data['weight'])
        edge_weight.append(data['weight'])
    
    edge_index = torch.tensor(edge_index, dtype=torch.long).t()
    edge_weight = torch.tensor(edge_weight, dtype=torch.float)
    
    # Create PyTorch Geometric Data object
    data = Data(x=x, edge_index=edge_index, edge_weight=edge_weight)
    
    return data
