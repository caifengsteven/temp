class FederatedLogisticRegression:
    def __init__(self, learning_rate=0.01, max_iter=100, local_iter=10, agg_method='mean', C=1.0):
        self.learning_rate = learning_rate
        self.max_iter = max_iter
        self.local_iter = local_iter
        self.agg_method = agg_method
        self.C = C  # L2 regularization parameter
        self.W = None
        self.b = None
        
    def fit(self, client_data, client_labels, M, percent_outliers=0.0):
        num_features = client_data[0].shape[1]
        num_clients = len(client_data)
        
        # Initialize global model parameters
        self.W = np.zeros(num_features)
        self.b = 0
        
        # Calculate number of outlier clients
        num_outliers = int(num_clients * percent_outliers)
        
        for t in range(self.max_iter):
            # Get local updates from all clients
            W_updates = []
            b_updates = []
            
            for i in range(num_clients):
                # For outlier clients, generate random parameters
                if i < num_outliers:
                    # Outliers generate random parameters that are far from true values
                    W_local = np.random.normal(0, 10, num_features)
                    b_local = np.random.normal(0, 10)
                else:
                    # Regular clients perform local training
                    W_local, b_local = self._local_update(client_data[i], client_labels[i])
                
                W_updates.append(W_local)
                b_updates.append(b_local)
            
            # Aggregate updates using specified method
            self.W, self.b = self._aggregate_updates(W_updates, b_updates)
        
        return self
    
    def _local_update(self, X, y):
        W_local = np.copy(self.W)
        b_local = self.b
        
        for _ in range(self.local_iter):
            # Calculate predicted probabilities
            z = X.dot(W_local) + b_local
            y_pred = 1 / (1 + np.exp(-z))
            
            # Calculate gradients with L2 regularization
            dW = (1/len(X)) * (X.T.dot(y_pred - y) + (W_local / self.C))
            db = (1/len(X)) * np.sum(y_pred - y)
            
            # Update parameters
            W_local = W_local - self.learning_rate * dW
            b_local = b_local - self.learning_rate * db
        
        return W_local, b_local
    
    def _aggregate_updates(self, W_updates, b_updates):
        if self.agg_method == 'mean':
            W_agg = np.mean(W_updates, axis=0)
            b_agg = np.mean(b_updates)
        elif self.agg_method == 'median':
            W_agg = np.median(W_updates, axis=0)
            b_agg = np.median(b_updates)
        elif self.agg_method == 'trim_mean':
            # Trimmed mean - remove top and bottom 10%
            trim_percent = 0.1
            W_agg = self._trimmed_mean(W_updates, trim_percent)
            b_agg = self._trimmed_mean(b_updates, trim_percent)
        else:
            raise ValueError("Aggregation method not supported")
        
        return W_agg, b_agg
    
    def _trimmed_mean(self, values, trim_percent):
        if isinstance(values[0], np.ndarray):
            # For array parameters like W
            stacked = np.vstack(values)
            sorted_indices = np.argsort(stacked, axis=0)
            
            n = len(values)
            k = int(n * trim_percent)
            
            if k > 0:
                trimmed = []
                for j in range(stacked.shape[1]):
                    col = stacked[:, j]
                    sorted_col = np.sort(col)
                    trimmed_col = sorted_col[k:n-k]
                    trimmed.append(np.mean(trimmed_col))
                return np.array(trimmed)
            else:
                return np.mean(stacked, axis=0)
        else:
            # For scalar parameters like b
            sorted_values = np.sort(values)
            n = len(sorted_values)
            k = int(n * trim_percent)
            
            if k > 0:
                return np.mean(sorted_values[k:n-k])
            else:
                return np.mean(sorted_values)
    
    def predict_proba(self, X):
        z = X.dot(self.W) + self.b
        return 1 / (1 + np.exp(-z))
    
    def predict(self, X, threshold=0.5):
        return (self.predict_proba(X) >= threshold).astype(int)


def create_iid_data(X, y, num_clients, client_sample_size=None):
    """Create IID data distribution across clients"""
    if client_sample_size is None:
        # Distribute all data evenly among clients
        indices = np.random.permutation(len(X))
        client_data = []
        client_labels = []
        
        # Stratified partition by label
        for i in range(num_clients):
            client_idx = indices[i::num_clients]
            client_data.append(X[client_idx])
            client_labels.append(y[client_idx])
    else:
        # Sample fixed size from data for each client
        client_data = []
        client_labels = []
        
        for _ in range(num_clients):
            idx = np.random.choice(len(X), client_sample_size, replace=False)
            client_data.append(X[idx])
            client_labels.append(y[idx])
    
    return client_data, client_labels


def create_non_iid_data(X, y, num_clients, num_classes, client_sample_size=None):
    """Create non-IID data distribution across clients"""
    unique_classes = np.unique(y)
    
    if client_sample_size is None:
        # Distribute all data with class imbalance
        client_data = [[] for _ in range(num_clients)]
        client_labels = [[] for _ in range(num_clients)]

        for c in unique_classes:
            class_indices = np.where(y == c)[0]
            np.random.shuffle(class_indices)

            base_samples = 100  # Base samples per class per client

                if i * base_samples < len(class_indices):
                    end_idx = min((i + 1) * base_samples, len(class_indices))
                    idx = class_indices[i * base_samples:end_idx]
                    
                    if len(client_data[i]) == 0:
                        client_data[i] = X[idx]
                        client_labels[i] = y[idx]
                    else:
                        client_data[i] = np.vstack([client_data[i], X[idx]])
                        client_labels[i] = np.hstack([client_labels[i], y[idx]])
        
        # Then, assign remaining data based on class groups
        clients_per_group = num_clients // num_classes
        if clients_per_group == 0:
            clients_per_group = 1
            
        for c_idx, c in enumerate(unique_classes):
            class_indices = np.where(y == c)[0]
            np.random.shuffle(class_indices)
            
            # Skip the first samples already distributed
            already_distributed = min(num_clients * 100, len(class_indices))
            remaining_indices = class_indices[already_distributed:]
            
            # Assign to clients in this group
            group_start = (c_idx % num_clients) * clients_per_group
            group_clients = [i % num_clients for i in range(group_start, group_start + clients_per_group)]
            
            samples_per_client = len(remaining_indices) // len(group_clients)
            if samples_per_client > 0:
                for i, client_idx in enumerate(group_clients):
                    start_idx = i * samples_per_client
                    end_idx = (i + 1) * samples_per_client if i < len(group_clients) - 1 else len(remaining_indices)
                    
                    if start_idx < len(remaining_indices):
                        idx = remaining_indices[start_idx:end_idx]
                        client_data[client_idx] = np.vstack([client_data[client_idx], X[idx]])
                        client_labels[client_idx] = np.hstack([client_labels[client_idx], y[idx]])
    else:
        # Sample fixed size for each client with class imbalance
        client_data = []
        client_labels = []
        
        # Split clients into groups, each focused on different classes
        clients_per_class = max(1, num_clients // num_classes)
        
        for i in range(num_clients):
            # Determine the primary class for this client
            primary_class = unique_classes[i % len(unique_classes)]
            
            # Create a probability distribution favoring the primary class
            class_probs = np.ones(len(unique_classes)) * 0.1  # 10% chance for other classes
            primary_idx = np.where(unique_classes == primary_class)[0][0]
            
            # Adjust to make sure probabilities sum to 1
            remaining_prob = 1.0 - (len(unique_classes) - 1) * 0.1
            class_probs[primary_idx] = remaining_prob
            
            # Sample based on this distribution
            samples = []
            labels = []
            
            for c_idx, c in enumerate(unique_classes):
                # Number of samples to take from this class
                n_samples = int(client_sample_size * class_probs[c_idx])
                if n_samples > 0:
                    class_indices = np.where(y == c)[0]
                    
                    # Ensure we don't request more samples than available
                    n_samples = min(n_samples, len(class_indices))
                    
                    # Randomly select indices
                    selected_indices = np.random.choice(class_indices, n_samples, replace=False)
                    
                    samples.append(X[selected_indices])
                    labels.append(y[selected_indices])
            
            # Combine samples from all classes
            if samples:
                client_data.append(np.vstack(samples))
                client_labels.append(np.hstack(labels))
            else:
                # Fallback if no samples were selected
                idx = np.random.choice(len(X), client_sample_size, replace=False)
                client_data.append(X[idx])
                client_labels.append(y[idx])
    
    # Convert to numpy arrays
    client_data = [np.array(data) for data in client_data]
    client_labels = [np.array(labels) for labels in client_labels]
    
    return client_data, client_labels


def evaluate_model(model, X_test, y_test):
    """Evaluate model performance with multiple metrics"""
    if hasattr(model, 'predict_proba'):
        if hasattr(model, 'classes_') and len(model.classes_) > 2:
            # Multi-class case
            y_prob = model.predict_proba(X_test)
            auc = roc_auc_score(y_test, y_prob, multi_class='ovr')
        else:
            # Binary case
            if isinstance(model, FederatedLogisticRegression):
                y_prob = model.predict_proba(X_test)
            else:
                y_prob = model.predict_proba(X_test)[:, 1]
            auc = roc_auc_score(y_test, y_prob)
    else:
        # If model doesn't have predict_proba, use default AUC of 0.5
        auc = 0.5
    
    y_pred = model.predict(X_test)
    acc = accuracy_score(y_test, y_pred)
    
    if len(np.unique(y_test)) > 2:
        # Multi-class F1
        f1 = f1_score(y_test, y_pred, average='macro')
    else:
        # Binary F1
        f1 = f1_score(y_test, y_pred)
    
    return {'accuracy': acc, 'f1': f1, 'auc': auc}
