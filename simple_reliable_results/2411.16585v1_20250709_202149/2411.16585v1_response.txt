class OrderMessage:
    """Represents a single order message in the limit order book."""
    
    def __init__(
        self, 
        timestamp: int,  # nanoseconds since midnight
        message_type: str,  # 'add', 'execute', 'execute_price_diff', 'cancel', 'replace'
        order_id: int,
        direction: str,  # 'buy' or 'sell'
        size: int,
        price: float,
        reference_id: Optional[int] = None,  # for referential messages
        old_price: Optional[float] = None,   # for replace messages
    ):
        self.timestamp = timestamp
        self.message_type = message_type
        self.order_id = order_id
        self.direction = direction
        self.size = size
        self.price = price
        self.reference_id = reference_id
        self.old_price = old_price
        
    def __str__(self) -> str:
        return (f"Message({self.message_type}, ID: {self.order_id}, "
                f"{self.direction}, {self.size} @ ${self.price:.2f})")


class LimitOrderBook:
    """Simulates a limit order book that processes order messages."""
    
    def __init__(self):
        # Maps price levels to queues of orders
        self.bids = defaultdict(list)  # Buy orders
        self.asks = defaultdict(list)  # Sell orders
        # Maps order IDs to (price, direction) for quick lookup
        self.orders = {}
        
    def add_order(self, message: OrderMessage) -> None:
        """Add a new limit order to the book."""
        if message.message_type != 'add':
            raise ValueError("Expected 'add' message type")
            
        self.orders[message.order_id] = (message.price, message.direction)
        
        if message.direction == 'buy':
            self.bids[message.price].append((message.order_id, message.size))
        else:
            self.asks[message.price].append((message.order_id, message.size))
    
    def execute_order(self, message: OrderMessage) -> None:
        """Execute (fully or partially) an existing order."""
        if message.message_type not in ['execute', 'execute_price_diff']:
            raise ValueError("Expected 'execute' message type")
            
        if message.reference_id not in self.orders:
            raise ValueError(f"Order {message.reference_id} not found")
            
        price, direction = self.orders[message.reference_id]
        
        if direction == 'buy':
            orders = self.bids[price]
        else:
            orders = self.asks[price]
            
        # Find the order and update its size
        for i, (order_id, size) in enumerate(orders):
            if order_id == message.reference_id:
                if message.size >= size:
                    # Full execution, remove the order
                    orders.pop(i)
                    del self.orders[message.reference_id]
                else:
                    # Partial execution, update size
                    orders[i] = (order_id, size - message.size)
                break
    
    def cancel_order(self, message: OrderMessage) -> None:
        """Cancel (fully or partially) an existing order."""
        if message.message_type != 'cancel':
            raise ValueError("Expected 'cancel' message type")
            
        if message.reference_id not in self.orders:
            raise ValueError(f"Order {message.reference_id} not found")
            
        price, direction = self.orders[message.reference_id]
        
        if direction == 'buy':
            orders = self.bids[price]
        else:
            orders = self.asks[price]
            
        # Find the order and update its size
        for i, (order_id, size) in enumerate(orders):
            if order_id == message.reference_id:
                if message.size >= size:
                    # Full cancellation, remove the order
                    orders.pop(i)
                    del self.orders[message.reference_id]
                else:
                    # Partial cancellation, update size
                    orders[i] = (order_id, size - message.size)
                break
    
    def replace_order(self, message: OrderMessage) -> None:
        """Replace an existing order (update price and/or size)."""
        if message.message_type != 'replace':
            raise ValueError("Expected 'replace' message type")
            
        if message.reference_id not in self.orders:
            raise ValueError(f"Order {message.reference_id} not found")
            
        old_price, direction = self.orders[message.reference_id]
        
        # Remove the old order
        if direction == 'buy':
            orders = self.bids[old_price]
        else:
            orders = self.asks[old_price]
            
        for i, (order_id, size) in enumerate(orders):
            if order_id == message.reference_id:
                orders.pop(i)
                del self.orders[message.reference_id]
                break
                
        # Add the new order
        self.orders[message.order_id] = (message.price, direction)
        
        if direction == 'buy':
            self.bids[message.price].append((message.order_id, message.size))
        else:
            self.asks[message.price].append((message.order_id, message.size))
    
    def process_message(self, message: OrderMessage) -> None:
        """Process an order message and update the book."""
        if message.message_type == 'add':
            self.add_order(message)
        elif message.message_type in ['execute', 'execute_price_diff']:
            self.execute_order(message)
        elif message.message_type == 'cancel':
            self.cancel_order(message)
        elif message.message_type == 'replace':
            self.replace_order(message)
        else:
            raise ValueError(f"Unknown message type: {message.message_type}")
    
    def get_best_bid(self) -> Tuple[float, int]:
        """Get the best (highest) bid price and total size."""
        if not self.bids:
            return 0.0, 0
        best_price = max(self.bids.keys())
        total_size = sum(size for _, size in self.bids[best_price])
        return best_price, total_size
    
    def get_best_ask(self) -> Tuple[float, int]:
        """Get the best (lowest) ask price and total size."""
        if not self.asks:
            return float('inf'), 0
        best_price = min(self.asks.keys())
        total_size = sum(size for _, size in self.asks[best_price])
        return best_price, total_size
    
    def get_mid_price(self) -> float:
        """Get the mid price (average of best bid and ask)."""
        best_bid, _ = self.get_best_bid()
        best_ask, _ = self.get_best_ask()
        
        if best_bid == 0.0 or best_ask == float('inf'):
            # If either side is empty, return the non-empty side
            return best_bid if best_ask == float('inf') else best_ask
        
        return (best_bid + best_ask) / 2
    
    def get_spread(self) -> float:
        """Get the bid-ask spread."""
        best_bid, _ = self.get_best_bid()
        best_ask, _ = self.get_best_ask()
        
        if best_bid == 0.0 or best_ask == float('inf'):
            return float('inf')
        
        return best_ask - best_bid


class MarketSimulator:

    def __init__(self, initial_price: float = 100.0, price_volatility: float = 0.01):

        else:
            orders = self.order_book.asks[price]
            
        size = next((s for oid, s in orders if oid == order_id), 0)
        
        return order_id, price, direction, size

        """Calculate log returns at the specified time scale."""
        prices = self.mid_price_history
        if len(prices) <= time_scale:
            return []
            
        returns = []
        for i in range(time_scale, len(prices)):
            if prices[i-time_scale] > 0 and prices[i] > 0:
                r = np.log(prices[i] / prices[i-time_scale])
                returns.append(r)

    def test_volatility_clustering(self) -> Dict:
        """Test for volatility clustering in the returns."""
        returns = self.calculate_returns()
        if len(returns) < 100:
            return {"error": "Not enough data to test volatility clustering"}
            
        squared_returns = [r**2 for r in returns]
        abs_returns = [abs(r) for r in returns]
        
        # Calculate autocorrelation of squared returns
        max_lag = min(20, len(returns) // 5)
        sq_autocorr = [stats.pearsonr(squared_returns[:-i], squared_returns[i:])[0] 
                       for i in range(1, max_lag)]
        
        # Calculate autocorrelation of absolute returns
        abs_autocorr = [stats.pearsonr(abs_returns[:-i], abs_returns[i:])[0] 
                        for i in range(1, max_lag)]
        
        return {
            "squared_returns_autocorr": sq_autocorr,
            "abs_returns_autocorr": abs_autocorr,
            "has_volatility_clustering": all(ac > 0 for ac in sq_autocorr[:5])
        }
    
    def test_heavy_tails(self) -> Dict:
        """Test for heavy tails in the returns distribution."""
        returns = self.calculate_returns()
        if len(returns) < 100:

        kurtosis = stats.kurtosis(returns)
        skewness = stats.skew(returns)
        
        # Compare to normal distribution
        norm_samples = np.random.normal(size=len(returns))
        norm_kurtosis = stats.kurtosis(norm_samples)
        
        return {
            "returns_kurtosis": kurtosis,
            "returns_skewness": skewness,
            "normal_kurtosis": norm_kurtosis,
            "has_heavy_tails": kurtosis > norm_kurtosis + 1.0
