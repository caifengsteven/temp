import numpy as np
import pandas as pd

# Data models based on the paper
@dataclass
class UniqueInstrument:
    """Base class for unique instruments"""
    ticker: str
    value: float
    delta: float
    vega: float
    gamma: float
    pl_scenarios: np.ndarray  # Historical P&L across scenarios
    
    def get_cost(self, notional: float) -> float:
        """Calculate trading cost for this instrument"""
        raise NotImplementedError("Subclasses must implement this method")


@dataclass
class Stock(UniqueInstrument):
    """Stock instrument"""
    def get_cost(self, notional: float) -> float:
        # Cost based on delta as per equation (6)
        return 0.5 * abs(self.delta) * abs(notional) * 0.0005  # Assuming 0.05% spread


@dataclass
class Option(UniqueInstrument):
    """Option instrument"""
    payoff_type: str  # 'c' for call, 'p' for put
    strike: float
    maturity: int  # Days to maturity
    
    def get_cost(self, notional: float) -> float:
        # Cost based on delta and vega as per equations (6) and (7)
        delta_cost = 0.5 * abs(self.delta) * abs(notional) * 0.001  # Assuming 0.1% spread
        vega_cost = 0.5 * abs(self.vega) * abs(notional) * 0.002  # Assuming 0.2% strike spread
        return delta_cost + vega_cost


@dataclass
class Future(UniqueInstrument):
    """Future instrument"""
    maturity: int  # Days to maturity
    
    def get_cost(self, notional: float) -> float:
        # Cost based on equation (6)
        return 0.5 * abs(notional) * 0.0002  # Assuming 0.02% spread


class Portfolio:
    """Portfolio class to manage a collection of instruments with notional amounts"""
    
    def __init__(self, name: str = "Portfolio"):
        self.name = name
        self.instruments = []
        self.notionals = []
        
    def add_instrument(self, instrument: UniqueInstrument, notional: float):
        """Add an instrument to the portfolio with its notional amount"""
        self.instruments.append(instrument)
        self.notionals.append(notional)
    
    def get_value(self) -> float:
        """Calculate total portfolio value"""
        return sum(instr.value * notional for instr, notional in zip(self.instruments, self.notionals))
    
    def get_delta(self) -> float:
        """Calculate total portfolio delta"""
        return sum(instr.delta * notional for instr, notional in zip(self.instruments, self.notionals))
    
    def get_vega(self) -> float:
        """Calculate total portfolio vega"""
        return sum(instr.vega * notional for instr, notional in zip(self.instruments, self.notionals))
    
    def get_gamma(self) -> float:
        """Calculate total portfolio gamma"""
        return sum(instr.gamma * notional for instr, notional in zip(self.instruments, self.notionals))
    
    def get_pl_scenarios(self) -> np.ndarray:
        """Calculate P&L across scenarios"""
        if not self.instruments:
            return np.array([])
        
        # Sum P&L across all instruments, weighted by notional
        pl = np.zeros_like(self.instruments[0].pl_scenarios)
        for instr, notional in zip(self.instruments, self.notionals):
            pl += instr.pl_scenarios * notional
        return pl
    
    def get_var(self, percentile: float = 0.01, decay_factor: float = 0.94) -> float:
        """Calculate Value at Risk at given percentile with exponential weighting"""
        pl = self.get_pl_scenarios()
        if len(pl) == 0:
            return 0
            
        # Sort P&L scenarios in ascending order
        sorted_pl = np.sort(pl)
        
        # Calculate VaR index based on decay factor as in equation (11)
        s = len(sorted_pl)
        alpha = 1 - percentile * (1 - decay_factor/s)
        i_star = int(np.ceil(np.log(alpha) / np.log(decay_factor)))
        
        # Ensure index is within bounds
        i_star = max(0, min(i_star, s-1))
        
        # Return the VaR value (negative of the P&L at the index)
        return -sorted_pl[i_star]
    
    def get_trading_cost(self) -> float:
        """Calculate trading cost for all instruments"""
        return sum(instr.get_cost(notional) for instr, notional in 
                  zip(self.instruments, self.notionals))
    
    def get_expected_pl(self) -> float:
        """Calculate expected P&L"""
        pl = self.get_pl_scenarios()
        if len(pl) == 0:
            return 0
        return np.mean(pl)


class RATS:
    """Risk-Aware Trading Swarm algorithm implementation"""
    
    def __init__(self, 
                 initial_portfolio: Portfolio,
                 eligible_instruments: List[UniqueInstrument],
                 max_instruments: int = 3,
                 n_particles: int = 100,
                 max_iterations: int = 50,
                 personal_coef: float = 0.5,
                 social_coef: float = 1.2,
                 inertia_weight: float = 0.9,
                 min_inertia_weight: float = 0.4,
                 delta_constraint: float = 0.5,
                 vega_constraint: float = 0.5,
                 gamma_constraint: float = 0.5):
        
        self.initial_portfolio = initial_portfolio
        self.eligible_instruments = eligible_instruments
        self.max_instruments = max_instruments
        self.n_particles = n_particles
        self.max_iterations = max_iterations
        self.personal_coef = personal_coef
        self.social_coef = social_coef
        self.inertia_weight = inertia_weight
        self.min_inertia_weight = min_inertia_weight
        
        # Constraints as proportions of initial portfolio sensitivities
        self.delta_constraint = delta_constraint
        self.vega_constraint = vega_constraint
        self.gamma_constraint = gamma_constraint
        
        # Initialize particles
        self.particles_positions = []
        self.particles_velocities = []
        self.particles_best_positions = []
        self.particles_best_values = np.full(n_particles, np.inf)
        self.global_best_position = None
        self.global_best_value = np.inf
        
        # Tracking optimization progress
        self.fitness_history = []
        
    def _initialize_particles(self):
        """Initialize particles with random positions and velocities"""
        for _ in range(self.n_particles):
            # Position: first part selects instruments, second part selects notionals
            position = []
            
            # Randomly select up to max_instruments
            num_instruments = random.randint(1, self.max_instruments)
            selected_indices = random.sample(range(len(self.eligible_instruments)), num_instruments)
            
            # Fill with -1 for unused slots
            position.extend(selected_indices + [-1] * (self.max_instruments - num_instruments))
            
            # Random notionals for selected instruments (-500 to 500 as an example)
            notionals = []
            for i in range(self.max_instruments):
                if i < num_instruments:
                    notionals.append(random.randint(-500, 500))
                else:
                    notionals.append(0)
            
            position.extend(notionals)
            self.particles_positions.append(position)
            
            # Initialize velocities randomly between -1 and 1
            self.particles_velocities.append(
                np.random.uniform(-1, 1, 2 * self.max_instruments)
            )
            
            # Initialize personal best position to current position
            self.particles_best_positions.append(position.copy())
    
    def _create_portfolio_from_position(self, position) -> Portfolio:
        """Create a portfolio from a particle position"""
        optimization_portfolio = Portfolio("Optimization")
        
        # First half of position contains instrument indices, second half contains notionals
        for i in range(self.max_instruments):
            idx = position[i]
            notional = position[i + self.max_instruments]
            
            # Skip if index is -1 (no instrument selected) or notional is 0
            if idx == -1 or notional == 0:
                continue
                
            # Add instrument to the optimization portfolio
            optimization_portfolio.add_instrument(self.eligible_instruments[idx], notional)
        
        return optimization_portfolio
    
    def _combine_portfolios(self, opt_portfolio: Portfolio) -> Portfolio:
        """Combine initial portfolio with optimization portfolio"""
        combined = Portfolio("Combined")
        
        # Add instruments from initial portfolio
        for instr, notional in zip(self.initial_portfolio.instruments, 
                                   self.initial_portfolio.notionals):
            combined.add_instrument(instr, notional)
            
        # Add instruments from optimization portfolio
        for instr, notional in zip(opt_portfolio.instruments, opt_portfolio.notionals):
            combined.add_instrument(instr, notional)
            
        return combined
    
    def _evaluate_fitness(self, position) -> Tuple[float, bool]:
        """Evaluate the fitness of a particle position and check constraints"""
        # Create optimization portfolio from position
        opt_portfolio = self._create_portfolio_from_position(position)
        
        # Create combined portfolio
        combined_portfolio = self._combine_portfolios(opt_portfolio)
        
        # Check constraints
        initial_delta = abs(self.initial_portfolio.get_delta())
        initial_vega = abs(self.initial_portfolio.get_vega())
        initial_gamma = abs(self.initial_portfolio.get_gamma())
        
        opt_delta = abs(opt_portfolio.get_delta())
        opt_vega = abs(opt_portfolio.get_vega())
        opt_gamma = abs(opt_portfolio.get_gamma())
        
        # Check if optimization portfolio satisfies sensitivity constraints
        delta_satisfied = opt_delta <= self.delta_constraint * initial_delta
        vega_satisfied = opt_vega <= self.vega_constraint * initial_vega
        gamma_satisfied = opt_gamma <= self.gamma_constraint * initial_gamma
        
        constraints_satisfied = delta_satisfied and vega_satisfied and gamma_satisfied
        
        # Calculate fitness value: cost-adjusted risk measure from equation (18)
        expected_pl = combined_portfolio.get_expected_pl()
        initial_expected_pl = self.initial_portfolio.get_expected_pl()
        var = combined_portfolio.get_var()
        trading_cost = opt_portfolio.get_trading_cost()
        
        # Avoid division by zero
        denominator = var - trading_cost
        if denominator == 0:
            fitness = float('inf')
        else:
            fitness = (expected_pl - initial_expected_pl - trading_cost) / denominator
            
            # We want to minimize the objective function in equation (18)
            # More negative is better (as explained in Section 5.1)
            fitness = -fitness
        
        # Penalize constraint violations
        if not constraints_satisfied:
            fitness += 1000  # Large penalty for constraint violation
        
        return fitness, constraints_satisfied
    
    def optimize(self) -> Portfolio:
        """Run the RATS optimization algorithm"""
        # Initialize particles
        self._initialize_particles()
        
        # Evaluate initial fitness for each particle
        for i in range(self.n_particles):
            fitness, _ = self._evaluate_fitness(self.particles_positions[i])
            self.particles_best_values[i] = fitness
            
            # Update global best if this particle is better
            if fitness < self.global_best_value:
                self.global_best_value = fitness
                self.global_best_position = self.particles_positions[i].copy()
        
        # Optimization loop
        for iteration in range(self.max_iterations):
            # Update inertia weight (linear decay)
            self.inertia_weight = self.inertia_weight - (
                (self.inertia_weight - self.min_inertia_weight) * iteration / self.max_iterations
            )
            
            # For each particle
            for i in range(self.n_particles):
                # Update velocity and position
                position = np.array(self.particles_positions[i])
                best_position = np.array(self.particles_best_positions[i])
                global_best = np.array(self.global_best_position)
                
                # Personal and social components for velocity update
                r1 = np.random.random(2 * self.max_instruments)
                r2 = np.random.random(2 * self.max_instruments)
                
                personal_component = self.personal_coef * r1 * (best_position - position)
                social_component = self.social_coef * r2 * (global_best - position)
                
                # Update velocity using equation (R1)
                self.particles_velocities[i] = (
                    self.inertia_weight * self.particles_velocities[i] + 
                    personal_component + social_component
                )
                
                # Update position
                new_position = position + self.particles_velocities[i]
                
                # Ensure indices stay within bounds
                for j in range(self.max_instruments):
                    # Instrument indices (first half)
                    if new_position[j] >= 0:  # Only round valid indices
                        new_position[j] = round(min(max(0, new_position[j]), 
                                                   len(self.eligible_instruments)-1))
                    
                    # Notionals (second half) - bound between -500 and 500 as an example
                    notional_idx = j + self.max_instruments
                    new_position[notional_idx] = round(min(max(-500, new_position[notional_idx]), 500))
                
                # Update particle position
                self.particles_positions[i] = new_position.tolist()
                
                # Evaluate new position
                fitness, _ = self._evaluate_fitness(self.particles_positions[i])
                
                # Update personal best if improved
                if fitness < self.particles_best_values[i]:
                    self.particles_best_values[i] = fitness
                    self.particles_best_positions[i] = self.particles_positions[i].copy()
                    
                    # Update global best if this is the new best
                    if fitness < self.global_best_value:
                        self.global_best_value = fitness
                        self.global_best_position = self.particles_positions[i].copy()
            
            # Track best fitness in this iteration
            self.fitness_history.append(self.global_best_value)
            
            # Early stopping if converged
            if iteration > 10 and abs(self.fitness_history[-1] - self.fitness_history[-10]) < 1e-6:
                print(f"Converged at iteration {iteration}")
                break
        
        # Create and return the optimal portfolio
        opt_portfolio = self._create_portfolio_from_position(self.global_best_position)
        return self._combine_portfolios(opt_portfolio)
