import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
from scipy.optimize import minimize

class VolatilityParametrization:
    """Base class for volatility parametrization models"""
    
    def __init__(self):
        pass
        
    def implied_volatility(self, T, K):
        """Calculate implied volatility for given T and K"""
        raise NotImplementedError
        
    def option_price(self, T, K, is_call=True):
        """Calculate option price using Black-Scholes formula"""
        vol = self.implied_volatility(T, K)
        return self._black_scholes(T, K, vol, is_call)
        
    def _black_scholes(self, T, K, sigma, is_call=True):
        """Black-Scholes formula implementation"""
        S0 = self.S0
        r = self.r
        
        d1 = (np.log(S0/K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        d2 = d1 - sigma * np.sqrt(T)
        
        if is_call:
            return S0 * stats.norm.cdf(d1) - K * np.exp(-r * T) * stats.norm.cdf(d2)
        else:
            return K * np.exp(-r * T) * stats.norm.cdf(-d2) - S0 * stats.norm.cdf(-d1)


class SABRModel(VolatilityParametrization):
    """Standard SABR volatility parametrization"""
    
    def __init__(self, S0, r, alpha, beta, rho, nu):
        """
        Initialize SABR model with parameters
        
        Parameters:
        - S0: spot price
        - r: risk-free rate
        - alpha: volatility level parameter
        - beta: skew parameter (0 <= beta <= 1)
        - rho: correlation parameter (-1 <= rho <= 1)
        - nu: volatility of volatility parameter
        """
        super().__init__()
        self.S0 = S0
        self.r = r
        self.alpha = alpha
        self.beta = beta
        self.rho = rho
        self.nu = nu
        
    def implied_volatility(self, T, K):
        """
        SABR implied volatility formula based on Hagan et al.
        
        Parameters:
        - T: time to expiry
        - K: strike price
        
        Returns:
        - implied volatility
        """
        # Forward price
        F = self.S0 * np.exp(self.r * T)
        
        # Handle ATM case separately to avoid numerical issues
        if abs(F - K) < 1e-10:
            return self._sabr_atm_vol(T, F)
        
        # SABR parameters
        alpha = self.alpha
        beta = self.beta
        rho = self.rho
        nu = self.nu
        
        # Compute the volatility
        log_FK = np.log(F/K)
        F_K_mid = (F * K)**(1-beta)/2
        
        z = nu/alpha * F_K_mid * log_FK
        
        # Compute x(z)
        if abs(z) < 1e-6:  # For small z, use approximation
            x_z = 1 - 0.5 * rho * z + (2 - 3*rho**2)/6 * z**2
        else:
            x_z = np.log((np.sqrt(1 - 2*rho*z + z**2) + z - rho)/(1 - rho))
        
        # Main formula components
        term1 = alpha / (F_K_mid * (1 + (1-beta)**2/24 * log_FK**2 + (1-beta)**4/1920 * log_FK**4))
        term2 = z / x_z
        term3 = 1 + ((1-beta)**2/24 * alpha**2/(F_K_mid)**2 + 
                     1/4 * rho*beta*nu*alpha/(F_K_mid) + 
                     (2-3*rho**2)/24 * nu**2) * T
        
        return term1 * term2 * term3
    
    def _sabr_atm_vol(self, T, F):
        """Special case for ATM volatility in SABR model"""
        alpha = self.alpha
        beta = self.beta
        rho = self.rho
        nu = self.nu
        
        # ATM formula
        atm_vol = alpha * (1 + 
                          ((1-beta)**2/24 * alpha**2/F**(2-2*beta) + 
                           1/4 * rho*beta*nu*alpha/F**(1-beta) + 
                           (2-3*rho**2)/24 * nu**2) * T)
        
        return atm_vol


class RandomizedSABRModel(SABRModel):
    """SABR model with randomized nu parameter using Gamma distribution"""
    
    def __init__(self, S0, r, alpha, beta, rho, k, theta, Nq=2):
        """
        Initialize Randomized SABR model
        
        Parameters:
        - S0, r, alpha, beta, rho: same as SABR
        - k: shape parameter for Gamma distribution
        - theta: scale parameter for Gamma distribution
        - Nq: number of quadrature points (default=2)
        """
        # Initialize with dummy nu - will be randomized
        super().__init__(S0, r, alpha, beta, rho, 1.0)
        self.k = k
        self.theta = theta
        self.Nq = Nq
        
        # Calculate quadrature points and weights
        self._calculate_quadrature()
        
    def _calculate_quadrature(self):
        """Calculate Gauss quadrature points and weights for Gamma distribution"""
        # For simplicity, we'll use a basic quadrature approach
        # In a real implementation, the method described in the paper should be used
        
        k, theta = self.k, self.theta
        Nq = self.Nq
        
        # For Gamma distribution with parameters k, theta
        # Simple approach: use percentiles as quadrature points
        percentiles = np.linspace(1/(2*Nq), 1-1/(2*Nq), Nq)
        self.quad_points = stats.gamma.ppf(percentiles, k, scale=theta)
        
        # Equal weights for simplicity
        self.quad_weights = np.ones(Nq) / Nq
        
    def implied_volatility(self, T, K):
        """
        Calculate randomized SABR implied volatility
        
        Parameters:
        - T: time to expiry
        - K: strike price
        
        Returns:
        - implied volatility from randomized model
        """
        # For analytical expansion, we would implement the formula from Theorem 2.4
        # For simplicity, we use the quadrature approach directly on prices
        
        prices = 0
        for i in range(self.Nq):
            nu_i = self.quad_points[i]
            weight_i = self.quad_weights[i]
            
            # Create temporary SABR model with this nu value
            temp_model = SABRModel(self.S0, self.r, self.alpha, self.beta, self.rho, nu_i)
            
            # Add weighted price
            prices += weight_i * temp_model.option_price(T, K)
            
        # Solve for implied volatility using Newton's method
        return self._solve_implied_volatility(prices, T, K)
        
    def _solve_implied_volatility(self, price, T, K):
        """Solve for implied volatility given option price"""
        # Initial guess - use ATM vol
        sigma_guess = self.alpha
        
        # Simple iterative approach
        for _ in range(10):  # Usually converges in a few iterations
            temp_price = self._black_scholes(T, K, sigma_guess)
            vega = self._calculate_vega(T, K, sigma_guess)
            
            if abs(vega) < 1e-10:
                break
                
            # Newton step
            sigma_guess = sigma_guess + (price - temp_price) / vega
            
            if abs(price - temp_price) < 1e-8:
                break
                
        return max(0.001, sigma_guess)  # Ensure positive vol
        
    def _calculate_vega(self, T, K, sigma):
        """Calculate option vega for Newton's method"""
        S0 = self.S0
        
        d1 = (np.log(S0/K) + (self.r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
        
        return S0 * np.sqrt(T) * stats.norm.pdf(d1)


def generate_market_data(S0, expiries, strike_range, true_model, noise_level=0.005):
    """
    Generate synthetic market data with some noise to simulate real market quotes
    
    Parameters:
    - S0: spot price
    - expiries: list of expiry times
    - strike_range: tuple (min_strike, max_strike, num_strikes)
    - true_model: model used to generate the "true" volatilities
    - noise_level: level of noise to add to volatilities
    
    Returns:
    - dictionary with market data
    """
    min_k, max_k, num_k = strike_range
    
    market_data = []
    
    # Generate strikes as percentages of S0
    strikes = np.linspace(min_k, max_k, num_k) * S0
    
    for T in expiries:
        for K in strikes:
            # Generate true volatility
            true_vol = true_model.implied_volatility(T, K)
            
            # Add some noise
            market_vol = true_vol * (1 + noise_level * (np.random.random() - 0.5))
            
            market_data.append({
                'expiry': T,
                'strike': K,
                'implied_vol': market_vol
            })
    
    return market_data


def calibrate_model(model_class, market_data, initial_params, param_bounds):
    """
    Calibrate model to market data
    
    Parameters:
    - model_class: class of model to calibrate
    - market_data: list of dictionaries with market quotes
    - initial_params: initial parameters for the model
    - param_bounds: bounds for parameters
    
    Returns:
    - calibrated model instance
    """
    # Extract spot price and rate from initial params
    S0, r = initial_params[:2]
    
    # Group market data by expiry
    expiry_data = {}
    for quote in market_data:
        T = quote['expiry']
        if T not in expiry_data:
            expiry_data[T] = []
        expiry_data[T].append(quote)
    
    # Calibrate by expiry
    calibrated_params = {}
    
    for T, quotes in expiry_data.items():
        def objective_function(params):
            # Create model with these parameters
            if model_class == SABRModel:
                model = model_class(S0, r, *params)
            else:  # RandomizedSABRModel
                model = model_class(S0, r, *params)
            
            # Calculate squared errors
            total_error = 0
            for quote in quotes:
                K = quote['strike']
                market_vol = quote['implied_vol']
                model_vol = model.implied_volatility(T, K)
                total_error += (model_vol - market_vol)**2
                
            return total_error
        
        # Initial params for optimization (without S0, r)
        if model_class == SABRModel:
            init_params = initial_params[2:6]
            bounds = param_bounds[2:6]
        else:
            init_params = initial_params[2:7]
            bounds = param_bounds[2:7]
        
        # Run optimization
        result = minimize(
            objective_function, 
            init_params,
            bounds=bounds,
            method='L-BFGS-B'
        )
        
        calibrated_params[T] = result.x
    
    # Return calibrated parameters for each expiry
    return calibrated_params


def plot_volatility_surface(models, expiries, strikes, title):
    """Plot volatility surfaces for comparison"""
    fig = plt.figure(figsize=(12, 8))
    ax = fig.add_subplot(111, projection='3d')
    
    # Create meshgrid for surface
    X, Y = np.meshgrid(expiries, strikes)
    
    # Plot each model's surface
    for i, (name, model) in enumerate(models.items()):
        Z = np.zeros_like(X)
        for i in range(X.shape[0]):
            for j in range(X.shape[1]):
                Z[i,j] = model.implied_volatility(X[i,j], Y[i,j])
        
        # Plot surface with transparency
        ax.plot_surface(X, Y, Z, label=name, alpha=0.7)
    
    ax.set_xlabel('Time to Expiry')
    ax.set_ylabel('Strike')
    ax.set_zlabel('Implied Volatility')
    ax.set_title(title)
    
    plt.tight_layout()
    plt.show()
