        }
        self.end_patterns = {
            "1": [r"ITEM\s+1A", r"ITEM\s+2"],
            "1A": [r"ITEM\s+1B", r"ITEM\s+2"],
            "2": [r"ITEM\s+3"],
            "3": [r"ITEM\s+4"],
            "7": [r"ITEM\s+7A", r"ITEM\s+8"]
        }
    
    def segment(self, report: Report) -> List[str]:
        text = report.get_text()
        lines = [line.content for line in report.lines]
        result = ["O"] * len(lines)
        
        for item_num, patterns in self.item_patterns.items():
            # Find item start
            start_idx = None
            for pattern in patterns:
                for i, line in enumerate(lines):
                    if re.search(pattern, line, re.IGNORECASE):
                        start_idx = i
                        break
                if start_idx is not None:
                    break
            
            if start_idx is None:
                continue
                
            # Find item end
            end_idx = len(lines)
            for pattern in self.end_patterns.get(item_num, []):
                for i, line in enumerate(lines[start_idx+1:], start_idx+1):
                    if re.search(pattern, line, re.IGNORECASE):
                        end_idx = i
                        break
                if end_idx < len(lines):
                    break
            
            # Label lines
            result[start_idx] = f"B{item_num}"
            for i in range(start_idx+1, end_idx):
                result[i] = f"I{item_num}"
                
        return result

def evaluate_segmenter(segmenter: ItemSegmenter, reports: List[Report]) -> Dict:
    """Evaluate a segmenter on a set of reports"""
    all_true = []
    all_pred = []
    
    for report in reports:
        true_labels = [line.label for line in report.lines]
        pred_labels = segmenter.segment(report)
        
        all_true.extend(true_labels)
        all_pred.extend(pred_labels)
    
    # Get items present in the dataset
    items = set()
    for label in all_true:
        if label.startswith('B') or label.startswith('I'):
            items.add(label[1:])  # Extract item number
    
    # Calculate F1 for each item
    results = {}
    for item in sorted(items):
        # Convert to binary classification for this item
        y_true = [1 if label.endswith(item) else 0 for label in all_true]
        y_pred = [1 if label.endswith(item) else 0 for label in all_pred]
        
        # Calculate metrics
        p = precision_score(y_true, y_pred, zero_division=0)
        r = recall_score(y_true, y_pred, zero_division=0)
        f1 = f1_score(y_true, y_pred, zero_division=0)
        
        results[f"Item {item}"] = {
            "precision": p,
            "recall": r,
            "f1": f1
        }
    
    # Calculate macro average
    f1_values = [item_results["f1"] for item_results in results.values()]
    results["Macro Avg"] = {
        "f1": sum(f1_values) / len(f1_values) if f1_values else 0
    }
    
    return results
