class PortfolioEnvironment:
    def __init__(self, data, window_size=10, transaction_cost_rate=0.0025):
        """
        Initialize the portfolio environment
        
        Args:
            data: DataFrame with columns for open, high, low, close for each asset
            window_size: Number of time steps to look back
            transaction_cost_rate: Transaction cost per asset
        """
        self.data = data
        self.window_size = window_size
        self.transaction_cost_rate = transaction_cost_rate
        self.n_assets = (len(data.columns) // 4)  # Assumes 4 features per asset (open, high, low, close)
        self.current_step = window_size
        self.returns = []
        
    def reset(self):
        """Reset environment and return initial state"""
        self.current_step = self.window_size
        self.returns = []
        self.portfolio_value = 1.0
        self.weights = np.zeros(self.n_assets + 1)  # +1 for cash
        self.weights[0] = 1.0  # Start with all cash
        return self._get_state()
    
    def _get_state(self):
        """
        Get the current state (price tensor) as described in the paper
        """
        # Extract the window of data
        window_data = self.data.iloc[self.current_step-self.window_size:self.current_step]
        
        # Initialize price tensor
        X_t = np.zeros((4, self.window_size, self.n_assets))
        
        # Current closing prices for normalization
        current_close = np.array([window_data.iloc[-1][f'close_{i}'] for i in range(self.n_assets)])
        
        # Fill the price tensor as per equation (1) in the paper
        for t in range(self.window_size):
            for i in range(self.n_assets):
                X_t[0, t, i] = window_data.iloc[t][f'open_{i}'] / current_close[i]
                X_t[1, t, i] = window_data.iloc[t][f'low_{i}'] / current_close[i]
                X_t[2, t, i] = window_data.iloc[t][f'high_{i}'] / current_close[i]
                X_t[3, t, i] = window_data.iloc[t][f'close_{i}'] / current_close[i]
        
        return X_t
    
    def step(self, action):
        """
        Take an action (portfolio weights) and move to the next state
        
        Args:
            action: Portfolio weights (must sum to 1)
        
        Returns:
            next_state, reward, done, info
        """
        # Ensure action sums to 1 (apply softmax if needed)
        action = np.array(action)
        if np.sum(action) != 1:
            action = action / np.sum(action)
        
        # Get relative price vector Y_t as per equation (4)
        current_close = np.array([self.data.iloc[self.current_step-1][f'close_{i}'] for i in range(self.n_assets)])
        next_close = np.array([self.data.iloc[self.current_step][f'close_{i}'] for i in range(self.n_assets)])
        Y_t = np.ones(self.n_assets + 1)
        Y_t[1:] = next_close / current_close
        
        # Calculate W_t' as per equation (11)
        W_t_prime = (Y_t * self.weights) / np.dot(Y_t, self.weights)
        
        # Calculate transaction cost as per equation (10)
        transaction_cost = self.transaction_cost_rate * np.sum(np.abs(W_t_prime[1:] - action[1:]))
        
        # Update portfolio value as per equation (5)
        log_return = np.log(Y_t).dot(self.weights)
        self.portfolio_value *= (1 - transaction_cost) * np.exp(log_return)
        
        # Store daily log return for Sharpe ratio calculation
        self.returns.append(log_return)
        
        # Update weights to new action
        self.weights = action
        
        # Move to next step
        self.current_step += 1
        done = self.current_step >= len(self.data) - 1
        
        # Calculate reward (average Sharpe ratio) as per equation (9)
        if len(self.returns) > 1:  # Need at least 2 returns for standard deviation
            mean_return = np.mean(self.returns)
            std_return = np.std(self.returns) if np.std(self.returns) > 0 else 1e-6  # Avoid division by zero
            steps = len(self.returns)
            reward = (252 * mean_return) / (np.sqrt(252) * std_return) / steps
        else:
            reward = 0
        
        next_state = self._get_state() if not done else None
        
        info = {
            'portfolio_value': self.portfolio_value,
            'weights': self.weights,
            'transaction_cost': transaction_cost,
            'log_return': log_return
        }
        
        return next_state, reward, done, info

class DRLAgent:
    def __init__(self, state_shape, n_assets, learning_rate=0.001, gamma=0.99):
        """
        Initialize the DRL agent with a VGG-like network as described in the paper
        
        Args:
            state_shape: Shape of the state (price tensor)
            n_assets: Number of assets including cash
            learning_rate: Learning rate for the optimizer
            gamma: Discount factor for future rewards
        """
        self.state_shape = state_shape
        self.n_assets = n_assets
        self.learning_rate = learning_rate
        self.gamma = gamma
        self.actor = self._build_actor()
        
        # Experience replay buffer
        self.memory = deque(maxlen=2000)
        
    def _build_actor(self):
        """Build the actor network (VGG-like) as described in Figure 3 of the paper"""
        input_shape = self.state_shape
        
        # Reshape input to match typical image format for Conv2D (batch, height, width, channels)
        inputs = Input(shape=(input_shape[0], input_shape[1], input_shape[2]))
        
        # VGG-like architecture
        x = Conv2D(64, (3, 3), padding='same', activation='relu')(inputs)
        x = MaxPooling2D((2, 2))(x)
        
        x = Conv2D(64, (3, 3), padding='same', activation='relu')(x)
        x = Conv2D(128, (3, 3), padding='same', activation='relu')(x)
        x = MaxPooling2D((2, 2))(x)
        
        x = Conv2D(128, (3, 3), padding='same', activation='relu')(x)
        x = Conv2D(256, (3, 3), padding='same', activation='relu')(x)
        x = MaxPooling2D((2, 2))(x)
        
        x = Flatten()(x)
        x = Dense(128, activation='relu')(x)
        x = Dropout(0.1)(x)
        x = Dense(128, activation='relu')(x)
        
        # Output layer with softmax for portfolio weights
        outputs = Dense(self.n_assets, activation='softmax')(x)
        
        model = Model(inputs=inputs, outputs=outputs)
        model.compile(optimizer=Adam(learning_rate=self.learning_rate), 
                      loss='mse')  # Using MSE for simplicity in this demo
        
        return model
    
    def act(self, state, explore=True):
        """
        Choose an action (portfolio weights) given the current state
        
        Args:
            state: Current state (price tensor)
            explore: Whether to add exploration noise
        
        Returns:
            action: Portfolio weights
        """
        # Reshape state for the network
        state = np.expand_dims(state, axis=0)
        
        # Get action from the actor network
        action = self.actor.predict(state)[0]
        
        # Add exploration noise if needed
        if explore:
            noise = np.random.normal(0, 0.1, size=action.shape)
            action = action + noise
            action = np.clip(action, 0, 1)  # Ensure non-negative weights (long only)
            action = action / np.sum(action)  # Normalize to sum to 1
        
        return action
    
    def train(self, batch_size=32):
        """
        Train the network using experience replay
        
        Args:
            batch_size: Number of samples to use for training
        """
        if len(self.memory) < batch_size:
            return
        
        # Sample batch from memory
        batch = random.sample(self.memory, batch_size)
        states, actions, rewards, next_states, dones = zip(*batch)
        
        # Convert to numpy arrays
        states = np.array(states)
        actions = np.array(actions)
        rewards = np.array(rewards)
        
        # Train actor network
        self.actor.fit(states, actions, verbose=0, batch_size=batch_size)
    
    def remember(self, state, action, reward, next_state, done):
        """Store experience in memory"""
        self.memory.append((state, action, reward, next_state, done))
    
    def save(self, filepath):
        """Save model weights"""
        self.actor.save_weights(filepath)
    
    def load(self, filepath):
        """Load model weights"""
        self.actor.load_weights(filepath)

def generate_simulated_data(n_assets=10, n_days=1000):
    """
    Generate simulated price data for testing
    
    Args:
        n_assets: Number of assets
        n_days: Number of trading days
    
    Returns:
        DataFrame with simulated price data
    """
    # Start date
    start_date = datetime(2020, 1, 1)
    dates = [start_date + timedelta(days=i) for i in range(n_days)]
    
    # Initialize DataFrame
    columns = []
    for i in range(n_assets):
        columns.extend([f'open_{i}', f'high_{i}', f'low_{i}', f'close_{i}'])
    
    data = pd.DataFrame(index=dates, columns=columns)
    
    # Generate price paths
    for i in range(n_assets):
        # Initial price
        price = 100 * (1 + 0.5 * np.random.rand())
        prices = [price]
        
        # Generate price path
        for _ in range(1, n_days):
            # Daily return with mean=0.0002 (5% annual) and std=0.01 (16% annual volatility)
            daily_return = np.random.normal(0.0002, 0.01)
            price *= np.exp(daily_return)
            prices.append(price)
        
        # Generate OHLC data
        for j, price in enumerate(prices):
            # Random intraday variation
            high_low_range = price * np.random.uniform(0.01, 0.03)
            high = price + high_low_range / 2
            low = price - high_low_range / 2
            open_price = np.random.uniform(low, high)
            close_price = np.random.uniform(low, high)
            
            data.iloc[j][f'open_{i}'] = open_price
            data.iloc[j][f'high_{i}'] = high
            data.iloc[j][f'low_{i}'] = low
            data.iloc[j][f'close_{i}'] = close_price
    
    return data

def calculate_performance_metrics(returns, portfolio_values):
    """
    Calculate performance metrics as described in the paper
    
    Args:
        returns: List of daily log returns
        portfolio_values: List of portfolio values
    
    Returns:
        Dictionary of performance metrics
    """
    returns = np.array(returns)
    
    # Annualized average return
    annual_return = np.mean(returns) * 252
    
    # Annualized volatility
    annual_volatility = np.std(returns) * np.sqrt(252)
    
    # Sharpe ratio
    sharpe_ratio = annual_return / annual_volatility if annual_volatility > 0 else 0
    
    # Sortino ratio (using downside deviation)
    downside_returns = returns[returns < 0]
    downside_deviation = np.std(downside_returns) * np.sqrt(252) if len(downside_returns) > 0 else 1e-6
    sortino_ratio = annual_return / downside_deviation
    
    # Maximum drawdown
    portfolio_values = np.array(portfolio_values)
    max_values = np.maximum.accumulate(portfolio_values)
    drawdowns = (max_values - portfolio_values) / max_values
    max_drawdown = np.max(drawdowns)
    
    # Calmar ratio
    calmar_ratio = annual_return / max_drawdown if max_drawdown > 0 else 0
    
    # Percentage of positive returns
    pct_positive = np.sum(returns > 0) / len(returns)
    
    # Average profit-loss ratio
    profits = returns[returns > 0]
    losses = returns[returns < 0]
    avg_profit = np.mean(profits) if len(profits) > 0 else 0
    avg_loss = np.abs(np.mean(losses)) if len(losses) > 0 else 1e-6
    pl_ratio = avg_profit / avg_loss
    
    return {
        'E(R)': annual_return,
        'Std(R)': annual_volatility,
        'Sharpe': sharpe_ratio,
        'Sortino': sortino_ratio,
        'MDD': max_drawdown,
        'Calmar': calmar_ratio,
        '%of+Ret': pct_positive,
        'Ave P/Ave L': pl_ratio
    }
