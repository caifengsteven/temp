class DeepLearningFinanceModel:
    """
    Deep Learning model for financial asset management based on the systematic review paper.
    Implements LSTM for price forecasting and a simple portfolio management strategy.
    """
    
    def __init__(self, sequence_length=60, epochs=50, batch_size=32, lstm_units=50):
        self.sequence_length = sequence_length
        self.epochs = epochs
        self.batch_size = batch_size
        self.lstm_units = lstm_units
        self.model = None
        self.scaler = MinMaxScaler(feature_range=(0, 1))
        
    def create_model(self, input_shape):
        """Creates an LSTM model as mentioned in the paper as the most common approach"""
        model = Sequential()
        model.add(LSTM(units=self.lstm_units, return_sequences=True, input_shape=input_shape))
        model.add(Dropout(0.2))
        model.add(LSTM(units=self.lstm_units, return_sequences=False))
        model.add(Dropout(0.2))
        model.add(Dense(units=1))
        model.compile(optimizer='adam', loss='mean_squared_error')
        return model
    
    def prepare_data(self, data):
        """Prepare time series data for LSTM model"""
        scaled_data = self.scaler.fit_transform(data.reshape(-1, 1))
        
        X, y = [], []

            X.append(scaled_data[i-self.sequence_length:i, 0])

        train_size = int(len(X) * 0.8)
        X_train, X_test = X[:train_size], X[train_size:]

        return X_train, y_train, X_test, y_test, scaled_data
    
    def train(self, data):
        """Train the LSTM model on financial time series data"""
        X_train, y_train, X_test, y_test, scaled_data = self.prepare_data(data)
        
        self.model = self.create_model((X_train.shape[1], 1))
        
        history = self.model.fit(
            X_train, y_train,
            epochs=self.epochs,
            batch_size=self.batch_size,
            validation_data=(X_test, y_test),
            verbose=0
        )
        
        return history, X_test, y_test, scaled_data
    
    def predict(self, X_test):
        """Generate predictions from the trained model"""
        if self.model is None:
            raise ValueError("Model has not been trained yet")
        
        predictions = self.model.predict(X_test)
        predictions = self.scaler.inverse_transform(predictions)

    def evaluate(self, predictions, actual_data):
        """Evaluate model performance using common metrics from the paper"""
        # Convert actual data to the original scale
        actual_data = self.scaler.inverse_transform(actual_data.reshape(-1, 1))
        
        # Calculate metrics mentioned in the paper
        rmse = np.sqrt(mean_squared_error(actual_data, predictions))

class PortfolioManager:
    """
    Simple portfolio management strategy using the price forecast from the DL model.
    Implements a basic trading strategy based on the predictions.
    """
    
    def __init__(self, initial_capital=10000):
        self.initial_capital = initial_capital
        self.capital = initial_capital
        self.holdings = 0
        self.portfolio_values = [initial_capital]
        self.trade_history = []
        
    def reset(self):
        """Reset the portfolio to initial state"""
        self.capital = self.initial_capital
        self.holdings = 0
        self.portfolio_values = [self.initial_capital]
        self.trade_history = []
        
    def calculate_signals(self, actual_prices, predicted_prices):
        """Generate buy/sell signals based on predicted price movements"""
        signals = np.zeros(len(actual_prices))
        
        for i in range(1, len(predicted_prices)):
            # Buy signal: predicted price is higher than current price
            if predicted_prices[i] > actual_prices[i-1]:
                signals[i] = 1  # Buy signal
            # Sell signal: predicted price is lower than current price
            elif predicted_prices[i] < actual_prices[i-1]:
                signals[i] = -1  # Sell signal
                
        return signals
    
    def backtest(self, actual_prices, predicted_prices):
        """Backtest trading strategy with actual and predicted prices"""
        self.reset()
        
        signals = self.calculate_signals(actual_prices, predicted_prices)
        
        for i in range(1, len(actual_prices)):
            current_price = actual_prices[i]
            signal = signals[i]
            
            # Execute trades based on signals
            if signal == 1 and self.capital > 0:  # Buy signal
                shares_to_buy = self.capital // current_price
                cost = shares_to_buy * current_price
                self.holdings += shares_to_buy
                self.capital -= cost
                self.trade_history.append(('BUY', i, shares_to_buy, current_price, cost))
                
            elif signal == -1 and self.holdings > 0:  # Sell signal
                proceeds = self.holdings * current_price
                self.trade_history.append(('SELL', i, self.holdings, current_price, proceeds))
                self.capital += proceeds
                self.holdings = 0
            
            # Calculate portfolio value
            portfolio_value = self.capital + (self.holdings * current_price)
            self.portfolio_values.append(portfolio_value)
            
        return self.portfolio_values
    
    def calculate_performance_metrics(self, benchmark_returns=None):
        """Calculate portfolio performance metrics as mentioned in the paper"""
        portfolio_values = np.array(self.portfolio_values)
        
        # Calculate returns
        returns = np.diff(portfolio_values) / portfolio_values[:-1]
        
        # Total return
        total_return = (portfolio_values[-1] - self.initial_capital) / self.initial_capital * 100
        
        # Compound Annual Growth Rate (CAGR)
        days = len(portfolio_values)
        cagr = (portfolio_values[-1] / self.initial_capital) ** (365 / days) - 1
        
        # Risk metrics
        std_dev = np.std(returns) * np.sqrt(252)  # Annualized
        
        # Sharpe Ratio (assuming risk-free rate = 0 for simplicity)
        mean_return = np.mean(returns) * 252  # Annualized
        sharpe_ratio = mean_return / std_dev if std_dev != 0 else 0
        
        # Maximum Drawdown
        cumulative_returns = np.cumprod(1 + returns)
        running_max = np.maximum.accumulate(cumulative_returns)
        drawdown = (running_max - cumulative_returns) / running_max
        max_drawdown = np.max(drawdown) * 100
        
        metrics = {
            'Total Return (%)': total_return,
            'CAGR (%)': cagr * 100,
            'Annualized Volatility (%)': std_dev * 100,
            'Sharpe Ratio': sharpe_ratio,
            'Maximum Drawdown (%)': max_drawdown,
            'Number of Trades': len(self.trade_history)
        }
        
        return metrics

def generate_synthetic_price_data(n_days=1000, volatility=0.01, starting_price=100, trend=0.0001, seed=None):
    """
    Generate synthetic price data for testing.
    Implements a geometric Brownian motion with trend.
    """
    if seed is not None:
        np.random.seed(seed)
        
    returns = np.random.normal(loc=trend, scale=volatility, size=n_days)
    price_path = starting_price * np.cumprod(1 + returns)
    
    # Add some seasonality
    t = np.arange(n_days)
    seasonality = 0.05 * np.sin(2 * np.pi * t / 252) + 0.03 * np.sin(2 * np.pi * t / 63)
    price_path = price_path * (1 + seasonality)
    
    return price_path

def generate_multi_asset_data(n_assets=5, n_days=1000, correlations=None, seed=None):
    """Generate correlated multi-asset data for portfolio testing"""
    if seed is not None:
        np.random.seed(seed)
    
    if correlations is None:
        # Generate random correlation matrix
        random_matrix = np.random.random((n_assets, n_assets))
        # Make it symmetric
        correlation_matrix = (random_matrix + random_matrix.T) / 2
        # Set diagonal to 1
        np.fill_diagonal(correlation_matrix, 1)
    else:
        correlation_matrix = correlations
    
    # Generate correlated returns
    mean_returns = np.random.uniform(0.0001, 0.0005, n_assets)
    volatilities = np.random.uniform(0.01, 0.02, n_assets)
    
    # Cholesky decomposition
    L = np.linalg.cholesky(correlation_matrix)
    
    # Generate uncorrelated returns
    uncorrelated_returns = np.random.normal(size=(n_days, n_assets))
    
    # Generate correlated returns
    correlated_returns = uncorrelated_returns @ L.T
    
    # Scale returns
    for i in range(n_assets):
        correlated_returns[:, i] = mean_returns[i] + volatilities[i] * correlated_returns[:, i]
    
    # Generate price paths
    prices = np.zeros((n_days, n_assets))
    starting_prices = np.random.uniform(50, 200, n_assets)
    
    for i in range(n_assets):
        prices[0, i] = starting_prices[i]
        for t in range(1, n_days):
            prices[t, i] = prices[t-1, i] * (1 + correlated_returns[t, i])
    
    return prices
