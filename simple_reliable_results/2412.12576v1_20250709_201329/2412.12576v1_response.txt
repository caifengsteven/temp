    def _winsorize_column(self, series, z_threshold=3):
        """
        Winsorize a column using Z-scores.
        
        Parameters:
        series (Series): Data column
        z_threshold (float): Z-score threshold for winsorization
        
        Returns:
        Series: Winsorized data
        """
        z_scores = stats.zscore(series, nan_policy='omit')
        series_winsorized = series.copy()
        
        # Clip values beyond threshold
        series_winsorized[(z_scores > z_threshold) & ~np.isnan(z_scores)] = series.mean() + z_threshold * series.std()
        series_winsorized[(z_scores < -z_threshold) & ~np.isnan(z_scores)] = series.mean() - z_threshold * series.std()
        
        return series_winsorized
